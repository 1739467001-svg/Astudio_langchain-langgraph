## 二、实现

```python
from langchain_core.documents import Document
from dataclasses import dataclass
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
import json, re

@dataclass
class ExplainedResult:
    """带解释的检索结果。"""
    content: str
    source: str
    relevance_score: float
    explanation: str     # 为什么检索到这个
    citation_id: int = 0

class RetrievalExplainer:
    """检索结果解释器。"""

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def explain(self, query: str, docs: list[Document]) -> list[ExplainedResult]:
        """为每个检索结果生成解释。"""
        results = []

        # 批量生成解释（简化：逐个）
        for i, doc in enumerate(docs[:5]):
            # 计算相关度（关键词重叠）
            query_words = set(query.lower().split())
            doc_words = set(doc.page_content.lower().split())
            overlap = len(query_words & doc_words) / max(len(query_words), 1)

            # LLM生成解释
            explanation = await self._generate_explanation(query, doc.page_content[:500])

            results.append(ExplainedResult(
                content=doc.page_content[:300],
                source=doc.metadata.get("source", "未知"),
                relevance_score=round(overlap, 4),
                explanation=explanation,
                citation_id=i + 1,
            ))

        return results

    async def _generate_explanation(self, query: str, content: str) -> str:
        """LLM生成：为什么这个文档与查询相关。"""
        prompt = f"""用一句话解释以下文档为什么与用户查询相关。

查询: {query}
文档: {content[:300]}

解释:"""
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content[:150]

    @staticmethod
    def format_for_display(results: list[ExplainedResult]) -> str:
        """格式化展示文本。"""
        lines = ["## 检索结果（附解释）"]
        for r in results:
            lines.append(f"\n### [{r.citation_id}] 来源: {r.source}")
            lines.append(f"相关度: {r.relevance_score:.0%}")
            lines.append(f"解释: {r.explanation}")
            lines.append(f"内容: {r.content[:100]}...")
        return "\n".join(lines)
```
