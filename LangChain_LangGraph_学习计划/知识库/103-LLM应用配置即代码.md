> 把模型选择、Prompt、Chain 结构等"配置"用声明式文件管理，而非硬编码在 Python 中。

## 二、配置文件结构

```yaml
# config/app_config.yaml

# 模型配置
model:
  primary:
    provider: openai
    name: gpt-4o-mini
    temperature: 0
    max_tokens: 500
  fallback:
    provider: ollama
    name: qwen2
    temperature: 0

# Prompt配置
prompts:
  qa_system:
    version: v1.1
    system: "你是友好的AI助手。用简洁的中文回答。"
    template: "{input}"

  rag_system:
    version: v2.0
    system: |
      你是知识库问答助手。
      规则：
      1. 只基于背景知识回答
      2. 不知道就说不知道
      3. 标注来源
    template: "背景：{context}\n问题：{question}"

# RAG配置
rag:
  top_k: 3
  chunk_size: 500
  chunk_overlap: 50
  embedding_model: text-embedding-3-small

# Agent配置
agent:
  max_iterations: 5
  verbose: true
  tools:
    - search_web
    - calculator

# 安全配置
security:
  input_guardrails: true
  output_guardrails: true
  rate_limit_per_minute: 20
  pii_masking: true

# 功能开关
feature_flags:
  streaming: true
  caching: true
  feedback: true
  multimodal: false
```

## 三、配置加载器

```python
import yaml
import os
from typing import Any

class ConfigAsCode:
    """配置即代码管理器"""
    def __init__(self, config_path: str = "config/app_config.yaml"):
        self.config_path = config_path
        self._config = {}
        self._last_modified = 0
        self._load()

    def _load(self):
        """加载YAML配置"""
        if not os.path.exists(self.config_path):
            return

        mtime = os.path.getmtime(self.config_path)
        if mtime == self._last_modified:
            return

        with open(self.config_path, "r", encoding="utf-8") as f:
            self._config = yaml.safe_load(f) or {}
        self._last_modified = mtime
        print(f"✅ 配置已加载: {self.config_path}")

    def get(self, path: str, default: Any = None) -> Any:
        """用点号路径获取配置: get('model.primary.name')"""
        self._load()  # 热加载
        keys = path.split(".")
        value = self._config
        for key in keys:
            if isinstance(value, dict):
                value = value.get(key)
            else:
                return default
        return value if value is not None else default

    def get_model_config(self) -> dict:
        return self.get("model.primary", {})

    def get_prompt(self, name: str) -> str:
        return self.get(f"prompts.{name}.template", "")

    def get_rag_config(self) -> dict:
        return self.get("rag", {})

    def is_enabled(self, feature: str) -> bool:
        return self.get(f"feature_flags.{feature}", False)

# 全局实例
config = ConfigAsCode()
```

## 四、在应用中使用

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate

def build_app_from_config():
    """从配置文件构建应用"""
    model_cfg = config.get_model_config()
    llm = ChatOpenAI(
        model=model_cfg.get("name", "gpt-4o-mini"),
        temperature=model_cfg.get("temperature", 0),
        max_tokens=model_cfg.get("max_tokens", 500),
    )

    prompt_template = config.get_prompt("qa_system")
    prompt = ChatPromptTemplate.from_template(prompt_template)

    chain = prompt | llm

    # 根据配置决定是否启用功能
    if config.is_enabled("streaming"):
        # 包装为流式
        pass

    return chain

# 配置变更时只需改YAML文件，无需改代码
# 例如：把model从gpt-4o-mini改成gpt-4o
# 只需改YAML: name: gpt-4o
```
