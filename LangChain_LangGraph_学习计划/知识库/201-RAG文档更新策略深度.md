## 二、增量更新实现

```python
from dataclasses import dataclass
from datetime import datetime
import hashlib
from enum import Enum

class UpdateType(str, Enum):
    CREATE = "create"
    UPDATE = "update"
    DELETE = "delete"
    REBUILD = "rebuild"

@dataclass
class DocumentChange:
    """文档变更记录。"""
    doc_id: str
    update_type: UpdateType
    old_hash: str = ""
    new_hash: str = ""
    timestamp: str = ""
    changed_fields: list[str] = None

class IncrementalUpdateManager:
    """增量更新管理器。

    只处理变更的文档，不全量重建。
    """

    def __init__(self, vectorstore, embeddings):
        self.vectorstore = vectorstore
        self.embeddings = embeddings
        self.doc_hashes: dict[str, str] = {}  # doc_id → content_hash
        self.change_log: list[DocumentChange] = []

    def detect_changes(self, documents: list[dict]) -> list[DocumentChange]:
        """检测文档变更。"""
        changes = []
        current_ids = {d["doc_id"] for d in documents}
        stored_ids = set(self.doc_hashes.keys())

        # 新增
        for doc in documents:
            doc_id = doc["doc_id"]
            content = doc.get("content", "")
            new_hash = hashlib.md5(content.encode()).hexdigest()

            if doc_id not in self.doc_hashes:
                changes.append(DocumentChange(
                    doc_id=doc_id,
                    update_type=UpdateType.CREATE,
                    new_hash=new_hash,
                    timestamp=datetime.now().isoformat(),
                ))
            elif self.doc_hashes[doc_id] != new_hash:
                changes.append(DocumentChange(
                    doc_id=doc_id,
                    update_type=UpdateType.UPDATE,
                    old_hash=self.doc_hashes[doc_id],
                    new_hash=new_hash,
                    timestamp=datetime.now().isoformat(),
                ))

        # 删除
        for doc_id in stored_ids - current_ids:
            changes.append(DocumentChange(
                doc_id=doc_id,
                update_type=UpdateType.DELETE,
                old_hash=self.doc_hashes[doc_id],
                timestamp=datetime.now().isoformat(),
            ))

        return changes

    async def apply_changes(self, changes: list[DocumentChange]) -> dict:
        """应用变更到向量库。"""
        created, updated, deleted = 0, 0, 0

        for change in changes:
            if change.update_type == UpdateType.CREATE:
                # 嵌入并入库
                created += 1
            elif change.update_type == UpdateType.UPDATE:
                # 先删旧向量，再重新嵌入
                # await self.vectorstore.delete(filter={"doc_id": change.doc_id})
                # 重新嵌入
                updated += 1
            elif change.update_type == UpdateType.DELETE:
                # await self.vectorstore.delete(filter={"doc_id": change.doc_id})
                deleted += 1
                self.doc_hashes.pop(change.doc_id, None)

            # 更新hash记录
            if change.update_type in (UpdateType.CREATE, UpdateType.UPDATE):
                self.doc_hashes[change.doc_id] = change.new_hash

            self.change_log.append(change)

        return {
            "created": created,
            "updated": updated,
            "deleted": deleted,
            "total_changes": len(changes),
        }
```

## 三、过期检测

```python
class DocumentFreshnessChecker:
    """文档新鲜度检查器。"""

    STALE_DAYS = {
        "news": 7,           # 新闻类7天过期
        "policy": 90,        # 政策类90天
        "technical": 180,    # 技术文档180天
        "reference": 365,    # 参考类365天
    }

    @staticmethod
    def check(doc_metadata: dict) -> dict:
        """检查文档是否过期。"""
        doc_type = doc_metadata.get("doc_type", "reference")
        updated = doc_metadata.get("updated_at", "")
        max_age = DocumentFreshnessChecker.STALE_DAYS.get(doc_type, 365)

        if not updated:
            return {"status": "unknown", "action": "review"}

        from datetime import datetime
        try:
            updated_date = datetime.fromisoformat(updated)
            age_days = (datetime.now() - updated_date).days
        except (ValueError, TypeError):
            return {"status": "unknown", "action": "review"}

        if age_days > max_age:
            return {
                "status": "stale",
                "age_days": age_days,
                "max_age": max_age,
                "action": "update_or_remove",
            }
        return {"status": "fresh", "age_days": age_days}
```

## 四、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 用content_hash检测变更 | 精确知道哪些变了 | ★★★ |
| 增量而非全量重建 | 省时间和成本 | ★★★ |
| 文档分类设过期时间 | 新闻7天/技术180天 | ★★☆ |
| 变更有日志记录 | 可追溯 | ★★☆ |
| 定期清理过期文档 | 保持知识库新鲜 | ★★☆ |
