## 二、指标导出格式

```python
class PrometheusExporter:
    """将LLM应用指标导出为Prometheus格式。

    可被Grafana直接消费。
    """

    def __init__(self):
        self.metrics: dict[str, list] = {}

    def record(self, name: str, value: float, **labels):
        """记录指标。"""
        if name not in self.metrics:
            self.metrics[name] = []
        label_str = ",".join(f'{k}="{v}"' for k, v in sorted(labels.items()))
        self.metrics[name].append((value, label_str))

    def export(self) -> str:
        """导出为Prometheus文本格式。"""
        lines = []
        for name, records in self.metrics.items():
            for value, labels in records:
                if labels:
                    lines.append(f"{name}{{{labels}}} {value}")
                else:
                    lines.append(f"{name} {value}")
        return "\n".join(lines)

# 使用示例
exporter = PrometheusExporter()
exporter.record("llm_request_total", 1, model="gpt-4o", status="success")
exporter.record("llm_latency_seconds", 1.5, model="gpt-4o")
exporter.record("llm_tokens_total", 350, model="gpt-4o", type="input")
exporter.record("llm_cost_usd", 0.002, model="gpt-4o")
# 输出:
# llm_request_total{model="gpt-4o",status="success"} 1
# llm_latency_seconds{model="gpt-4o"} 1.5
```

## 三、FastAPI 指标端点

```python
from fastapi import FastAPI, Response
import time
from collections import defaultdict

app = FastAPI()

class MetricsCollector:
    """简化版指标采集器。"""
    def __init__(self):
        self.counters = defaultdict(float)
        self.histograms = defaultdict(list)

    def inc(self, name: str, value: float = 1, **labels):
        key = f"{name}{{{','.join(f'{k}={v}' for k,v in sorted(labels.items()))}}}"
        self.counters[key] += value

    def observe(self, name: str, value: float, **labels):
        self.histograms[name].append(value)
        self.inc(name + "_count", 1, **labels)
        self.inc(name + "_sum", value, **labels)

    def export_prometheus(self) -> str:
        lines = []
        for key, val in self.counters.items():
            lines.append(f"{key} {val}")
        for name, values in self.histograms.items():
            if values:
                lines.append(f"{name}_count {len(values)}")
                lines.append(f"{name}_sum {sum(values)}")
                lines.append(f"{name}_avg {sum(values)/len(values):.4f}")
        return "\n".join(lines)

metrics = MetricsCollector()

@app.get("/metrics")
async def prometheus_metrics():
    """Prometheus指标端点。"""
    return Response(
        content=metrics.export_prometheus(),
        media_type="text/plain",
    )

@app.post("/chat")
async def chat(request: dict):
    """带指标采集的聊天端点。"""
    start = time.time()
    # 调用LLM...
    latency = time.time() - start

    metrics.inc("llm_request_total", 1, model="gpt-4o", status="success")
    metrics.observe("llm_latency_seconds", latency, model="gpt-4o")
    metrics.inc("llm_tokens_total", 350, model="gpt-4o", type="input")

    return {"response": "回答"}
```

## 五、告警可视化

```python
class AlertDashboard:
    """告警可视化数据生成器。"""

    def __init__(self):
        self.alerts: list[dict] = []

    def add_alert(
        self,
        name: str,
        severity: str,
        message: str,
        value: float,
        threshold: float,
    ):
        """添加告警。"""
        self.alerts.append({
            "name": name,
            "severity": severity,
            "message": message,
            "value": value,
            "threshold": threshold,
            "timestamp": time.time(),
            "status": "firing",
        })

    def resolve_alert(self, name: str):
        """解除告警。"""
        for alert in self.alerts:
            if alert["name"] == name and alert["status"] == "firing":
                alert["status"] = "resolved"
                alert["resolved_at"] = time.time()

    def active_alerts(self) -> list[dict]:
        """活跃告警。"""
        return [a for a in self.alerts if a["status"] == "firing"]

    def summary(self) -> dict:
        """告警摘要。"""
        active = self.active_alerts()
        by_severity = {}
        for a in active:
            by_severity[a["severity"]] = by_severity.get(a["severity"], 0) + 1

        return {
            "total_active": len(active),
            "by_severity": by_severity,
            "critical": by_severity.get("critical", 0),
            "high": by_severity.get("high", 0),
            "warning": by_severity.get("warning", 0),
        }
```

## 七、通知渠道集成

```python
class NotificationSender:
    """多渠道告警通知。"""

    @staticmethod
    async def send_slack(webhook_url: str, alert: dict):
        """发送Slack通知。"""
        import httpx
        message = {
            "text": f"🚨 {alert['severity'].upper()}: {alert['name']}",
            "attachments": [{
                "color": "danger" if alert["severity"] == "critical" else "warning",
                "fields": [
                    {"title": "告警", "value": alert["message"], "short": False},
                    {"title": "当前值", "value": str(alert["value"]), "short": True},
                    {"title": "阈值", "value": str(alert["threshold"]), "short": True},
                ],
            }],
        }
        async with httpx.AsyncClient() as client:
            await client.post(webhook_url, json=message)

    @staticmethod
    async def send_webhook(url: str, alert: dict):
        """发送Webhook通知。"""
        import httpx
        async with httpx.AsyncClient() as client:
            await client.post(url, json=alert)
```
