## 二、实现

```python
from dataclasses import dataclass, field
from pathlib import Path
from datetime import datetime
import yaml, time

@dataclass
class AppConfig:
    """应用配置。"""
    model: str = "gpt-4o-mini"
    temperature: float = 0.7
    max_tokens: int = 1000
    cache_enabled: bool = True
    cache_ttl: int = 3600
    rate_limit_qps: int = 10
    rag_k: int = 5
    rerank: bool = True
    security: dict = field(default_factory=lambda: {"input_validation": True, "pii_redaction": True})

class ConfigLoader:
    """配置加载器。"""

    @staticmethod
    def load(path: str) -> AppConfig:
        """从YAML加载。"""
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or {}
        return AppConfig(**data)

    @staticmethod
    def validate(config: AppConfig) -> dict:
        """验证配置。"""
        issues = []
        if not config.model: issues.append("缺少模型")
        if config.max_tokens < 1: issues.append("max_tokens太小")
        if config.rag_k > 20: issues.append("rag_k太大")
        return {"valid": len(issues) == 0, "issues": issues}

    @staticmethod
    def diff(old: AppConfig, new: AppConfig) -> dict:
        """对比两个版本配置。"""
        changes = {}
        for field_name in ["model", "temperature", "max_tokens", "cache_enabled", "rate_limit_qps", "rag_k"]:
            old_val = getattr(old, field_name)
            new_val = getattr(new, field_name)
            if old_val != new_val:
                changes[field_name] = {"old": old_val, "new": new_val}
        return {"has_changes": len(changes) > 0, "changes": changes}


class HotReloadManager:
    """热更新管理器。"""

    def __init__(self, config_path: str):
        self.path = Path(config_path)
        self.config: AppConfig = ConfigLoader.load(config_path)
        self._last_modified: float = self.path.stat().st_mtime
        self._callbacks: list = []

    def on_change(self, callback):
        self._callbacks.append(callback)

    def check_and_reload(self) -> bool:
        """检查文件变更并热加载。"""
        current = self.path.stat().st_mtime
        if current > self._last_modified:
            old = self.config
            self.config = ConfigLoader.load(str(self.path))
            self._last_modified = current
            for cb in self._callbacks:
                try:
                    cb(ConfigLoader.diff(old, self.config))
                except Exception:
                    pass
            return True
        return False


class ConfigVersionManager:
    """配置版本管理器。"""

    def __init__(self):
        self.versions: dict[str, AppConfig] = {}
        self.active: str = ""

    def save_version(self, version: str, config: AppConfig):
        self.versions[version] = config

    def activate(self, version: str) -> bool:
        if version in self.versions:
            self.active = version
            return True
        return False

    def rollback(self) -> str | None:
        """回滚到上一版本。"""
        versions = sorted(self.versions.keys())
        if self.active in versions:
            idx = versions.index(self.active)
            if idx > 0:
                self.active = versions[idx - 1]
                return self.active
        return None
```
