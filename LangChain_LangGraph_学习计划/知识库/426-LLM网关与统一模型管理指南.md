# LLM 网关与统一模型管理指南

> 应用直接调 OpenAI API、直接调 Anthropic API、直接调本地模型——每家 SDK 不同、参数不同、错误处理不同。换一家模型要改一堆代码。LLM 网关在应用和模型之间加一层代理，统一接口、统一计费、统一限流、统一监控。

---

## 1. 什么是 LLM 网关

### 没有网关的架构

```
应用 A → OpenAI SDK → OpenAI API
应用 B → Anthropic SDK → Anthropic API
应用 C → 自定义 HTTP → 本地 vLLM

问题：
  - 三套 SDK、三套配置、三套错误处理
  - 换模型 = 改代码
  - 无法统一限流和计费
  - 无法统一监控
  - API Key 散落各处
```

### 有网关的架构

```
应用 A ─┐
应用 B ─┼→ LLM 网关 ─┬→ OpenAI API
应用 C ─┘            ├→ Anthropic API
                     ├→ 本地 vLLM
                     └→ Azure OpenAI

网关职责：统一接口 + 路由 + 限流 + 计费 + 监控 + 缓存 + 降级
应用只需要一个 SDK、一个 Key、一个端点
```

## 2. 网关核心功能

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
import time
import hashlib
import json

class ModelProvider(Enum):
    OPENAI = "openai"
    ANTHROPIC = "anthropic"
    AZURE = "azure"
    LOCAL = "local"
    GOOGLE = "google"


@dataclass
class ModelConfig:
    """模型配置"""
    name: str                        # "gpt-4o-mini"
    provider: ModelProvider          # OPENAI
    # API 配置
    api_key: str = ""
    base_url: str = ""
    # 参数
    max_tokens: int = 4096
    temperature: float = 0.0
    # 成本
    input_price_per_m: float = 0.15   # 每百万 Token 输入价格
    output_price_per_m: float = 0.60
    # 限制
    rate_limit_rpm: int = 500        # 每分钟请求限制
    rate_limit_tpm: int = 150000     # 每分钟 Token 限制
    # 状态
    enabled: bool = True
    priority: int = 1                # 优先级（降级链用）


@dataclass
class ChatRequest:
    """统一聊天请求"""
    messages: list[dict]
    model: str = "gpt-4o-mini"
    temperature: float = 0.0
    max_tokens: int = 4096
    stream: bool = False
    # 网关用元数据
    user_id: str = ""
    session_id: str = ""
    request_id: str = field(default_factory=lambda: str(hashlib.md5(str(time.time()).encode()).hexdigest()[:8]))


@dataclass
class ChatResponse:
    """统一聊天响应"""
    content: str
    model: str = ""
    provider: str = ""
    # Token 统计
    input_tokens: int = 0
    output_tokens: int = 0
    # 成本
    cost_usd: float = 0.0
    # 性能
    latency_ms: float = 0.0
    # 元数据
    request_id: str = ""
    from_cache: bool = False
    # 错误
    error: str | None = None
```

## 3. LLM 网关实现

```python
from langchain_openai import ChatOpenAI
from langchain_anthropic import ChatAnthropic
from langchain_core.messages import HumanMessage, SystemMessage, AIMessage
from collections import defaultdict
import asyncio

class LLMGateway:
    """LLM 统一网关"""

    def __init__(self):
        self.models: dict[str, ModelConfig] = {}
        self.providers: dict[str, Any] = {}  # provider_name → llm instance
        # 限流
        self.request_counts: dict[str, list[float]] = defaultdict(list)  # model → [timestamps]
        self.token_counts: dict[str, list[float]] = defaultdict(list)
        # 缓存
        self.cache: dict[str, ChatResponse] = {}
        # 统计
        self.stats: dict[str, dict] = defaultdict(lambda: {
            "requests": 0, "tokens": 0, "cost": 0.0, "errors": 0, "cache_hits": 0,
        })

    def register_model(self, config: ModelConfig):
        """注册模型"""
        self.models[config.name] = config

    def get_model(self, name: str) -> ModelConfig | None:
        """获取模型配置"""
        return self.models.get(name)

    def chat(self, request: ChatRequest) -> ChatResponse:
        """统一聊天接口"""
        start = time.time()

        # 1. 缓存检查
        cache_key = self._cache_key(request)
        if cache_key in self.cache:
            cached = self.cache[cache_key]
            cached.from_cache = True
            self.stats[request.model]["cache_hits"] += 1
            return cached

        # 2. 获取模型配置
        config = self.models.get(request.model)
        if not config:
            return ChatResponse(
                content="", error=f"模型 {request.model} 未注册",
                request_id=request.request_id,
            )

        if not config.enabled:
            # 尝试降级
            fallback = self._get_fallback(request.model)
            if fallback:
                config = fallback
                request.model = config.name
            else:
                return ChatResponse(
                    content="", error=f"模型 {request.model} 不可用且无降级",
                    request_id=request.request_id,
                )

        # 3. 限流检查
        if not self._check_rate_limit(config):
            return ChatResponse(
                content="", error=f"模型 {config.name} 限流",
                request_id=request.request_id,
            )

        # 4. 调用模型
        try:
            response = self._call_model(config, request)
            latency = (time.time() - start) * 1000

            # 5. 计算成本
            input_tokens = self._estimate_tokens(request.messages)
            output_tokens = self._estimate_tokens([{"content": response}])
            cost = (
                input_tokens / 1_000_000 * config.input_price_per_m +
                output_tokens / 1_000_000 * config.output_price_per_m
            )

            result = ChatResponse(
                content=response,
                model=config.name,
                provider=config.provider.value,
                input_tokens=input_tokens,
                output_tokens=output_tokens,
                cost_usd=cost,
                latency_ms=latency,
                request_id=request.request_id,
            )

            # 6. 缓存
            if request.temperature == 0:  # 只有 temperature=0 才缓存
                self.cache[cache_key] = result

            # 7. 统计
            self.stats[config.name]["requests"] += 1
            self.stats[config.name]["tokens"] += input_tokens + output_tokens
            self.stats[config.name]["cost"] += cost

            return result

        except Exception as e:
            self.stats[config.name]["errors"] += 1
            return ChatResponse(
                content="", error=str(e),
                request_id=request.request_id,
            )

    def _call_model(self, config: ModelConfig, request: ChatRequest) -> str:
        """调用具体模型"""
        messages = []
        for msg in request.messages:
            role = msg.get("role", "user")
            content = msg.get("content", "")
            if role == "system":
                messages.append(SystemMessage(content=content))
            elif role == "assistant":
                messages.append(AIMessage(content=content))
            else:
                messages.append(HumanMessage(content=content))

        if config.provider == ModelProvider.OPENAI:
            llm = ChatOpenAI(
                model=config.name,
                temperature=request.temperature,
                max_tokens=request.max_tokens,
                api_key=config.api_key,
                base_url=config.base_url or None,
            )
        elif config.provider == ModelProvider.ANTHROPIC:
            llm = ChatAnthropic(
                model=config.name,
                temperature=request.temperature,
                max_tokens=request.max_tokens,
                api_key=config.api_key,
            )
        else:
            # 通用 OpenAI 兼容接口（vLLM/Ollama 等）
            llm = ChatOpenAI(
                model=config.name,
                temperature=request.temperature,
                max_tokens=request.max_tokens,
                api_key=config.api_key or "dummy",
                base_url=config.base_url,
            )

        response = llm.invoke(messages)
        return response.content

    def _get_fallback(self, model_name: str) -> ModelConfig | None:
        """获取降级模型"""
        # 按优先级找下一个可用模型
        available = [
            m for m in self.models.values()
            if m.enabled and m.name != model_name
        ]
        available.sort(key=lambda m: m.priority)
        return available[0] if available else None

    def _check_rate_limit(self, config: ModelConfig) -> bool:
        """检查限流"""
        now = time.time()
        # 清理 60 秒前的记录
        self.request_counts[config.name] = [
            t for t in self.request_counts[config.name] if now - t < 60
        ]
        # 检查 RPM
        if len(self.request_counts[config.name]) >= config.rate_limit_rpm:
            return False

        self.request_counts[config.name].append(now)
        return True

    @staticmethod
    def _cache_key(request: ChatRequest) -> str:
        """生成缓存键"""
        key_data = json.dumps({
            "model": request.model,
            "messages": request.messages,
            "temperature": request.temperature,
        }, sort_keys=True, ensure_ascii=False)
        return hashlib.sha256(key_data.encode()).hexdigest()[:32]

    @staticmethod
    def _estimate_tokens(messages: list[dict]) -> int:
        """粗略估算 Token 数"""
        total = 0
        for msg in messages:
            content = msg.get("content", "")
            chinese = sum(1 for c in content if '\u4e00' <= c <= '\u9fff')
            other = len(content) - chinese
            total += int(chinese * 1.5 + other / 4)
        return total

    def dashboard(self) -> dict:
        """网关仪表盘"""
        return {
            "models": [
                {
                    "name": m.name,
                    "provider": m.provider.value,
                    "enabled": m.enabled,
                    "priority": m.priority,
                }
                for m in self.models.values()
            ],
            "stats": dict(self.stats),
            "cache_size": len(self.cache),
        }
```

## 4. 使用示例

```python
# 创建网关
gateway = LLMGateway()

# 注册模型
gateway.register_model(ModelConfig(
    name="gpt-4o-mini",
    provider=ModelProvider.OPENAI,
    api_key="sk-xxx",
    input_price_per_m=0.15,
    output_price_per_m=0.60,
    rate_limit_rpm=500,
    priority=1,
))

gateway.register_model(ModelConfig(
    name="gpt-4o",
    provider=ModelProvider.OPENAI,
    api_key="sk-xxx",
    input_price_per_m=2.50,
    output_price_per_m=10.00,
    rate_limit_rpm=500,
    priority=2,  # 降级时使用
))

gateway.register_model(ModelConfig(
    name="claude-3-5-sonnet",
    provider=ModelProvider.ANTHROPIC,
    api_key="sk-ant-xxx",
    input_price_per_m=3.00,
    output_price_per_m=15.00,
    rate_limit_rpm=50,
    priority=3,
))

# 统一调用（应用层不关心底层模型）
response = gateway.chat(ChatRequest(
    messages=[
        {"role": "system", "content": "你是助手"},
        {"role": "user", "content": "什么是 RAG？"},
    ],
    model="gpt-4o-mini",
    temperature=0,
    user_id="user_001",
))

print(f"回答：{response.content[:100]}")
print(f"模型：{response.model}")
print(f"Token：{response.input_tokens}+{response.output_tokens}")
print(f"成本：${response.cost_usd:.4f}")
print(f"延迟：{response.latency_ms:.0f}ms")
print(f"缓存：{response.from_cache}")

# 仪表盘
print(f"\n网关状态：{gateway.dashboard()}")
```

## 5. 高级功能

### 智能路由

```python
class SmartRouter:
    """智能路由器：根据请求特征选择最优模型"""

    def __init__(self, gateway: LLMGateway):
        self.gateway = gateway
        self.routing_rules: list[dict] = []

    def add_rule(self, condition: Callable, model: str):
        """添加路由规则"""
        self.routing_rules.append({"condition": condition, "model": model})

    def route(self, request: ChatRequest) -> str:
        """选择模型"""
        for rule in self.routing_rules:
            if rule["condition"](request):
                return rule["model"]
        return request.model  # 默认

# 路由规则
router = SmartRouter(gateway)
router.add_rule(
    condition=lambda r: len(str(r.messages)) > 5000,  # 长上下文用大模型
    model="gpt-4o",
)
router.add_rule(
    condition=lambda r: any("code" in m.get("content", "") for m in r.messages),  # 代码任务
    model="gpt-4o",
)
router.add_rule(
    condition=lambda r: r.user_id.startswith("free_"),  # 免费用户用小模型
    model="gpt-4o-mini",
)
```

### 成本追踪

```python
class CostTracker:
    """成本追踪器"""

    def __init__(self, gateway: LLMGateway):
        self.gateway = gateway
        self.cost_log: list[dict] = []

    def record(self, response: ChatResponse, request: ChatRequest):
        """记录成本"""
        self.cost_log.append({
            "user_id": request.user_id,
            "model": response.model,
            "input_tokens": response.input_tokens,
            "output_tokens": response.output_tokens,
            "cost": response.cost_usd,
            "timestamp": time.time(),
        })

    def report(self, group_by: str = "model") -> dict:
        """成本报告"""
        from collections import defaultdict
        grouped = defaultdict(lambda: {"cost": 0, "tokens": 0, "requests": 0})

        for entry in self.cost_log:
            key = entry[group_by]
            grouped[key]["cost"] += entry["cost"]
            grouped[key]["tokens"] += entry["input_tokens"] + entry["output_tokens"]
            grouped[key]["requests"] += 1

        return {
            "total_cost": sum(e["cost"] for e in self.cost_log),
            "by_group": dict(grouped),
            "avg_cost_per_request": sum(e["cost"] for e in self.cost_log) / max(len(self.cost_log), 1),
        }
```

## 6. 网关功能矩阵

| 功能 | 说明 | 价值 |
|------|------|------|
| 统一接口 | 一套 SDK 调用所有模型 | 降低开发成本 |
| 模型路由 | 按请求特征选模型 | 优化成本 |
| 降级链 | 模型不可用时自动切换 | 提高可用性 |
| 限流 | RPM/TPM 限制 | 防止超额 |
| 缓存 | 相同请求返回缓存 | 降本加速 |
| 计费 | 统一成本追踪 | 成本透明 |
| 监控 | 统一延迟/错误统计 | 可观测 |
| 脱敏 | 敏感信息过滤 | 安全合规 |
| 审计 | 所有调用记录 | 合规 |
