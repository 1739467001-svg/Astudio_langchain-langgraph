## 二、实现

```python
import hashlib
from dataclasses import dataclass
from datetime import datetime

@dataclass
class CanaryConfig:
    """金丝雀配置。"""
    stages: list = None
    current_stage: int = 0
    rollback_thresholds: dict = None

    def __post_init__(self):
        self.stages = self.stages or [
            ("5%", 1800), ("25%", 3600), ("50%", 7200), ("100%", 0),
        ]
        self.rollback_thresholds = self.rollback_thresholds or {
            "error_rate": 0.05, "latency_p95": 5000,
        }

class CanaryReleaseManager:
    """金丝雀发布管理器。"""

    def __init__(self, config: CanaryConfig = CanaryConfig()):
        self.config = config

    def should_route_new(self, user_id: str) -> bool:
        """判断用户是否路由到新版本。"""
        if self.config.current_stage >= len(self.config.stages):
            return True
        pct = int(self.config.stages[self.config.current_stage][0].replace("%", ""))
        h = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 100
        return h < pct

    def check_health(self, metrics: dict) -> dict:
        """健康检查。"""
        checks = {
            "error_rate": metrics.get("error_rate", 0) <= self.config.rollback_thresholds["error_rate"],
            "latency": metrics.get("latency_p95", 0) <= self.config.rollback_thresholds["latency_p95"],
        }
        all_pass = all(checks.values())
        return {"healthy": all_pass, "action": "advance" if all_pass else "rollback", "checks": checks}

    def advance(self) -> bool:
        if self.config.current_stage < len(self.config.stages) - 1:
            self.config.current_stage += 1
            return True
        return False

    def rollback(self):
        self.config.current_stage = 0


class FeatureFlagManager:
    """特性开关管理器。"""

    def __init__(self):
        self.flags: dict[str, dict] = {}

    def set(self, name: str, enabled: bool, percentage: float = 1.0):
        self.flags[name] = {"enabled": enabled, "percentage": percentage}

    def is_enabled(self, name: str, user_id: str = "") -> bool:
        flag = self.flags.get(name)
        if not flag or not flag["enabled"]:
            return False
        if flag["percentage"] >= 1.0:
            return True
        h = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 100
        return h < flag["percentage"] * 100

    def disable(self, name: str):
        """秒级禁用。"""
        if name in self.flags:
            self.flags[name]["enabled"] = False
```
