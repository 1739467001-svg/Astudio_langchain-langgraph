## 二、数据质量标准

```python
from dataclasses import dataclass
from enum import Enum

class DataQuality(str, Enum):
    HIGH = "high"       # 可直接入库
    MEDIUM = "medium"   # 需清洗后入库
    LOW = "low"          # 不建议入库
    REJECT = "reject"    # 拒绝

@dataclass
class QualityReport:
    """数据质量报告。"""
    total_documents: int = 0
    high_quality: int = 0
    medium_quality: int = 0
    low_quality: int = 0
    rejected: int = 0
    issues: list[str] = None

    def __post_init__(self):
        if self.issues is None:
            self.issues = []

    @property
    def quality_score(self) -> float:
        """质量评分 = 高质量占比。"""
        total = max(self.total_documents, 1)
        return round(self.high_quality / total, 4)

class DataQualityChecker:
    """数据质量检查器。"""

    @staticmethod
    def check(text: str, metadata: dict = None) -> tuple[DataQuality, list[str]]:
        """检查单个文档质量。"""
        issues = []

        # 1. 长度检查
        if len(text) < 50:
            issues.append("内容太短（<50字符）")
        elif len(text) > 50000:
            issues.append("内容过长（>50000字符），需分块")

        # 2. 空白检查
        if text.strip() != text:
            issues.append("首尾有多余空白")
        if "  " in text:
            issues.append("含连续空格")
        if "\n\n\n" in text:
            issues.append("含多余空行")

        # 3. 编码检查
        if "\ufffd" in text:  # 替换字符
            issues.append("含编码错误字符")
        if any(ord(c) < 32 and c not in "\n\r\t" for c in text):
            issues.append("含控制字符")

        # 4. 语言检查
        chinese_ratio = sum(1 for c in text if '\u4e00' <= c <= '\u9fff') / max(len(text), 1)
        if 0 < chinese_ratio < 0.1:
            issues.append("中文占比过低，可能是乱码")

        # 5. 重复检查
        lines = text.split("\n")
        if len(lines) > 5:
            unique_lines = set(lines)
            if len(unique_lines) / len(lines) < 0.5:
                issues.append("行重复率高（>50%）")

        # 6. 元数据检查
        if metadata:
            if not metadata.get("source"):
                issues.append("缺少来源元数据")
            if not metadata.get("created_at") and not metadata.get("date"):
                issues.append("缺少时间元数据")

        # 评级
        if len(issues) == 0:
            return DataQuality.HIGH, issues
        elif len(issues) <= 2:
            return DataQuality.MEDIUM, issues
        elif len(issues) <= 4:
            return DataQuality.LOW, issues
        else:
            return DataQuality.REJECT, issues
```

## 三、数据清洗规则

```python
import re
from langchain_core.documents import Document

class DataCleaner:
    """数据清洗器。"""

    @staticmethod
    def clean(text: str) -> str:
        """完整清洗流程。"""
        # 1. 去除HTML标签
        text = re.sub(r'<[^>]+>', '', text)

        # 2. 去除URL（可选保留）
        text = re.sub(r'https?://\S+', '[URL]', text)

        # 3. 去除控制字符
        text = ''.join(c for c in text if ord(c) >= 32 or c in '\n\r\t')

        # 4. 标准化空白
        text = re.sub(r'[ \t]+', ' ', text)          # 连续空格→单个
        text = re.sub(r'\n{3,}', '\n\n', text)        # 多个空行→双空行
        text = text.strip()

        # 5. 去除零宽字符
        text = text.replace('\u200b', '').replace('\u200c', '').replace('\u200d', '')
        text = text.replace('\ufeff', '')

        # 6. 修复常见编码问题
        text = text.replace('\u201c', '"').replace('\u201d', '"')  # 弯引号
        text = text.replace('\u2018', "'").replace('\u2019', "'")
        text = text.replace('—', '--').replace('…', '...')

        return text

    @staticmethod
    def clean_documents(docs: list[Document]) -> list[Document]:
        """批量清洗文档。"""
        cleaned = []
        for doc in docs:
            clean_text = DataCleaner.clean(doc.page_content)
            if len(clean_text.strip()) >= 50:  # 太短丢弃
                cleaned.append(Document(
                    page_content=clean_text,
                    metadata=doc.metadata,
                ))
        return cleaned
```

## 四、数据增强

```python
class DataAugmenter:
    """数据增强器——提升数据质量。"""

    @staticmethod
    def add_metadata(doc: Document) -> Document:
        """自动添加元数据。"""
        text = doc.page_content

        # 添加字符/词数统计
        doc.metadata["char_count"] = len(text)
        doc.metadata["word_count"] = len(text.split())

        # 添加语言检测
        chinese_count = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        doc.metadata["language"] = "zh" if chinese_count > len(text) * 0.1 else "en"

        # 添加内容hash（去重用）
        import hashlib
        doc.metadata["content_hash"] = hashlib.md5(text.encode()).hexdigest()

        return doc

    @staticmethod
    def deduplicate(docs: list[Document]) -> list[Document]:
        """去重——基于内容hash。"""
        seen = set()
        result = []
        for doc in docs:
            h = doc.metadata.get("content_hash") or hashlib.md5(
                doc.page_content.encode()
            ).hexdigest()
            if h not in seen:
                seen.add(h)
                result.append(doc)
        return result
```
