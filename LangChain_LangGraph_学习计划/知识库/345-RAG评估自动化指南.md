## 二、实现

```python
import asyncio
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class EvalTestSuite:
    """评估测试套件。"""
    name: str
    test_cases: list[dict] = field(default_factory=list)  # [{question, expected, keywords}]
    baseline: dict = field(default_factory=dict)

@dataclass
class EvalResult:
    """评估结果。"""
    total: int
    passed: int
    pass_rate: float
    avg_latency_ms: float
    details: list[dict] = field(default_factory=list)

class AutomatedEvalRunner:
    """自动化评估运行器。"""

    def __init__(self, rag_pipeline, judge_llm=None):
        self.pipeline = rag_pipeline
        self.judge = judge_llm

    async def run_suite(self, suite: EvalTestSuite) -> EvalResult:
        """运行评估套件。"""
        results = []
        semaphore = asyncio.Semaphore(5)  # 限制并发

        async def eval_one(tc: dict):
            async with semaphore:
                response = await self.pipeline.answer(tc["question"])
                answer = response.get("answer", "")

                # 检查关键词覆盖
                keywords = tc.get("keywords", [])
                keyword_coverage = sum(1 for kw in keywords if kw in answer) / max(len(keywords), 1)

                # 检查答案长度
                has_content = len(answer) > 20

                passed = keyword_coverage >= 0.5 and has_content

                return {"question": tc["question"][:50], "passed": passed, "coverage": round(keyword_coverage, 2)}

        tasks = [eval_one(tc) for tc in suite.test_cases]
        results = await asyncio.gather(*tasks)

        passed = sum(1 for r in results if r["passed"])
        n = len(results)

        return EvalResult(
            total=n, passed=passed,
            pass_rate=round(passed / max(n, 1), 4),
            avg_latency_ms=0,
            details=results,
        )

    def compare_baseline(self, result: EvalResult, baseline: dict) -> dict:
        """对比基线。"""
        baseline_rate = baseline.get("pass_rate", 0)
        current_rate = result.pass_rate
        delta = current_rate - baseline_rate

        return {
            "baseline_pass_rate": baseline_rate,
            "current_pass_rate": current_rate,
            "delta": round(delta, 4),
            "regressed": delta < 0,
            "should_block": delta < -0.05,  # 退化>5%阻止
        }

    def generate_report(self, result: EvalResult, comparison: dict) -> str:
        """生成报告。"""
        status = "✅ 通过" if not comparison["regressed"] else "❌ 退化"
        return f"""# RAG评估报告

## 结果
- 通过率: {result.pass_rate:.1%} ({result.passed}/{result.total})
- 基线对比: {comparison['baseline_pass_rate']:.1%} → {comparison['current_pass_rate']:.1%} (Δ={comparison['delta']:+.1%})
- 状态: {status}

## 详情
"""
        for d in result.details[:10]:
            icon = "✅" if d["passed"] else "❌"
            report += f"- {icon} {d['question']} (覆盖: {d['coverage']:.0%})\n"
        return report
```

## 三、CI/CD 集成

```python
class CIEvalIntegration:
    """CI/CD评估集成。"""

    @staticmethod
    async def run_ci_check(runner: AutomatedEvalRunner, suite: EvalTestSuite) -> dict:
        """CI中运行的评估检查。"""
        result = await runner.run_suite(suite)
        comparison = runner.compare_baseline(result, suite.baseline)

        return {
            "ci_passed": not comparison["should_block"],
            "pass_rate": result.pass_rate,
            "regressed": comparison["regressed"],
            "delta": comparison["delta"],
            "report": runner.generate_report(result, comparison),
        }
```
