## 二、评审检查器

```python
class DesignDocChecker:
    """设计文档评审检查器。"""

    MUST_HAVE = [
        "目标可量化", "有架构图", "技术栈选择",
        "数据安全方案", "降级策略", "SLO定义",
        "风险缓解", "里程碑",
    ]

    @classmethod
    def review(cls, doc: str) -> dict:
        """检查文档完整性。"""
        keywords = {
            "目标可量化": ["成功标准", "量化", "指标"],
            "有架构图": ["mermaid", "graph", "架构"],
            "技术栈选择": ["LangGraph", "技术栈", "模型"],
            "数据安全方案": ["脱敏", "权限", "安全"],
            "降级策略": ["降级", "fallback", "备用"],
            "SLO定义": ["SLO", "可用性", "延迟"],
            "风险缓解": ["风险", "缓解", "措施"],
            "里程碑": ["里程碑", "MVP", "阶段"],
        }
        missing = []
        for item in cls.MUST_HAVE:
            found = any(kw in doc for kw in keywords.get(item, [item]))
            if not found:
                missing.append(item)
        return {
            "complete": len(missing) == 0,
            "missing": missing,
            "score": round(1 - len(missing) / len(cls.MUST_HAVE), 2),
        }
```
