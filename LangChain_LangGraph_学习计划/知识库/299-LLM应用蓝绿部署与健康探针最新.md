## 二、实现

```python
import asyncio, time
from dataclasses import dataclass
from enum import Enum

class DeployStatus(str, Enum):
    BLUE = "blue"
    GREEN = "green"
    SWITCHING = "switching"

@dataclass
class HealthCheck:
    """健康探针。"""
    endpoint: str = "/health"
    timeout: int = 5
    interval: int = 10
    healthy_threshold: int = 3  # 连续成功次数

class BlueGreenDeployer:
    """蓝绿部署管理器。"""

    def __init__(self):
        self.current: DeployStatus = DeployStatus.BLUE
        self.health = HealthCheck()
        self.consecutive_ok: int = 0

    async def health_check(self, target: str) -> bool:
        """执行健康检查。"""
        # 模拟健康检查
        return True  # 实际调用target的/health端点

    async def deploy_green(self, green_ready: bool = True) -> dict:
        """部署Green环境并切换。"""
        # 1. 等待Green就绪
        if not green_ready:
            return {"status": "green_not_ready"}

        # 2. 健康检查
        for _ in range(self.health.healthy_threshold):
            ok = await self.health_check("green")
            if ok:
                self.consecutive_ok += 1
            else:
                self.consecutive_ok = 0
                return {"status": "health_check_failed"}
            await asyncio.sleep(self.health.interval)

        # 3. 切换流量
        self.current = DeployStatus.SWITCHING
        old = "blue" if self.current != DeployStatus.GREEN else "green"
        self.current = DeployStatus.GREEN
        return {"status": "switched", "from": old, "to": "green"}

    def rollback(self) -> dict:
        """回滚到Blue。"""
        self.current = DeployStatus.BLUE
        return {"status": "rolled_back", "to": "blue"}


class HealthProbe:
    """健康探针——三层检查。"""

    @staticmethod
    async def liveness(app) -> bool:
        """存活探针——进程是否活着。"""
        return True

    @staticmethod
    async def readiness(app) -> bool:
        """就绪探针——是否能处理请求。"""
        # 检查LLM/向量库/数据库是否可用
        checks = {
            "llm": True,
            "vector_db": True,
            "database": True,
        }
        return all(checks.values())

    @staticmethod
    async def startup(app) -> bool:
        """启动探针——启动是否完成。"""
        return True
```
