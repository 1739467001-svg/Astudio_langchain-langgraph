## 三、实现

### 3.1 文档元数据带权限

```python
from langchain_core.documents import Document

# 每个文档都带权限元数据
docs = [
    Document(
        page_content="销售月度报告...",
        metadata={
            "source": "sales_report.pdf",
            "department": "sales",         # 部门权限
            "classification": "internal",   # 级别: public/internal/confidential
            "allowed_users": ["user_001", "user_002"],  # 精确ACL
            "allowed_roles": ["sales_manager", "ceo"],
        }
    ),
    Document(
        page_content="财务季度报告...",
        metadata={
            "source": "finance_report.pdf",
            "department": "finance",
            "classification": "confidential",
            "allowed_users": ["user_003"],
            "allowed_roles": ["finance_manager", "ceo"],
        }
    ),
]
```

### 3.2 权限感知检索

```python
class PermissionAwareRetriever:
    """带权限控制的检索器"""
    def __init__(self, vectorstore):
        self.vectorstore = vectorstore

    def search_with_permissions(
        self,
        query: str,
        user_id: str,
        user_department: str,
        user_level: int,  # 1=public, 2=internal, 3=confidential
        k: int = 5,
    ) -> list:
        """带权限过滤的检索"""
        # 先检索更多结果
        raw_results = self.vectorstore.similarity_search(query, k=k*3)

        # 过滤：只返回用户有权访问的
        filtered = []
        for doc in raw_results:
            if self._has_access(doc, user_id, user_department, user_level):
                filtered.append(doc)
            if len(filtered) >= k:
                break

        return filtered

    def _has_access(self, doc, user_id, user_department, user_level) -> bool:
        """检查用户是否有权访问文档"""
        meta = doc.metadata

        # 检查1: 部门匹配
        doc_dept = meta.get("department")
        if doc_dept and doc_dept != user_department and doc_dept != "all":
            return False

        # 检查2: 级别匹配
        level_map = {"public": 1, "internal": 2, "confidential": 3}
        doc_level = level_map.get(meta.get("classification", "public"), 1)
        if doc_level > user_level:
            return False

        # 检查3: ACL（如果有allowed_users列表）
        allowed_users = meta.get("allowed_users", [])
        if allowed_users and user_id not in allowed_users:
            # 检查角色
            allowed_roles = meta.get("allowed_roles", [])
            user_roles = get_user_roles(user_id)  # 从外部获取用户角色
            if not any(role in allowed_roles for role in user_roles):
                return False

        return True

# 使用
retriever = PermissionAwareRetriever(vectorstore)
results = retriever.search_with_permissions(
    query="季度报告",
    user_id="user_001",
    user_department="sales",
    user_level=2,  # internal级别
    k=3,
)
# 只返回销售部+internal级别+user_001有权访问的文档
```

## 四、权限检查清单

| 检查项 | 说明 | 状态 |
|--------|------|------|
| 文档带权限元数据 | 每个文档有department/classification/ACL | ☐ |
| 检索前过滤 | 检索结果按权限过滤 | ☐ |
| 部门隔离 | 不同部门不能互相检索 | ☐ |
| 级别控制 | 机密文档只给高级别用户 | ☐ |
| ACL精确控制 | 特定文档限制特定用户 | ☐ |
| 缓存隔离 | 缓存Key包含用户权限信息 | ☐ |
| 审计日志 | 记录谁检索了什么 | ☐ |
