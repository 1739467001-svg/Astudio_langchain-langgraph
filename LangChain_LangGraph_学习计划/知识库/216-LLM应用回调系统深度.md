# LLM 应用回调系统深度

> 回调系统让你在不修改核心代码的情况下，监听 LLM 应用的关键事件——请求开始、Token 生成、工具调用、错误发生。这是监控、日志、追踪的基础设施。

## 二、回调实现

```python
from typing import Callable, Any
from dataclasses import dataclass, field
from datetime import datetime
from langchain_core.callbacks import BaseCallbackHandler

@dataclass
class CallbackEvent:
    """回调事件记录。"""
    event: str
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())
    data: dict = field(default_factory=dict)

class LoggingCallback(BaseCallbackHandler):
    """日志回调——记录所有事件。"""

    def __init__(self):
        self.events: list[CallbackEvent] = []

    def on_llm_start(self, serialized, prompts, **kwargs):
        self.events.append(CallbackEvent(
            event="llm_start",
            data={"model": serialized.get("name", "unknown") if serialized else "unknown"},
        ))

    def on_llm_new_token(self, token: str, **kwargs):
        self.events.append(CallbackEvent(
            event="token",
            data={"token": token[:50]},
        ))

    def on_llm_end(self, response, **kwargs):
        self.events.append(CallbackEvent(
            event="llm_end",
            data={"tokens": len(response.generations) if response else 0},
        ))

    def on_tool_start(self, serialized, input_str, **kwargs):
        self.events.append(CallbackEvent(
            event="tool_start",
            data={"tool": serialized.get("name", "unknown") if serialized else "unknown"},
        ))

    def on_tool_end(self, output, **kwargs):
        self.events.append(CallbackEvent(
            event="tool_end",
            data={"output": str(output)[:100]},
        ))

    def on_error(self, error, **kwargs):
        self.events.append(CallbackEvent(
            event="error",
            data={"error": str(error)[:200]},
        ))


class MetricsCallback(BaseCallbackHandler):
    """指标回调——采集性能指标。"""

    def __init__(self):
        self.llm_calls = 0
        self.token_count = 0
        self.tool_calls = 0
        self.errors = 0
        self._start_time = None

    def on_llm_start(self, serialized, prompts, **kwargs):
        self.llm_calls += 1
        self._start_time = datetime.now()

    def on_llm_new_token(self, token, **kwargs):
        self.token_count += 1

    def on_tool_start(self, serialized, input_str, **kwargs):
        self.tool_calls += 1

    def on_error(self, error, **kwargs):
        self.errors += 1

    def summary(self) -> dict:
        return {
            "llm_calls": self.llm_calls,
            "token_count": self.token_count,
            "tool_calls": self.tool_calls,
            "errors": self.errors,
        }


class TracingCallback(BaseCallbackHandler):
    """追踪回调——生成执行轨迹。"""

    def __init__(self):
        self.trace: list[dict] = []
        self._depth = 0

    def on_chain_start(self, serialized, inputs, **kwargs):
        name = serialized.get("name", "chain") if serialized else "chain"
        self.trace.append({
            "type": "chain_start",
            "name": name,
            "depth": self._depth,
            "timestamp": datetime.now().isoformat(),
        })
        self._depth += 1

    def on_chain_end(self, outputs, **kwargs):
        self._depth = max(0, self._depth - 1)
        self.trace.append({
            "type": "chain_end",
            "depth": self._depth,
            "timestamp": datetime.now().isoformat(),
        })

    def on_tool_start(self, serialized, input_str, **kwargs):
        name = serialized.get("name", "tool") if serialized else "tool"
        self.trace.append({
            "type": "tool_start",
            "name": name,
            "input": input_str[:100] if input_str else "",
            "depth": self._depth,
        })

    def on_tool_end(self, output, **kwargs):
        self.trace.append({
            "type": "tool_end",
            "output": str(output)[:100] if output else "",
            "depth": self._depth,
        })

    def get_trace_text(self) -> str:
        """生成可读的追踪文本。"""
        lines = []
        for event in self.trace:
            indent = "  " * event.get("depth", 0)
            etype = event["type"]
            name = event.get("name", "")
            lines.append(f"{indent}{etype}: {name}")
        return "\n".join(lines)
```

## 三、回调组合使用

```python
class CompositeCallback(BaseCallbackHandler):
    """组合回调——同时运行多个回调。"""

    def __init__(self, callbacks: list[BaseCallbackHandler]):
        self.callbacks = callbacks

    def on_llm_start(self, serialized, prompts, **kwargs):
        for cb in self.callbacks:
            try:
                cb.on_llm_start(serialized, prompts, **kwargs)
            except Exception:
                pass  # 一个回调失败不影响其他

    def on_llm_new_token(self, token, **kwargs):
        for cb in self.callbacks:
            try:
                cb.on_llm_new_token(token, **kwargs)
            except Exception:
                pass

    def on_tool_start(self, serialized, input_str, **kwargs):
        for cb in self.callbacks:
            try:
                cb.on_tool_start(serialized, input_str, **kwargs)
            except Exception:
                pass

    def on_tool_end(self, output, **kwargs):
        for cb in self.callbacks:
            try:
                cb.on_tool_end(output, **kwargs)
            except Exception:
                pass

    def on_error(self, error, **kwargs):
        for cb in self.callbacks:
            try:
                cb.on_error(error, **kwargs)
            except Exception:
                pass
```

## 四、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 回调不修改主流程 | 只观察不改数据 | ★★★ |
| 回调要容错 | 一个失败不影响其他 | ★★★ |
| Token回调用于流式 | 打字机效果 | ★★★ |
| 指标回调用于监控 | 采集性能数据 | ★★☆ |
| 追踪回调用于调试 | 可视化执行链 | ★★☆ |
