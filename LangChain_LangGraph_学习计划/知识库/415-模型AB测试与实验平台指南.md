# 模型 AB 测试与实验平台指南

> 换了个新模型，怎么知道是变好了还是变差了？靠感觉不行，需要科学的 A/B 测试——一部分流量用模型 A，一部分用模型 B，用统计方法比较两个版本的差异，让数据说话。

---

## 1. 为什么需要 A/B 测试

### 常见误区

```
误区 1："新模型评测分数更高，一定更好"
→ 评测集和真实用户分布不同，高分不代表好体验

误区 2："我手动看了几十个回答，感觉更好了"
→ 样本太小，主观偏差严重

误区 3："全量切换后满意度下降了"
→ 没有对照组，无法区分是模型问题还是其他因素
```

### 科学的 A/B 测试

```
用户流量 → 随机分流
  ├── 50% → 模型 A（对照组，当前版本）
  └── 50% → 模型 B（实验组，新版本）

收集指标 → 统计检验 → 结论是否有显著差异
```

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
import uuid
import time
import hashlib


class ExperimentStatus(Enum):
    DRAFT = "draft"
    RUNNING = "running"
    PAUSED = "paused"
    COMPLETED = "completed"
    CANCELLED = "cancelled"


class MetricType(Enum):
    CONTINUOUS = "continuous"   # 连续型（延迟、Token 数）
    BINARY = "binary"           # 二元型（满意/不满意）
    CATEGORICAL = "categorical"  # 分类型（评分 1-5）
    RATIO = "ratio"             # 比率型（错误率）


@dataclass
class ExperimentMetric:
    """实验指标"""
    name: str
    type: MetricType
    description: str = ""
    # 统计参数
    min_sample_size: int = 100       # 最小样本量
    significance_level: float = 0.05  # 显著性水平
    power: float = 0.8               # 统计功效
    # 方向
    higher_is_better: bool = True     # 指标越高越好


@dataclass
class ExperimentVariant:
    """实验变体"""
    name: str                          # "control" / "treatment"
    description: str = ""
    config: dict = field(default_factory=dict)  # 模型配置
    traffic_percentage: float = 0.5    # 流量比例
    # 结果
    samples: list[float] = field(default_factory=list)
    metrics_summary: dict = field(default_factory=dict)


@dataclass
class Experiment:
    """A/B 测试实验"""
    id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    name: str = ""
    description: str = ""
    status: ExperimentStatus = ExperimentStatus.DRAFT
    # 变体
    variants: dict[str, ExperimentVariant] = field(default_factory=dict)
    # 指标
    metrics: list[ExperimentMetric] = field(default_factory=list)
    # 时序
    start_time: float | None = None
    end_time: float | None = None
    duration_hours: int = 72           # 默认运行 3 天
    # 结果
    results: dict = field(default_factory=dict)
    conclusion: str = ""
```

### 分流策略

```python
class TrafficSplitter:
    """流量分流器：确定性地将用户分配到变体"""

    def __init__(self, experiment: Experiment):
        self.experiment = experiment

    def assign(self, user_id: str) -> str:
        """确定性地分配用户到变体

        使用 hash(user_id + experiment_id) 确保同一用户始终分配到同一变体
        """
        # 计算哈希
        hash_input = f"{user_id}:{self.experiment.id}"
        hash_value = int(hashlib.md5(hash_input.encode()).hexdigest(), 16)
        # 归一化到 0-1
        normalized = (hash_value % 10000) / 10000

        # 累积分配
        cumulative = 0
        for variant_name, variant in self.experiment.variants.items():
            cumulative += variant.traffic_percentage
            if normalized < cumulative:
                return variant_name

        # 兜底
        return list(self.experiment.variants.keys())[0]

    def assign_batch(self, user_ids: list[str]) -> dict[str, list[str]]:
        """批量分配"""
        assignments: dict[str, list[str]] = {v: [] for v in self.experiment.variants}
        for uid in user_ids:
            variant = self.assign(uid)
            assignments[variant].append(uid)
        return assignments
```

## 3. 指标收集

```python
class MetricsCollector:
    """实验指标收集器"""

    def __init__(self):
        self.data: dict[str, dict[str, list[float]]] = {}
        # data[experiment_id][variant_name] = {metric_name: [values]}

    def record(
        self,
        experiment_id: str,
        variant: str,
        metric_name: str,
        value: float,
    ):
        """记录指标值"""
        if experiment_id not in self.data:
            self.data[experiment_id] = {}
        if variant not in self.data[experiment_id]:
            self.data[experiment_id][variant] = {}
        if metric_name not in self.data[experiment_id][variant]:
            self.data[experiment_id][variant][metric_name] = []

        self.data[experiment_id][variant][metric_name].append(value)

    def get_metric_data(
        self,
        experiment_id: str,
        metric_name: str,
    ) -> dict[str, list[float]]:
        """获取某个指标在所有变体上的数据"""
        result = {}
        for variant, metrics in self.data.get(experiment_id, {}).items():
            if metric_name in metrics:
                result[variant] = metrics[metric_name]
        return result

    def summary(
        self,
        experiment_id: str,
        variant: str,
    ) -> dict[str, float]:
        """获取变体的指标摘要"""
        import numpy as np
        metrics = self.data.get(experiment_id, {}).get(variant, {})
        summary = {}
        for name, values in metrics.items():
            if values:
                summary[name] = {
                    "count": len(values),
                    "mean": float(np.mean(values)),
                    "std": float(np.std(values)),
                    "median": float(np.median(values)),
                    "p95": float(np.percentile(values, 95)),
                }
        return summary
```

## 4. 统计检验

```python
import numpy as np
from scipy import stats

class StatisticalAnalyzer:
    """统计检验器"""

    @staticmethod
    def t_test(
        control: list[float],
        treatment: list[float],
        alpha: float = 0.05,
    ) -> dict:
        """两样本 t 检验（连续型指标）

        假设：两组数据近似正态分布
        """
        if len(control) < 2 or len(treatment) < 2:
            return {
                "test": "t_test",
                "significant": False,
                "reason": "样本量不足",
            }

        t_stat, p_value = stats.ttest_ind(treatment, control)

        # 效应量 (Cohen's d)
        pooled_std = np.sqrt(
            ((len(control) - 1) * np.var(control, ddof=1) +
             (len(treatment) - 1) * np.var(treatment, ddof=1)) /
            (len(control) + len(treatment) - 2)
        )
        cohens_d = (np.mean(treatment) - np.mean(control)) / pooled_std if pooled_std > 0 else 0

        # 置信区间
        se = np.sqrt(np.var(control, ddof=1) / len(control) +
                     np.var(treatment, ddof=1) / len(treatment))
        diff = np.mean(treatment) - np.mean(control)
        ci_lower = diff - 1.96 * se
        ci_upper = diff + 1.96 * se

        return {
            "test": "t_test",
            "control_mean": float(np.mean(control)),
            "treatment_mean": float(np.mean(treatment)),
            "difference": float(diff),
            "difference_pct": float(diff / np.mean(control) * 100) if np.mean(control) != 0 else 0,
            "t_statistic": float(t_stat),
            "p_value": float(p_value),
            "significant": p_value < alpha,
            "confidence_interval": [float(ci_lower), float(ci_upper)],
            "effect_size": float(cohens_d),
            "effect_interpretation": StatisticalAnalyzer._interpret_effect(cohens_d),
        }

    @staticmethod
    def chi_square(
        control_success: int,
        control_total: int,
        treatment_success: int,
        treatment_total: int,
        alpha: float = 0.05,
    ) -> dict:
        """卡方检验（二元型指标：转化率/满意率）"""
        contingency = [
            [control_success, control_total - control_success],
            [treatment_success, treatment_total - treatment_success],
        ]
        chi2, p_value, dof, expected = stats.chi2_contingency(contingency)

        control_rate = control_success / control_total
        treatment_rate = treatment_success / treatment_total

        return {
            "test": "chi_square",
            "control_rate": float(control_rate),
            "treatment_rate": float(treatment_rate),
            "difference": float(treatment_rate - control_rate),
            "chi2_statistic": float(chi2),
            "p_value": float(p_value),
            "significant": p_value < alpha,
        }

    @staticmethod
    def mann_whitney(
        control: list[float],
        treatment: list[float],
        alpha: float = 0.05,
    ) -> dict:
        """Mann-Whitney U 检验（非参数，不假设正态分布）"""
        if len(control) < 2 or len(treatment) < 2:
            return {"test": "mann_whitney", "significant": False, "reason": "样本不足"}

        statistic, p_value = stats.mannwhitneyu(
            treatment, control, alternative="two-sided"
        )

        return {
            "test": "mann_whitney",
            "control_median": float(np.median(control)),
            "treatment_median": float(np.median(treatment)),
            "statistic": float(statistic),
            "p_value": float(p_value),
            "significant": p_value < alpha,
        }

    @staticmethod
    def sample_size(
        baseline: float,
        mde: float,
        alpha: float = 0.05,
        power: float = 0.8,
    ) -> int:
        """计算所需样本量

        Args:
            baseline: 基准指标值（如当前满意度 0.8）
            mde: 最小可检测效应（如 0.05 表示 5% 的变化）
            alpha: 显著性水平
            power: 统计功效
        """
        from scipy.stats import norm

        z_alpha = norm.ppf(1 - alpha / 2)
        z_beta = norm.ppf(power)

        # 比率型样本量公式
        p1 = baseline
        p2 = baseline + mde
        p_avg = (p1 + p2) / 2

        n = ((z_alpha + z_beta) ** 2 * 2 * p_avg * (1 - p_avg)) / (p2 - p1) ** 2
        return int(np.ceil(n))

    @staticmethod
    def _interpret_effect(d: float) -> str:
        if abs(d) < 0.2:
            return "微小效应"
        elif abs(d) < 0.5:
            return "小效应"
        elif abs(d) < 0.8:
            return "中效应"
        else:
            return "大效应"
```

## 5. 完整实验平台

```python
class ExperimentPlatform:
    """A/B 测试实验平台"""

    def __init__(self):
        self.experiments: dict[str, Experiment] = {}
        self.splitters: dict[str, TrafficSplitter] = {}
        self.collector = MetricsCollector()
        self.analyzer = StatisticalAnalyzer()

    def create_experiment(
        self,
        name: str,
        control_config: dict,
        treatment_config: dict,
        metrics: list[ExperimentMetric],
        traffic_split: float = 0.5,
    ) -> Experiment:
        """创建实验"""
        exp = Experiment(
            name=name,
            variants={
                "control": ExperimentVariant(
                    name="control",
                    description="当前版本",
                    config=control_config,
                    traffic_percentage=1 - traffic_split,
                ),
                "treatment": ExperimentVariant(
                    name="treatment",
                    description="新版本",
                    config=treatment_config,
                    traffic_percentage=traffic_split,
                ),
            },
            metrics=metrics,
        )
        self.experiments[exp.id] = exp
        self.splitters[exp.id] = TrafficSplitter(exp)
        return exp

    def start(self, experiment_id: str):
        """启动实验"""
        exp = self.experiments[experiment_id]
        exp.status = ExperimentStatus.RUNNING
        exp.start_time = time.time()

    def get_variant(self, experiment_id: str, user_id: str) -> str:
        """获取用户应该分配到的变体"""
        return self.splitters[experiment_id].assign(user_id)

    def get_config(self, experiment_id: str, user_id: str) -> dict:
        """获取用户应该使用的模型配置"""
        variant_name = self.get_variant(experiment_id, user_id)
        exp = self.experiments[experiment_id]
        return exp.variants[variant_name].config

    def record_metric(
        self,
        experiment_id: str,
        user_id: str,
        metric_name: str,
        value: float,
    ):
        """记录指标"""
        variant = self.get_variant(experiment_id, user_id)
        self.collector.record(experiment_id, variant, metric_name, value)

    def analyze(self, experiment_id: str) -> dict:
        """分析实验结果"""
        exp = self.experiments[experiment_id]
        results = {}

        for metric in exp.metrics:
            data = self.collector.get_metric_data(experiment_id, metric.name)

            if "control" not in data or "treatment" not in data:
                continue

            control_vals = data["control"]
            treatment_vals = data["treatment"]

            # 选择检验方法
            if metric.type == MetricType.CONTINUOUS:
                test_result = self.analyzer.t_test(control_vals, treatment_vals)
            elif metric.type == MetricType.BINARY:
                control_success = sum(control_vals)
                treatment_success = sum(treatment_vals)
                test_result = self.analyzer.chi_square(
                    int(control_success), len(control_vals),
                    int(treatment_success), len(treatment_vals),
                )
            else:
                test_result = self.analyzer.mann_whitney(control_vals, treatment_vals)

            results[metric.name] = test_result

        # 生成结论
        conclusion = self._generate_conclusion(results, exp.metrics)
        exp.results = results
        exp.conclusion = conclusion

        return {
            "experiment": exp.name,
            "status": exp.status.value,
            "results": results,
            "conclusion": conclusion,
        }

    def _generate_conclusion(self, results: dict, metrics: list[ExperimentMetric]) -> str:
        """生成实验结论"""
        significant_metrics = []
        for metric in metrics:
            r = results.get(metric.name, {})
            if r.get("significant"):
                diff = r.get("difference", 0)
                direction = "提升" if diff > 0 else "下降"
                significant_metrics.append(
                    f"{metric.name} {direction} {abs(diff):.4f}"
                    f"（p={r.get('p_value', 0):.4f}）"
                )

        if not significant_metrics:
            return "未发现显著差异，建议延长实验或增加样本量"

        return "显著差异：" + "; ".join(significant_metrics)
```

## 6. 使用示例

```python
# 创建实验平台
platform = ExperimentPlatform()

# 定义指标
metrics = [
    ExperimentMetric(
        name="satisfaction_rate",
        type=MetricType.BINARY,
        description="用户满意度（点赞率）",
        higher_is_better=True,
    ),
    ExperimentMetric(
        name="response_latency_ms",
        type=MetricType.CONTINUOUS,
        description="响应延迟（毫秒）",
        higher_is_better=False,  # 越低越好
    ),
    ExperimentMetric(
        name="token_cost",
        type=MetricType.CONTINUOUS,
        description="每次调用 Token 成本",
        higher_is_better=False,
    ),
]

# 创建实验：对比 gpt-4o-mini 和 gpt-4o
exp = platform.create_experiment(
    name="GPT-4o vs Mini 模型对比",
    control_config={"model": "gpt-4o-mini", "temperature": 0},
    treatment_config={"model": "gpt-4o", "temperature": 0},
    metrics=metrics,
    traffic_split=0.5,
)

# 启动
platform.start(exp.id)

# 模拟数据收集
import random
for i in range(500):
    user_id = f"user_{i}"
    variant = platform.get_variant(exp.id, user_id)

    if variant == "treatment":
        satisfaction = random.random() < 0.85  # 85% 满意
        latency = random.gauss(1500, 300)     # 1.5s
        cost = random.gauss(2000, 200)        # 2000 tokens
    else:
        satisfaction = random.random() < 0.78  # 78% 满意
        latency = random.gauss(800, 200)       # 0.8s
        cost = random.gauss(1200, 150)         # 1200 tokens

    platform.record_metric(exp.id, user_id, "satisfaction_rate", float(satisfaction))
    platform.record_metric(exp.id, user_id, "response_latency_ms", latency)
    platform.record_metric(exp.id, user_id, "token_cost", cost)

# 分析结果
results = platform.analyze(exp.id)
print(f"实验: {results['experiment']}")
print(f"结论: {results['conclusion']}")
for metric_name, result in results["results"].items():
    print(f"\n{metric_name}:")
    for k, v in result.items():
        print(f"  {k}: {v}")
```

## 7. 最佳实践

| 实践 | 说明 |
|------|------|
| 先算样本量 | 确保有足够数据检测到差异 |
| 一次只改一个变量 | 否则无法归因 |
| 随机分流 | 用 hash(user_id) 确保随机性 |
| 运行足够长 | 至少 3-7 天，覆盖周期性波动 |
| 看多个指标 | 满意度上升但延迟也上升未必好 |
| 防止偷看 | 不要中途提前下结论 |
| 灰度发布 | 从 5% 开始逐步扩大 |

### 常见陷阱

| 陷阱 | 后果 | 解决 |
|------|------|------|
| 样本量不足 | 假阴性，漏掉真实差异 | 提前计算样本量 |
| 辛普森悖论 | 分层结果和总体矛盾 | 检查子群体分布 |
| 幸存者偏差 | 只看完成用户 | 关注中途流失 |
| 网络效应 | 用户互相影响 | 隔离实验组 |
| 新奇效应 | 新东西初期表现好 | 等效应消退 |
