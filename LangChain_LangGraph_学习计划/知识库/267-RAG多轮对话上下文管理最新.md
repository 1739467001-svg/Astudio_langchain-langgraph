## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class ConversationTurn:
    turn_id: int
    user_message: str
    assistant_response: str
    entities: list[str] = field(default_factory=list)

class ConversationContext:
    """多轮对话上下文管理器。"""

    def __init__(self, max_turns: int = 10):
        self.turns: list[ConversationTurn] = []
        self.max_turns = max_turns
        self.summary: str = ""

    def add_turn(self, turn: ConversationTurn):
        self.turns.append(turn)
        if len(self.turns) > self.max_turns:
            old = self.turns[:len(self.turns) // 2]
            self.turns = self.turns[len(self.turns) // 2:]
            self._update_summary(old)

    def _update_summary(self, old_turns: list[ConversationTurn]):
        parts = [f"第{t.turn_id}轮: {t.user_message[:50]}" for t in old_turns]
        self.summary = "; ".join(parts)

    def resolve_query(self, current_query: str) -> str:
        """指代消解——消除'它''这个'等代词。"""
        if not self.turns:
            return current_query

        pronouns = ["它", "它们", "这个", "那个", "上面"]
        has_pronoun = any(p in current_query for p in pronouns)
        if not has_pronoun:
            return current_query

        last_entities = self.turns[-1].entities
        if not last_entities:
            return current_query

        resolved = current_query
        for pronoun in pronouns:
            if pronoun in resolved:
                resolved = resolved.replace(pronoun, last_entities[0])
                break
        return resolved

    def get_context_text(self) -> str:
        """为LLM生成上下文文本。"""
        parts = []
        if self.summary:
            parts.append(f"[历史摘要] {self.summary}")
        for turn in self.turns[-3:]:  # 最近3轮
            parts.append(f"用户: {turn.user_message[:100]}")
            parts.append(f"助手: {turn.assistant_response[:100]}")
        return "\n".join(parts)

    def detect_topic_shift(self, current_query: str) -> bool:
        """检测话题切换。"""
        if not self.turns:
            return False
        last_words = set(self.turns[-1].user_message.lower().split())
        current_words = set(current_query.lower().split())
        return len(last_words & current_words) == 0 and len(current_words) > 3
```
