## 二、种子数据来源

```python
class SeedDataSources:
    """种子数据来源。"""

    SOURCES = {
        "已有文档": {"优先": "最高", "量": "50-200条", "特点": "真实准确"},
        "人工构造": {"优先": "高", "量": "30-50条", "特点": "可控覆盖"},
        "LLM生成": {"优先": "中", "量": "50-100条", "特点": "快需审核"},
        "公开数据集": {"优先": "低", "量": "视情况", "特点": "需适配"},
    }

    @classmethod
    def recommended(cls):
        return [
            "1. 已有文档50条（最高质量）",
            "2. 人工构造30条（覆盖核心场景）",
            "3. LLM生成50条（扩充边界）",
            "4. RAGAS评估效果",
            "5. 效果达标→扩充到500条",
        ]


class ColdStartValidator:
    """冷启动验证器。"""

    MIN = {"docs": 50, "recall": 0.7, "faithfulness": 0.8}

    @classmethod
    def check_ready(cls, stats: dict) -> dict:
        checks = {
            "文档数": stats.get("docs", 0) >= cls.MIN["docs"],
            "召回率": stats.get("recall", 0) >= cls.MIN["recall"],
            "忠实度": stats.get("faithfulness", 0) >= cls.MIN["faithfulness"],
        }
        return {
            "ready": all(checks.values()),
            "checks": checks,
            "action": "可上线" if all(checks.values()) else "需补充数据",
        }
```
