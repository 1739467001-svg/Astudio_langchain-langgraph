## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import hashlib
from collections import defaultdict

class DocStatus(str, Enum):
    ACTIVE = "active"
    STALE = "stale"
    EXPIRED = "expired"
    DELETED = "deleted"

@dataclass
class ManagedDocument:
    """受管理的文档。"""
    doc_id: str
    content: str
    source: str
    content_hash: str = ""
    status: DocStatus = DocStatus.ACTIVE
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.now().isoformat())
    access_count: int = 0
    last_accessed: str = ""
    metadata: dict = field(default_factory=dict)

class KnowledgeBaseManager:
    """知识库管理系统——全生命周期管理。"""

    def __init__(self, vectorstore=None, embeddings=None):
        self.vectorstore = vectorstore
        self.embeddings = embeddings
        self.documents: dict[str, ManagedDocument] = {}
        self.doc_hashes: dict[str, str] = {}  # source_path → hash

    async def ingest(self, source: str, content: str, metadata: dict = None) -> str:
        """摄入文档。"""
        doc_id = hashlib.md5(f"{source}{datetime.now()}".encode()).hexdigest()[:12]
        content_hash = hashlib.md5(content.encode()).hexdigest()

        doc = ManagedDocument(
            doc_id=doc_id, content=content, source=source,
            content_hash=content_hash, metadata=metadata or {},
        )
        self.documents[doc_id] = doc
        self.doc_hashes[source] = content_hash

        # 入向量库
        if self.vectorstore:
            from langchain_core.documents import Document
            await self.vectorstore.aadd_documents([
                Document(page_content=content, metadata={"doc_id": doc_id, "source": source, **(metadata or {})})
            ])

        return doc_id

    async def update(self, source: str, new_content: str) -> dict:
        """更新文档（检测变更）。"""
        old_hash = self.doc_hashes.get(source)
        new_hash = hashlib.md5(new_content.encode()).hexdigest()

        if old_hash == new_hash:
            return {"status": "no_change"}

        # 找到旧文档并删除
        old_docs = [d for d in self.documents.values() if d.source == source]
        for doc in old_docs:
            doc.status = DocStatus.DELETED

        # 摄入新版本
        doc_id = await self.ingest(source, new_content)
        return {"status": "updated", "new_doc_id": doc_id}

    async def delete(self, source: str) -> dict:
        """删除文档。"""
        deleted = 0
        for doc in self.documents.values():
            if doc.source == source:
                doc.status = DocStatus.DELETED
                deleted += 1
        return {"deleted": deleted}

    def check_freshness(self, max_age_days: int = 90) -> list[dict]:
        """检查文档新鲜度。"""
        now = datetime.now()
        stale_docs = []

        for doc in self.documents.values():
            if doc.status != DocStatus.ACTIVE:
                continue
            updated = datetime.fromisoformat(doc.updated_at)
            age_days = (now - updated).days

            if age_days > max_age_days:
                doc.status = DocStatus.EXPIRED
                stale_docs.append({"doc_id": doc.doc_id, "source": doc.source, "age_days": age_days})
            elif age_days > max_age_days * 0.8:
                doc.status = DocStatus.STALE

        return stale_docs

    def stats(self) -> dict:
        """统计报告。"""
        by_status = defaultdict(int)
        for doc in self.documents.values():
            by_status[doc.status.value] += 1

        return {
            "total": len(self.documents),
            "active": by_status[DocStatus.ACTIVE.value],
            "stale": by_status[DocStatus.STALE.value],
            "expired": by_status[DocStatus.EXPIRED.value],
            "deleted": by_status[DocStatus.DELETED.value],
            "total_access": sum(d.access_count for d in self.documents.values()),
        }
```
