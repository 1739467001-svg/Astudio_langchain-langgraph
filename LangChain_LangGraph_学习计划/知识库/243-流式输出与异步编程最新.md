## 二、astream_events 完整用法

```python
import asyncio
from langgraph.prebuilt import create_react_agent
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool

@tool
def search(query: str) -> str:
    """搜索信息"""
    return f"结果: {query}"

agent = create_react_agent(
    ChatOpenAI(model="gpt-4o", streaming=True),  # 必须streaming=True
    [search],
)

async def stream_complete(question: str):
    """完整事件流消费。"""
    async for event in agent.astream_events(
        {"messages": [{"role": "user", "content": question}]},
        version="v2",  # 必须指定v2
    ):
        kind = event["event"]
        name = event.get("name", "")
        data = event.get("data", {})

        if kind == "on_chat_model_stream":
            # LLM Token流
            chunk = data.get("chunk")
            if chunk and chunk.content:
                print(chunk.content, end="", flush=True)

        elif kind == "on_tool_start":
            print(f"\n🔍 工具开始: {name}")

        elif kind == "on_tool_end":
            print(f"✅ 工具完成: {name}")

        elif kind == "on_chain_end" and name == "LangGraph":
            print("\n[完成]")

asyncio.run(stream_complete("搜索AI最新进展"))
```

## 三、SSE 端点

```python
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
import json

app = FastAPI()

@app.post("/chat/stream")
async def chat_stream(request: dict):
    """SSE流式端点。"""
    question = request["message"]

    async def generate():
        async for event in agent.astream_events(
            {"messages": [{"role": "user", "content": question}]},
            version="v2",
        ):
            if event["event"] == "on_chat_model_stream":
                chunk = event["data"].get("chunk")
                if chunk and chunk.content:
                    yield f"data: {json.dumps({'type': 'token', 'content': chunk.content})}\n\n"
            elif event["event"] == "on_tool_start":
                yield f"data: {json.dumps({'type': 'tool_start', 'name': event['name']})}\n\n"
        yield f"data: {json.dumps({'type': 'done'})}\n\n"

    return StreamingResponse(generate(), media_type="text/event-stream")
```

## 四、stream_mode 多模式

```python
async def stream_modes_demo():
    """stream_mode四种模式演示。"""

    # Mode 1: values — 完整状态
    async for state in agent.astream(input, stream_mode="values"):
        print(f"状态: {list(state.keys())}")

    # Mode 2: updates — 只输出变更
    async for update in agent.astream(input, stream_mode="updates"):
        print(f"更新: {update}")

    # Mode 3: messages — 消息级
    async for msg, metadata in agent.astream(input, stream_mode="messages"):
        print(f"[{msg.__class__.__name__}] {msg.content[:50]}")

    # Mode 4: 多模式同时
    async for output in agent.astream(input, stream_mode=["messages", "updates"]):
        mode, data = output
        if mode == "messages":
            msg, meta = data
            print(msg.content, end="")
```

class StreamingPitfalls:

## 六、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 用astream_events | 最灵活 | ★★★ |
| streaming=True必设 | 否则无Token流 | ★★★ |
| version='v2'必设 | v1已过时 | ★★★ |
| 过滤事件类型 | 防止刷屏 | ★★☆ |
| Nginx关缓冲 | 防SSE延迟 | ★★☆ |
