## 二、故障转移

```python
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime
import time

class FailoverStatus(str, Enum):
    HEALTHY = "healthy"
    DEGRADED = "degraded"
    FAILING = "failing"
    FAILED = "failed"

@dataclass
class HealthCheck:
    """健康检查。"""
    component: str
    is_healthy: bool = True
    latency_ms: float = 0
    last_check: str = field(default_factory=lambda: datetime.now().isoformat())
    consecutive_failures: int = 0

class FailoverManager:
    """故障转移管理器。"""

    def __init__(self, max_failures: int = 3):
        self.max_failures = max_failures
        self.health: dict[str, HealthCheck] = {}
        self.failover_count: int = 0

    def check_health(self, component: str, is_healthy: bool, latency_ms: float = 0):
        """记录健康检查。"""
        hc = self.health.get(component, HealthCheck(component=component))

        if is_healthy:
            hc.consecutive_failures = 0
            hc.is_healthy = True
        else:
            hc.consecutive_failures += 1
            if hc.consecutive_failures >= self.max_failures:
                hc.is_healthy = False
                self.failover_count += 1

        hc.latency_ms = latency_ms
        hc.last_check = datetime.now().isoformat()
        self.health[component] = hc

    def get_status(self, component: str) -> FailoverStatus:
        """获取组件状态。"""
        hc = self.health.get(component)
        if not hc:
            return FailoverStatus.HEALTHY
        if not hc.is_healthy:
            return FailoverStatus.FAILED
        if hc.consecutive_failures > 0:
            return FailoverStatus.DEGRADED
        return FailoverStatus.HEALTHY

    def should_failover(self, component: str) -> bool:
        """是否应该故障转移。"""
        return self.get_status(component) == FailoverStatus.FAILED


class LLMFailoverChain:
    """LLM故障转移链。"""

    def __init__(self):
        self.chain: list[dict] = []
        self.current_index: int = 0

    def add_provider(self, name: str, invoke_func, priority: int = 0):
        """添加LLM提供者。"""
        self.chain.append({
            "name": name,
            "invoke": invoke_func,
            "priority": priority,
        })
        self.chain.sort(key=lambda x: x["priority"])

    async def invoke(self, messages: list) -> str:
        """带故障转移的调用。"""
        last_error = None

        for provider in self.chain:
            try:
                result = await provider["invoke"](messages)
                return result
            except Exception as e:
                last_error = e
                continue

        # 全部失败→返回兜底
        return f"所有LLM提供者不可用: {str(last_error)[:100]}"
```

## 三、数据一致性

```python
class DataSyncManager:
    """多活数据同步管理器。"""

    def __init__(self):
        self.sync_log: list[dict] = []
        self.conflict_count: int = 0

    async def sync_vectorstore(self, primary, replica) -> dict:
        """同步向量库。"""
        # 实际：增量同步变更
        return {"status": "synced", "replicas": 1}

    async def sync_database(self, primary, replica) -> dict:
        """同步数据库。"""
        return {"status": "synced", "lag_seconds": 0}

    def detect_conflict(self, primary_data, replica_data) -> bool:
        """检测数据冲突。"""
        if primary_data != replica_data:
            self.conflict_count += 1
            return True
        return False
```
