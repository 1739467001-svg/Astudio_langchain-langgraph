## 二、请求队列实现

```python
import asyncio
from asyncio import Semaphore, Queue
from datetime import datetime

class RequestQueue:
    """请求队列：平滑流量+削峰填谷"""
    def __init__(self, max_concurrent: int = 5, max_queue_size: int = 100):
        self.semaphore = Semaphore(max_concurrent)  # 最大并发
        self.queue = Queue(maxsize=max_queue_size)    # 等待队列
        self.stats = {"total": 0, "completed": 0, "rejected": 0, "avg_wait": 0}

    async def submit(self, func, *args, **kwargs):
        """提交请求到队列"""
        self.stats["total"] += 1
        start_time = datetime.now()

        # 检查队列是否满
        if self.queue.full():
            self.stats["rejected"] += 1
            raise QueueFullError("请求队列已满，请稍后重试")

        # 入队等待
        await self.queue.put(1)

        try:
            # 获取并发许可
            async with self.semaphore:
                wait_time = (datetime.now() - start_time).total_seconds()
                self._update_avg_wait(wait_time)

                # 执行请求
                result = await func(*args, **kwargs)
                self.stats["completed"] += 1
                return result
        finally:
            await self.queue.get()

    def _update_avg_wait(self, wait: float):
        n = self.stats["completed"]
        old_avg = self.stats["avg_wait"]
        self.stats["avg_wait"] = (old_avg * n + wait) / (n + 1)

    def get_stats(self) -> dict:
        return {
            **self.stats,
            "queue_size": self.queue.qsize(),
            "avg_wait_seconds": round(self.stats["avg_wait"], 2),
        }


class QueueFullError(Exception):
    pass
```

## 三、速率限制器

```python
import time
from collections import defaultdict

class TokenBucket:
    """令牌桶：平滑QPS"""
    def __init__(self, rate: float = 5.0, capacity: int = 10):
        """
        rate: 每秒生成令牌数(QPS)
        capacity: 桶容量(允许突发)
        """
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_update = time.time()
        self.lock = asyncio.Lock()

    async def acquire(self) -> bool:
        """获取一个令牌"""
        async with self.lock:
            now = time.time()
            # 补充令牌
            elapsed = now - self.last_update
            self.tokens = min(self.capacity, self.tokens + elapsed * self.rate)
            self.last_update = now

            if self.tokens >= 1:
                self.tokens -= 1
                return True
            return False

    async def wait_and_acquire(self, max_wait: float = 10.0) -> bool:
        """等待直到获取令牌"""
        start = time.time()
        while not await self.acquire():
            if time.time() - start > max_wait:
                return False
            await asyncio.sleep(0.1)
        return True
```

## 四、完整流量整形管线

```python
class TrafficShaper:
    """完整流量整形：队列+令牌桶+断路器"""
    def __init__(self, max_concurrent=5, rate=5.0, capacity=10):
        self.queue = RequestQueue(max_concurrent=max_concurrent)
        self.bucket = TokenBucket(rate=rate, capacity=capacity)

    async def process(self, func, *args, **kwargs):
        """带流量整形的请求处理"""
        # Step 1: 令牌桶限流
        if not await self.bucket.wait_and_acquire(max_wait=5):
            return {"error": "请求过于频繁，请稍后重试"}

        # Step 2: 队列排队
        try:
            return await self.queue.submit(func, *args, **kwargs)
        except QueueFullError:
            return {"error": "系统繁忙，请稍后重试"}

    def stats(self) -> dict:
        return {
            "queue": self.queue.get_stats(),
            "bucket_tokens": round(self.bucket.tokens, 1),
        }
```
