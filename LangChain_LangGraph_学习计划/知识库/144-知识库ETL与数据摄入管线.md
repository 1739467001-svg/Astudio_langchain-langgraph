## 二、Extract：多源数据接入

```python
from langchain_core.documents import Document
from typing import AsyncIterator
import asyncio

class DataExtractor:
    """多源数据提取器。"""

    @staticmethod
    async def extract_pdf(file_path: str) -> list[Document]:
        """提取PDF文档。"""
        from langchain_community.document_loaders import PyMuPDFLoader
        loader = PyMuPDFLoader(file_path)
        docs = await loader.aload()
        # 添加来源元数据
        for doc in docs:
            doc.metadata["source"] = file_path
            doc.metadata["file_type"] = "pdf"
        return docs

    @staticmethod
    async def extract_web(url: str) -> list[Document]:
        """提取网页内容。"""
        from langchain_community.document_loaders import AsyncHtmlLoader
        loader = AsyncHtmlLoader(url)
        docs = await loader.aload()
        for doc in docs:
            doc.metadata["source"] = url
            doc.metadata["file_type"] = "web"
        return docs

    @staticmethod
    async def extract_database(
        query: str,
        connection_string: str,
    ) -> list[Document]:
        """从数据库提取数据。"""
        from langchain_community.document_loaders import SQLDatabaseLoader
        from langchain_community.utilities import SQLDatabase
        db = SQLDatabase.from_uri(connection_string)
        loader = SQLDatabaseLoader(query, db=db)
        docs = loader.load()
        for doc in docs:
            doc.metadata["source"] = "database"
            doc.metadata["query"] = query
        return docs

    @staticmethod
    async def extract_directory(
        dir_path: str,
        glob_pattern: str = "**/*",
    ) -> AsyncIterator[list[Document]]:
        """批量提取目录下的文件。"""
        from langchain_community.document_loaders import DirectoryLoader
        from pathlib import Path

        files = list(Path(dir_path).glob(glob_pattern))
        batch_size = 20

        for i in range(0, len(files), batch_size):
            batch = files[i:i + batch_size]
            tasks = []
            for f in batch:
                if f.suffix.lower() == ".pdf":
                    tasks.append(DataExtractor.extract_pdf(str(f)))
                # 可扩展其他格式
            results = await asyncio.gather(*tasks, return_exceptions=True)
            for result in results:
                if isinstance(result, list):
                    yield result
```

```python
import re
import hashlib
from dataclasses import dataclass

@dataclass
class TransformConfig:
    """数据转换配置。"""
    chunk_size: int = 500
    chunk_overlap: int = 50
    min_chunk_size: int = 50        # 太小的块丢弃
    max_chunk_size: int = 2000      # 太大的块强制拆分
    dedupe: bool = True
    strip_html: bool = True
    normalize_whitespace: bool = True

class DataTransformer:
    """数据清洗与分块转换器。"""

    def __init__(self, config: TransformConfig = TransformConfig()):
        self.config = config

    def transform(self, documents: list[Document]) -> list[Document]:
        """完整转换流程。"""
        # 1. 清洗
        cleaned = [self._clean(doc) for doc in documents]
        cleaned = [d for d in cleaned if d.page_content.strip()]

        # 2. 分块
        from langchain_text_splitters import RecursiveCharacterTextSplitter
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=self.config.chunk_size,
            chunk_overlap=self.config.chunk_overlap,
            separators=["\n\n", "\n", "。", ".", " ", ""],
        )
        chunks = splitter.split_documents(cleaned)

        # 3. 过滤太小/太大的块
        chunks = [
            d for d in chunks
            if self.config.min_chunk_size <= len(d.page_content) <= self.config.max_chunk_size
        ]

        # 4. 元数据增强
        chunks = [self._enrich_metadata(d) for d in chunks]

        # 5. 去重
        if self.config.dedupe:
            chunks = self._deduplicate(chunks)

        return chunks

    def _clean(self, doc: Document) -> Document:
        """清洗文档。"""
        content = doc.page_content

        # 去除HTML标签
        if self.config.strip_html:
            content = re.sub(r'<[^>]+>', '', content)

        # 标准化空白
        if self.config.normalize_whitespace:
            content = re.sub(r'\s+', ' ', content)
            content = re.sub(r'\n{3,}', '\n\n', content)

        return Document(page_content=content.strip(), metadata=doc.metadata)

    def _enrich_metadata(self, doc: Document) -> Document:
        """增强元数据。"""
        content = doc.page_content
        doc.metadata["content_hash"] = hashlib.md5(
            content.encode()
        ).hexdigest()
        doc.metadata["char_count"] = len(content)
        doc.metadata["word_count"] = len(content.split())
        return doc

    def _deduplicate(self, docs: list[Document]) -> list[Document]:
        """按内容哈希去重。"""
        seen = set()
        result = []
        for doc in docs:
            h = doc.metadata.get("content_hash")
            if h not in seen:
                seen.add(h)
                result.append(doc)
        return result
```

## 四、Load：嵌入与入库

```python
from langchain_core.embeddings import Embeddings
from langchain_core.vectorstores import VectorStore

class DataLoader:
    """数据加载器：嵌入+入库。"""

    def __init__(
        self,
        embeddings: Embeddings,
        vectorstore: VectorStore,
        batch_size: int = 100,
    ):
        self.embeddings = embeddings
        self.vectorstore = vectorstore
        self.batch_size = batch_size

    async def load(self, documents: list[Document]) -> dict:
        """批量嵌入并入库。"""
        total = len(documents)
        loaded = 0
        errors = []

        for i in range(0, total, self.batch_size):
            batch = documents[i:i + self.batch_size]
            try:
                await self.vectorstore.aadd_documents(batch)
                loaded += len(batch)
            except Exception as e:
                errors.append({"batch": i, "error": str(e)})

        return {
            "total": total,
            "loaded": loaded,
            "failed": len(errors),
            "errors": errors[:5],
        }
```

## 五、完整 ETL 管线

```python
class KnowledgeBaseETL:
    """知识库ETL管线：提取→转换→加载。"""

    def __init__(
        self,
        embeddings: Embeddings,
        vectorstore: VectorStore,
        config: TransformConfig = TransformConfig(),
    ):
        self.extractor = DataExtractor()
        self.transformer = DataTransformer(config)
        self.loader = DataLoader(embeddings, vectorstore)

    async def run_pdf_pipeline(self, file_path: str) -> dict:
        """PDF→向量库完整管线。"""
        # 1. Extract
        docs = await self.extractor.extract_pdf(file_path)

        # 2. Transform
        chunks = self.transformer.transform(docs)

        # 3. Load
        result = await self.loader.load(chunks)

        return {
            "source": file_path,
            "extracted_docs": len(docs),
            "transformed_chunks": len(chunks),
            "loaded": result["loaded"],
            "failed": result["failed"],
        }

    async def run_directory_pipeline(
        self,
        dir_path: str,
    ) -> list[dict]:
        """目录批量处理管线。"""
        results = []
        async for batch in self.extractor.extract_directory(dir_path):
            chunks = self.transformer.transform(batch)
            load_result = await self.loader.load(chunks)
            results.append({
                "batch_docs": len(batch),
                "chunks": len(chunks),
                "loaded": load_result["loaded"],
            })
        return results
```

```python
class IncrementalUpdater:
    """增量更新管理器。"""

    def __init__(self, etl: KnowledgeBaseETL):
        self.etl = etl
        self.file_hashes: dict[str, str] = {}  # 文件路径→hash

    async def sync_directory(self, dir_path: str) -> dict:
        """同步目录变更。"""
        from pathlib import Path
        current_files = {str(f): self._file_hash(f) for f in Path(dir_path).rglob("*") if f.is_file()}

        added = []
        modified = []
        deleted = []

        # 检测新增和修改
        for path, h in current_files.items():
            if path not in self.file_hashes:
                added.append(path)
            elif self.file_hashes[path] != h:
                modified.append(path)
            self.file_hashes[path] = h

        # 检测删除
        for path in list(self.file_hashes):
            if path not in current_files:
                deleted.append(path)
                del self.file_hashes[path]

        # 处理变更
        for path in added + modified:
            await self.etl.run_pdf_pipeline(path)

        return {"added": len(added), "modified": len(modified), "deleted": len(deleted)}

    @staticmethod
    def _file_hash(file_path) -> str:
        import hashlib
        h = hashlib.md5()
        with open(file_path, "rb") as f:
            h.update(f.read())
        return h.hexdigest()
```

## 七、调度与监控

```python
class ETLScheduler:
    """ETL调度器：定时同步。"""

    def __init__(self, updater: IncrementalUpdater):
        self.updater = updater
        self.history: list[dict] = []

    async def run_scheduled_sync(self, dir_path: str) -> dict:
        """执行定时同步。"""
        import time
        start = time.time()

        try:
            result = await self.updater.sync_directory(dir_path)
            result["status"] = "success"
        except Exception as e:
            result = {"status": "error", "error": str(e)}

        result["duration_s"] = round(time.time() - start, 2)
        result["timestamp"] = time.time()
        self.history.append(result)
        return result

    def stats(self) -> dict:
        """ETL统计。"""
        if not self.history:
            return {"total_runs": 0}
        last = self.history[-1]
        return {
            "total_runs": len(self.history),
            "last_status": last["status"],
            "last_duration": last.get("duration_s", 0),
            "last_added": last.get("added", 0),
            "last_modified": last.get("modified", 0),
            "last_deleted": last.get("deleted", 0),
        }
```

## 八、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 分批处理大数据集 | 避免内存溢出 | ★★★ |
| 增量更新而非全量 | 只处理变更文件 | ★★★ |
| 文件hash追踪变更 | 精确检测修改 | ★★☆ |
| 去重防止冗余 | 内容哈希去重 | ★★☆ |
| 元数据要完整 | 来源/日期/页码便于溯源 | ★★☆ |
| 过滤太小/太大块 | 太小无用太大稀释 | ★★☆ |
| 监控ETL执行 | 失败要告警 | ★★☆ |
