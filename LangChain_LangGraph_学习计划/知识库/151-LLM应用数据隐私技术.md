### 2.1 规则脱敏

```python
import re
from dataclasses import dataclass

@dataclass
class PIIPattern:
    """PII识别模式。"""
    name: str
    pattern: str
    replacement: str

PII_PATTERNS = [
    PIIPattern("手机号", r'\b1[3-9]\d{9}\b', "[PHONE]"),
    PIIPattern("身份证", r'\b\d{17}[\dXx]\b', "[ID_CARD]"),
    PIIPattern("邮箱", r'[\w.+-]+@[\w-]+\.[\w.-]+', "[EMAIL]"),
    PIIPattern("银行卡", r'\b\d{16,19}\b', "[BANK_CARD]"),
    PIIPattern("API Key", r'sk-[a-zA-Z0-9]{40,}', "[API_KEY]"),
    PIIPattern("地址", r'(?:省|市|区|县|路|街|号|楼|室).{0,30}', "[ADDRESS]"),
]

class DataMasker:
    """数据脱敏器。"""

    def __init__(self, patterns: list[PIIPattern] = None):
        self.patterns = patterns or PII_PATTERNS
        self.mapping: dict[str, str] = {}  # 占位符→原文（用于恢复）

    def mask(self, text: str) -> str:
        """脱敏：替换PII为占位符。"""
        for p in self.patterns:
            matches = re.findall(p.pattern, text)
            for match in matches:
                placeholder = f"{p.replacement}_{len(self.mapping)}"
                self.mapping[placeholder] = match
                text = text.replace(match, placeholder, 1)
        return text

    def unmask(self, text: str) -> str:
        """恢复：占位符→原文。"""
        for placeholder, original in self.mapping.items():
            text = text.replace(placeholder, original)
        return text

    def detect_pii(self, text: str) -> list[dict]:
        """检测PII但不替换。"""
        findings = []
        for p in self.patterns:
            matches = re.findall(p.pattern, text)
            if matches:
                findings.append({"type": p.name, "count": len(matches)})
        return findings
```

### 2.2 LLM 辅助脱敏

```python
from langchain_core.messages import HumanMessage

LLM_MASK_PROMPT = """请识别以下文本中的所有个人敏感信息（PII），并替换为占位符。

需要识别的PII类型：
- 姓名（用[NAME]替换）
- 电话号码（用[PHONE]替换）
- 邮箱地址（用[EMAIL]替换）
- 身份证号（用[ID]替换）
- 银行卡号（用[CARD]替换）
- 家庭住址（用[ADDRESS]替换）
- 公司名（用[COMPANY]替换）

原文:
{text}

脱敏后:"""

async def llm_assisted_mask(llm, text: str) -> str:
    """用LLM识别和脱敏——比规则更灵活。"""
    prompt = LLM_MASK_PROMPT.format(text=text)
    response = await llm.ainvoke([HumanMessage(content=prompt)])
    return response.content
```

```python
import numpy as np

class DifferentialPrivacy:
    """LLM场景的差分隐私。"""

    @staticmethod
    def add_noise_to_embeddings(
        embeddings: np.ndarray,
        epsilon: float = 1.0,
    ) -> np.ndarray:
        """对嵌入向量加差分隐私噪声。

        在向量入库前加入拉普拉斯噪声，
        使得攻击者无法从向量库中精确推断原始文本。

        Args:
            embeddings: 原始嵌入向量 [N, dim]
            epsilon: 隐私预算，越小越隐私但精度损失越大

        Returns:
            加噪后的向量
        """
        sensitivity = 1.0  # 嵌入向量的敏感度
        scale = sensitivity / epsilon
        noise = np.random.laplace(0, scale, size=embeddings.shape)
        noisy = embeddings + noise

        # 重新归一化
        norms = np.linalg.norm(noisy, axis=1, keepdims=True)
        noisy = noisy / np.clip(norms, 1e-8, None)

        return noisy

    @staticmethod
    def evaluate_privacy_utility_tradeoff(
        original: np.ndarray,
        epsilons: list[float],
    ) -> list[dict]:
        """评估隐私-效用权衡。

        不同epsilon下的隐私保护和检索质量。
        """
        results = []
        for eps in epsilons:
            noisy = DifferentialPrivacy.add_noise_to_embeddings(original, eps)
            # 计算原始和加噪向量的相似度
            cosine_sim = np.mean([
                np.dot(original[i], noisy[i])
                for i in range(len(original))
            ])
            results.append({
                "epsilon": eps,
                "utility": round(float(cosine_sim), 4),
                "privacy": "高" if eps < 0.5 else "中" if eps < 2 else "低",
            })
        return results
```

## 五、隐私合规检查

```python
class PrivacyComplianceChecker:
    """隐私合规检查器。"""

    CHECKS = {
        "数据传输": [
            "API调用使用TLS加密",
            "不在URL中传敏感数据",
            "API Key安全存储不在代码中",
        ],
        "数据存储": [
            "对话历史脱敏后存储",
            "向量库不存储原文PII",
            "日志不记录PII",
            "数据库有访问控制",
        ],
        "数据处理": [
            "输入发LLM前脱敏",
            "LLM输出过滤PII",
            "不将用户数据用于训练",
        ],
        "用户权利": [
            "用户可查看其数据",
            "用户可删除其数据",
            "用户可导出其数据",
        ],
    }

    @classmethod
    def audit(cls, status: dict) -> dict:
        """执行隐私合规审计。"""
        results = {}
        total_pass = 0
        total = 0

        for category, checks in cls.CHECKS.items():
            passed = sum(1 for check in checks if status.get(check, False))
            total_pass += passed
            total += len(checks)
            results[category] = {
                "passed": passed,
                "total": len(checks),
                "items": {check: status.get(check, False) for check in checks},
            }

        return {
            "overall": f"{total_pass}/{total}",
            "compliance_rate": round(total_pass / total * 100, 1) if total else 0,
            "details": results,
        }
```
