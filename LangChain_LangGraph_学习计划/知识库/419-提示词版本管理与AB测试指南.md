# 提示词版本管理与 AB 测试指南

> Prompt 是 LLM 应用中"代码"一样的东西——改一个字就可能让输出天差地别。但绝大多数团队把 Prompt 写在代码里，没有版本号、没有变更记录、没有回滚能力。一旦改坏了，没人知道改了什么、为什么改、怎么回滚。

---

## 1. 为什么需要 Prompt 版本管理

### 常见问题

```
问题 1：Prompt 改了但没人记录改了什么
  v1: "你是客服助手" → v2: "你是专业客服助手" → v3: "你是资深客服助手"
  三天后：满意度下降了，谁改的？改了什么？为什么改？

问题 2：A/B 测试时无法切换 Prompt 版本
  想对比两个 Prompt 的效果，但只有一个硬编码在代码里

问题 3：改坏了无法回滚
  新 Prompt 导致幻觉率飙升，旧版本已经被覆盖了
```

### 版本管理的价值

| 能力 | 无版本管理 | 有版本管理 |
|------|-----------|-----------|
| 变更追踪 | 无 | 每次改动有记录 |
| 回滚能力 | 无 | 一键回滚 |
| A/B 测试 | 难 | 一行切换 |
| 变更审查 | 无 | PR 审核 |
| 效果对比 | 靠记忆 | 版本间对比 |

## 2. Prompt 版本模型

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
import time
import hashlib

class PromptStatus(Enum):
    DRAFT = "draft"           # 草稿
    REVIEW = "review"         # 待审核
    ACTIVE = "active"         # 线上使用
    ARCHIVED = "archived"     # 已归档
    ROLLED_BACK = "rolled_back"  # 已回滚


@dataclass
class PromptVersion:
    """Prompt 版本"""
    id: str = ""
    prompt_id: str = ""           # Prompt 模板 ID（同一 Prompt 的不同版本共享）
    version: str = ""             # 语义版本号 "1.0.0"
    # 内容
    system_prompt: str = ""
    user_template: str = ""       # 用户消息模板
    variables: list[str] = field(default_factory=list)  # 模板变量
    # 元数据
    author: str = ""
    description: str = ""         # 本次变更说明
    changelog: str = ""           # 变更内容
    status: PromptStatus = PromptStatus.DRAFT
    # 时序
    created_at: float = field(default_factory=time.time)
    activated_at: float | None = None
    # 性能指标
    metrics: dict = field(default_factory=dict)
    # 哈希（用于快速比对）
    content_hash: str = ""

    def __post_init__(self):
        if not self.content_hash:
            content = f"{self.system_prompt}:{self.user_template}"
            self.content_hash = hashlib.sha256(content.encode()).hexdigest()[:16]

    def to_dict(self) -> dict:
        return {
            "id": self.id,
            "prompt_id": self.prompt_id,
            "version": self.version,
            "system_prompt": self.system_prompt,
            "user_template": self.user_template,
            "variables": self.variables,
            "author": self.author,
            "description": self.description,
            "status": self.status.value,
            "created_at": self.created_at,
            "metrics": self.metrics,
            "content_hash": self.content_hash,
        }
```

## 3. Prompt 注册中心

```python
class PromptRegistry:
    """Prompt 版本注册中心"""

    def __init__(self):
        self.prompts: dict[str, list[PromptVersion]] = {}  # prompt_id → versions
        self.active: dict[str, str] = {}  # prompt_id → version_id
        self.versions_by_id: dict[str, PromptVersion] = {}

    def register(self, version: PromptVersion) -> PromptVersion:
        """注册新版本"""
        if not version.id:
            import uuid
            version.id = str(uuid.uuid4())[:8]

        if version.prompt_id not in self.prompts:
            self.prompts[version.prompt_id] = []

        self.prompts[version.prompt_id].append(version)
        self.versions_by_id[version.id] = version

        # 如果是第一个版本，自动激活
        if len(self.prompts[version.prompt_id]) == 1:
            self.activate(version.prompt_id, version.id)

        return version

    def activate(self, prompt_id: str, version_id: str) -> bool:
        """激活指定版本"""
        version = self.versions_by_id.get(version_id)
        if not version or version.prompt_id != prompt_id:
            return False

        # 归档当前活跃版本
        if prompt_id in self.active:
            old_id = self.active[prompt_id]
            old = self.versions_by_id[old_id]
            old.status = PromptStatus.ARCHIVED

        # 激活新版本
        version.status = PromptStatus.ACTIVE
        version.activated_at = time.time()
        self.active[prompt_id] = version_id
        return True

    def rollback(self, prompt_id: str) -> bool:
        """回滚到上一个版本"""
        versions = self.prompts.get(prompt_id, [])
        if len(versions) < 2:
            return False

        # 找到当前活跃版本的前一个
        current_id = self.active.get(prompt_id)
        versions_sorted = sorted(versions, key=lambda v: v.created_at)

        prev_version = None
        for i, v in enumerate(versions_sorted):
            if v.id == current_id and i > 0:
                prev_version = versions_sorted[i - 1]
                break

        if prev_version:
            # 标记当前为已回滚
            current = self.versions_by_id[current_id]
            current.status = PromptStatus.ROLLED_BACK
            # 激活前一个
            return self.activate(prompt_id, prev_version.id)

        return False

    def get_active(self, prompt_id: str) -> PromptVersion | None:
        """获取当前活跃版本"""
        version_id = self.active.get(prompt_id)
        if version_id:
            return self.versions_by_id[version_id]
        return None

    def get_version(self, version_id: str) -> PromptVersion | None:
        """获取指定版本"""
        return self.versions_by_id.get(version_id)

    def get_history(self, prompt_id: str) -> list[PromptVersion]:
        """获取版本历史"""
        return sorted(
            self.prompts.get(prompt_id, []),
            key=lambda v: v.created_at,
        )

    def diff(self, prompt_id: str, v1_id: str, v2_id: str) -> dict:
        """对比两个版本"""
        v1 = self.versions_by_id[v1_id]
        v2 = self.versions_by_id[v2_id]

        return {
            "system_changed": v1.system_prompt != v2.system_prompt,
            "template_changed": v1.user_template != v2.user_template,
            "v1_hash": v1.content_hash,
            "v2_hash": v2.content_hash,
            "v1_system": v1.system_prompt[:200],
            "v2_system": v2.system_prompt[:200],
        }
```

## 4. Prompt 渲染与使用

```python
from langchain_core.prompts import ChatPromptTemplate
from langchain_openai import ChatOpenAI

class PromptRenderer:
    """Prompt 渲染器"""

    def __init__(self, registry: PromptRegistry):
        self.registry = registry

    def render(
        self,
        prompt_id: str,
        variables: dict,
        version_id: str | None = None,
    ) -> ChatPromptTemplate:
        """渲染 Prompt

        Args:
            prompt_id: Prompt ID
            variables: 模板变量
            version_id: 指定版本（None 则用活跃版本）
        """
        if version_id:
            version = self.registry.get_version(version_id)
        else:
            version = self.registry.get_active(prompt_id)

        if not version:
            raise ValueError(f"Prompt {prompt_id} 版本 {version_id or 'active'} 不存在")

        prompt = ChatPromptTemplate.from_messages([
            ("system", version.system_prompt),
            ("human", version.user_template),
        ])

        return prompt

    def invoke(
        self,
        prompt_id: str,
        variables: dict,
        llm: ChatOpenAI,
        version_id: str | None = None,
    ):
        """渲染并调用 LLM"""
        prompt = self.render(prompt_id, variables, version_id)
        chain = prompt | llm
        return chain.invoke(variables)


# 使用示例
registry = PromptRegistry()
renderer = PromptRenderer(registry)

# 注册 Prompt
v1 = PromptVersion(
    prompt_id="customer_service",
    version="1.0.0",
    system_prompt="你是客服助手。礼貌回答用户问题。",
    user_template="{question}",
    variables=["question"],
    author="alice",
    description="初始版本",
)
registry.register(v1)

# 注册改进版
v2 = PromptVersion(
    prompt_id="customer_service",
    version="1.1.0",
    system_prompt="""你是专业客服助手。请遵循：
1. 礼貌回答
2. 准确引用知识库
3. 无法回答时说明并转人工

知识库：
{knowledge_base}""",
    user_template="{question}",
    variables=["question", "knowledge_base"],
    author="bob",
    description="增加知识库和规则约束",
    changelog="添加3条客服规则和知识库变量",
)
registry.register(v2)

# 使用活跃版本
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
response = renderer.invoke("customer_service", {
    "question": "退货流程是什么？",
    "knowledge_base": "退货：登录→找到订单→点击退货",
}, llm)

# 使用指定版本（A/B 测试用）
response_v1 = renderer.invoke("customer_service", {
    "question": "退货流程是什么？",
    "knowledge_base": "退货：登录→找到订单→点击退货",
}, llm, version_id=v1.id)

# 回滚
registry.rollback("customer_service")
```

## 5. Prompt A/B 测试

```python
class PromptABTest:
    """Prompt A/B 测试"""

    def __init__(self, registry: PromptRegistry):
        self.registry = registry
        self.experiments: dict[str, dict] = {}

    def create_experiment(
        self,
        experiment_id: str,
        prompt_id: str,
        version_a_id: str,
        version_b_id: str,
        traffic_split: float = 0.5,
    ) -> dict:
        """创建 A/B 实验"""
        self.experiments[experiment_id] = {
            "prompt_id": prompt_id,
            "version_a": version_a_id,
            "version_b": version_b_id,
            "traffic_split": traffic_split,
            "results": {"a": [], "b": []},
            "active": True,
        }
        return self.experiments[experiment_id]

    def get_version(self, experiment_id: str, user_id: str) -> str:
        """根据用户获取 Prompt 版本"""
        exp = self.experiments.get(experiment_id)
        if not exp or not exp["active"]:
            # 无实验，返回活跃版本
            active = self.registry.get_active(exp["prompt_id"]) if exp else None
            return active.id if active else ""

        # 确定性分流
        import hashlib
        h = int(hashlib.md5(f"{user_id}:{experiment_id}".encode()).hexdigest(), 16) % 100
        return exp["version_b"] if h < exp["traffic_split"] * 100 else exp["version_a"]

    def record(
        self,
        experiment_id: str,
        user_id: str,
        satisfaction: float,  # 0-1
        latency_ms: float,
        token_count: int,
    ):
        """记录实验结果"""
        exp = self.experiments.get(experiment_id)
        if not exp:
            return

        variant = "b" if self.get_version(experiment_id, user_id) == exp["version_b"] else "a"
        exp["results"][variant].append({
            "satisfaction": satisfaction,
            "latency_ms": latency_ms,
            "token_count": token_count,
        })

    def analyze(self, experiment_id: str) -> dict:
        """分析实验结果"""
        exp = self.experiments.get(experiment_id)
        if not exp:
            return {"error": "实验不存在"}

        results_a = exp["results"]["a"]
        results_b = exp["results"]["b"]

        if len(results_a) < 10 or len(results_b) < 10:
            return {
                "status": "数据不足",
                "a_samples": len(results_a),
                "b_samples": len(results_b),
            }

        import numpy as np
        from scipy import stats

        sat_a = [r["satisfaction"] for r in results_a]
        sat_b = [r["satisfaction"] for r in results_b]

        t_stat, p_value = stats.ttest_ind(sat_b, sat_a)

        return {
            "version_a": exp["version_a"],
            "version_b": exp["version_b"],
            "a_satisfaction": f"{np.mean(sat_a):.1%}",
            "b_satisfaction": f"{np.mean(sat_b):.1%}",
            "difference": f"{np.mean(sat_b) - np.mean(sat_a):+.1%}",
            "p_value": f"{p_value:.4f}",
            "significant": p_value < 0.05,
            "a_samples": len(results_a),
            "b_samples": len(results_b),
        }


# 使用
ab_test = PromptABTest(registry)
ab_test.create_experiment("exp_001", "customer_service", v1.id, v2.id, 0.5)

# 模拟数据
import random
for i in range(200):
    user_id = f"user_{i}"
    variant = ab_test.get_version("exp_001", user_id)
    if variant == v2.id:
        sat = random.gauss(0.85, 0.1)
    else:
        sat = random.gauss(0.75, 0.1)
    ab_test.record("exp_001", user_id, max(0, min(1, sat)),
                   random.gauss(1000, 200), random.gauss(500, 50))

print(ab_test.analyze("exp_001"))
```

## 6. Prompt 变更审查流程

```python
class PromptChangeReview:
    """Prompt 变更审查流程"""

    STEPS = [
        "draft",       # 1. 创建草稿
        "review",      # 2. 提交审核
        "approved",    # 3. 审核通过
        "tested",      # 4. 测试通过
        "deployed",    # 5. 部署上线
    ]

    def __init__(self, registry: PromptRegistry):
        self.registry = registry
        self.reviews: dict[str, dict] = {}

    def submit_review(
        self,
        version_id: str,
        reviewer: str,
        notes: str = "",
    ) -> dict:
        """提交审核"""
        version = self.registry.get_version(version_id)
        if not version:
            return {"error": "版本不存在"}

        review = {
            "version_id": version_id,
            "reviewer": reviewer,
            "notes": notes,
            "status": "pending",
            "submitted_at": time.time(),
            "checks": self._auto_checks(version),
        }
        self.reviews[version_id] = review
        version.status = PromptStatus.REVIEW
        return review

    def _auto_checks(self, version: PromptVersion) -> dict:
        """自动检查项"""
        return {
            "has_system_prompt": bool(version.system_prompt),
            "has_user_template": bool(version.user_template),
            "variables_defined": len(version.variables) > 0,
            "system_prompt_length": len(version.system_prompt),
            "template_length": len(version.user_template),
            "estimated_tokens": len(version.system_prompt) // 4 + len(version.user_template) // 4,
            "warnings": self._warnings(version),
        }

    def _warnings(self, version: PromptVersion) -> list[str]:
        warnings = []
        if len(version.system_prompt) > 4000:
            warnings.append("系统提示过长(>4000字符)，可能影响成本")
        if "{question}" not in version.user_template:
            warnings.append("用户模板缺少 {question} 变量")
        if "ignore" in version.system_prompt.lower():
            warnings.append("系统提示包含 'ignore'，可能有注入风险")
        return warnings

    def approve(self, version_id: str, approved_by: str) -> bool:
        """审核通过"""
        version = self.registry.get_version(version_id)
        if not version or version.status != PromptStatus.REVIEW:
            return False
        version.status = PromptStatus.ACTIVE
        if version_id in self.reviews:
            self.reviews[version_id]["status"] = "approved"
            self.reviews[version_id]["approved_by"] = approved_by
        return True
```

## 7. 最佳实践

| 实践 | 说明 |
|------|------|
| 语义版本号 | MAJOR.MINOR.PATCH，大改 MAJOR，小改 MINOR |
| 每次改有说明 | description 字段必填 |
| 上线前测试 | 至少跑 50 条评估集 |
| 灰度发布 | 5% → 25% → 50% → 100% |
| 有回滚预案 | 确认回滚版本可用 |
| 定期清理 | 归档超过 3 个月的旧版本 |

### Prompt 管理 vs 代码管理

| 方面 | 代码 | Prompt |
|------|------|--------|
| 版本号 | Git tag | 语义版本 |
| 审查 | PR Review | 变更审查流程 |
| 测试 | 单元测试 | 评估集测试 |
| 部署 | CI/CD | 灰度发布 |
| 回滚 | Git revert | 一键回滚 |
| 监控 | APM | 满意度+质量指标 |
