## 二、基准测试器

```python
import asyncio
import time
import statistics
from dataclasses import dataclass, field
from typing import Callable
import tiktoken

@dataclass
class LLMPerformanceMetrics:
    """LLM性能指标。"""
    ttft_ms: float = 0          # 首Token延迟
    total_latency_ms: float = 0  # 端到端延迟
    input_tokens: int = 0        # 输入Token数
    output_tokens: int = 0       # 输出Token数
    tps: float = 0              # Token/秒
    cost_usd: float = 0         # 成本
    error: str = ""

class LLMBenchmark:
    """LLM性能基准测试器。"""

    def __init__(self, llm):
        self.llm = llm
        self.encoder = None
        try:
            self.encoder = tiktoken.encoding_for_model("gpt-4o")
        except Exception:
            pass

    def _count_tokens(self, text: str) -> int:
        """计算Token数。"""
        if self.encoder:
            return len(self.encoder.encode(text))
        return len(text) // 3

    async def benchmark_single(
        self,
        prompt: str,
        model: str = "gpt-4o",
        streaming: bool = True,
    ) -> LLMPerformanceMetrics:
        """基准测试单次调用。"""
        from langchain_core.messages import HumanMessage

        input_tokens = self._count_tokens(prompt)
        start = time.time()
        ttft = None

        try:
            if streaming:
                # 流式：测量TTFT
                first_token = True
                output_text = ""
                async for chunk in self.llm.astream([HumanMessage(content=prompt)]):
                    if first_token and chunk.content:
                        ttft = (time.time() - start) * 1000
                        first_token = False
                    output_text += chunk.content

                output_tokens = self._count_tokens(output_text)
                total_latency = (time.time() - start) * 1000
            else:
                # 非流式
                response = await self.llm.ainvoke([HumanMessage(content=prompt)])
                output_tokens = self._count_tokens(response.content)
                total_latency = (time.time() - start) * 1000
                ttft = total_latency  # 非流式TTFT=总延迟

            tps = output_tokens / (total_latency / 1000) if total_latency > 0 else 0

            # 成本估算
            pricing = {"gpt-4o": {"input": 2.50, "output": 10.00}, "gpt-4o-mini": {"input": 0.15, "output": 0.60}}
            prices = pricing.get(model, pricing["gpt-4o"])
            cost = (input_tokens / 1e6 * prices["input"]) + (output_tokens / 1e6 * prices["output"])

            return LLMPerformanceMetrics(
                ttft_ms=round(ttft, 2),
                total_latency_ms=round(total_latency, 2),
                input_tokens=input_tokens,
                output_tokens=output_tokens,
                tps=round(tps, 2),
                cost_usd=round(cost, 6),
            )

        except Exception as e:
            return LLMPerformanceMetrics(
                total_latency_ms=round((time.time() - start) * 1000, 2),
                error=str(e)[:200],
            )

    async def benchmark_batch(
        self,
        prompts: list[str],
        model: str = "gpt-4o",
        concurrency: int = 5,
    ) -> dict:
        """批量基准测试。"""
        semaphore = asyncio.Semaphore(concurrency)

        async def run_one(prompt):
            async with semaphore:
                return await self.benchmark_single(prompt, model)

        results = await asyncio.gather(*[run_one(p) for p in prompts])

        # 统计
        latencies = [r.total_latency_ms for r in results if not r.error]
        ttfts = [r.ttft_ms for r in results if r.ttft_ms > 0 and not r.error]
        tokens = [r.output_tokens for r in results if not r.error]
        costs = [r.cost_usd for r in results if not r.error]

        n = len(latencies)
        if n == 0:
            return {"error": "无成功结果"}

        latencies.sort()
        ttfts.sort()

        return {
            "total_requests": len(prompts),
            "successful": n,
            "failed": len(prompts) - n,
            "ttft_p50_ms": round(ttfts[n // 2], 2) if ttfts else 0,
            "ttft_p95_ms": round(ttfts[int(n * 0.95)], 2) if ttfts else 0,
            "latency_p50_ms": round(latencies[n // 2], 2),
            "latency_p95_ms": round(latencies[int(n * 0.95)], 2),
            "avg_tps": round(statistics.mean([r.tps for r in results if r.tps > 0]), 2),
            "total_tokens": sum(tokens),
            "total_cost_usd": round(sum(costs), 6),
            "cost_per_request": round(sum(costs) / n, 6),
        }


class RAGBenchmark:
    """RAG全链路基准测试器。"""

    def __init__(self, rag_pipeline):
        self.pipeline = rag_pipeline

    async def benchmark(
        self,
        queries: list[str],
        concurrency: int = 3,
    ) -> dict:
        """RAG全链路基准。"""
        semaphore = asyncio.Semaphore(concurrency)

        async def run_one(query):
            async with semaphore:
                start = time.time()
                try:
                    result = await self.pipeline.answer(query)
                    latency = (time.time() - start) * 1000
                    return {
                        "latency_ms": latency,
                        "success": True,
                        "answer_length": len(result.get("answer", "")),
                        "docs_retrieved": len(result.get("contexts", [])),
                    }
                except Exception:
                    return {"latency_ms": (time.time() - start) * 1000, "success": False}

        results = await asyncio.gather(*[run_one(q) for q in queries])

        latencies = sorted([r["latency_ms"] for r in results])
        n = len(latencies)

        return {
            "total": n,
            "successful": sum(1 for r in results if r["success"]),
            "latency_p50": round(latencies[n // 2], 2),
            "latency_p95": round(latencies[int(n * 0.95)], 2),
            "qps": round(n / (sum(latencies) / 1000 / concurrency), 2),
        }
```

## 三、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 测TTFT不只测总延迟 | 用户感知首Token | ★★★ |
| 用P95不用平均 | 平均被异常值拉偏 | ★★★ |
| 流式比非流式感知好 | TTFT低3-5倍 | ★★★ |
| 并发基准测QPS | 知道系统上限 | ★★☆ |
| 记录成本 | 性能和成本平衡 | ★★☆ |
