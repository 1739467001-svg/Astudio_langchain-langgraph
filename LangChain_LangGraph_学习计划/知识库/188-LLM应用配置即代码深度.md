# LLM 应用配置即代码深度

> Prompt、模型参数、工具列表——这些配置散在各处，改一个 Prompt 要翻代码、上线靠手动改。配置即代码（IaC）让所有配置版本化、可审计、可回滚。

## 二、配置文件设计

```yaml
# llm_app_config.yaml — LLM应用完整配置
version: "1.2.0"
environment: production

# 模型配置
models:
  default: gpt-4o-mini
  fallback: gpt-4o
  embedding: BAAI/bge-large-zh-v1.5
  params:
    temperature: 0.7
    max_tokens: 1000
    streaming: true

# Prompt配置（版本化）
prompts:
  customer_service:
    version: "2.1"
    system: |
      你是专业客服助手。礼貌、准确、高效。
    few_shot:
      - input: "退货政策"
        output: "7天内可退货..."

# 工具配置
tools:
  enabled:
    - search_web
    - query_database
    - send_email
  disabled:
    - execute_code  # 生产环境禁用
  permissions:
    send_email:
      require_approval: true
    query_database:
      sql_only_select: true

# RAG配置
rag:
  chunk_size: 500
  chunk_overlap: 50
  retrieval_k: 5
  rerank: true
  rerank_k: 3
  hybrid_search: true
  vector_weight: 0.6
  keyword_weight: 0.4

# 缓存配置

  semantic_cache:
    enabled: true
    similarity_threshold: 0.92
    ttl_seconds: 3600

# 安全配置
```

## 三、配置加载器

```python
import yaml
from pathlib import Path
from dataclasses import dataclass, field
from typing import Any
import json

@dataclass
class AppConfig:
    """应用配置（从YAML加载）。"""
    version: str = "1.0.0"
    environment: str = "development"
    models: dict = field(default_factory=dict)
    prompts: dict = field(default_factory=dict)
    tools: dict = field(default_factory=dict)
    rag: dict = field(default_factory=dict)
    cache: dict = field(default_factory=dict)
    security: dict = field(default_factory=dict)
    monitoring: dict = field(default_factory=dict)

class ConfigLoader:
    """配置加载器。"""

    @staticmethod
    def load(path: str) -> AppConfig:
        """从YAML文件加载配置。"""
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)
        return AppConfig(**data)

    @staticmethod
    def validate(config: AppConfig) -> dict:
        """验证配置完整性。"""
        issues = []

        # 检查模型配置
        if not config.models.get("default"):
            issues.append("缺少默认模型配置")
        if not config.models.get("embedding"):
            issues.append("缺少嵌入模型配置")

        # 检查Prompt配置
        for name, prompt in config.prompts.items():
            if not prompt.get("system"):
                issues.append(f"Prompt '{name}' 缺少system提示")

        # 检查工具权限
        for tool, perm in config.tools.get("permissions", {}).items():
            if perm.get("require_approval") and tool not in config.tools.get("enabled", []):
                issues.append(f"工具'{tool}'设置了权限但未启用")

        # 检查RAG参数
        rag = config.rag
        if rag.get("chunk_size", 0) < 100:
            issues.append("chunk_size太小（<100）")
        if rag.get("retrieval_k", 0) > 20:
            issues.append("retrieval_k太大（>20），可能影响延迟")

        return {
            "valid": len(issues) == 0,
            "issues": issues,
        }

    @staticmethod
    def diff(old: AppConfig, new: AppConfig) -> dict:
        """对比两个版本配置的差异。"""
        changes = {}
        for field_name in ["models", "prompts", "tools", "rag", "cache", "security"]:
            old_val = getattr(old, field_name)
            new_val = getattr(new, field_name)
            if old_val != new_val:
                changes[field_name] = {
                    "old": old_val,
                    "new": new_val,
                }
        return {
            "has_changes": len(changes) > 0,
            "changed_sections": list(changes.keys()),
            "details": changes,
        }
```

## 四、热更新

```python
import time
from pathlib import Path

class HotReloadConfigManager:
    """配置热更新管理器。"""

    def __init__(self, config_path: str):
        self.path = Path(config_path)
        self.config: AppConfig = ConfigLoader.load(config_path)
        self._last_modified: float = self.path.stat().st_mtime
        self._callbacks: list[callable] = []

    def on_change(self, callback: callable):
        """注册配置变更回调。"""
        self._callbacks.append(callback)

    def check_and_reload(self) -> bool:
        """检查文件是否变更，如变更则重新加载。"""
        current_mtime = self.path.stat().st_mtime
        if current_mtime > self._last_modified:
            old_config = self.config
            self.config = ConfigLoader.load(str(self.path))
            self._last_modified = current_mtime

            # 计算差异
            diff = ConfigLoader.diff(old_config, self.config)

            # 通知回调
            for callback in self._callbacks:
                callback(diff)

            return True
        return False

    def get(self) -> AppConfig:
        """获取当前配置。"""
        return self.config
```

## 五、配置版本管理

```python
class ConfigVersionManager:
    """配置版本管理器。"""

    def __init__(self):
        self.versions: dict[str, AppConfig] = {}  # version → config
        self.active_version: str = ""

    def save_version(self, config: AppConfig):
        """保存配置版本。"""
        self.versions[config.version] = config

    def activate(self, version: str) -> bool:
        """激活指定版本。"""
        if version in self.versions:
            self.active_version = version
            return True
        return False

    def rollback(self) -> str | None:
        """回滚到上一个版本。"""
        versions = sorted(self.versions.keys())
        current_idx = versions.index(self.active_version) if self.active_version in versions else -1
        if current_idx > 0:
            prev_version = versions[current_idx - 1]
            self.active_version = prev_version
            return prev_version
        return None

    def get_active(self) -> AppConfig | None:
        """获取当前活跃配置。"""
        return self.versions.get(self.active_version)
```
