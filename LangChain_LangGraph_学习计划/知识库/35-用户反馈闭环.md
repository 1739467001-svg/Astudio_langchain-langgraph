### 2.2 前端反馈组件

```python
# 后端API
from fastapi import FastAPI
from pydantic import BaseModel
import sqlite3
from datetime import datetime

app = FastAPI()

class Feedback(BaseModel):
    message_id: str
    rating: int  # 1-5
    feedback_type: str = None  # "too_long" / "irrelevant" / "wrong" / "good"
    comment: str = None

@app.post("/api/feedback")
async def submit_feedback(fb: Feedback):
    """收集用户反馈"""
    conn = sqlite3.connect("data/app.db")
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS feedback (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            message_id TEXT,
            rating INTEGER,
            feedback_type TEXT,
            comment TEXT,
            created_at TEXT
        )
    """)
    cursor.execute(
        "INSERT INTO feedback (message_id, rating, feedback_type, comment, created_at) VALUES (?,?,?,?,?)",
        (fb.message_id, fb.rating, fb.feedback_type, fb.comment, datetime.now().isoformat())
    )
    conn.commit()
    conn.close()
    return {"status": "received"}
```

        headers: { 'Content-Type': 'application/json' },

## 三、反馈分析

### 3.1 反馈分类

```python
def analyze_feedback(feedbacks: list) -> dict:
    """分析反馈数据"""
    total = len(feedbacks)
    if total == 0:
        return {"total": 0}

    # 评分统计
    ratings = [f["rating"] for f in feedbacks]
    avg_rating = sum(ratings) / total

    # 好评/差评比例
    positive = sum(1 for r in ratings if r >= 4)
    negative = sum(1 for r in ratings if r <= 2)

    # 问题类型统计
    type_counts = {}
    for f in feedbacks:
        ft = f.get("feedback_type")
        if ft:
            type_counts[ft] = type_counts.get(ft, 0) + 1

    return {
        "total": total,
        "avg_rating": round(avg_rating, 2),
        "positive_rate": f"{positive/total:.1%}",
        "negative_rate": f"{negative/total:.1%}",
        "top_issues": sorted(type_counts.items(), key=lambda x: -x[1])[:3],
    }
```

### 3.2 Bad Case 提取

```python
def extract_bad_cases(feedbacks: list, threshold: int = 2) -> list:
    """提取差评案例用于改进"""
    bad_cases = []
    for f in feedbacks:
        if f["rating"] <= threshold:
            bad_cases.append({
                "message_id": f["message_id"],
                "rating": f["rating"],
                "issue": f.get("feedback_type"),
                "comment": f.get("comment"),
            })
    return bad_cases

# 将Bad Case转为测试用例
def bad_cases_to_test_cases(bad_cases: list) -> list:
    """把差评案例转为回归测试用例"""
    test_cases = []
    for case in bad_cases:
        # 需要从历史中找到原始问题和回答
        # 这里简化处理
        test_cases.append({
            "id": case["message_id"],
            "issue": case["issue"],
            "note": case["comment"],
            "status": "pending",  # 待修复
        })
    return test_cases
```

## 七、反馈闭环指标

| 指标 | 含义 | 目标 |
|------|------|------|
| 满意度 | 平均评分(1-5) | ≥4.0 |
| 好评率 | 4-5星占比 | ≥80% |
| Bad Case率 | 1-2星占比 | ≤5% |
| 反馈收集率 | 留下反馈的用户占比 | ≥10% |
| 修复速度 | 从反馈到修复的天数 | ≤7天 |
| 回归率 | 修复后问题重现的比例 | ≤5% |
