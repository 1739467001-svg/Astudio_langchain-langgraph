```python
import time
from functools import wraps
from dataclasses import dataclass, field
from collections import defaultdict

@dataclass
class TraceSpan:
    """性能追踪Span。"""
    name: str
    start: float
    end: float = 0
    duration_ms: float = 0

@dataclass
class PerformanceTrace:
    """全链路性能追踪。"""
    spans: list[TraceSpan] = field(default_factory=list)
    _current: TraceSpan | None = None

    def start_span(self, name: str):
        if self._current:
            self._current.end = time.time()
            self._current.duration_ms = round((self._current.end - self._current.start) * 1000, 2)
            self.spans.append(self._current)
        self._current = TraceSpan(name=name, start=time.time())

    def end_trace(self) -> dict:
        if self._current:
            self._current.end = time.time()
            self._current.duration_ms = round((self._current.end - self._current.start) * 1000, 2)
            self.spans.append(self._current)
            self._current = None

        total = sum(s.duration_ms for s in self.spans)
        return {
            "total_ms": round(total, 2),
            "spans": [
                {"name": s.name, "ms": s.duration_ms, "pct": round(s.duration_ms / total * 100, 1)}
                for s in self.spans
            ],
            "bottleneck": max(self.spans, key=lambda s: s.duration_ms).name if self.spans else "none",
        }

def trace_performance(func):
    """性能追踪装饰器。"""
    @wraps(func)
    async def wrapper(*args, **kwargs):
        trace = kwargs.get("trace") or PerformanceTrace()
        trace.start_span(func.__name__)
        result = await func(*args, **kwargs)
        trace.end_trace()
        return result
    return wrapper
```

## 三、性能基准测试

```python
import asyncio
import statistics

class PerformanceBenchmark:
    """性能基准测试器。"""

    @staticmethod
    async def benchmark(
        func,
        inputs: list,
        concurrency: int = 1,
    ) -> dict:
        """基准测试。

        Args:
            func: 异步函数
            inputs: 输入列表
            concurrency: 并发数
        """
        latencies = []
        errors = 0

        # 并发执行
        semaphore = asyncio.Semaphore(concurrency)

        async def run_one(input_data):
            nonlocal errors
            async with semaphore:
                start = time.time()
                try:
                    await func(input_data)
                    latencies.append((time.time() - start) * 1000)
                except Exception:
                    errors += 1

        tasks = [run_one(inp) for inp in inputs]
        await asyncio.gather(*tasks)

        if not latencies:
            return {"error": "无成功请求"}

        latencies.sort()
        n = len(latencies)

        return {
            "total_requests": len(inputs),
            "successful": n,
            "failed": errors,
            "avg_latency_ms": round(statistics.mean(latencies), 2),
            "p50_ms": round(latencies[n // 2], 2),
            "p95_ms": round(latencies[int(n * 0.95)], 2),
            "p99_ms": round(latencies[int(n * 0.99)], 2),
            "qps": round(n / (sum(latencies) / 1000 / concurrency), 2),
        }
```
