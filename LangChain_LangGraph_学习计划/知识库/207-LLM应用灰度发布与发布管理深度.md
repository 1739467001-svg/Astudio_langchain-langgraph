```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import hashlib

class CanaryStage(str, Enum):
    PERCENT_5 = "5%"
    PERCENT_25 = "25%"
    PERCENT_50 = "50%"
    PERCENT_100 = "100%"

@dataclass
class CanaryConfig:
    """金丝雀发布配置。"""
    new_version: str
    old_version: str
    stages: list = field(default_factory=lambda: [
        ("5%", 1800),    # 5%流量，观察30分钟
        ("25%", 3600),   # 25%流量，观察1小时
        ("50%", 7200),   # 50%流量，观察2小时
        ("100%", 0),     # 全量
    ])
    current_stage: int = 0
    rollback_thresholds: dict = field(default_factory=lambda: {
        "error_rate": 0.05,      # 错误率>5%回滚
        "latency_p95": 5000,      # P95>5秒回滚
        "satisfaction_drop": 0.2,  # 满意度降20%回滚
    })

class CanaryReleaseManager:
    """金丝雀发布管理器。"""

    def __init__(self, config: CanaryConfig):
        self.config = config
        self.metrics_history: list[dict] = []

    def should_route_to_new(self, user_id: str) -> bool:
        """决定用户是否路由到新版本。"""
        if self.config.current_stage >= len(self.config.stages):
            return True

        target_percent = int(self.config.stages[self.config.current_stage][0].replace("%", ""))
        hash_val = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 100
        return hash_val < target_percent

    def check_health(self, current_metrics: dict, baseline: dict) -> dict:
        """检查金丝雀健康度。"""
        checks = {}

        # 错误率
        error_rate = current_metrics.get("error_rate", 0)
        threshold = self.config.rollback_thresholds["error_rate"]
        checks["error_rate"] = {
            "value": error_rate,
            "threshold": threshold,
            "passed": error_rate <= threshold,
        }

        # 延迟
        latency = current_metrics.get("latency_p95", 0)
        latency_threshold = self.config.rollback_thresholds["latency_p95"]
        checks["latency_p95"] = {
            "value": latency,
            "threshold": latency_threshold,
            "passed": latency <= latency_threshold,
        }

        # 满意度
        satisfaction = current_metrics.get("satisfaction", 1.0)
        baseline_satisfaction = baseline.get("satisfaction", 1.0)
        drop = (baseline_satisfaction - satisfaction) / baseline_satisfaction if baseline_satisfaction > 0 else 0
        checks["satisfaction"] = {
            "value": satisfaction,
            "baseline": baseline_satisfaction,
            "drop": round(drop, 4),
            "passed": drop < self.config.rollback_thresholds["satisfaction_drop"],
        }

        all_passed = all(c["passed"] for c in checks.values())

        return {
            "healthy": all_passed,
            "action": "advance" if all_passed else "rollback",
            "checks": checks,
            "stage": self.config.stages[self.config.current_stage][0] if self.config.current_stage < len(self.config.stages) else "100%",
        }

    def advance(self) -> bool:
        """推进到下一阶段。"""
        if self.config.current_stage < len(self.config.stages) - 1:
            self.config.current_stage += 1
            return True
        return False

    def rollback(self) -> str:
        """回滚。"""
        self.config.current_stage = 0
        # 交换版本
        self.config.old_version, self.config.new_version = self.config.new_version, self.config.old_version
        return f"已回滚到 {self.config.old_version}"
```

## 三、特性开关

```python
class FeatureFlagManager:
    """特性开关管理器。

    运行时控制功能开关，
    无需重新部署即可启用/禁用功能。
    """

    def __init__(self):
        self.flags: dict[str, dict] = {}

    def set(self, name: str, enabled: bool, percentage: float = 1.0):
        """设置特性开关。"""
        self.flags[name] = {
            "enabled": enabled,
            "percentage": percentage,
            "updated_at": datetime.now().isoformat(),
        }

    def is_enabled(self, name: str, user_id: str = "") -> bool:
        """检查特性是否启用。"""
        flag = self.flags.get(name)
        if not flag or not flag["enabled"]:
            return False

        if flag["percentage"] >= 1.0:
            return True

        hash_val = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 100
        return hash_val < flag["percentage"] * 100

    def enable(self, name: str):
        """启用特性。"""
        if name in self.flags:
            self.flags[name]["enabled"] = True

    def disable(self, name: str):
        """禁用特性（秒级生效，无需部署）。"""
        if name in self.flags:
            self.flags[name]["enabled"] = False
```

## 四、回滚自动化

```python
class AutoRollbackManager:
    """自动回滚管理器。"""

    def __init__(self, canary: CanaryReleaseManager):
        self.canary = canary
        self.rollback_count: int = 0

    async def monitor_and_act(self, metrics: dict, baseline: dict):
        """监控并自动决策。"""
        health = self.canary.check_health(metrics, baseline)

        if not health["healthy"]:
            self.rollback_count += 1
            action = self.canary.rollback()
            return {
                "action": "rollback",
                "reason": health["checks"],
                "message": action,
                "total_rollbacks": self.rollback_count,
            }

        # 健康则推进
        advanced = self.canary.advance()
        return {
            "action": "advance" if advanced else "complete",
            "stage": health["stage"],
        }
```
