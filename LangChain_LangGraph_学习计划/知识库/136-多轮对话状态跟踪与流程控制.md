# 多轮对话状态跟踪与流程控制

> 单轮问答简单：输入→输出。但真实对话是多轮的：用户先订机票，再选座位，再付钱——每一步都依赖前一步的结果。对话状态跟踪（DST）管理这些跨轮信息，流程控制决定对话的走向。这份指南覆盖对话状态机设计、槽位填充和流程控制。

## 三、对话状态数据结构

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any

class DialogState(str, Enum):
    IDLE = "idle"
    COLLECTING = "collecting"
    CONFIRMING = "confirming"
    EXECUTING = "executing"
    COMPLETED = "completed"
    ERROR = "error"

@dataclass
class Slot:
    """槽位：对话中需要收集的信息字段。"""
    name: str
    description: str
    value: Any = None
    required: bool = True
    options: list[str] | None = None  # 可选值
    ask_message: str = ""             # 询问用户的提示语
    confirmed: bool = False           # 是否已确认

@dataclass
class ConversationState:
    """对话状态：跨轮维护的上下文。"""
    state: DialogState = DialogState.IDLE
    intent: str = ""                   # 用户意图
    slots: dict[str, Slot] = field(default_factory=dict)
    history: list[dict] = field(default_factory=list)  # 对话历史
    context: dict = field(default_factory=dict)        # 通用上下文
    turn_count: int = 0               # 对话轮次

    def get_slot(self, name: str) -> Slot | None:
        return self.slots.get(name)

    def set_slot(self, name: str, value: Any):
        if name in self.slots:
            self.slots[name].value = value

    def get_missing_slots(self) -> list[Slot]:
        """获取尚未填充的必需槽位。"""
        return [
            s for s in self.slots.values()
            if s.required and s.value is None
        ]

    def all_required_filled(self) -> bool:
        """所有必需槽位是否已填充。"""
        return len(self.get_missing_slots()) == 0

    def get_summary(self) -> str:
        """生成当前状态的摘要。"""
        filled = {s.name: s.value for s in self.slots.values() if s.value is not None}
        return f"意图: {self.intent}, 状态: {self.state.value}, 已填: {filled}"
```

## 四、对话流程引擎

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage

class DialogFlowEngine:
    """对话流程引擎。

    管理对话状态机：
    1. 识别用户意图
    2. 收集必需信息（槽位填充）
    3. 确认信息
    4. 执行操作
    5. 返回结果
    """

    def __init__(self, llm: BaseChatModel):
        self.llm = llm
        # 预定义流程（意图→槽位模板）
        self.flows: dict[str, list[Slot]] = {}
        self.conversations: dict[str, ConversationState] = {}

    def register_flow(self, intent: str, slots: list[Slot]):
        """注册对话流程模板。"""
        self.flows[intent] = slots

    async def process(
        self,
        session_id: str,
        user_input: str,
    ) -> str:
        """处理用户输入，返回Agent响应。"""
        # 获取或创建对话状态
        state = self.conversations.get(session_id)
        if state is None:
            state = ConversationState()
            self.conversations[session_id] = state

        state.turn_count += 1
        state.history.append({"role": "user", "content": user_input})

        # 状态机处理
        if state.state == DialogState.IDLE:
            return await self._handle_idle(state, user_input)
        elif state.state == DialogState.COLLECTING:
            return await self._handle_collecting(state, user_input)
        elif state.state == DialogState.CONFIRMING:
            return await self._handle_confirming(state, user_input)
        else:
            # 重置对话
            state.state = DialogState.IDLE
            return await self._handle_idle(state, user_input)

    async def _handle_idle(self, state: ConversationState, user_input: str) -> str:
        """空闲态：识别意图。"""
        intent = await self._identify_intent(user_input)

        if intent and intent in self.flows:
            # 初始化槽位
            state.intent = intent
            state.slots = {s.name: Slot(**s.__dict__) for s in self.flows[intent]}
            state.state = DialogState.COLLECTING

            # 尝试从第一句话提取槽位
            await self._extract_slots(state, user_input)

            # 检查是否所有必需槽位已填充
            if state.all_required_filled():
                return await self._transition_to_confirming(state)
            else:
                return self._ask_next_slot(state)
        else:
            return "你好！我可以帮你订机票、查天气或回答问题。请告诉我你需要什么？"

    async def _handle_collecting(self, state: ConversationState, user_input: str) -> str:
        """收集态：提取槽位信息。"""
        await self._extract_slots(state, user_input)

        if state.all_required_filled():
            return await self._transition_to_confirming(state)
        else:
            return self._ask_next_slot(state)

    async def _handle_confirming(self, state: ConversationState, user_input: str) -> str:
        """确认态：用户确认或修改。"""
        lower_input = user_input.lower()

        if any(w in lower_input for w in ["确认", "对", "是", "好的", "没错", "yes", "ok"]):
            # 确认→执行
            state.state = DialogState.EXECUTING
            return await self._execute(state)
        elif any(w in lower_input for w in ["修改", "不对", "不是", "改"]):
            # 修改→回到收集
            state.state = DialogState.COLLECTING
            return "好的，请告诉我需要修改什么？"
        else:
            # 不明确的回答
            return "请确认以上信息是否正确？回复'确认'或说明需要修改的内容。"

    async def _execute(self, state: ConversationState) -> str:
        """执行态：调用实际操作。"""
        # 这里调用实际的业务逻辑
        result = f"已执行 {state.intent}："
        for slot in state.slots.values():
            if slot.value is not None:
                result += f"\n  {slot.description}: {slot.value}"

        state.state = DialogState.COMPLETED
        result += "\n操作完成！还有其他需要帮助的吗？"
        return result

    async def _identify_intent(self, user_input: str) -> str | None:
        """用LLM识别用户意图。"""
        prompt = f"""判断用户输入的意图。可选意图: {list(self.flows.keys())}

用户输入: {user_input}

只输出意图名称（如book_flight），如果无法判断则输出unknown:"""

        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        intent = response.content.strip().lower()
        return intent if intent in self.flows else None

    async def _extract_slots(self, state: ConversationState, user_input: str):
        """从用户输入中提取槽位。"""
        missing = state.get_missing_slots()
        if not missing:
            return

        slot_desc = "\n".join(
            f"- {s.name}: {s.description}" + (f" (可选值: {s.options})" if s.options else "")
            for s in missing
        )

        prompt = f"""从用户输入中提取以下信息。如果未提及则留空。

需要提取的字段:
{slot_desc}

用户输入: {user_input}

输出JSON格式:
```json
{{"字段名": "值"}}
```
只输出提取到的字段。"""

        response = await self.llm.ainvoke([HumanMessage(content=prompt)])

        import json, re
        json_match = re.search(r'\{.*\}', response.content, re.DOTALL)
        if json_match:
            data = json.loads(json_match.group())
            for slot_name, value in data.items():
                if slot_name in state.slots and value:
                    state.slots[slot_name].value = value

    def _ask_next_slot(self, state: ConversationState) -> str:
        """询问下一个未填充的槽位。"""
        missing = state.get_missing_slots()
        if not missing:
            return "信息已收集完整。"

        next_slot = missing[0]
        return next_slot.ask_message or f"请提供{next_slot.description}："

    async def _transition_to_confirming(self, state: ConversationState) -> str:
        """转入确认态。"""
        state.state = DialogState.CONFIRMING

        summary = "请确认以下信息：\n"
        for slot in state.slots.values():
            if slot.value is not None:
                summary += f"  {slot.description}: {slot.value}\n"
        summary += "\n回复'确认'或说明需要修改的内容。"
        return summary
```

## 五、注册对话流程

```python
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o", temperature=0)
engine = DialogFlowEngine(llm)

# 注册订机票流程
engine.register_flow("book_flight", [
    Slot(name="departure", description="出发城市", ask_message="请问从哪个城市出发？"),
    Slot(name="destination", description="目的城市", ask_message="请问目的地是哪个城市？"),
    Slot(name="date", description="出发日期", ask_message="请问出发日期是哪天？"),
    Slot(name="passenger", description="乘客姓名", ask_message="请问乘客姓名？"),
    Slot(name="seat_type", description="座位类型", required=False,
         options=["aisle", "window"], ask_message="需要靠过道还是靠窗的座位？"),
])

# 注册查天气流程
engine.register_flow("check_weather", [
    Slot(name="city", description="城市", ask_message="请问查询哪个城市的天气？"),
    Slot(name="date", description="日期", required=False, ask_message="查询哪天的天气？"),
])

# 使用
async def demo():
    session = "user-001"

    # 第一轮：用户意图+部分信息
    r1 = await engine.process(session, "我想订一张北京到上海的机票")
    print(r1)
    # → "请问出发日期是哪天？"

    # 第二轮：补充日期
    r2 = await engine.process(session, "明天")
    print(r2)
    # → "请问乘客姓名？"

    # 第三轮：补充姓名
    r3 = await engine.process(session, "张三")
    print(r3)
    # → "请确认以下信息：..."

    # 第四轮：确认
    r4 = await engine.process(session, "确认")
    print(r4)
    # → "已执行 book_flight：..."
```

## 六、与 LangGraph 集成

```python
from langgraph.graph import StateGraph, START, END
from typing import TypedDict, Annotated

class DialogGraphState(TypedDict):
    conversation_state: ConversationState
    user_input: str
    response: str

async def dialog_node(state: DialogGraphState) -> dict:
    """对话节点：调用对话引擎处理。"""
    cs = state["conversation_state"]
    response = await engine.process(cs.intent, state["user_input"])
    return {"response": response}

def route_dialog(state: DialogGraphState) -> str:
    """根据对话状态路由。"""
    cs = state["conversation_state"]
    if cs.state == DialogState.COMPLETED:
        return "end"
    return "continue"

# 构建对话图
graph = StateGraph(DialogGraphState)
graph.add_node("dialog", dialog_node)
graph.add_edge(START, "dialog")
graph.add_conditional_edges("dialog", route_dialog, {
    "continue": "dialog",
    "end": END,
})

dialog_app = graph.compile()
```

## 七、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 槽位定义要明确 | 必填/选填、可选值、提示语 | ★★★ |
| 意图识别用LLM | 比规则更灵活 | ★★★ |
| 一次输入可填充多个槽位 | 减少对话轮次 | ★★☆ |
| 必须有确认步骤 | 防止信息错误 | ★★★ |
| 对话超时要重置 | 避免跨意图混淆 | ★★☆ |
| 支持中途修改 | 用户可回退修改 | ★★☆ |
