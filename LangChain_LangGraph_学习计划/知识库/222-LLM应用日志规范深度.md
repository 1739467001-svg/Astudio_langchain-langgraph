## 二、结构化日志

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import json
import re

class LogLevel(str, Enum):
    DEBUG = "DEBUG"
    INFO = "INFO"
    WARN = "WARN"
    ERROR = "ERROR"
    AUDIT = "AUDIT"

@dataclass
class LLMLogRecord:
    """LLM日志记录。"""
    timestamp: str
    level: LogLevel
    event: str             # 事件类型
    request_id: str        # 请求ID
    user_id: str = ""
    session_id: str = ""
    model: str = ""
    prompt_tokens: int = 0
    completion_tokens: int = 0
    latency_ms: float = 0
    tool_name: str = ""
    error: str = ""
    sanitized_content: str = ""  # 脱敏后的内容（截断）

    def to_json(self) -> str:
        return json.dumps({
            k: v for k, v in self.__dict__.items() if v
        }, ensure_ascii=False)

class StructuredLogger:
    """结构化日志记录器。"""

    def __init__(self):
        self.records: list[LLMLogRecord] = []

    def log(self, level: LogLevel, event: str, request_id: str, **kwargs):
        """记录日志。"""
        # 脱敏处理
        if "content" in kwargs:
            kwargs["sanitized_content"] = self._sanitize(kwargs.pop("content"))

        record = LLMLogRecord(
            timestamp=datetime.now().isoformat(),
            level=level,
            event=event,
            request_id=request_id,
            **kwargs,
        )
        self.records.append(record)

    @staticmethod
    def _sanitize(text: str) -> str:
        """脱敏。"""
        text = re.sub(r'\b1[3-9]\d{9}\b', '[PHONE]', text)
        text = re.sub(r'\b\d{17}[\dXx]\b', '[ID]', text)
        text = re.sub(r'sk-[a-zA-Z0-9]{40,}', '[API_KEY]', text)
        return text[:500]  # 截断

    def get_by_request(self, request_id: str) -> list[LLMLogRecord]:
        """获取请求的完整日志链。"""
        return [r for r in self.records if r.request_id == request_id]

    def get_errors(self, limit: int = 20) -> list[LLMLogRecord]:
        """获取错误日志。"""
        return [r for r in self.records if r.level == LogLevel.ERROR][-limit:]

    def get_audit_trail(self, user_id: str) -> list[LLMLogRecord]:
        """获取用户审计轨迹。"""
        return [r for r in self.records if r.user_id == user_id and r.level == LogLevel.AUDIT]

    def token_usage_report(self) -> dict:
        """Token使用报告。"""
        total_prompt = sum(r.prompt_tokens for r in self.records)
        total_completion = sum(r.completion_tokens for r in self.records)

        by_model = {}
        for r in self.records:
            if r.model:
                if r.model not in by_model:
                    by_model[r.model] = {"prompt": 0, "completion": 0, "calls": 0}
                by_model[r.model]["prompt"] += r.prompt_tokens
                by_model[r.model]["completion"] += r.completion_tokens
                by_model[r.model]["calls"] += 1

        return {
            "total_prompt_tokens": total_prompt,
            "total_completion_tokens": total_completion,
            "total_tokens": total_prompt + total_completion,
            "by_model": by_model,
            "total_requests": len(self.records),
        }
```

## 三、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 每条日志有request_id | 可追溯全链路 | ★★★ |
| 日志脱敏 | PII不记录 | ★★★ |
| 记录Token消耗 | 成本追踪 | ★★★ |
| 审计级别日志 | 用户操作可追溯 | ★★☆ |
| 日志截断 | 防止超长日志 | ★★☆ |
| 支持按请求ID查询 | 快速排障 | ★★★ |
