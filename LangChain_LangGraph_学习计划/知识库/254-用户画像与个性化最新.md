## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
import json, re

@dataclass
class UserProfile:
    """用户画像。"""
    user_id: str
    name: str = ""
    language: str = "zh"
    region: str = ""
    preferences: dict = field(default_factory=dict)  # {category: value}
    history: list[dict] = field(default_factory=list)
    ratings: dict = field(default_factory=dict)
    skills: list[str] = field(default_factory=list)  # 用户技能水平

    def add_preference(self, key: str, value: str):
        self.preferences[key] = {"value": value, "updated_at": datetime.now().isoformat()}

    def get_profile_text(self) -> str:
        """生成画像文本注入Prompt。"""
        lines = []
        if self.name:
            lines.append(f"用户名: {self.name}")
        if self.language:
            lines.append(f"语言: {self.language}")
        if self.preferences:
            for k, v in self.preferences.items():
                lines.append(f"{k}: {v['value']}")
        if self.skills:
            lines.append(f"技能水平: {', '.join(self.skills)}")
        return "\n".join(lines) if lines else "无已知信息"


class ProfileLearningEngine:
    """画像学习引擎——从对话中自动学习。"""

    EXTRACT_PROMPT = """从以下对话中提取用户信息。只提取明确提及的。

对话:
{conversation}

输出JSON:
```json
{{
  "facts": [{{"key": "姓名", "value": "张三"}}],
  "preferences": [{{"key": "回答风格", "value": "简洁技术性"}}]
}}
```
无新信息返回空数组。"""

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def learn(self, profile: UserProfile, conversation: str) -> UserProfile:
        """从对话中学习用户信息。"""
        prompt = self.EXTRACT_PROMPT.format(conversation=conversation[:1000])
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])

        match = re.search(r'\{.*\}', response.content, re.DOTALL)
        if match:
            data = json.loads(match.group())
            for fact in data.get("facts", []):
                if fact["key"] == "姓名":
                    profile.name = fact["value"]
            for pref in data.get("preferences", []):
                profile.add_preference(pref["key"], pref["value"])

        return profile


class PersonalizationInjector:
    """个性化注入器——将画像注入系统提示。"""

    @staticmethod
    def build_system_prompt(base_prompt: str, profile: UserProfile) -> str:
        """构建个性化系统提示。"""
        profile_text = profile.get_profile_text()
        if profile_text == "无已知信息":
            return base_prompt

        return f"""{base_prompt}

## 用户画像
{profile_text}

## 个性化要求
- 用用户的偏好语言回答
- 适配用户技能水平
- 记住用户已知信息，不重复问"""
```

## 三、最佳实践

| 实践 | 说明 | 优先级 |
