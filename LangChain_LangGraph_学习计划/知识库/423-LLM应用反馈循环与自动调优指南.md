# LLM 应用反馈循环与自动调优指南

> 上线后系统不是一成不变的——用户反馈说"回答太啰嗦"，你改了 Prompt；又说"有时候不够详细"，你又改回去。手动调优累且慢。反馈循环自动调优让系统自己从反馈中学习，自动调整参数、优化 Prompt、改进检索策略。

---

## 1. 反馈循环全景

### 什么是自动调优

```
传统：人工发现 → 人工分析 → 人工修改 → 人工部署
      周期：天/周

自动调优：系统收集反馈 → 自动分析 → 自动生成优化方案 → A/B 验证 → 自动部署
         周期：小时/天
```

### 调优的三个层次

| 层次 | 调什么 | 频率 | 自动化程度 |
|------|--------|------|-----------|
| 参数调优 | temperature/top_p/检索数量 | 每小时 | 全自动 |
| Prompt 调优 | 系统提示/Few-Shot | 每天 | 半自动 |
| 策略调优 | 检索策略/分块策略/路由 | 每周 | 人工审核 |

## 2. 反馈信号采集

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
from collections import defaultdict
import time

class SignalType(Enum):
    EXPLICIT = "explicit"     # 显式：点赞/踩/评分
    IMPLICIT = "implicit"     # 隐式：复制/重生成/停留时间
    OUTCOME = "outcome"       # 结果：转化率/完成率
    EXPERT = "expert"         # 专家标注


@dataclass
class FeedbackSignal:
    """反馈信号"""
    id: str = ""
    session_id: str = ""
    user_id: str = ""
    signal_type: SignalType = SignalType.EXPLICIT
    # 评分
    score: float = 0.0          # 0-1，越高越好
    # 上下文
    query: str = ""
    response: str = ""
    retrieved_docs: list[dict] = field(default_factory=list)
    # 配置快照
    config_snapshot: dict = field(default_factory=dict)  # 当时的参数
    # 元数据
    timestamp: float = field(default_factory=time.time)
    processed: bool = False


class FeedbackPipeline:
    """反馈采集管道"""

    def __init__(self):
        self.signals: list[FeedbackSignal] = []
        self.aggregated: dict[str, list[float]] = defaultdict(list)

    def collect(self, signal: FeedbackSignal):
        """采集反馈信号"""
        self.signals.append(signal)
        # 按配置快照聚合
        config_key = self._config_key(signal.config_snapshot)
        self.aggregated[config_key].append(signal.score)

    def get_performance_by_config(self) -> dict[str, dict]:
        """按配置分组查看性能"""
        import numpy as np
        result = {}
        for config_key, scores in self.aggregated.items():
            if len(scores) >= 5:  # 至少 5 个样本
                result[config_key] = {
                    "count": len(scores),
                    "mean": float(np.mean(scores)),
                    "std": float(np.std(scores)),
                    "p25": float(np.percentile(scores, 25)),
                    "p75": float(np.percentile(scores, 75)),
                }
        return result

    def get_underperforming(self, threshold: float = 0.6) -> dict[str, dict]:
        """获取表现不佳的配置"""
        perf = self.get_performance_by_config()
        return {
            k: v for k, v in perf.items()
            if v["mean"] < threshold
        }

    @staticmethod
    def _config_key(config: dict) -> str:
        """将配置转为可比较的键"""
        return f"temp={config.get('temperature')}_topk={config.get('top_k')}_model={config.get('model')}"
```

## 3. 自动参数调优器

```python
import numpy as np
from itertools import product

class AutoParameterTuner:
    """自动参数调优器：贝叶斯优化风格"""

    def __init__(self):
        self.param_space = {
            "temperature": [0.0, 0.1, 0.3, 0.5, 0.7],
            "top_k": [3, 5, 7, 10],
            "max_tokens": [256, 512, 1024],
        }
        self.trials: list[dict] = []  # 试验记录
        self.best_params: dict = {}
        self.best_score: float = 0.0

    def suggest(self) -> dict:
        """建议下一组参数"""
        if len(self.trials) < 5:
            # 初始阶段：随机采样
            return self._random_sample()

        # 利用阶段：选择表现好的参数附近的值
        return self._exploit()

    def _random_sample(self) -> dict:
        import random
        return {
            param: random.choice(values)
            for param, values in self.param_space.items()
        }

    def _exploit(self) -> dict:
        """基于历史表现选择"""
        # 找到表现最好的试验
        sorted_trials = sorted(self.trials, key=lambda t: t["score"], reverse=True)
        best = sorted_trials[0]

        # 在最优参数附近探索
        params = {}
        for param, values in self.param_space.items():
            current = best["params"].get(param)
            if current in values:
                idx = values.index(current)
                # 50% 保持，50% 探索邻近值
                import random
                if random.random() < 0.5:
                    params[param] = current
                else:
                    neighbors = []
                    if idx > 0:
                        neighbors.append(values[idx - 1])
                    if idx < len(values) - 1:
                        neighbors.append(values[idx + 1])
                    params[param] = random.choice(neighbors) if neighbors else current
            else:
                params[param] = values[0]

        return params

    def record(self, params: dict, score: float):
        """记录试验结果"""
        self.trials.append({
            "params": params,
            "score": score,
            "timestamp": time.time(),
        })
        if score > self.best_score:
            self.best_score = score
            self.best_params = params.copy()

    def report(self) -> dict:
        return {
            "total_trials": len(self.trials),
            "best_score": self.best_score,
            "best_params": self.best_params,
            "top_3_trials": sorted(self.trials, key=lambda t: t["score"], reverse=True)[:3],
        }
```

## 4. Prompt 自动优化器

```python
class PromptAutoOptimizer:
    """Prompt 自动优化器"""

    def __init__(self, llm: ChatOpenAI | None = None):
        self.llm = llm or ChatOpenAI(temperature=0.3, model="gpt-4o")

    def analyze_issues(self, feedbacks: list[FeedbackSignal]) -> list[dict]:
        """从反馈中分析问题"""
        from langchain_core.prompts import ChatPromptTemplate
        import json

        negative = [f for f in feedbacks if f.score < 0.5]
        if not negative:
            return []

        samples = "\n".join([
            f"- 查询：{f.query[:100]}\n  回答：{f.response[:200]}\n  评分：{f.score}"
            for f in negative[:10]
        ])

        prompt = ChatPromptTemplate.from_messages([
            ("system", """分析以下负面反馈的共同问题，输出 JSON 列表：
[
  {"issue": "问题描述", "frequency": "high/medium/low", "suggestion": "改进建议"}
]"""),
            ("human", "{samples}"),
        ])

        chain = prompt | self.llm
        result = chain.invoke({"samples": samples})

        try:
            return json.loads(result.content)
        except json.JSONDecodeError:
            return [{"issue": "解析失败", "frequency": "unknown", "suggestion": ""}]

    def generate_optimized_prompt(
        self,
        current_prompt: str,
        issues: list[dict],
    ) -> str:
        """基于问题生成优化后的 Prompt"""
        from langchain_core.prompts import ChatPromptTemplate

        issues_text = "\n".join([
            f"- {i['issue']}（频率：{i.get('frequency', 'unknown')}）→ 建议：{i.get('suggestion', '')}"
            for i in issues
        ])

        prompt = ChatPromptTemplate.from_messages([
            ("system", """你是 Prompt 工程专家。根据反馈问题优化 Prompt。

优化原则：
1. 针对每个问题添加对应约束
2. 不要破坏原有有效部分
3. 保持 Prompt 简洁
4. 添加必要的 Few-Shot 示例

直接输出优化后的完整 Prompt，不要加解释。"""),
            ("human", "当前 Prompt：\n{current_prompt}\n\n需要解决的问题：\n{issues}"),
        ])

        chain = prompt | self.llm
        result = chain.invoke({
            "current_prompt": current_prompt,
            "issues": issues_text,
        })

        return result.content

    def compare_prompts(
        self,
        old_prompt: str,
        new_prompt: str,
        test_cases: list[dict],
        llm: ChatOpenAI,
    ) -> dict:
        """对比新旧 Prompt 效果"""
        from langchain_core.prompts import ChatPromptTemplate
        from langchain_core.messages import HumanMessage, SystemMessage

        old_scores = []
        new_scores = []

        for case in test_cases:
            query = case["query"]
            expected = case.get("expected", "")

            # 用旧 Prompt
            old_response = llm.invoke([
                SystemMessage(content=old_prompt),
                HumanMessage(content=query),
            ])
            old_score = self._score_response(old_response.content, expected, query)

            # 用新 Prompt
            new_response = llm.invoke([
                SystemMessage(content=new_prompt),
                HumanMessage(content=query),
            ])
            new_score = self._score_response(new_response.content, expected, query)

            old_scores.append(old_score)
            new_scores.append(new_score)

        return {
            "old_avg_score": float(np.mean(old_scores)),
            "new_avg_score": float(np.mean(new_scores)),
            "improvement": float(np.mean(new_scores) - np.mean(old_scores)),
            "old_scores": old_scores,
            "new_scores": new_scores,
        }

    @staticmethod
    def _score_response(response: str, expected: str, query: str) -> float:
        """简单评分：关键词匹配"""
        if not expected:
            return 0.5  # 无期望答案，给中性分
        expected_words = set(expected.lower().split())
        response_words = set(response.lower().split())
        overlap = len(expected_words & response_words)
        return overlap / max(len(expected_words), 1)
```

## 5. 完整自动调优循环

```python
class AutoTuningLoop:
    """完整自动调优循环"""

    def __init__(
        self,
        llm: ChatOpenAI,
        current_config: dict,
        current_prompt: str,
    ):
        self.llm = llm
        self.feedback = FeedbackPipeline()
        self.param_tuner = AutoParameterTuner()
        self.prompt_optimizer = PromptAutoOptimizer(llm)
        self.current_config = current_config
        self.current_prompt = current_prompt
        self.history: list[dict] = []

    def run_cycle(self) -> dict:
        """执行一轮调优"""
        results = {}

        # 1. 参数调优
        new_params = self.param_tuner.suggest()
        perf = self.feedback.get_performance_by_config()

        if perf:
            config_key = self.feedback._config_key(self.current_config)
            current_perf = perf.get(config_key, {})
            if current_perf.get("mean", 0) < 0.7:
                # 当前配置表现不佳，切换
                self.current_config.update(new_params)
                results["param_tuning"] = {
                    "action": "updated",
                    "new_params": new_params,
                    "reason": f"当前性能 {current_perf.get('mean', 0):.1%} < 70%",
                }
            else:
                results["param_tuning"] = {"action": "no_change"}
        else:
            results["param_tuning"] = {"action": "insufficient_data"}

        # 2. Prompt 调优
        negatives = [s for s in self.feedback.signals if s.score < 0.5]
        if len(negatives) >= 5:
            issues = self.prompt_optimizer.analyze_issues(negatives)
            if issues:
                new_prompt = self.prompt_optimizer.generate_optimized_prompt(
                    self.current_prompt, issues
                )
                # 需要人工审核（半自动）
                results["prompt_tuning"] = {
                    "action": "suggested",
                    "issues_found": len(issues),
                    "new_prompt": new_prompt[:200] + "...",
                    "needs_review": True,
                }
        else:
            results["prompt_tuning"] = {"action": "insufficient_negative_feedback"}

        # 3. 记录历史
        self.history.append({
            "timestamp": time.time(),
            "current_config": self.current_config.copy(),
            "feedback_count": len(self.feedback.signals),
            "avg_score": self._avg_score(),
            "results": results,
        })

        return results

    def _avg_score(self) -> float:
        if not self.feedback.signals:
            return 0.0
        return sum(s.score for s in self.feedback.signals) / len(self.feedback.signals)

    def dashboard(self) -> dict:
        return {
            "total_feedback": len(self.feedback.signals),
            "avg_score": f"{self._avg_score():.1%}",
            "tuning_cycles": len(self.history),
            "current_config": self.current_config,
            "underperforming_configs": self.feedback.get_underperforming(0.6),
            "param_tuner_report": self.param_tuner.report(),
            "recent_cycles": self.history[-5:],
        }
```

## 6. 使用示例

```python
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

# 初始化调优循环
tuning_loop = AutoTuningLoop(
    llm=llm,
    current_config={"temperature": 0.0, "top_k": 5, "model": "gpt-4o-mini"},
    current_prompt="你是客服助手。请礼貌回答用户问题。",
)

# 模拟反馈数据
import random
for i in range(50):
    signal = FeedbackSignal(
        session_id=f"session_{i}",
        user_id=f"user_{i}",
        signal_type=SignalType.EXPLICIT,
        score=random.gauss(0.65, 0.2),
        query=f"问题 {i}",
        response=f"回答 {i}",
        config_snapshot={"temperature": 0.0, "top_k": 5, "model": "gpt-4o-mini"},
    )
    signal.score = max(0, min(1, signal.score))
    tuning_loop.feedback.collect(signal)

# 运行调优循环
result = tuning_loop.run_cycle()
print(f"参数调优: {result.get('param_tuning', {}).get('action')}")
print(f"Prompt调优: {result.get('prompt_tuning', {}).get('action')}")
print(f"仪表盘: {tuning_loop.dashboard()}")
```

## 7. 调优安全边界

| 约束 | 说明 |
|------|------|
| 参数范围 | temperature 只能在 0-1 之间 |
| Prompt 审核 | 自动生成的 Prompt 需人工审核 |
| 灰度验证 | 新配置先 5% 灰度测试 |
| 回滚机制 | 调优后效果下降自动回滚 |
| 频率限制 | 每天最多调优 1 次 |
| 最低样本量 | 至少 50 条反馈才调优 |
