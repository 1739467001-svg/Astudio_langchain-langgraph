## 二、QPS预估与资源计算

```python
from dataclasses import dataclass

@dataclass
class CapacityPlan:
    """容量规划结果。"""
    peak_qps: float
    concurrent_requests: int
    api_containers: int
    vector_memory_gb: int
    cache_memory_gb: int
    daily_storage_gb: float
    daily_token_cost: float

class CapacityPlanner:
    """容量规划器。"""

    @staticmethod
    def plan(
        daily_active_users: int = 1000,
        sessions_per_user: int = 3,
        messages_per_session: int = 5,
        peak_factor: float = 3.0,
        avg_latency_ms: float = 2000,
        avg_tokens_per_req: int = 500,
    ) -> CapacityPlan:
        """完整容量规划。"""
        # QPS预估
        daily_requests = daily_active_users * sessions_per_user * messages_per_session
        avg_qps = daily_requests / (8 * 3600)  # 8小时活跃
        peak_qps = avg_qps * peak_factor

        # 资源计算
        concurrent = peak_qps * (avg_latency_ms / 1000)
        api_containers = max(1, int(concurrent / 100) + 1)
        vector_mem = max(2, int(100000 * 512 * 4 / 1e9) + 2)
        cache_mem = max(1, int(peak_qps * 30))
        daily_storage = max(1, peak_qps * 86400 * 0.001 / 1024)

        # 成本
        daily_tokens = peak_qps * 86400 * avg_tokens_per_req
        daily_cost = daily_tokens / 1e6 * 0.15  # mini定价

        return CapacityPlan(
            peak_qps=round(peak_qps, 2),
            concurrent_requests=round(concurrent, 1),
            api_containers=api_containers,
            vector_memory_gb=vector_mem,
            cache_memory_gb=cache_mem,
            daily_storage_gb=round(daily_storage, 2),
            daily_token_cost=round(daily_cost, 2),
        )

    @staticmethod
    def identify_bottlenecks(peak_qps: float) -> list[dict]:
        """识别瓶颈。"""
        return [
            {"component": "LLM API", "limit": "RPM限制(如500)", "max_qps": 8.3, "first_bottleneck": True},
            {"component": "向量库", "limit": "内存不足", "max_qps": "取决于HNSW"},
            {"component": "API层", "limit": "容器并发", "max_qps": "容器数×100"},
        ]
```
