## 二、基准测试器

```python
import asyncio, time, statistics
from dataclasses import dataclass

@dataclass
class BenchmarkResult:
    """基准测试结果。"""
    total: int
    successful: int
    p50_ms: float
    p95_ms: float
    avg_tps: float
    total_cost_usd: float

async def benchmark_llm(llm, prompts: list[str], concurrency: int = 5) -> BenchmarkResult:
    """LLM基准测试。"""
    from langchain_core.messages import HumanMessage
    import tiktoken

    encoder = tiktoken.encoding_for_model("gpt-4o")
    semaphore = asyncio.Semaphore(concurrency)
    latencies = []
    total_tokens = 0

    async def run_one(prompt):
        nonlocal total_tokens
        async with semaphore:
            start = time.time()
            response = await llm.ainvoke([HumanMessage(content=prompt)])
            elapsed = (time.time() - start) * 1000
            latencies.append(elapsed)
            total_tokens += len(encoder.encode(response.content))

    await asyncio.gather(*[run_one(p) for p in prompts])

    latencies.sort()
    n = len(latencies)
    avg_tps = total_tokens / (sum(latencies) / 1000) if latencies else 0

    return BenchmarkResult(
        total=len(prompts),
        successful=n,
        p50_ms=round(latencies[n // 2], 2) if n else 0,
        p95_ms=round(latencies[int(n * 0.95)], 2) if n else 0,
        avg_tps=round(avg_tps, 2),
        total_cost_usd=round(total_tokens / 1e6 * 2.50, 6),  # 简化估算
    )
```

## 三、调优建议

```python
class PerformanceTuning:
    """性能调优建议。"""

    @staticmethod
    def recommend(current_metrics: dict) -> list[str]:
        """根据当前指标推荐优化。"""
        suggestions = []

        if current_metrics.get("ttft_ms", 0) > 500:
            suggestions.append("启用流式输出降低TTFT")

        if current_metrics.get("p95_ms", 0) > 3000:
            suggestions.append("模型路由: 80%走mini")
            suggestions.append("启用语义缓存")

        if current_metrics.get("cost_per_req", 0) > 0.01:
            suggestions.append("模型路由省成本")
            suggestions.append("语义缓存减少调用")

        if not suggestions:
            suggestions.append("性能达标，保持现状")

        return suggestions
```
