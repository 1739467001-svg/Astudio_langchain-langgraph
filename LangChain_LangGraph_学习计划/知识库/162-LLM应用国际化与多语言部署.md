## 二、语言检测与路由

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
from dataclasses import dataclass
from enum import Enum

class Language(str, Enum):
    ZH = "zh"
    EN = "en"
    JA = "ja"
    KO = "ko"
    AR = "ar"  # 阿拉伯语
    AUTO = "auto"

@dataclass
class LanguageConfig:
    """语言配置。"""
    code: str
    system_prompt: str          # 对应语言的系统提示
    embedding_model: str        # 对应语言的嵌入模型
    llm_model: str             # 对应语言的LLM
    text_direction: str = "ltr"  # ltr/rtl

LANGUAGE_CONFIGS = {
    "zh": LanguageConfig(
        code="zh",
        system_prompt="你是一个专业的AI助手。用中文回答，简洁有条理。",
        embedding_model="BAAI/bge-large-zh-v1.5",
        llm_model="gpt-4o",
    ),
    "en": LanguageConfig(
        code="en",
        system_prompt="You are a professional AI assistant. Answer in English, clear and concise.",
        embedding_model="text-embedding-3-small",
        llm_model="gpt-4o",
    ),
    "ja": LanguageConfig(
        code="ja",
        system_prompt="あなたはプロのAIアシスタントです。日本語で簡潔に答えてください。",
        embedding_model="text-embedding-3-small",
        llm_model="gpt-4o",
    ),
    "ar": LanguageConfig(
        code="ar",
        system_prompt="أنت مساعد ذكاء اصطناعي محترف. أجب باللغة العربية.",
        embedding_model="text-embedding-3-small",
        llm_model="gpt-4o",
        text_direction="rtl",
    ),
}

class LanguageRouter:
    """语言检测与路由器。"""

    @staticmethod
    def detect(text: str) -> str:
        """检测文本语言。"""
        # 简化版：基于Unicode范围
        if any('\u4e00' <= c <= '\u9fff' for c in text):
            return "zh"
        elif any('\u3040' <= c <= '\u30ff' for c in text):
            return "ja"
        elif any('\uac00' <= c <= '\ud7af' for c in text):
            return "ko"
        elif any('\u0600' <= c <= '\u06ff' for c in text):
            return "ar"
        else:
            return "en"

    @staticmethod
    def get_config(language: str) -> LanguageConfig:
        """获取语言配置。"""
        return LANGUAGE_CONFIGS.get(language, LANGUAGE_CONFIGS["en"])

    @staticmethod
    async def detect_and_route(
        llm: BaseChatModel,
        text: str,
    ) -> tuple[str, LanguageConfig]:
        """检测语言并返回配置。"""
        lang = LanguageRouter.detect(text)
        config = LanguageRouter.get_config(lang)
        return lang, config
```

```python
from langchain_core.vectorstores import VectorStore
from langchain_core.embeddings import Embeddings
from langchain_core.documents import Document

class MultilingualRAG:
    """多语言RAG系统。

    策略1：按语言分库——每种语言独立向量库
    策略2：统一嵌入——用一个多语言嵌入模型
    """

    def __init__(
        self,
        vectorstores: dict[str, VectorStore],  # {language: vectorstore}
        llm: BaseChatModel,
    ):
        self.vectorstores = vectorstores
        self.llm = llm

    async def retrieve(self, query: str) -> list[Document]:
        """多语言检索。"""
        lang = LanguageRouter.detect(query)
        vs = self.vectorstores.get(lang) or self.vectorstores.get("en")
        return await vs.asimilarity_search(query, k=5)

    async def answer(self, query: str) -> str:
        """多语言问答。"""
        lang = LanguageRouter.detect(query)
        config = LanguageRouter.get_config(lang)

        docs = await self.retrieve(query)
        context = "\n".join(d.page_content for d in docs)

        prompt = f"{config.system_prompt}\n\n信息:\n{context[:2000]}\n\n问题: {query}"
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content
```

## 四、统一嵌入方案

```python
class UnifiedEmbeddingRAG:
    """统一嵌入方案：用一个多语言嵌入模型。

    优点：简单，不需要分库
    缺点：特定语言精度可能不如专用模型
    """

    def __init__(self, vectorstore: VectorStore, llm: BaseChatModel):
        # 用多语言嵌入模型
        # 如: Cohere embed-multilingual-v3
        # 或: BGE-m3 (支持100+语言)
        self.vectorstore = vectorstore
        self.llm = llm

    async def answer(self, query: str) -> str:
        """统一检索+多语言生成。"""
        docs = await self.vectorstore.asimilarity_search(query, k=5)
        context = "\n".join(d.page_content for d in docs)

        # LLM用用户查询的语言回答
        lang = LanguageRouter.detect(query)
        lang_instruction = {
            "zh": "请用中文回答",
            "en": "Please answer in English",
            "ja": "日本語で答えてください",
        }.get(lang, "Please answer in English")

        prompt = f"基于以下信息回答问题。{lang_instruction}。\n\n信息:\n{context[:2000]}\n\n问题: {query}"
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content
```

## 五、系统提示翻译

```python
class PromptTranslator:
    """系统提示翻译器。"""

    # 核心提示的多语言版本
    PROMPTS = {
        "customer_service": {
            "zh": "你是专业客服助手。礼貌、准确、高效地回答用户问题。",
            "en": "You are a professional customer service assistant. Answer politely, accurately, and efficiently.",
            "ja": "あなたはプロのカスタマーサービスアシスタントです。丁寧、正確、効率的に回答してください。",
        },
        "data_analyst": {
            "zh": "你是数据分析专家。基于数据给出准确分析和洞察。",
            "en": "You are a data analysis expert. Provide accurate analysis and insights based on data.",
            "ja": "あなたはデータ分析の専門家です。データに基づいて正確な分析と洞察を提供してください。",
        },
    }

    @classmethod
    def get_prompt(cls, prompt_key: str, language: str) -> str:
        """获取指定语言的提示。"""
        prompts = cls.PROMPTS.get(prompt_key, {})
        return prompts.get(language, prompts.get("en", ""))

    @classmethod
    async def translate_prompt(
        cls,
        llm: BaseChatModel,
        prompt: str,
        target_lang: str,
    ) -> str:
        """动态翻译提示。"""
        lang_names = {"zh": "中文", "en": "English", "ja": "日本語", "ko": "한국어"}
        target = lang_names.get(target_lang, "English")

        translate_prompt = f"将以下文本翻译为{target}，保持语义和语气不变:\n\n{prompt}"
        response = await llm.ainvoke([HumanMessage(content=translate_prompt)])
        return response.content
```
