# 评测数据集构建与管理

```python
from dataclasses import dataclass, field
from typing import Any
from enum import Enum

class Difficulty(str, Enum):
    EASY = "easy"
    MEDIUM = "medium"
    HARD = "hard"

@dataclass
class EvalDataPoint:
    """单条评测数据点。"""
    id: str
    input: str                          # 用户输入
    expected_output: str | None = None  # 期望输出（可选）
    context: list[str] | None = None    # 检索上下文（RAG场景）
    expected_trajectory: list[dict] | None = None  # 参考轨迹（Agent场景）
    difficulty: Difficulty = Difficulty.MEDIUM
    category: str = ""                  # 分类标签
    source: str = ""                    # 数据来源
    eval_criteria: dict = field(default_factory=dict)  # 评判标准
    tags: list[str] = field(default_factory=list)

@dataclass
class EvalDataset:
    """评测数据集。"""
    name: str
    version: str
    description: str
    data_points: list[EvalDataPoint]
    created_at: str
    metadata: dict = field(default_factory=dict)

    def filter_by(self, **kwargs) -> list[EvalDataPoint]:
        """按条件筛选数据点。"""
        result = self.data_points
        for key, value in kwargs.items():
            result = [dp for dp in result if getattr(dp, key, None) == value]
        return result

    def stats(self) -> dict:
        """数据集统计。"""
        return {
            "total": len(self.data_points),
            "by_difficulty": {
                d.value: sum(1 for dp in self.data_points if dp.difficulty == d)
                for d in Difficulty
            },
            "by_category": self._count_by("category"),
            "with_expected": sum(1 for dp in self.data_points if dp.expected_output),
            "with_trajectory": sum(1 for dp in self.data_points if dp.expected_trajectory),
        }

    def _count_by(self, field_name: str) -> dict:
        counts = {}
        for dp in self.data_points:
            val = getattr(dp, field_name, "")
            counts[val] = counts.get(val, 0) + 1
        return counts
```

### 3.1 LLM 自动生成测试用例

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
import json

GENERATE_PROMPT = """你是测试数据生成专家。请为以下系统生成{num}个测试用例。

系统描述: {system_description}

要求：
1. 覆盖不同难度（简单/中等/困难）
2. 包含正常流程和边界情况
3. 包含可能触发错误的输入
4. 每个用例都有期望输出

输出JSON数组:
```json
[
  {{
    "input": "用户输入",
    "expected_output": "期望输出",
    "difficulty": "easy/medium/hard",
    "category": "分类",
    "tags": ["标签1"]
  }}
]
```"""

async def generate_eval_cases(
    llm: BaseChatModel,
    system_description: str,
    num: int = 20,
) -> list[dict]:
    """用LLM自动生成测试用例。"""
    prompt = GENERATE_PROMPT.format(
        num=num,
        system_description=system_description,
    )
    response = await llm.ainvoke([HumanMessage(content=prompt)])

    import re
    json_match = re.search(r'\[.*\]', response.content, re.DOTALL)
    if json_match:
        return json.loads(json_match.group())
    return []

async def generate_diverse_cases(
    llm: BaseChatModel,
    system_description: str,
    categories: list[str],
    per_category: int = 5,
) -> list[EvalDataPoint]:
    """按分类生成多样化测试用例。"""
    all_points = []
    for category in categories:
        desc = f"{system_description}\n测试分类: {category}"
        cases = await generate_eval_cases(llm, desc, per_category)

        for i, case in enumerate(cases):
            all_points.append(EvalDataPoint(
                id=f"{category}_{i}",
                input=case.get("input", ""),
                expected_output=case.get("expected_output"),
                difficulty=Difficulty(case.get("difficulty", "medium")),
                category=category,
                source="llm_generated",
                tags=case.get("tags", []),
            ))

    return all_points
```

### 3.2 从线上日志提取

```python
class LogToEvalConverter:
    """从线上日志提取评测数据点。"""

    async def extract_from_logs(
        self,
        logs: list[dict],  # 线上交互日志
        quality_threshold: float = 0.8,
    ) -> list[EvalDataPoint]:
        """从用户交互日志中提取高质量测试用例。

        只提取用户满意度高的交互（如有点赞/ thumbs up）。
        """
        points = []
        for log in logs:
            # 只提取正反馈的交互
            if log.get("feedback", {}).get("rating", 0) < quality_threshold:
                continue

            points.append(EvalDataPoint(
                id=f"log_{log['session_id']}_{log['turn']}",
                input=log["user_input"],
                expected_output=log["assistant_response"],  # 正反馈的回答作为参考
                context=log.get("retrieved_context"),
                difficulty=Difficulty.MEDIUM,
                category=log.get("intent", "general"),
                source="user_log",
                tags=["user_feedback", "positive"],
            ))

        return points
```

```python
import json
from pathlib import Path
from datetime import datetime

class DatasetVersionManager:
    """数据集版本管理器。

    功能：
    - 保存/加载数据集
    - 版本对比
    - 防止数据泄露
    """

    def __init__(self, base_dir: str = "./eval_datasets"):
        self.base_dir = Path(base_dir)
        self.base_dir.mkdir(parents=True, exist_ok=True)

    def save(self, dataset: EvalDataset):
        """保存数据集到文件。"""
        dir_path = self.base_dir / dataset.name / dataset.version
        dir_path.mkdir(parents=True, exist_ok=True)

        # 数据文件
        data_file = dir_path / "data.jsonl"
        with open(data_file, "w", encoding="utf-8") as f:
            for dp in dataset.data_points:
                f.write(json.dumps({
                    "id": dp.id,
                    "input": dp.input,
                    "expected_output": dp.expected_output,
                    "context": dp.context,
                    "difficulty": dp.difficulty.value,
                    "category": dp.category,
                    "source": dp.source,
                    "tags": dp.tags,
                }, ensure_ascii=False) + "\n")

        # 元数据
        meta_file = dir_path / "meta.json"
        meta_file.write_text(json.dumps({
            "name": dataset.name,
            "version": dataset.version,
            "description": dataset.description,
            "created_at": datetime.now().isoformat(),
            "stats": dataset.stats(),
        }, ensure_ascii=False, indent=2))

        print(f"数据集已保存: {dir_path}")

    def load(self, name: str, version: str) -> EvalDataset:
        """加载数据集。"""
        dir_path = self.base_dir / name / version
        data_file = dir_path / "data.jsonl"

        data_points = []
        with open(data_file, "r", encoding="utf-8") as f:
            for line in f:
                data = json.loads(line.strip())
                data_points.append(EvalDataPoint(
                    id=data["id"],
                    input=data["input"],
                    expected_output=data.get("expected_output"),
                    context=data.get("context"),
                    difficulty=Difficulty(data.get("difficulty", "medium")),
                    category=data.get("category", ""),
                    source=data.get("source", ""),
                    tags=data.get("tags", []),
                ))

        return EvalDataset(
            name=name,
            version=version,
            description="",
            data_points=data_points,
            created_at="",
        )

    def list_versions(self, name: str) -> list[str]:
        """列出一个数据集的所有版本。"""
        dir_path = self.base_dir / name
        if not dir_path.exists():
            return []
        return sorted([d.name for d in dir_path.iterdir() if d.is_dir()])

    def diff_versions(
        self, name: str, v1: str, v2: str
    ) -> dict:
        """对比两个版本的数据集差异。"""
        d1 = self.load(name, v1)
        d2 = self.load(name, v2)

        ids1 = {dp.id for dp in d1.data_points}
        ids2 = {dp.id for dp in d2.data_points}

        added = ids2 - ids1
        removed = ids1 - ids2
        common = ids1 & ids2

        # 检查共同项是否有修改
        modified = []
        dp1_map = {dp.id: dp for dp in d1.data_points}
        dp2_map = {dp.id: dp for dp in d2.data_points}

        for dp_id in common:
            if dp1_map[dp_id].input != dp2_map[dp_id].input:
                modified.append(dp_id)
            elif dp1_map[dp_id].expected_output != dp2_map[dp_id].expected_output:
                modified.append(dp_id)

        return {
            "v1_count": len(ids1),
            "v2_count": len(ids2),
            "added": len(added),
            "removed": len(removed),
            "modified": len(modified),
            "added_ids": list(added)[:10],
            "removed_ids": list(removed)[:10],
        }
```

```python
class DataLeakageGuard:
    """数据泄露防护。"""

    @staticmethod
    def split_dataset(
        dataset: EvalDataset,
        train_ratio: float = 0.6,
        eval_ratio: float = 0.2,
        holdout_ratio: float = 0.2,
    ) -> dict:
        """将数据集分为训练集、评估集和留出集。

        留出集永不用于训练或Prompt工程，
        只在最终评估时使用。
        """
        import random
        data = dataset.data_points.copy()
        random.shuffle(data)

        n = len(data)
        train_end = int(n * train_ratio)
        eval_end = int(n * (train_ratio + eval_ratio))

        return {
            "train": data[:train_end],
            "eval": data[train_end:eval_end],
            "holdout": data[eval_end:],  # 永不公开
        }

    @staticmethod
    def check_prompt_for_leakage(
        prompt: str,
        eval_inputs: list[str],
    ) -> list[str]:
        """检查Prompt是否泄露了测试集内容。"""
        leaked = []
        for eval_input in eval_inputs:
            # 如果Prompt中直接包含测试输入
            if eval_input[:50] in prompt:
                leaked.append(eval_input[:50])
        return leaked
```

## 七、评估结果管理

```python
@dataclass
class EvalResult:
    """单次评估结果。"""
    dataset_name: str
    dataset_version: str
    system_version: str          # 被测系统版本
    timestamp: str
    results: list[dict]          # 每条数据点的评估结果
    aggregate_scores: dict      # 聚合分数

class EvalResultTracker:
    """评估结果跟踪器：记录历史，对比版本。"""

    def __init__(self):
        self.history: list[EvalResult] = []

    def add_result(self, result: EvalResult):
        self.history.append(result)

    def compare_versions(
        self,
        system_v1: str,
        system_v2: str,
        dataset_name: str,
    ) -> dict:
        """对比两个系统版本的评估结果。"""
        r1 = self._find_result(system_v1, dataset_name)
        r2 = self._find_result(system_v2, dataset_name)

        if not r1 or not r2:
            return {"error": "结果不存在"}

        deltas = {}
        for key in r1.aggregate_scores:
            old = r1.aggregate_scores[key]
            new = r2.aggregate_scores.get(key, 0)
            deltas[key] = {
                "old": old,
                "new": new,
                "delta": round(new - old, 4),
                "improved": new > old,
            }

        return {
            "v1": system_v1,
            "v2": system_v2,
            "deltas": deltas,
            "overall_improved": sum(1 for d in deltas.values() if d["improved"]),
            "overall_regressed": sum(1 for d in deltas.values() if not d["improved"] and d["delta"] != 0),
        }

    def _find_result(self, system_version: str, dataset_name: str) -> EvalResult | None:
        for r in reversed(self.history):
            if r.system_version == system_version and r.dataset_name == dataset_name:
                return r
        return None
```

## 八、最佳实践
