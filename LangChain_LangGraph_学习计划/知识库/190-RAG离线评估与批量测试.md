# RAG 离线评估与批量测试

## 三、实现

```python
from dataclasses import dataclass, field
from typing import Optional
import asyncio
import time
import json

@dataclass
class EvalTestCase:
    """评估测试用例。"""
    query: str
    expected_answer: Optional[str] = None
    expected_sources: Optional[list[str]] = None
    tags: list[str] = field(default_factory=list)

@dataclass
class EvalTestResult:
    """单条评估结果。"""
    query: str
    actual_answer: str
    latency_ms: float
    tokens_used: int
    faithfulness: float = 0      # 忠实度
    relevance: float = 0        # 相关性
    source_recall: float = 0     # 来源召回
    passed: bool = False

class OfflineEvaluator:
    """离线评估器。"""

    def __init__(self, rag_pipeline, test_cases: list[EvalTestCase]):
        self.pipeline = rag_pipeline
        self.test_cases = test_cases
        self.results: list[EvalTestResult] = []

    async def run(self, concurrency: int = 5) -> dict:
        """批量评估。"""
        semaphore = asyncio.Semaphore(concurrency)

        async def evaluate_one(tc: EvalTestCase) -> EvalTestResult:
            async with semaphore:
                start = time.time()
                result = await self.pipeline.answer(tc.query)
                latency = (time.time() - start) * 1000

                # 简化评估
                answer = result.get("answer", "")
                faithfulness = self._score_faithfulness(answer, tc.expected_answer)
                relevance = self._score_relevance(answer, tc.query)
                source_recall = self._score_source_recall(result, tc.expected_sources)

                passed = faithfulness >= 0.7 and relevance >= 0.7

                return EvalTestResult(
                    query=tc.query,
                    actual_answer=answer[:200],
                    latency_ms=round(latency, 2),
                    tokens_used=result.get("tokens", 0),
                    faithfulness=faithfulness,
                    relevance=relevance,
                    source_recall=source_recall,
                    passed=passed,
                )

        tasks = [evaluate_one(tc) for tc in self.test_cases]
        self.results = await asyncio.gather(*tasks)
        return self.report()

    def _score_faithfulness(self, answer: str, expected: Optional[str]) -> float:
        """忠实度评分（简化版）。"""
        if not expected:
            return 0.8  # 无参考答案给默认分
        # 关键词重叠率
        answer_words = set(answer.lower().split())
        expected_words = set(expected.lower().split())
        if not expected_words:
            return 0.8
        overlap = len(answer_words & expected_words) / len(expected_words)
        return min(1.0, overlap * 1.5)

    def _score_relevance(self, answer: str, query: str) -> float:
        """相关性评分。"""
        if not answer:
            return 0
        # 查询关键词在答案中的覆盖
        query_words = set(query.lower().split())
        answer_lower = answer.lower()
        covered = sum(1 for w in query_words if w in answer_lower)
        return covered / max(len(query_words), 1)

    def _score_source_recall(self, result: dict, expected_sources: Optional[list[str]]) -> float:
        """来源召回率。"""
        if not expected_sources:
            return 1.0
        actual_sources = set(result.get("sources", []))
        expected = set(expected_sources)
        return len(actual_sources & expected) / len(expected)

    def report(self) -> dict:
        """生成评估报告。"""
        if not self.results:
            return {"error": "无结果"}

        total = len(self.results)
        passed = sum(1 for r in self.results if r.passed)

        return {
            "total": total,
            "passed": passed,
            "pass_rate": round(passed / total, 4),
            "avg_faithfulness": round(sum(r.faithfulness for r in self.results) / total, 4),
            "avg_relevance": round(sum(r.relevance for r in self.results) / total, 4),
            "avg_source_recall": round(sum(r.source_recall for r in self.results) / total, 4),
            "avg_latency_ms": round(sum(r.latency_ms for r in self.results) / total, 2),
            "total_tokens": sum(r.tokens_used for r in self.results),
            "p95_latency": sorted(r.latency_ms for r in self.results)[int(total * 0.95)],
        }
```

## 四、基线对比

```python
class BaselineComparator:
    """基线对比器。"""

    def __init__(self):
        self.baseline: dict | None = None

    def save_baseline(self, report: dict):
        """保存基线。"""
        self.baseline = report.copy()

    def compare(self, current: dict) -> dict:
        """对比当前结果与基线。"""
        if not self.baseline:
            return {"message": "无基线", "current": current}

        deltas = {}
        for key in ["pass_rate", "avg_faithfulness", "avg_relevance"]:
            old = self.baseline.get(key, 0)
            new = current.get(key, 0)
            delta = round(new - old, 4)
            deltas[key] = {
                "baseline": old,
                "current": new,
                "delta": delta,
                "improved": delta > 0,
            }

        latency_delta = current.get("avg_latency_ms", 0) - self.baseline.get("avg_latency_ms", 0)
        deltas["avg_latency_ms"] = {
            "baseline": self.baseline.get("avg_latency_ms"),
            "current": current.get("avg_latency_ms"),
            "delta": round(latency_delta, 2),
            "improved": latency_delta < 0,  # 延迟降低是改进
        }

        regressed = sum(1 for d in deltas.values() if not d["improved"] and d["delta"] != 0)
        return {
            "deltas": deltas,
            "regressed_count": regressed,
            "should_block_ci": regressed > 0,
        }
```

## 五、CI/CD 集成

```python
class CIEvaluator:
    """CI/CD集成的评估器。"""

    @staticmethod
    async def run_ci_eval(
        rag_pipeline,
        test_cases: list[EvalTestCase],
        baseline_report: dict | None = None,
        max_regression: int = 0,
    ) -> dict:
        """CI中运行的评估。"""
        evaluator = OfflineEvaluator(rag_pipeline, test_cases)
        report = await evaluator.run()

        result = {"report": report}

        if baseline_report:
            comparator = BaselineComparator()
            comparator.save_baseline(baseline_report)
            comparison = comparator.compare(report)
            result["comparison"] = comparison
            result["ci_passed"] = comparison["regressed_count"] <= max_regression
        else:
            result["ci_passed"] = report["pass_rate"] >= 0.7

        return result
```
