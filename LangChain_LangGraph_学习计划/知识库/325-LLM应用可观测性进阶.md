## 二、智能告警

```python
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from collections import defaultdict
from enum import Enum

class AlertSeverity(str, Enum):
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"

@dataclass
class SmartAlert:
    """智能告警。"""
    alert_id: str
    severity: AlertSeverity
    title: str
    description: str
    root_cause_hint: str = ""
    related_spans: list[str] = field(default_factory=list)
    auto_diagnosed: bool = False
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())

class SmartAlertEngine:
    """智能告警引擎——异常模式检测+根因提示。"""

    def __init__(self):
        self.metrics_window: list[dict] = []
        self.alerts: list[SmartAlert] = []

    def record(self, metrics: dict):
        """记录指标。"""
        self.metrics_window.append({**metrics, "timestamp": datetime.now().isoformat()})
        # 只保留最近100条
        if len(self.metrics_window) > 100:
            self.metrics_window.pop(0)

    def detect_anomalies(self) -> list[SmartAlert]:
        """检测异常模式。"""
        alerts = []

        # 1. 延迟突增
        recent = self.metrics_window[-10:]
        if len(recent) >= 5:
            avg_latency = sum(m.get("latency_ms", 0) for m in recent) / len(recent)
            if avg_latency > 5000:
                alerts.append(SmartAlert(
                    alert_id="latency_spike",
                    severity=AlertSeverity.HIGH,
                    title="延迟突增",
                    description=f"最近10次平均延迟{avg_latency:.0f}ms",
                    root_cause_hint="检查LLM API响应时间或向量库查询",
                ))

        # 2. 错误率突增
        errors = sum(1 for m in recent if m.get("error", False))
        if len(recent) >= 5 and errors / len(recent) > 0.3:
            alerts.append(SmartAlert(
                alert_id="error_spike",
                severity=AlertSeverity.CRITICAL,
                title="错误率突增",
                description=f"错误率{errors}/{len(recent)}",
                root_cause_hint="检查最近部署变更或API可用性",
            ))

        self.alerts.extend(alerts)
        return alerts

    async def auto_diagnose(self, alert: SmartAlert, llm) -> SmartAlert:
        """LLM辅助根因分析。"""
        from langchain_core.messages import HumanMessage
        prompt = f"""你是LLM应用运维专家。分析以下告警的根因。

告警: {alert.title}
描述: {alert.description}
提示: {alert.root_cause_hint}

给出:
1. 最可能的根因
2. 验证方法
3. 修复建议"""

        response = await llm.ainvoke([HumanMessage(content=prompt)])
        alert.description += f"\n[自动诊断] {response.content[:300]}"
        alert.auto_diagnosed = True
        return alert
```
