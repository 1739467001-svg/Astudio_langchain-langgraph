## 一、基础调用

### 1.1 最简单的 LLM 调用

```python
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()
llm = ChatOpenAI(model="gpt-4o-mini")
response = llm.invoke("用一句话解释什么是量子计算")
print(response.content)
```

### 1.2 带参数的调用

```python
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.9, max_tokens=100)
response = llm.invoke("写一首关于春天的诗")
print(response.content)
print(f"Token: {response.usage_metadata}")
```

### 1.3 流式输出

```python
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o-mini", streaming=True)
for chunk in llm.stream("讲一个短故事"):
    print(chunk.content, end="", flush=True)
print()
```

## 三、输出解析

### 3.1 JSON 输出

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import JsonOutputParser
from pydantic import BaseModel, Field

class Book(BaseModel):
    title: str = Field(description="书名")
    author: str = Field(description="作者")
    year: int = Field(description="出版年份")
    rating: float = Field(description="评分1-10")

parser = JsonOutputParser(pydantic_object=Book)

prompt = ChatPromptTemplate.from_template(
    "推荐一本关于{topic}的书。\n{format_instructions}"
)

llm = ChatOpenAI(model="gpt-4o-mini")
chain = prompt | llm | parser

result = chain.invoke({
    "topic": "人工智能",
    "format_instructions": parser.get_format_instructions()
})
print(result)  # dict
```

## 四、记忆与对话

### 4.1 简单多轮对话

```python
from langchain_core.messages import HumanMessage, AIMessage

llm = ChatOpenAI(model="gpt-4o-mini")
history = []

def chat(user_input):
    history.append(HumanMessage(content=user_input))
    response = llm.invoke(history)
    history.append(response)
    return response.content

print(chat("我叫张三"))
print(chat("我叫什么？"))
```

## 五、Chain 组合

### 5.1 并行生成

```python
from langchain_core.runnables import RunnableParallel, RunnableLambda

parallel = RunnableParallel(

```

### 5.2 多步骤 Chain

```python
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnableLambda

llm = ChatOpenAI(model="gpt-4o-mini")

# 步骤1：生成关键词
kw_prompt = ChatPromptTemplate.from_template("为'{topic}'生成3个关键词，逗号分隔")
kw_chain = kw_prompt | llm | StrOutputParser()

# 步骤2：根据关键词写摘要
sum_prompt = ChatPromptTemplate.from_template("根据关键词'{keywords}'写一段摘要")
sum_chain = sum_prompt | llm | StrOutputParser()

# 串联
final = kw_chain | RunnableLambda(lambda kws: {"keywords": kws}) | sum_chain
print(final.invoke({"topic": "量子计算"}))
```

## 六、工具与 Agent

### 6.1 自定义工具

```python
from langchain_core.tools import tool

@tool
def add_numbers(a: float, b: float) -> float:
    """将两个数字相加。当用户需要做加法时使用。"""
    return a + b

@tool
def get_weather(city: str) -> str:
    """查询城市天气。当用户问天气时使用。
    
    Args:
        city: 城市名称，如"北京"
    """
    # 模拟
    weather_db = {"北京": "晴 25度", "上海": "雨 20度"}
    return weather_db.get(city, "未知城市")
```

### 6.2 Agent 完整示例

```python
from langchain.agents import create_tool_calling_agent, AgentExecutor
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder

tools = [add_numbers, get_weather]
llm = ChatOpenAI(model="gpt-4o-mini")

prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一个有用的助手，可以使用工具。"),
    ("human", "{input}"),
    MessagesPlaceholder(variable_name="agent_scratchpad"),
])

agent = create_tool_calling_agent(llm, tools, prompt)
executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

print(executor.invoke({"input": "北京和上海天气怎么样？2+3等于几？"})["output"])
```

## 七、RAG 完整示例

### 7.1 从文件构建知识库

```python
from langchain_community.document_loaders import TextLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import FAISS

# 1. 加载
docs = TextLoader("knowledge.txt", encoding="utf-8").load()

# 2. 分割
splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
chunks = splitter.split_documents(docs)

# 3. 向量化 + 存储
embeddings = OpenAIEmbeddings()
db = FAISS.from_documents(chunks, embeddings)
db.save_local("my_kb")

# 4. 检索
results = db.similarity_search("查询内容", k=3)
for doc in results:
    print(doc.page_content[:100])
    print(doc.metadata)
    print("---")
```

### 7.2 RAG 问答链

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

retriever = db.as_retriever(search_kwargs={"k": 3})

prompt = ChatPromptTemplate.from_template("""
基于以下知识回答问题。知识中没有则说"我不知道"。

知识：{context}
问题：{question}
""")

def format_docs(docs):
    return "\n\n".join(d.page_content for d in docs)

rag = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | prompt | llm | StrOutputParser()
)

print(rag.invoke("你的问题"))
```

### 7.3 带来源引用的 RAG

```python
from langchain_core.runnables import RunnableLambda

def format_with_sources(docs):
    text = "\n\n".join(d.page_content for d in docs)
    sources = [d.metadata.get("source", "unknown") for d in docs]
    return {"context": text, "sources": sources}

prompt = ChatPromptTemplate.from_template("""
基于知识回答。知识：{context}
问题：{question}
参考来源：{sources}
""")

rag_with_sources = (
    {
        "context": retriever | RunnableLambda(lambda d: format_with_sources(d)["context"]),
        "sources": retriever | RunnableLambda(lambda d: format_with_sources(d)["sources"]),
        "question": RunnablePassthrough(),
    }
    | prompt | llm | StrOutputParser()
)
```

## 八、LangGraph 示例

### 8.1 线性工作流

```python
from typing import TypedDict
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    text: str
    processed: str

def upper_node(state: State) -> dict:
    return {"processed": state["text"].upper()}

def add_exclaim_node(state: State) -> dict:
    return {"processed": state["processed"] + "!"}

g = StateGraph(State)
g.add_node("upper", upper_node)
g.add_node("exclaim", add_exclaim_node)
g.add_edge(START, "upper")
g.add_edge("upper", "exclaim")
g.add_edge("exclaim", END)

app = g.compile()
print(app.invoke({"text": "hello"}))  # {"text": "hello", "processed": "HELLO!"}
```

### 8.2 条件分支

```python
from typing import TypedDict
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    number: int
    result: str

def check_node(state: State) -> dict:
    return {}  # 只做路由判断，不修改 State

def even_node(state: State) -> dict:
    return {"result": f"{state['number']} 是偶数"}

def odd_node(state: State) -> dict:
    return {"result": f"{state['number']} 是奇数"}

def route(state: State) -> str:
    return "even" if state["number"] % 2 == 0 else "odd"

g = StateGraph(State)
g.add_node("check", check_node)
g.add_node("even", even_node)
g.add_node("odd", odd_node)
g.add_edge(START, "check")
g.add_conditional_edges("check", route, {"even": "even", "odd": "odd"})
g.add_edge("even", END)
g.add_edge("odd", END)

app = g.compile()
print(app.invoke({"number": 4, "result": ""}))   # 偶数
print(app.invoke({"number": 7, "result": ""}))   # 奇数
```

### 8.3 循环（重试机制）

```python
from typing import TypedDict
from langgraph.graph import StateGraph, START, END

class State(TypedDict):
    question: str
    answer: str
    attempts: int

def generate_node(state: State) -> dict:
    """模拟生成回答"""
    attempts = state.get("attempts", 0) + 1
    # 模拟：第3次才成功
    if attempts >= 3:
        answer = "合格回答"
    else:
        answer = "不合格"
    return {"answer": answer, "attempts": attempts}

def check_node(state: State) -> dict:
    return {}

def route(state: State) -> str:
    if "合格" in state["answer"]:
        return "done"
    elif state["attempts"] >= 5:
        return "done"
    return "retry"

g = StateGraph(State)
g.add_node("generate", generate_node)
g.add_node("check", check_node)
g.add_edge(START, "generate")
g.add_edge("generate", "check")
g.add_conditional_edges("check", route, {"retry": "generate", "done": END})

app = g.compile()
result = app.invoke({"question": "test", "answer": "", "attempts": 0})
print(f"结果: {result['answer']}, 尝试: {result['attempts']}次")
```

### 8.4 Human-in-the-Loop

```python
from typing import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

class State(TypedDict):
    draft: str
    approved: bool

def generate_draft(state: State) -> dict:
    return {"draft": "这是AI生成的草稿内容"}

def review_node(state: State) -> dict:
    # 这个节点会在运行时被跳过（因为 interrupt_before）
    return {"approved": True}

g = StateGraph(State)
g.add_node("generate", generate_draft)
g.add_node("review", review_node)
g.add_edge(START, "generate")
g.add_edge("generate", "review")
g.add_edge("review", END)

app = g.compile(
    checkpointer=MemorySaver(),
    interrupt_before=["review"]
)

config = {"configurable": {"thread_id": "1"}}

# 第一次调用：停在 review 之前
result = app.invoke({"draft": "", "approved": False}, config=config)
print(f"草稿: {result['draft']}")

# 人工修改草稿
app.update_state(config, {"draft": "人工修改后的草稿"})

# 继续执行
result = app.invoke(None, config=config)
print(f"最终: {result['draft']}, 已批准: {result['approved']}")
```

## 九、实用技巧

### 9.1 批量处理

```python
llm = ChatOpenAI(model="gpt-4o-mini")
prompt = ChatPromptTemplate.from_template("翻译为英文：{text}")
chain = prompt | llm | StrOutputParser()

texts = [{"text": "你好"}, {"text": "世界"}, {"text": "再见"}]
results = chain.batch(texts)
for r in results:
    print(r)
```

### 9.2 异步并发

```python
import asyncio

async def main():
    tasks = [chain.ainvoke({"text": t}) for t in ["你好", "世界", "再见"]]
    results = await asyncio.gather(*tasks)
    print(results)

asyncio.run(main())
```

### 9.3 缓存

```python
from langchain_core.globals import set_llm_cache
from langchain_community.cache import InMemoryCache

set_llm_cache(InMemoryCache())

# 第一次调用（实际请求）

assert r1.content == r2.content
```
