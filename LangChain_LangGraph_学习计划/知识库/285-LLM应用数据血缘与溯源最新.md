## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Any

@dataclass
class DataLineageNode:
    """血缘节点。"""
    node_id: str
    node_type: str        # source/process/storage/output
    name: str
    inputs: list[str] = field(default_factory=list)   # 上游节点ID
    outputs: list[str] = field(default_factory=list)   # 下游节点ID
    metadata: dict = field(default_factory=dict)
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())

class DataLineageTracker:
    """数据血缘追踪器。"""

    def __init__(self):
        self.nodes: dict[str, DataLineageNode] = {}

    def add_node(self, node: DataLineageNode):
        """添加节点。"""
        self.nodes[node.node_id] = node
        # 更新上下游关系
        for input_id in node.inputs:
            if input_id in self.nodes:
                self.nodes[input_id].outputs.append(node.node_id)

    def get_lineage(self, node_id: str) -> dict:
        """获取节点血缘。"""
        node = self.nodes.get(node_id)
        if not node:
            return {"error": "节点不存在"}

        upstream = [self.nodes[i].name for i in node.inputs if i in self.nodes]
        downstream = [self.nodes[o].name for o in node.outputs if o in self.nodes]

        return {
            "node": node.name,
            "type": node.node_type,
            "upstream": upstream,
            "downstream": downstream,
        }

    def impact_analysis(self, source_id: str) -> list[str]:
        """影响分析——如果某数据源变更，影响哪些下游。"""
        affected = []
        to_visit = [source_id]
        visited = set()

        while to_visit:
            current = to_visit.pop(0)
            if current in visited:
                continue
            visited.add(current)
            node = self.nodes.get(current)
            if node:
                affected.append(node.name)
                to_visit.extend(node.outputs)

        return affected[1:]  # 排除自身

    def trace_answer_to_source(self, answer_id: str) -> list[str]:
        """从答案追溯到数据源。"""
        chain = []
        current = answer_id
        while current:
            node = self.nodes.get(current)
            if not node:
                break
            chain.append(node.name)
            if node.node_type == "source":
                break
            current = node.inputs[0] if node.inputs else None
        return chain
```
