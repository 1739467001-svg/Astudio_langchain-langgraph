## 二、隔离方案

```python
from dataclasses import dataclass, field
from enum import Enum

class IsolationLevel(str, Enum):
    SHARED = "shared"          # 共享（按ID隔离）
    COLLECTION = "collection"   # 按集合隔离
    DATABASE = "database"      # 按数据库隔离

@dataclass
class TenantConfig:
    """租户配置。"""
    tenant_id: str
    name: str
    plan: str = "basic"        # basic/premium/enterprise
    isolation: IsolationLevel = IsolationLevel.SHARED
    qps_limit: int = 10
    token_budget_daily: int = 100000
    vector_collection: str = ""  # 向量库集合名
    db_schema: str = ""          # 数据库schema
    cache_prefix: str = ""       # 缓存前缀

class TenantManager:
    """租户管理器。"""

    def __init__(self):
        self.tenants: dict[str, TenantConfig] = {}

    def register(self, config: TenantConfig):
        """注册租户。"""
        config.vector_collection = f"tenant_{config.tenant_id}"
        config.db_schema = f"tenant_{config.tenant_id}"
        config.cache_prefix = f"t:{config.tenant_id}:"
        self.tenants[config.tenant_id] = config

    def get(self, tenant_id: str) -> TenantConfig | None:
        return self.tenants.get(tenant_id)

    def check_quota(self, tenant_id: str, tokens_used: int) -> dict:
        """检查配额。"""
        tenant = self.tenants.get(tenant_id)
        if not tenant:
            return {"allowed": False, "reason": "租户不存在"}

        if tokens_used >= tenant.token_budget_daily:
            return {"allowed": False, "reason": "超出日Token预算"}

        return {"allowed": True, "remaining": tenant.token_budget_daily - tokens_used}


class TenantIsolation:
    """租户数据隔离器。"""

    @staticmethod
    def build_vector_filter(tenant: TenantConfig) -> dict:
        """构建向量库的租户过滤器。"""
        if tenant.isolation == IsolationLevel.COLLECTION:
            return {"collection": tenant.vector_collection}
        return {"tenant_id": tenant.tenant_id}  # 共享模式下按字段过滤

    @staticmethod
    def build_cache_key(tenant: TenantConfig, key: str) -> str:
        """构建租户缓存键。"""
        return f"{tenant.cache_prefix}{key}"

    @staticmethod
    def build_thread_id(tenant: TenantConfig, user_id: str) -> str:
        """构建租户内线程ID。"""
        return f"{tenant.tenant_id}:{user_id}"
```

## 三、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 数据按租户隔离 | 防止数据泄露 | ★★★ |
| 按租户设QPS限制 | 防资源耗尽 | ★★★ |
| 按租户设Token预算 | 成本控制 | ★★★ |
| thread_id带租户前缀 | 对话隔离 | ★★☆ |
| 缓存键带租户前缀 | 防缓存混 | ★★☆ |
