## 三、种子数据准备

```python
# 预置种子数据——让空知识库立即有用
SEED_DOCUMENTS = [
    {
        "content": """产品常见问题(FAQ)
Q: 如何重置密码？
A: 点击登录页的"忘记密码"，输入注册邮箱，按邮件指引重置。

Q: 如何联系客服？
A: 拨打400-XXX-XXXX或发送邮件至support@example.com。

Q: 退换货政策？
A: 7天无理由退换，15天换货，商品需保持完好。""",
        "metadata": {"source": "faq.md", "category": "FAQ", "type": "seed"}
    },
    {
        "content": """产品使用手册
第一章: 快速开始
1. 下载并安装客户端
2. 使用手机号注册账号
3. 完成实名认证
4. 开始使用核心功能

第二章: 常见操作
1. 搜索: 在首页搜索框输入关键词
2. 收藏: 点击商品右上角星标
3. 下单: 选择商品→加入购物车→结算""",
        "metadata": {"source": "manual.md", "category": "手册", "type": "seed"}
    },
]

def init_seed_data(vectorstore):
    """初始化种子数据"""
    from langchain_core.documents import Document
    docs = [
        Document(page_content=d["content"], metadata=d["metadata"])
        for d in SEED_DOCUMENTS
    ]
    vectorstore.add_documents(docs)
    print(f"✅ 已预置 {len(docs)} 个种子文档")
    return vectorstore
```

## 四、搜索兜底机制

```python
def cold_start_rag(question: str, vectorstore, llm, web_search_tool=None):
    """冷启动RAG：知识库无结果时搜索互联网"""
    # Step 1: 先查知识库
    results = vectorstore.similarity_search(question, k=3)

    if results:
        # 知识库有结果→正常RAG
        context = "\n".join(d.page_content for d in results)
        prompt = f"基于以下知识回答：{context}\n问题：{question}"
        return llm.invoke(prompt).content

    # Step 2: 知识库无结果→搜索互联网
    if web_search_tool:
        search_result = web_search_tool.invoke(question)
        prompt = f"基于搜索结果回答：{search_result}\n问题：{question}"
        return llm.invoke(prompt).content

    # Step 3: 都没有→引导用户
    return """抱歉，知识库中暂无相关信息。

建议：
1. 上传相关文档到知识库
2. 或尝试换一种问法
3. 或访问我们的帮助中心：https://help.example.com"""
```

## 五、引导上传

```python
def suggest_upload_topics(question: str, llm) -> str:
    """根据用户问题建议上传什么文档"""
    prompt = f"""用户问了以下问题但知识库中没有答案。
请建议用户上传什么类型的文档能帮助回答这类问题。

用户问题：{question}

建议上传的文档类型（简短说明）："""
    return llm.invoke(prompt).content
```
