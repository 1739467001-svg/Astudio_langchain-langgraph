```python
def tune_hnsw_params(
```

```python
from datetime import datetime

class IndexRebuildManager:
    """索引重建管理器：蓝绿策略零停机重建。"""

    def __init__(self, vectorstore_client):
        self.client = vectorstore_client

    async def rebuild_with_new_model(
        self,
        collection_name: str,
        new_embeddings,
        documents: list,
        new_index_params: dict,
    ):
        """用新嵌入模型重建索引。

        流程：
        1. 创建新集合（Green）
        2. 用新模型嵌入所有文档
        3. 验证新索引
        4. 切换流量
        5. 删除旧集合（Blue）
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        green_name = f"{collection_name}_v{timestamp}"

        # 1. 创建新集合
        print(f"创建新集合: {green_name}")
        await self.client.create_collection(
            collection_name=green_name,
            schema=new_index_params,
        )

        # 2. 重新嵌入并插入
        print(f"重新嵌入 {len(documents)} 个文档...")
        embeddings = await new_embeddings.aembed_documents(
            [doc.page_content for doc in documents]
        )

        await self.client.insert(
            collection_name=green_name,
            data=[
                {"id": i, "vector": emb, "text": doc.page_content, "metadata": doc.metadata}
                for i, (doc, emb) in enumerate(zip(documents, embeddings))
            ],
        )

        # 3. 验证
        print("验证新索引...")
        test_queries = ["测试查询1", "测试查询2"]
        for q in test_queries:
            results = await self.client.search(green_name, q, k=3)
            if not results:
                raise Exception(f"新索引验证失败: {q}")

        # 4. 切换流量（更新路由配置）
        print(f"切换流量: {collection_name} → {green_name}")
        await self.client.update_alias(
            alias=collection_name,
            collection=green_name,
        )

        # 5. 删除旧集合
        old_collections = await self.client.list_collections(
            prefix=f"{collection_name}_v"
        )
        for old in old_collections:
            if old != green_name:
                print(f"删除旧集合: {old}")
                await self.client.drop_collection(old)

        print(f"重建完成: {green_name}")
```

### 3.2 增量重建

```python
class IncrementalRebuild:
    """增量重建：不重新嵌入全部数据，只处理变更部分。"""

    async def rebuild_incremental(
        self,
        collection_name: str,
        new_documents: list,
        updated_documents: list,
        deleted_ids: list,
    ):
        """增量更新索引。

        适用于：嵌入模型不变，只是数据更新。
        """
        # 1. 新增文档
        if new_documents:
            embeddings = await self.embeddings.aembed_documents(
                [d.page_content for d in new_documents]
            )
            await self.client.insert(collection_name, [
                {"vector": emb, "text": d.page_content, "metadata": d.metadata}
                for d, emb in zip(new_documents, embeddings)
            ])

        # 2. 更新文档（先删后插）
        if updated_documents:
            old_ids = [d.metadata["doc_id"] for d in updated_documents]
            await self.client.delete(collection_name, old_ids)

            embeddings = await self.embeddings.aembed_documents(
                [d.page_content for d in updated_documents]
            )
            await self.client.insert(collection_name, [
                {"vector": emb, "text": d.page_content, "metadata": d.metadata}
                for d, emb in zip(updated_documents, embeddings)
            ])

        # 3. 删除文档
        if deleted_ids:
            await self.client.delete(collection_name, deleted_ids)
```

```python
class VectorStoreMigration:
    """向量库迁移管理器。"""

    def __init__(self, source_client, target_client):
        self.source = source_client
        self.target = target_client

    async def migrate(self, source_collection: str, target_collection: str):
        """执行迁移流程。"""

        # 1. 创建目标集合
        await self.target.create_collection(target_collection)

        # 2. 批量迁移历史数据
        batch_size = 1000
        total = await self.source.count(source_collection)
        migrated = 0

        for offset in range(0, total, batch_size):
            batch = await self.source.query(
                source_collection,
                offset=offset,
                limit=batch_size,
            )

            # 如果需要重新嵌入（模型变了）
            # new_embeddings = await self.new_embeddings.aembed_documents(
            #     [d["text"] for d in batch]
            # )

            await self.target.insert(target_collection, batch)
            migrated += len(batch)
            print(f"迁移进度: {migrated}/{total}")

        # 3. 验证
        test_queries = ["验证查询1", "验证查询2", "验证查询3"]
        for q in test_queries:
            source_results = await self.source.search(source_collection, q, k=5)
            target_results = await self.target.search(target_collection, q, k=5)
            # 对比结果（允许向量精度差异）
            source_ids = {r["id"] for r in source_results}
            target_ids = {r["id"] for r in target_results}
            overlap = len(source_ids & target_ids) / len(source_ids)
            print(f"查询'{q}': 结果重叠率={overlap:.2%}")

        print("迁移完成，验证通过")
```

```python
class VectorStoreMonitor:
    """向量库性能监控。"""

    def __init__(self, client):
        self.client = client
        self.metrics = {
            "query_latencies": [],
            "insert_latencies": [],
            "errors": [],
        }

    async def benchmark_query(
        self,
        collection: str,
        query_vector: list,
        k: int = 10,
    ) -> dict:
        """基准测试单次查询。"""
        import time
        start = time.time()
        try:
            results = await self.client.search(
                collection, query_vector, k=k
            )
            latency = (time.time() - start) * 1000
            self.metrics["query_latencies"].append(latency)
            return {"latency_ms": round(latency, 2), "results": len(results)}
        except Exception as e:
            self.metrics["errors"].append(str(e))
            return {"error": str(e)}

    def latency_stats(self) -> dict:
        """延迟统计。"""
        latencies = sorted(self.metrics["query_latencies"])
        if not latencies:
            return {}
        n = len(latencies)
        return {
            "count": n,
            "p50": round(latencies[n // 2], 2),
            "p95": round(latencies[int(n * 0.95)], 2),
            "p99": round(latencies[int(n * 0.99)], 2),
            "avg": round(sum(latencies) / n, 2),
        }
```
