## 二、三种处理器

```python
from langchain_core.callbacks import BaseCallbackHandler

class QuickLoggingCallback(BaseCallbackHandler):
    """快速日志回调——记录关键事件。"""
    def __init__(self):
        self.events = []
    def on_llm_start(self, serialized, prompts, **kwargs):
        self.events.append({"type": "llm_start"})
    def on_tool_start(self, serialized, input_str, **kwargs):
        self.events.append({"type": "tool_start", "name": serialized.get("name", "")})
    def on_tool_end(self, output, **kwargs):
        self.events.append({"type": "tool_end", "output": str(output)[:50]})
    def on_error(self, error, **kwargs):
        self.events.append({"type": "error", "error": str(error)[:100]})

class QuickMetricsCallback(BaseCallbackHandler):
    """快速指标回调——采集性能。"""
    def __init__(self):
        self.llm_calls = 0
        self.tool_calls = 0
        self.errors = 0
    def on_llm_start(self, serialized, prompts, **kwargs):
        self.llm_calls += 1
    def on_tool_start(self, serialized, input_str, **kwargs):
        self.tool_calls += 1
    def on_error(self, error, **kwargs):
        self.errors += 1
    def summary(self):
        return {"llm_calls": self.llm_calls, "tool_calls": self.tool_calls, "errors": self.errors}

class QuickTracingCallback(BaseCallbackHandler):
    """快速追踪回调——执行轨迹。"""
    def __init__(self):
        self.trace = []
        self._depth = 0
    def on_chain_start(self, serialized, inputs, **kwargs):
        self.trace.append({"event": "start", "name": serialized.get("name", ""), "depth": self._depth})
        self._depth += 1
    def on_chain_end(self, outputs, **kwargs):
        self._depth = max(0, self._depth - 1)
        self.trace.append({"event": "end", "depth": self._depth})
    def get_trace_text(self):
        return "\n".join(f"{'  '*t['depth']}{t['event']}: {t.get('name','')}" for t in self.trace)
```
