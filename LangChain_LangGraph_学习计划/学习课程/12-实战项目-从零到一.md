## 一、项目概述

### 1.1 项目名称

**智能知识库助手（Smart Knowledge Base Assistant）**

### 1.2 功能描述

用户上传文档（txt/md），系统自动构建知识库，用户可以与知识库进行多轮对话。系统会：

1. 加载和分割文档（RAG）
2. 向量化并存储（向量数据库）
3. 基于知识库回答问题（检索 + 生成）
4. 支持多轮对话（Memory）
5. 当知识库中找不到答案时，自动调用搜索工具（Agent + Tool）
6. 通过 LangGraph 编排整个流程（图式工作流）

### 1.3 技术栈

```
LangChain     → LLM 调用、Prompt、RAG 组件
LangGraph     → 工作流编排、多步决策、循环
FAISS         → 向量数据库
ChatOpenAI    → LLM 模型
```

```python
# ============ state.py ============
from typing import TypedDict, Annotated
from operator import add
from langchain_core.messages import AnyMessage

class KBState(TypedDict):
    messages: Annotated[list[AnyMessage], add]  # 对话历史
    question: str                                 # 当前问题
    context: str                                    # 检索到的上下文
    answer: str                                     # 生成的回答
    source: str                                     # 回答来源（知识库/搜索）
    retry_count: int                                # 重试次数


# ============ knowledge_base.py ============
from langchain_community.document_loaders import TextLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import FAISS

class KnowledgeBase:
    def __init__(self, docs_dir: str = "docs"):
        self.docs_dir = docs_dir
        self.embeddings = OpenAIEmbeddings()
        self.vectorstore = None
    
    def build(self):
        """加载文档并构建向量数据库"""
        import os
        documents = []
        for filename in os.listdir(self.docs_dir):
            if filename.endswith((".txt", ".md")):
                loader = TextLoader(os.path.join(self.docs_dir, filename))
                documents.extend(loader.load())
        
        if not documents:
            raise ValueError(f"在 {self.docs_dir} 中没有找到文档文件")
        
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=500,
            chunk_overlap=50
        )
        chunks = splitter.split_documents(documents)
        
        self.vectorstore = FAISS.from_documents(chunks, self.embeddings)
        print(f"✅ 知识库构建完成：{len(chunks)} 个文档片段")
    
    def search(self, query: str, k: int = 3) -> str:
        """检索相关内容"""
        if not self.vectorstore:
            return ""
        results = self.vectorstore.similarity_search(query, k=k)
        return "\n\n".join(doc.page_content for doc in results)


# ============ tools.py ============
from langchain_core.tools import tool

@tool
def web_search(query: str) -> str:
    """搜索互联网获取最新信息。当知识库中没有答案时使用。
    
    Args:
        query: 搜索关键词
    """
    # 教学示例：模拟搜索结果
    # 实际可用 DuckDuckGoSearchRun 或 TavilySearchResults
    return f"[模拟搜索结果] 关于'{query}'的最新信息：这是从互联网获取的内容。"


# ============ nodes.py ============
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.messages import AIMessage, HumanMessage
from state import KBState
from knowledge_base import KnowledgeBase
from tools import web_search

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
kb = KnowledgeBase()

def retrieve_node(state: KBState) -> dict:
    """检索节点：从知识库中检索相关内容"""
    question = state["question"]
    context = kb.search(question)
    return {"context": context, "retry_count": state.get("retry_count", 0)}

def answer_node(state: KBState) -> dict:
    """回答节点：基于检索到的内容生成回答"""
    prompt = ChatPromptTemplate.from_template("""
    基于以下背景知识回答用户的问题。
    如果背景知识中没有相关信息，请回复"知识库中未找到相关内容"。
    
    背景知识：
    {context}
    
    问题：{question}
    """)
    
    chain = prompt | llm | StrOutputParser()
    answer = chain.invoke({
        "context": state["context"],
        "question": state["question"]
    })
    
    return {"answer": answer, "source": "knowledge_base"}

def check_answer_node(state: KBState) -> dict:
    """检查节点：判断知识库回答是否有效"""
    answer = state.get("answer", "")
    if "未找到" in answer or len(answer) < 10:
        # 知识库没有答案，需要搜索
        return {"retry_count": state.get("retry_count", 0) + 1}
    return {}

def search_node(state: KBState) -> dict:
    """搜索节点：调用搜索工具获取信息"""
    search_result = web_search.invoke({"query": state["question"]})
    
    prompt = ChatPromptTemplate.from_template("""
    根据以下搜索结果回答用户的问题：
    
    搜索结果：{search_result}
    
    问题：{question}
    """)
    
    chain = prompt | llm | StrOutputParser()
    answer = chain.invoke({
        "search_result": search_result,
        "question": state["question"]
    })
    
    return {"answer": answer, "source": "web_search", "context": search_result}

def output_node(state: KBState) -> dict:
    """输出节点：格式化最终回答"""
    answer = state["answer"]
    source = state.get("source", "unknown")
    source_label = "📚 知识库" if source == "knowledge_base" else "🔍 互联网搜索"
    
    final = f"{answer}\n\n---\n来源：{source_label}"
    return {"answer": final}


# ============ graph.py ============
from langgraph.graph import StateGraph, START, END
from state import KBState
from nodes import retrieve_node, answer_node, check_answer_node, search_node, output_node

def route_after_check(state: KBState) -> str:
    """根据检查结果决定路由"""
    answer = state.get("answer", "")
    retry = state.get("retry_count", 0)
    
    if "未找到" in answer and retry <= 1:
        return "search"     # 知识库没答案，去搜索
    return "output"          # 有答案，直接输出

def build_graph():
    """构建 LangGraph 工作流"""
    graph = StateGraph(KBState)
    
    # 添加节点
    graph.add_node("retrieve", retrieve_node)
    graph.add_node("answer", answer_node)
    graph.add_node("check", check_answer_node)
    graph.add_node("search", search_node)
    graph.add_node("output", output_node)
    
    # 添加边
    graph.add_edge(START, "retrieve")
    graph.add_edge("retrieve", "answer")
    graph.add_edge("answer", "check")
    
    # 条件边：检查后决定走搜索还是输出
    graph.add_conditional_edges(
        "check",
        route_after_check,
        {
            "search": "search",   # 知识库没答案 → 搜索
            "output": "output",   # 有答案 → 输出
        }
    )
    
    graph.add_edge("search", "output")
    graph.add_edge("output", END)
    
    return graph.compile()


# ============ main.py ============
import os
from dotenv import load_dotenv
from graph import build_graph
from knowledge_base import kb
from langchain_core.messages import HumanMessage, AIMessage

load_dotenv()

def main():
    print("=" * 60)
    print("  智能知识库助手 (Smart Knowledge Base Assistant)")
    print("=" * 60)
    
    # 构建知识库
    docs_dir = "docs"
    if not os.path.exists(docs_dir):
        os.makedirs(docs_dir)
        # 创建示例文档
        with open(os.path.join(docs_dir, "sample.txt"), "w", encoding="utf-8") as f:
            f.write("""LangChain 是一个用于构建大语言模型（LLM）应用的开源框架。
它由 Harrison Chase 于 2022 年创建。
LangChain 的核心组件包括 Models、Prompts、Output Parsers、Memory、Chains、Agents 和 Retrieval。
LangChain 支持多种 LLM 提供商，包括 OpenAI、Anthropic、Google 等。
LangChain 使用 LCEL（LangChain Expression Language）作为其表达式语言。

LangGraph 是 LangChain 团队开发的图式工作流编排框架。
它允许开发者构建带有条件分支、循环和状态管理的复杂 AI 工作流。
LangGraph 的核心概念包括 State、Node、Edge 和 Conditional Edge。
""")
        print(f"✅ 已创建示例文档：{docs_dir}/sample.txt")
    
    kb.build()
    
    # 构建工作流
    app = build_graph()
    
    # 对话循环
    chat_history = []
    print("\n💡 提示：输入 'quit' 退出，输入 'clear' 清空对话历史\n")
    
    while True:
        user_input = input("👤 你: ").strip()
        
        if user_input.lower() == "quit":
            print("👋 再见！")
            break
        elif user_input.lower() == "clear":
            chat_history = []
            print("🧹 对话历史已清空\n")
            continue
        elif not user_input:
            continue
        
        # 运行工作流
        result = app.invoke({
            "question": user_input,
            "messages": chat_history + [HumanMessage(content=user_input)],
            "retry_count": 0,
        })
        
        # 输出回答
        print(f"\n🤖 助手: {result['answer']}\n")
        
        # 更新对话历史
        chat_history.append(HumanMessage(content=user_input))
        chat_history.append(AIMessage(content=result["answer"]))

if __name__ == "__main__":
    main()
```
