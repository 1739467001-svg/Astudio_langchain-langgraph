## 二、结构化日志

```python
from dataclasses import dataclass, field
from datetime import datetime
import json
from enum import Enum

class LogLevel(str, Enum):
    DEBUG = "DEBUG"
    INFO = "INFO"
    WARNING = "WARNING"
    ERROR = "ERROR"

@dataclass
class LLMLogEntry:
    """LLM应用日志条目。"""
    timestamp: str
    level: LogLevel
    request_id: str           # 请求ID（贯穿全链路）
    event: str                # 事件类型
    user_id: str = ""
    session_id: str = ""
    model: str = ""
    prompt_tokens: int = 0
    completion_tokens: int = 0
    latency_ms: float = 0
    tool_name: str = ""
    error: str = ""
    metadata: dict = field(default_factory=dict)

    def to_json(self) -> str:
        return json.dumps(&#123;
            k: v for k, v in self.__dict__.items() if v
        &#125;, ensure_ascii=False)

class LLMLogger:
    """LLM应用日志记录器。"""

    def __init__(self):
        self.logs: list[LLMLogEntry] = []

    def log(self, **kwargs):
        entry = LLMLogEntry(
            timestamp=datetime.now().isoformat(),
            **kwargs,
        )
        self.logs.append(entry)

    def get_by_request_id(self, request_id: str) -> list[LLMLogEntry]:
        """获取某请求的完整日志链。"""
        return [l for l in self.logs if l.request_id == request_id]

    def get_errors(self, limit: int = 50) -> list[LLMLogEntry]:
        """获取所有错误日志。"""
        return [l for l in self.logs if l.level == LogLevel.ERROR][-limit:]

    def get_slow_requests(self, threshold_ms: float = 5000) -> list[LLMLogEntry]:
        """获取慢请求。"""
        return [l for l in self.logs if l.latency_ms > threshold_ms]
```

## 三、智能诊断

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage

DIAGNOSE_PROMPT = """你是LLM应用运维专家。分析以下日志，诊断问题。

## 日志（按时间排序）
&#123;logs&#125;

## 诊断要求
1. 识别异常模式（延迟突增/错误聚集/工具反复失败）
2. 定位根因
3. 给出修复建议

## 输出
```json
&#123;&#123;
  "anomalies": ["异常1"],
  "root_cause": "根因",
  "severity": "high/medium/low",
  "fix_suggestion": "建议",
  "affected_requests": ["request_id1"]
&#125;&#125;
```"""

class LogDiagnosticsEngine:
    """日志智能诊断引擎。"""

    def __init__(self, llm: BaseChatModel, logger: LLMLogger):
        self.llm = llm
        self.logger = logger

    async def diagnose(self, request_id: str = None, time_range: tuple = None) -> dict:
        """诊断问题。"""
        # 获取相关日志
        if request_id:
            logs = self.logger.get_by_request_id(request_id)
        elif time_range:
            logs = [l for l in self.logger.logs if time_range[0] <= l.timestamp <= time_range[1]]
        else:
            logs = self.logger.get_errors(limit=20)

        if not logs:
            return &#123;"status": "no_logs", "message": "无相关日志"&#125;

        logs_text = "\n".join(l.to_json() for l in logs[:30])

        prompt = DIAGNOSE_PROMPT.format(logs=logs_text[:3000])
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])

        import re, json
        match = re.search(r'\&#123;.*\&#125;', response.content, re.DOTALL)
        if match:
            return json.loads(match.group())
        return &#123;"analysis": response.content[:500]&#125;

    async def auto_diagnose_recent(self, minutes: int = 30) -> dict:
        """自动诊断最近N分钟的问题。"""
        cutoff = (datetime.now().timestamp() - minutes * 60)
        recent_logs = [
            l for l in self.logger.logs
            if datetime.fromisoformat(l.timestamp).timestamp() > cutoff
        ]

        if not recent_logs:
            return &#123;"status": "no_recent_logs"&#125;

        # 检测异常模式
        errors = [l for l in recent_logs if l.level == LogLevel.ERROR]
        slow = [l for l in recent_logs if l.latency_ms > 5000]

        anomalies = []
        if len(errors) > 5:
            anomalies.append(f"错误数异常: &#123;len(errors)&#125;个错误在&#123;minutes&#125;分钟内")
        if len(slow) > 3:
            anomalies.append(f"延迟异常: &#123;len(slow)&#125;个请求超过5秒")

        if not anomalies:
            return &#123;"status": "healthy", "message": f"最近&#123;minutes&#125;分钟无异常"&#125;

        # LLM深入分析
        return await self.diagnose(time_range=None)
```

## 四、日志聚合分析

```python
from collections import defaultdict, Counter

class LogAggregator:
    """日志聚合分析器。"""

    @staticmethod
    def analyze_patterns(logs: list[LLMLogEntry]) -> dict:
        """分析日志模式。"""
        # 按事件类型统计
        event_counts = Counter(l.event for l in logs)

        # 按模型统计
        model_usage = Counter(l.model for l in logs if l.model)

        # 延迟分布
        latencies = [l.latency_ms for l in logs if l.latency_ms > 0]
        avg_latency = sum(latencies) / len(latencies) if latencies else 0

        # 错误模式
        errors = [l for l in logs if l.level == LogLevel.ERROR]
        error_patterns = Counter(l.error[:50] for l in errors if l.error)

        # Token消耗
        total_prompt = sum(l.prompt_tokens for l in logs)
        total_completion = sum(l.completion_tokens for l in logs)

        # 按时间分布
        hourly = defaultdict(int)
        for l in logs:
            hour = l.timestamp[:13]  # YYYY-MM-DDTHH
            hourly[hour] += 1

        return &#123;
            "total_logs": len(logs),
            "by_event": dict(event_counts.most_common(10)),
            "by_model": dict(model_usage),
            "avg_latency_ms": round(avg_latency, 2),
            "error_count": len(errors),
            "top_errors": dict(error_patterns.most_common(5)),
            "total_tokens": total_prompt + total_completion,
            "hourly_distribution": dict(sorted(hourly.items())),
        &#125;
```

## 五、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 每条日志有request_id | 可追溯全链路 | ★★★ |
| 日志结构化为JSON | 便于搜索聚合 | ★★★ |
| 记录延迟分解 | 检索/LLM/工具各自耗时 | ★★☆ |
| 用LLM智能诊断 | 比人工翻日志快 | ★★☆ |
| 定期聚合分析 | 发现趋势和模式 | ★★☆ |
| 日志脱敏 | PII不记录 | ★★★ |
