## 二、实现

```python
import hashlib
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from langchain_core.documents import Document

@dataclass
class FreshnessConfig:
    max_age_days: int = 90        # 最大有效天数
    stale_threshold: int = 60     # 超过标记为stale
    low_hit_threshold: int = 1     # 低于此命中次数标记

class DocumentFreshnessChecker:
    """文档新鲜度检查器。"""

    def __init__(self, config: FreshnessConfig = FreshnessConfig()):
        self.config = config

    def check(self, doc: Document, hit_count: int = 0) -> dict:
        """检查文档新鲜度。"""
        updated = doc.metadata.get("updated_at", "")
        if not updated:
            return &#123;"status": "unknown", "action": "review"&#125;

        try:
            updated_date = datetime.fromisoformat(updated)
            age_days = (datetime.now() - updated_date).days
        except (ValueError, TypeError):
            return &#123;"status": "unknown", "action": "review"&#125;

        if age_days > self.config.max_age_days:
            return &#123;"status": "expired", "age_days": age_days, "action": "delete_or_update"&#125;
        elif age_days > self.config.stale_threshold:
            return &#123;"status": "stale", "age_days": age_days, "action": "mark_stale"&#125;
        elif hit_count < self.config.low_hit_threshold and age_days > 30:
            return &#123;"status": "low_usage", "age_days": age_days, "action": "review"&#125;
        return &#123;"status": "fresh", "age_days": age_days&#125;


class DocumentCleanupManager:
    """文档清理管理器。"""

    def __init__(self):
        self.checker = DocumentFreshnessChecker()
        self.stats = &#123;"checked": 0, "fresh": 0, "stale": 0, "expired": 0, "removed": 0&#125;

    def scan_and_clean(self, docs: list[Document], hit_counts: dict = None) -> dict:
        """扫描并清理文档。"""
        hit_counts = hit_counts or &#123;&#125;
        to_keep = []
        to_remove = []

        for doc in docs:
            doc_id = doc.metadata.get("doc_id", "")
            hits = hit_counts.get(doc_id, 0)
            result = self.checker.check(doc, hits)

            self.stats["checked"] += 1

            if result["status"] == "expired":
                self.stats["expired"] += 1
                to_remove.append(doc)
            elif result["status"] == "stale":
                self.stats["stale"] += 1
                doc.metadata["stale"] = True
                to_keep.append(doc)
            else:
                self.stats["fresh"] += 1
                to_keep.append(doc)

        self.stats["removed"] = len(to_remove)
        return &#123;
            "kept": len(to_keep),
            "removed": len(to_remove),
            "stats": self.stats,
        &#125;
```
