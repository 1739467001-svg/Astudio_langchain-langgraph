## 2. 批处理（Batching）

### 为什么批处理有效

```
串行调用：
  请求1 → LLM → 2s → 响应1
  请求2 → LLM → 2s → 响应2
  请求3 → LLM → 2s → 响应3
  总耗时：6s

批处理：
  请求1, 2, 3 → LLM（batch=3）→ 2.5s → 响应1, 2, 3
  总耗时：2.5s（仅多 0.5s）
```

### 动态批处理器

```python
from dataclasses import dataclass, field
import asyncio
import time
from typing import Any, Callable

@dataclass
class PendingRequest:
    """待处理请求"""
    id: str
    input: str
    future: asyncio.Future
    timestamp: float = field(default_factory=time.time)


class DynamicBatcher:
    """动态批处理器：积攒请求后批量发送"""

    def __init__(
        self,
        batch_fn: Callable,
        max_batch_size: int = 32,
        max_wait_ms: float = 50,
    ):
        """
        Args:
            batch_fn: 批量处理函数，输入 list[str]，输出 list[str]
            max_batch_size: 最大批量大小
            max_wait_ms: 最大等待时间（毫秒）
        """
        self.batch_fn = batch_fn
        self.max_batch_size = max_batch_size
        self.max_wait_ms = max_wait_ms
        self.pending: list[PendingRequest] = []
        self.running = False

    async def submit(self, request_id: str, input: str) -> str:
        """提交单个请求，等待批量处理结果"""
        loop = asyncio.get_event_loop()
        future = loop.create_future()
        self.pending.append(PendingRequest(id=request_id, input=input, future=future))

        # 如果积累够了或第一次请求，触发处理
        if len(self.pending) >= self.max_batch_size:
            await self._flush()

        return await future

    async def _flush(self):
        """执行批量处理"""
        if not self.pending:
            return

        # 取出待处理请求
        batch = self.pending[:self.max_batch_size]
        self.pending = self.pending[self.max_batch_size:]

        # 提取输入
        inputs = [req.input for req in batch]

        try:
            # 批量调用
            results = await self.batch_fn(inputs)

            # 分发结果
            for req, result in zip(batch, results):
                if not req.future.done():
                    req.future.set_result(result)
        except Exception as e:
            # 批量失败，逐个通知
            for req in batch:
                if not req.future.done():
                    req.future.set_exception(e)

    async def run(self):
        """后台运行：定时刷新"""
        self.running = True
        while self.running:
            await asyncio.sleep(self.max_wait_ms / 1000)
            if self.pending:
                await self._flush()

    def stop(self):
        self.running = False


# 使用示例
async def batch_llm_call(inputs: list[str]) -> list[str]:
    """批量 LLM 调用"""
    from langchain_openai import ChatOpenAI
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    # OpenAI 支持批量调用
    results = await llm.abatch(inputs)
    return [r.content for r in results]


batcher = DynamicBatcher(
    batch_fn=batch_llm_call,
    max_batch_size=20,
    max_wait_ms=50,
)

async def main():
    import asyncio
    # 启动后台刷新
    task = asyncio.create_task(batcher.run())

    # 模拟 100 个并发请求
    requests = [batcher.submit(f"req_&#123;i&#125;", f"问题 &#123;i&#125;") for i in range(100)]
    results = await asyncio.gather(*requests)
    print(f"处理了 &#123;len(results)&#125; 个请求")

    batcher.stop()
    task.cancel()

# asyncio.run(main())
```

### 批处理 vs 串行对比

| 方式 | 100 请求耗时 | API 调用次数 | 吞吐量 |
|------|-------------|-------------|--------|
| 串行 | ~200s | 100 | 0.5 QPS |
| 并发(asyncio) | ~10s | 100 | 10 QPS |
| 批处理(batch=20) | ~10s | 5 | 10 QPS |
| 批处理+并发 | ~2s | 5 | 50 QPS |

## 3. 并发控制

```python
import asyncio
from dataclasses import dataclass
from typing import Any

class ConcurrencyManager:
    """并发控制管理器"""

    def __init__(
        self,
        max_concurrent: int = 10,
        rate_limit_per_minute: int = 60,
    ):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.rate_limiter = RateLimiter(rate_limit_per_minute)
        self.active_count = 0
        self.total_count = 0

    async def execute(self, func: Callable, *args, **kwargs) -> Any:
        """并发执行函数"""
        async with self.semaphore:
            await self.rate_limiter.acquire()
            self.active_count += 1
            self.total_count += 1
            try:
                result = await func(*args, **kwargs)
                return result
            finally:
                self.active_count -= 1

    def stats(self) -> dict:
        return &#123;
            "active": self.active_count,
            "total": self.total_count,
            "max_concurrent": self.semaphore._value,
        &#125;


class RateLimiter:
    """令牌桶限流器"""

    def __init__(self, rate_per_minute: int):
        self.rate = rate_per_minute / 60  # 每秒
        self.tokens = rate_per_minute
        self.max_tokens = rate_per_minute
        self.last_refill = time.time()
        self.lock = asyncio.Lock()

    async def acquire(self):
        """获取令牌"""
        async with self.lock:
            now = time.time()
            # 补充令牌
            elapsed = now - self.last_refill
            self.tokens = min(self.max_tokens, self.tokens + elapsed * self.rate)
            self.last_refill = now

            if self.tokens < 1:
                # 等待令牌
                wait = (1 - self.tokens) / self.rate
                await asyncio.sleep(wait)
                self.tokens = 0
            else:
                self.tokens -= 1


# 使用
concurrency = ConcurrencyManager(max_concurrent=10, rate_limit_per_minute=100)

async def call_llm(prompt: str) -> str:
    return await concurrency.execute(llm.ainvoke, prompt)
```

## 4. 语义缓存加速

```python
import numpy as np
from dataclasses import dataclass
from typing import Any

@dataclass
class SemanticCacheEntry:
    query: str
    response: str
    embedding: list[float]
    timestamp: float
    hit_count: int = 0


class SemanticCache:
    """语义缓存：相似查询命中缓存"""

    def __init__(self, threshold: float = 0.92, ttl: int = 3600):
        self.threshold = threshold
        self.ttl = ttl
        self.cache: list[SemanticCacheEntry] = []
        self.embeddings_model = None  # 延迟加载

    def _get_embeddings(self):
        if self.embeddings_model is None:
            from langchain_openai import OpenAIEmbeddings
            self.embeddings_model = OpenAIEmbeddings(model="text-embedding-3-small")
        return self.embeddings_model

    def _cosine_sim(self, a: list[float], b: list[float]) -> float:
        a_arr, b_arr = np.array(a), np.array(b)
        return float(np.dot(a_arr, b_arr) / (np.linalg.norm(a_arr) * np.linalg.norm(b_arr)))

    def get(self, query: str) -> str | None:
        """查找缓存"""
        emb = self._get_embeddings().embed_query(query)
        now = time.time()

        best_sim = 0
        best_entry = None

        for entry in self.cache:
            if now - entry.timestamp > self.ttl:
                continue
            sim = self._cosine_sim(emb, entry.embedding)
            if sim > best_sim:
                best_sim = sim
                best_entry = entry

        if best_entry and best_sim >= self.threshold:
            best_entry.hit_count += 1
            return best_entry.response

        return None

    def set(self, query: str, response: str):
        """写入缓存"""
        emb = self._get_embeddings().embed_query(query)
        self.cache.append(SemanticCacheEntry(
            query=query, response=response, embedding=emb, timestamp=time.time()
        ))
```

## 5. 自部署推理加速

### vLLM 推理服务器

```python
# 安装: pip install vllm
# 启动: python -m vllm.entrypoints.openai.api_server --model meta-llama/Llama-3-8B

from vllm import LLM, SamplingParams

class VLLMAccelerator:
    """vLLM 推理加速"""

    def __init__(self, model: str = "meta-llama/Llama-3-8B"):
        self.llm = LLM(
            model=model,
            # vLLM 核心优化
            enable_prefix_caching=True,   # 前缀缓存
            tensor_parallel_size=1,        # GPU 数量
            gpu_memory_utilization=0.9,    # GPU 内存使用率
            max_model_len=4096,            # 最大上下文长度
            # 量化
            quantization=None,             # awq / gptq / fp8
        )

    def generate(
        self,
        prompts: list[str],
        max_tokens: int = 512,
        temperature: float = 0.0,
    ) -> list[str]:
        """批量推理"""
        params = SamplingParams(
            max_tokens=max_tokens,
            temperature=temperature,
        )

        # vLLM 自动批处理 + PagedAttention
        outputs = self.llm.generate(prompts, params)
        return [output.outputs[0].text for output in outputs]
```

### vLLM 核心技术

| 技术 | 原理 | 加速效果 |
|------|------|---------|
| PagedAttention | KV Cache 分页管理 | 减少内存碎片 |
| Continuous Batching | 动态组装 batch | 提高 GPU 利用率 |
| Prefix Caching | 缓存公共前缀 | 减少重复计算 |
| Speculative Decoding | 小模型先猜+大模型验证 | 2-3x 加速 |
| Tensor Parallelism | 多 GPU 并行 | 大模型加速 |

### 推理框架对比

| 框架 | 特点 | 适用 |
|------|------|------|
| vLLM | PagedAttention+连续批处理 | 通用推荐 |
| TGI | HuggingFace 推理服务器 | HF 生态 |
| TensorRT-LLM | NVIDIA 优化 | 极致性能 |
| Ollama | 本地一键运行 | 开发调试 |
| llama.cpp | CPU/混合推理 | 无 GPU |

## 6. 推测解码（Speculative Decoding）

```python
# 概念：用小模型快速生成候选 token，大模型验证
# 小模型生成快（5x），大模型验证只需 1 次前向传播

"""
流程：
1. 小模型（draft model）生成 N 个候选 token（如 4 个）
2. 大模型（target model）一次性验证这 N 个 token
3. 如果 K 个正确（K≤N），保留 K 个，从第 K+1 个重新生成
4. 加速：N 次小模型 + 1 次大模型 vs N 次大模型

例：小模型 4 个 token 正确 3 个
  串行：4 × 大模型延迟 = 4 × 20ms = 80ms
  推测：4 × 小模型延迟 + 1 × 大模型延迟 = 4 × 4ms + 20ms = 36ms
  加速：2.2x
"""

# vLLM 中启用推测解码
from vllm import LLM

llm = LLM(
    model="meta-llama/Llama-3-70B",      # 大模型（主模型）
    speculative_model="meta-llama/Llama-3-8B",  # 小模型（草稿模型）
    num_speculative_tokens=5,             # 每次猜测 5 个 token
    # 效果：生成速度提升 1.5-3x
)
```

## 7. 加速策略选择

| 场景 | 推荐策略 | 预期加速 |

| API 调用 | 批处理+并发+语义缓存 | 5-10x |
