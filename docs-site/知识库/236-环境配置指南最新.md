## 二、安装

```bash
# 1. 安装Python 3.11+（推荐用uv）
curl -LsSf https://astral.sh/uv/install.sh | sh
uv python install 3.11

# 2. 创建项目
uv init my-llm-app
cd my-llm-app

# 3. 安装依赖
uv add langchain langgraph langchain-openai
uv add langchain-community  # 第三方集成（按需）
uv add langsmith  # 可观测性

# 4. 向量库（选一个）
uv add chromadb  # 轻量级
# 或
uv add pymilvus  # 生产级

# 5. 开发工具
uv add --dev pytest pytest-asyncio
```

```python
# config.py — 统一配置管理
import os
from dotenv import load_dotenv

load_dotenv()

class Config:
    OPENAI_API_KEY = os.getenv("OPENAI_API_KEY", "")
    LANGSMITH_TRACING = os.getenv("LANGSMITH_TRACING", "false").lower() == "true"
    LANGSMITH_API_KEY = os.getenv("LANGSMITH_API_KEY", "")
    LANGSMITH_PROJECT = os.getenv("LANGSMITH_PROJECT", "default")
    TAVILY_API_KEY = os.getenv("TAVILY_API_KEY", "")
```

## 四、验证配置

```python
# verify_setup.py — 验证环境配置
import asyncio
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

async def verify():
    """验证所有组件是否正常。"""
    results = &#123;&#125;

    # 1. LLM
    try:
        llm = ChatOpenAI(model="gpt-4o-mini")
        resp = await llm.ainvoke([HumanMessage(content="说'OK'")])
        results["llm"] = "✅ 正常" if "OK" in resp.content else "⚠️ 响应异常"
    except Exception as e:
        results["llm"] = f"❌ 失败: &#123;e&#125;"

    # 2. LangSmith
    import os
    if os.getenv("LANGSMITH_TRACING") == "true":
        results["langsmith"] = "✅ 已启用"
    else:
        results["langsmith"] = "⚠️ 未启用"

    # 3. 向量库
    try:
        import chromadb
        client = chromadb.Client()
        results["vector_db"] = "✅ 正常"
    except Exception:
        results["vector_db"] = "⚠️ 未安装"

    # 4. LangGraph
    try:
        from langgraph.graph import StateGraph
        results["langgraph"] = "✅ 正常"
    except Exception:
        results["langgraph"] = "❌ 未安装"

    print("=== 环境验证 ===")
    for component, status in results.items():
        print(f"&#123;component&#125;: &#123;status&#125;")

asyncio.run(verify())
```
