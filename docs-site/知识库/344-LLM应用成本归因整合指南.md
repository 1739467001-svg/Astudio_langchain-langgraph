## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from collections import defaultdict

PRICING = &#123;
    "gpt-4o": &#123;"input": 2.50, "output": 10.00&#125;,
    "gpt-4o-mini": &#123;"input": 0.15, "output": 0.60&#125;,
&#125;

@dataclass
class CostRecord:
    timestamp: str
    user_id: str
    tool_name: str
    model: str
    input_tokens: int
    output_tokens: int
    cost_usd: float

class CostAttributionSystem:
    """成本归因系统——追踪+分析+报告+建议。"""

    def __init__(self):
        self.records: list[CostRecord] = []

    def record(self, user_id: str, tool_name: str, model: str, input_tokens: int, output_tokens: int):
        """记录成本。"""
        pricing = PRICING.get(model, PRICING["gpt-4o-mini"])
        cost = (input_tokens / 1e6 * pricing["input"]) + (output_tokens / 1e6 * pricing["output"])
        self.records.append(CostRecord(
            timestamp=datetime.now().isoformat(),
            user_id=user_id, tool_name=tool_name, model=model,
            input_tokens=input_tokens, output_tokens=output_tokens,
            cost_usd=round(cost, 6),
        ))

    def analyze(self, dimension: str = "tool") -> dict:
        """按维度归因分析。"""
        groups = defaultdict(lambda: &#123;"cost": 0, "calls": 0, "tokens": 0&#125;)

        for r in self.records:
            key = getattr(r, dimension, "unknown")
            groups[key]["cost"] += r.cost_usd
            groups[key]["calls"] += 1
            groups[key]["tokens"] += r.input_tokens + r.output_tokens

        return &#123;k: &#123;**v, "cost": round(v["cost"], 4)&#125; for k, v in groups.items()&#125;

    def report(self) -> dict:
        """完整成本报告。"""
        total_cost = sum(r.cost_usd for r in self.records)
        return &#123;
            "total_cost": round(total_cost, 4),
            "total_calls": len(self.records),
            "cost_per_call": round(total_cost / max(len(self.records), 1), 6),
            "by_user": self.analyze("user_id"),
            "by_tool": self.analyze("tool_name"),
            "by_model": self.analyze("model"),
        &#125;

    def optimize_suggestions(self) -> list[str]:
        """生成优化建议。"""
        suggestions = []
        by_model = self.analyze("model")

        # 检查是否大部分成本在高价模型
        gpt4o_cost = by_model.get("gpt-4o", &#123;&#125;).get("cost", 0)
        total = sum(v["cost"] for v in by_model.values())
        if total > 0 and gpt4o_cost / total > 0.5:
            suggestions.append("建议启用模型路由，80%走mini可降60%成本")

        # 检查是否有高频低价值的工具调用
        by_tool = self.analyze("tool_name")
        for tool, data in by_tool.items():
            if data["calls"] > 100 and data["cost"] / data["calls"] > 0.01:
                suggestions.append(f"工具'&#123;tool&#125;'调用频繁且成本高，建议启用结果缓存")

        if not suggestions:
            suggestions.append("当前成本结构合理")

        return suggestions
```
