## 三、实现

### 3.1 带解释的检索

```python
from langchain_core.documents import Document
from langchain_community.vectorstores import FAISS

def retrieve_with_explanation(vectorstore, query: str, k: int = 3) -> dict:
    """带解释的检索"""
    # 检索（带分数）
    results_with_scores = vectorstore.similarity_search_with_score(query, k=k)

    explained = []
    for rank, (doc, score) in enumerate(results_with_scores, 1):
        explained.append(&#123;
            "rank": rank,
            "source": doc.metadata.get("source", "未知"),
            "page": doc.metadata.get("page", "?"),
            "score": round(float(score), 4),  # FAISS: 越低越相似
            "relevance": _score_to_label(float(score)),
            "preview": doc.page_content[:100],
            "content": doc.page_content,
        &#125;)

    return &#123;
        "query": query,
        "total_results": len(explained),
        "results": explained,
    &#125;

def _score_to_label(score: float) -> str:
    """分数转可读标签（FAISS距离，越低越好）"""
    if score < 0.3:
        return "高度相关 ⭐⭐⭐"
    elif score < 0.6:
        return "相关 ⭐⭐"
    elif score < 1.0:
        return "弱相关 ⭐"
    else:
        return "不太相关 ⚠️"
```

### 3.2 带解释的回答

```python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate

def rag_with_explanation(question: str, vectorstore, llm) -> dict:
    """带解释的RAG回答"""
    # Step 1: 带解释的检索
    retrieval = retrieve_with_explanation(vectorstore, question, k=3)

    # Step 2: 组装上下文（带来源标注）
    context_parts = []
    for r in retrieval["results"]:
        context_parts.append(
            f"[片段&#123;r['rank']&#125; | 来源: &#123;r['source']&#125; | 相关度: &#123;r['relevance']&#125;]\n&#123;r['content']&#125;"
        )
    context = "\n\n".join(context_parts)

    # Step 3: 生成回答
    prompt = ChatPromptTemplate.from_template(
        "基于以下知识回答。在关键信息后标注来源。\n\n&#123;context&#125;\n\n问题：&#123;question&#125;"
    )
    answer = (prompt | llm).invoke(&#123;"context": context, "question": question&#125;).content

    # Step 4: 组装解释性回答
    return &#123;
        "question": question,
        "answer": answer,
        "sources": [
            &#123;"source": r["source"], "relevance": r["relevance"], "score": r["score"]&#125;
            for r in retrieval["results"]
        ],
        "retrieval_count": retrieval["total_results"],
        "original_query": question,
    &#125;
```

### 3.3 解释性展示

```python
def format_explanation(result: dict) -> str:
    """格式化解释性回答"""
    output = f"📝 回答: &#123;result['answer']&#125;\n\n"
    output += f"📚 信息来源 (&#123;result['retrieval_count']&#125;个片段):\n"
    for src in result["sources"]:
        output += f"  &#123;src['source']&#125; - &#123;src['relevance']&#125; (分数: &#123;src['score']&#125;)\n"
    return output
```
