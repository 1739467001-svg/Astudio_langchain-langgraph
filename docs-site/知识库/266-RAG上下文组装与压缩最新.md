## 二、实现

```python
from langchain_core.documents import Document
from dataclasses import dataclass
import tiktoken

@dataclass
class AssemblyConfig:
    max_tokens: int = 3000
    max_docs: int = 5
    include_sources: bool = True
    compress: bool = True

class ContextAssembler:
    """上下文组装器。"""

    def __init__(self, config: AssemblyConfig = AssemblyConfig()):
        self.config = config
        self._encoder = None
        try:
            self._encoder = tiktoken.encoding_for_model("gpt-4o")
        except Exception:
            pass

    def assemble(self, docs: list[Document], query: str = "") -> str:
        """组装上下文。"""
        # 1. 排序（假设已按相关度排好）
        docs = docs[:self.config.max_docs]

        # 2. Token预算控制
        parts = []
        used = 0
        for doc in docs:
            content = doc.page_content
            tokens = self._count_tokens(content)
            if used + tokens > self.config.max_tokens:
                remaining = self.config.max_tokens - used
                if remaining > 100:
                    content = self._truncate(content, remaining)
                    parts.append(self._format(content, doc))
                break
            parts.append(self._format(content, doc))
            used += tokens

        return "\n\n---\n\n".join(parts)

    def _format(self, content: str, doc: Document) -> str:
        if self.config.include_sources:
            source = doc.metadata.get("source", "未知")
            return f"[来源: &#123;source&#125;]\n&#123;content&#125;"
        return content

    def _count_tokens(self, text: str) -> int:
        if self._encoder:
            return len(self._encoder.encode(text))
        return len(text) // 3

    def _truncate(self, text: str, max_tokens: int) -> str:
        if self._encoder:
            tokens = self._encoder.encode(text)[:max_tokens]
            return self._encoder.decode(tokens) + "\n[已截断]"
        return text[:max_tokens * 3] + "..."
```
