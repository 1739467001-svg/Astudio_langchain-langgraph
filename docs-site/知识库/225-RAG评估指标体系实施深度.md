## 一、评估指标实施

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
from dataclasses import dataclass
from typing import Optional
import json, re

@dataclass
class RAGEvalResult:
    """RAG评估结果。"""
    faithfulness: float = 0       # 忠实度: 回答是否基于检索内容
    answer_relevancy: float = 0   # 答案相关性: 是否切题
    context_precision: float = 0  # 上下文精确率: 检索内容是否相关
    context_recall: float = 0     # 上下文召回率: 是否检索到所有需要的信息
    overall_score: float = 0

class RAGEvaluator:
    """RAG评估器——用LLM-as-Judge实现四大指标。"""

    def __init__(self, judge_llm: BaseChatModel):
        self.judge = judge_llm

    async def evaluate(
        self,
        question: str,
        answer: str,
        contexts: list[str],
        ground_truth: Optional[str] = None,
    ) -> RAGEvalResult:
        """完整评估。"""
        faithfulness = await self._eval_faithfulness(answer, contexts)
        relevancy = await self._eval_relevancy(question, answer)
        precision = await self._eval_precision(question, contexts)
        recall = await self._eval_recall(contexts, ground_truth) if ground_truth else 0.8

        overall = (faithfulness + relevancy + precision + recall) / 4

        return RAGEvalResult(
            faithfulness=round(faithfulness, 4),
            answer_relevancy=round(relevancy, 4),
            context_precision=round(precision, 4),
            context_recall=round(recall, 4),
            overall_score=round(overall, 4),
        )

    async def _eval_faithfulness(self, answer: str, contexts: list[str]) -> float:
        """忠实度: 回答中有多少能从上下文推断。"""
        context = "\n".join(contexts)[:1500]
        prompt = f"""判断以下回答的每个陈述是否都能从上下文推断。

上下文:
&#123;context&#125;

回答:
&#123;answer[:500]&#125;

评分0-1: """
        response = await self.judge.ainvoke([HumanMessage(content=prompt)])
        match = re.search(r'0\.\d+|[01]', response.content)
        return float(match.group()) if match else 0.5

    async def _eval_relevancy(self, question: str, answer: str) -> float:
        """相关性: 回答是否切题。"""
        prompt = f"""判断回答是否直接回答了问题。

问题: &#123;question&#125;
回答: &#123;answer[:300]&#125;

评分0-1: """
        response = await self.judge.ainvoke([HumanMessage(content=prompt)])
        match = re.search(r'0\.\d+|[01]', response.content)
        return float(match.group()) if match else 0.5

    async def _eval_precision(self, question: str, contexts: list[str]) -> float:
        """精确率: 检索内容是否相关。"""
        context = "\n".join(contexts)[:1000]
        prompt = f"""判断检索到的上下文是否与问题相关。

问题: &#123;question&#125;
上下文: &#123;context[:500]&#125;

评分0-1: """
        response = await self.judge.ainvoke([HumanMessage(content=prompt)])
        match = re.search(r'0\.\d+|[01]', response.content)
        return float(match.group()) if match else 0.5

    async def _eval_recall(self, contexts: list[str], ground_truth: str) -> float:
        """召回率: 检索到所有需要的信息了吗。"""
        context = "\n".join(contexts)[:1000]
        prompt = f"""判断检索到的上下文是否包含标准答案中的所有关键信息。

标准答案: &#123;ground_truth[:300]&#125;
检索上下文: &#123;context[:500]&#125;

评分0-1: """
        response = await self.judge.ainvoke([HumanMessage(content=prompt)])
        match = re.search(r'0\.\d+|[01]', response.content)
        return float(match.group()) if match else 0.5


class BatchRAGEvaluator:
    """批量RAG评估器。"""

    def __init__(self, evaluator: RAGEvaluator):
        self.evaluator = evaluator
        self.results: list[dict] = []

    async def evaluate_batch(
        self,
        rag_pipeline,
        test_cases: list[dict],  # [&#123;question, ground_truth&#125;]
    ) -> dict:
        """批量评估。"""
        import asyncio

        async def eval_one(tc):
            result = await rag_pipeline.answer(tc["question"])
            answer = result.get("answer", "")
            contexts = result.get("contexts", [])

            eval_result = await self.evaluator.evaluate(
                question=tc["question"],
                answer=answer,
                contexts=contexts,
                ground_truth=tc.get("ground_truth"),
            )
            return &#123;
                "question": tc["question"][:50],
                "faithfulness": eval_result.faithfulness,
                "relevancy": eval_result.answer_relevancy,
                "precision": eval_result.context_precision,
                "recall": eval_result.context_recall,
                "overall": eval_result.overall_score,
            &#125;

        tasks = [eval_one(tc) for tc in test_cases]
        self.results = await asyncio.gather(*tasks)

        return self.report()

    def report(self) -> dict:
        """汇总报告。"""
        if not self.results:
            return &#123;"error": "无结果"&#125;

        n = len(self.results)
        return &#123;
            "total": n,
            "avg_faithfulness": round(sum(r["faithfulness"] for r in self.results) / n, 4),
            "avg_relevancy": round(sum(r["relevancy"] for r in self.results) / n, 4),
            "avg_precision": round(sum(r["precision"] for r in self.results) / n, 4),
            "avg_recall": round(sum(r["recall"] for r in self.results) / n, 4),
            "avg_overall": round(sum(r["overall"] for r in self.results) / n, 4),
        &#125;
```

## 二、基线管理

```python
class EvalBaselineManager:
    """评估基线管理器。"""

    def __init__(self):
        self.baseline: dict | None = None
        self.history: list[dict] = []

    def save_baseline(self, report: dict):
        """保存基线。"""
        self.baseline = report.copy()
        self.history.append(report)

    def compare(self, current: dict) -> dict:
        """对比基线。"""
        if not self.baseline:
            return &#123;"message": "无基线"&#125;

        deltas = &#123;&#125;
        for key in ["avg_faithfulness", "avg_relevancy", "avg_precision", "avg_recall", "avg_overall"]:
            old = self.baseline.get(key, 0)
            new = current.get(key, 0)
            delta = round(new - old, 4)
            deltas[key] = &#123;"baseline": old, "current": new, "delta": delta, "improved": delta > 0&#125;

        regressed = sum(1 for d in deltas.values() if not d["improved"] and d["delta"] != 0)
        return &#123;
            "deltas": deltas,
            "regressed_count": regressed,
            "should_alert": regressed > 0,
        &#125;
```
