## 二、知识沉淀实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional
from enum import Enum

class KnowledgeType(str, Enum):
    FAQ = "faq"               # 常见问答
    BEST_PRACTICE = "best_practice"  # 最佳实践
    FAILURE_CASE = "failure_case"    # 失败案例
    TERMINOLOGY = "terminology"     # 术语
    TROUBLESHOOTING = "troubleshooting"  # 故障排查

@dataclass
class KnowledgeEntry:
    """知识条目。"""
    id: str
    type: KnowledgeType
    title: str
    content: str
    tags: list[str] = field(default_factory=list)
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())
    updated_at: str = ""
    usage_count: int = 0       # 被引用次数
    helpful_count: int = 0      # 有帮助次数
    author: str = "system"

class KnowledgeBase:
    """知识沉淀库。"""

    def __init__(self):
        self.entries: dict[str, KnowledgeEntry] = &#123;&#125;

    def add(self, entry: KnowledgeEntry):
        """添加知识条目。"""
        self.entries[entry.id] = entry

    def search(self, query: str, k: int = 5) -> list[KnowledgeEntry]:
        """搜索知识（简化版：关键词匹配）。"""
        query_lower = query.lower()
        scored = []
        for entry in self.entries.values():
            title_match = query_lower in entry.title.lower()
            content_match = query_lower in entry.content.lower()
            tag_match = any(query_lower in t.lower() for t in entry.tags)

            score = sum([title_match * 3, content_match * 2, tag_match * 1])
            if score > 0:
                scored.append((entry, score))

        scored.sort(key=lambda x: x[1], reverse=True)
        return [e for e, _ in scored[:k]]

    def record_usage(self, entry_id: str, helpful: bool = True):
        """记录使用情况。"""
        entry = self.entries.get(entry_id)
        if entry:
            entry.usage_count += 1
            if helpful:
                entry.helpful_count += 1

    def get_popular(self, k: int = 10) -> list[KnowledgeEntry]:
        """获取最常引用的知识。"""
        return sorted(
            self.entries.values(),
            key=lambda e: e.usage_count,
            reverse=True,
        )[:k]

    def get_context_for_prompt(self, query: str) -> str:
        """为Prompt提供相关知识上下文。"""
        results = self.search(query, k=3)
        if not results:
            return ""

        parts = ["## 相关知识"]
        for entry in results:
            parts.append(f"### &#123;entry.title&#125;")
            parts.append(entry.content[:300])
            parts.append(f"(类型: &#123;entry.type.value&#125;, 引用次数: &#123;entry.usage_count&#125;)")

        return "\n\n".join(parts)


class KnowledgeExtractor:
    """知识提取器——从交互中自动沉淀知识。"""

    @staticmethod
    async def extract_from_conversation(
        llm,
        conversation: str,
        feedback: str = "",
    ) -> list[dict]:
        """从对话中提取可沉淀的知识。"""
        from langchain_core.messages import HumanMessage

        prompt = f"""从以下对话中提取可沉淀的知识。

对话:
&#123;conversation[:1500]&#125;

用户反馈: &#123;feedback[:200]&#125;

提取以下类型的知识:
1. FAQ: 如果用户问了常见问题
2. BEST_PRACTICE: 如果有成功的解决方案
3. FAILURE_CASE: 如果有失败/错误的案例
4. TROUBLESHOOTING: 如果有故障排查过程

输出JSON数组:
```json
[&#123;&#123;
  "type": "faq",
  "title": "标题",
  "content": "内容",
  "tags": ["标签"]
&#125;&#125;]
```
没有可沉淀的知识则返回空数组。"""

        response = await llm.ainvoke([HumanMessage(content=prompt)])
        import re, json
        match = re.search(r'\[.*\]', response.content, re.DOTALL)
        if match:
            return json.loads(match.group())
        return []
```

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 每次成功交互提取知识 | 积累组织记忆 | ★★★ |
