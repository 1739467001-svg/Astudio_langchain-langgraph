# 对话编排与多轮状态管理指南

> 单轮问答简单：用户问→Agent 答。但真实对话是多轮的：用户先问 A，追问 B，改主意了回到 A，又要 C。Agent 需要记住对话脉络、管理话题切换、保持上下文连贯——这就是对话编排与状态管理。

---

## 1. 对话编排核心概念

### 单轮 vs 多轮

```
单轮：
  用户 → 问题 → Agent → 回答 → 结束

多轮编排：
  用户 → 问题A → Agent → 回答A
  用户 → 追问B → Agent → 回答B（引用A的上下文）
  用户 → 修正 → Agent → 重新回答B
  用户 → 切换话题C → Agent → 回答C
  用户 → 回到A → Agent → 恢复A的上下文继续
```

### 对话状态包含什么

| 状态类型 | 示例 | 生命周期 |
|----------|------|---------|
| 对话历史 | 用户说了什么、Agent 回了什么 | 当前会话 |
| 用户画像 | 偏好、语言、知识水平 | 跨会话 |
| 任务状态 | 当前任务进行到哪一步 | 任务期间 |
| 话题上下文 | 当前讨论的话题和关键信息 | 话题期间 |
| 槽位状态 | 已收集的信息（姓名、日期等） | 任务期间 |
| 工具结果 | 上一步工具调用返回了什么 | 当前轮 |

## 2. 对话状态模型

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
import time

class DialogPhase(Enum):
    """对话阶段"""
    GREETING = "greeting"        # 问候
    INTENT_RECOGNITION = "intent"  # 意图识别
    SLOT_FILLING = "slot_filling"  # 槽位填充
    CONFIRMATION = "confirmation"  # 确认
    EXECUTION = "execution"        # 执行
    RESULT_DELIVERY = "result"     # 结果交付
    CLOSING = "closing"            # 结束


@dataclass
class Slot:
    """对话槽位"""
    name: str
    value: Any = None
    required: bool = True
    prompt: str = ""            # 缺失时追问什么
    confirmed: bool = False     # 用户是否确认


@dataclass
class TopicContext:
    """话题上下文"""
    topic: str = ""
    key_entities: dict[str, Any] = field(default_factory=dict)  # 提到的实体
    key_facts: list[str] = field(default_factory=list)          # 关键事实
    last_discussed: float = field(default_factory=time.time)


@dataclass
class DialogState:
    """完整对话状态"""
    # 标识
    session_id: str = ""
    user_id: str = ""
    # 阶段
    phase: DialogPhase = DialogPhase.GREETING
    # 槽位
    slots: dict[str, Slot] = field(default_factory=dict)
    # 话题
    current_topic: TopicContext = field(default_factory=TopicContext)
    topic_history: list[TopicContext] = field(default_factory=list)
    # 任务
    current_task: str = ""
    task_step: int = 0
    # 上下文
    context_variables: dict[str, Any] = field(default_factory=dict)
    last_tool_results: dict[str, Any] = field(default_factory=dict)
    # 时序
    turn_count: int = 0
    created_at: float = field(default_factory=time.time)
    last_active: float = field(default_factory=time.time)

    def get_missing_slots(self) -> list[Slot]:
        """获取缺失的必填槽位"""
        return [
            slot for slot in self.slots.values()
            if slot.required and slot.value is None
        ]

    def is_ready_to_execute(self) -> bool:
        """所有必填槽位已填充"""
        return len(self.get_missing_slots()) == 0

    def switch_topic(self, new_topic: str):
        """切换话题"""
        if self.current_topic.topic:
            self.topic_history.append(self.current_topic)
        self.current_topic = TopicContext(topic=new_topic)

    def restore_topic(self, topic_name: str) -> bool:
        """恢复之前的话题"""
        for i, t in enumerate(self.topic_history):
            if t.topic == topic_name:
                self.topic_history.append(self.current_topic)
                self.current_topic = self.topic_history.pop(i)
                return True
        return False
```

## 3. 对话编排引擎

```python
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from typing import TypedDict, Annotated
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
import json


class OrchestratorState(TypedDict):
    messages: Annotated[list, add_messages]
    dialog_state: dict
    current_response: str
    next_action: str


def build_dialog_orchestrator(llm: ChatOpenAI):
    """构建对话编排引擎"""

    def intent_recognition_node(state: OrchestratorState) -> dict:
        """意图识别：理解用户想做什么"""
        last_msg = state["messages"][-1].content
        ds = state["dialog_state"]

        prompt = ChatPromptTemplate.from_messages([
            ("system", """分析用户意图。输出 JSON：
&#123;
  "intent": "greeting/query/task/followup/correction/topic_switch/closing",
  "topic": "用户讨论的话题",
  "entities": &#123;"提取的实体": "值"&#125;,
  "slots_to_fill": ["需要追问的槽位"],
  "needs_tool": false
&#125;"""),
            ("human", "&#123;user_input&#125;"),
        ])

        chain = prompt | llm
        result = chain.invoke(&#123;"user_input": last_msg&#125;)

        try:
            intent_data = json.loads(result.content)
        except json.JSONDecodeError:
            intent_data = &#123;"intent": "query", "topic": last_msg[:50]&#125;

        # 更新对话状态
        ds["current_topic"] = intent_data.get("topic", "")
        ds["phase"] = "intent_recognition"
        ds["turn_count"] = ds.get("turn_count", 0) + 1

        # 更新槽位
        entities = intent_data.get("entities", &#123;&#125;)
        slots = ds.get("slots", &#123;&#125;)
        for name, value in entities.items():
            if name not in slots:
                slots[name] = &#123;"value": None, "required": True, "confirmed": False&#125;
            slots[name]["value"] = value

        ds["slots"] = slots

        # 决定下一步
        intent = intent_data.get("intent", "query")
        if intent == "greeting":
            return &#123;"next_action": "respond", "dialog_state": ds,
                    "current_response": "你好！有什么可以帮助你的？"&#125;
        elif intent == "closing":
            return &#123;"next_action": "end", "dialog_state": ds,
                    "current_response": "再见！祝你一切顺利。"&#125;
        elif intent == "slot_filling_needed" or intent_data.get("slots_to_fill"):
            return &#123;"next_action": "ask_slot", "dialog_state": ds&#125;
        elif intent_data.get("needs_tool"):
            return &#123;"next_action": "call_tool", "dialog_state": ds&#125;
        else:
            return &#123;"next_action": "respond", "dialog_state": ds&#125;

    def slot_filling_node(state: OrchestratorState) -> dict:
        """槽位填充：追问缺失信息"""
        ds = state["dialog_state"]
        slots = ds.get("slots", &#123;&#125;)

        # 找到第一个缺失的必填槽位
        missing = [
            (name, s) for name, s in slots.items()
            if p.get("required") and p.get("value") is None
        ]

        if not missing:
            return &#123;"next_action": "respond", "dialog_state": ds&#125;

        slot_name, slot = missing[0]
        prompt_text = slot.get("prompt", f"请提供&#123;slot_name&#125;：")
        ds["phase"] = "slot_filling"

        return &#123;
            "next_action": "respond",
            "dialog_state": ds,
            "current_response": prompt_text,
        &#125;

    def tool_execution_node(state: OrchestratorState) -> dict:
        """工具执行"""
        ds = state["dialog_state"]
        ds["phase"] = "execution"
        # 实际调用工具
        return &#123;
            "next_action": "respond",
            "dialog_state": ds,
            "current_response": "已执行操作，结果如下...",
        &#125;

    def response_generation_node(state: OrchestratorState) -> dict:
        """生成回复"""
        ds = state["dialog_state"]
        ds["phase"] = "result_delivery"
        ds["last_active"] = time.time()

        response = state.get("current_response", "我理解了你的问题。")

        return &#123;
            "messages": [AIMessage(content=response)],
            "dialog_state": ds,
            "next_action": "end_turn",
        &#125;

    def route(state: OrchestratorState) -> str:
        action = state.get("next_action", "respond")
        if action == "end":
            return END
        elif action == "ask_slot":
            return "slot_filling"
        elif action == "call_tool":
            return "tool_execution"
        elif action == "respond":
            return "response"
        elif action == "end_turn":
            return END
        return END

    # 构建图
    graph = StateGraph(OrchestratorState)
    graph.add_node("intent", intent_recognition_node)
    graph.add_node("slot_filling", slot_filling_node)
    graph.add_node("tool_execution", tool_execution_node)
    graph.add_node("response", response_generation_node)

    graph.add_edge(START, "intent")
    graph.add_conditional_edges("intent", route)
    graph.add_conditional_edges("slot_filling", route)
    graph.add_conditional_edges("tool_execution", route)
    graph.add_edge("response", END)

    return graph.compile()

    &#125;,

result = orchestrator.invoke(&#123;
```

## 4. 上下文窗口管理

```python
class ContextWindowManager:
    """上下文窗口管理器：防止对话历史超出窗口"""

    def __init__(self, max_tokens: int = 8000, reserved_for_response: int = 2000):
        self.max_tokens = max_tokens
        self.reserved = reserved_for_response
        self.available = max_tokens - reserved_for_response

    def trim_history(
        self,
        messages: list,
        system_prompt: str,
        current_input: str,
    ) -> list:
        """裁剪对话历史以适应窗口"""
        system_tokens = self._estimate_tokens(system_prompt)
        input_tokens = self._estimate_tokens(current_input)
        budget = self.available - system_tokens - input_tokens

        if budget <= 0:
            # 系统提示太长，只保留当前输入
            return []

        # 从最近的开始保留
        trimmed = []
        total = 0
        for msg in reversed(messages):
            msg_tokens = self._estimate_tokens(msg.content if hasattr(msg, 'content') else str(msg))
            if total + msg_tokens > budget:
                break
            trimmed.insert(0, msg)
            total += msg_tokens

        return trimmed

    def summarize_old_context(
        self,
        messages: list,
        llm: ChatOpenAI,
        keep_recent: int = 6,
    ) -> str:
        """将旧消息总结为摘要"""
        if len(messages) <= keep_recent:
            return ""

        old_messages = messages[:-keep_recent]
        old_text = "\n".join([
            f"&#123;'用户' if isinstance(m, HumanMessage) else '助手'&#125;: &#123;m.content[:200]&#125;"
            for m in old_messages
        ])

        prompt = f"总结以下对话要点（200字以内）：\n&#123;old_text&#125;"
        summary = llm.invoke(prompt).content

        return f"[之前对话摘要] &#123;summary&#125;"

    @staticmethod
    def _estimate_tokens(text: str) -> int:
        chinese = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        other = len(text) - chinese
        return int(chinese * 1.5 + other / 4)
```

## 5. 话题切换与恢复

```python
class TopicManager:
    """话题管理器"""

    def __init__(self):
        self.active_topics: dict[str, TopicContext] = &#123;&#125;  # topic → context
        self.current_topic: str = ""
        self.topic_stack: list[str] = []  # 话题栈（支持嵌套）

    def detect_topic_switch(self, user_input: str, llm: ChatOpenAI) -> str | None:
        """检测是否在切换话题"""
        if not self.current_topic:
            return None

        prompt = ChatPromptTemplate.from_messages([
            ("system", """判断用户是否在切换话题。
当前话题：&#123;current_topic&#125;
用户输入：&#123;user_input&#125;
只输出 "switch" 或 "continue" 或 "return"（回到之前的话题）"""),
            ("human", "&#123;user_input&#125;"),
        ])

        chain = prompt | llm
        result = chain.invoke(&#123;
            "current_topic": self.current_topic,
            "user_input": user_input,
        &#125;)

        action = result.content.strip().lower()
        if action == "switch":
            # 保存当前话题，切换到新的
            if self.current_topic:
                self.topic_stack.append(self.current_topic)
            return "switch"
        elif action == "return":
            # 恢复之前的话题
            if self.topic_stack:
                self.current_topic = self.topic_stack.pop()
                return "return"
        return None

    def get_context_for_topic(self, topic: str) -> TopicContext:
        """获取话题的上下文"""
        return self.active_topics.get(topic, TopicContext(topic=topic))
```

## 6. 对话编排模式

| 模式 | 适用 | 特点 |
|------|------|------|
| 状态机驱动 | 任务型对话 | 明确阶段，槽位填充 |
| 意图驱动 | 开放对话 | 每轮识别意图，动态路由 |
| 话题驱动 | 多话题对话 | 话题切换/恢复 |
| 混合模式 | 生产推荐 | 状态机+意图+话题 |

### 模式选择

```
任务明确（预约/下单）→ 状态机驱动
开放问答（闲聊/搜索）→ 意图驱动
多话题穿插        → 话题驱动
生产环境          → 混合模式
```

## 7. 配置参考

| 配置 | 推荐值 | 说明 |
|------|--------|------|
| 上下文窗口 | 8000-16000 Token | 保留 2000 给响应 |
| 保留最近消息 | 6-10 条 | 最近的对话最相关 |
| 旧消息摘要 | >10 条时触发 | 压缩历史 |
| 槽位超时 | 5 分钟 | 超时后重新收集 |
| 话题栈深度 | 3 层 | 太深用户会迷失 |
| 对话最大轮次 | 50 轮 | 超过自动结束 |
