# RAG 语义分块与自适应分块指南

> 固定大小分块简单粗暴——把一句话从中间切断，语义全丢。这篇指南讲透基于嵌入的语义分块、递归分块和自适应分块大小，让每个 chunk 都有完整的语义边界。

---

## 一、为什么固定分块不够好

```mermaid

    subgraph 固定分块 &#123;"固定500字符分块"&#125;
        F1["Chunk1: ...第三条 甲方应"] --> F2["Chunk2: 在30日内付款..."]
        F1_NOTE["❌ 语义断裂"]
    end

    subgraph 语义分块 &#123;"语义分块"&#125;
        S1["Chunk1: ...第三条 甲方应在30日内付款"]
        S2["Chunk2: 第四条 违约责任..."]
        S1_NOTE["✅ 语义完整"]
    end

    style F1_NOTE fill:#FFCDD2,stroke:#C62828
    style S1_NOTE fill:#C8E6C9
```

固定分块的问题：句子被截断、表格被拆散、标题与内容分离。语义分块通过**检测语义断点**来决定切分位置，确保每个 chunk 是一个完整的语义单元。

## 二、语义分块实现

```python
from dataclasses import dataclass, field
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_core.documents import Document
import numpy as np
from typing import Optional

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
embeddings = OpenAIEmbeddings()

@dataclass
class ChunkConfig:
    """分块配置。"""
    min_chunk_size: int = 100       # 最小分块字符数
    max_chunk_size: int = 800       # 最大分块字符数
    similarity_threshold: float = 0.5  # 语义断点阈值
    overlap: int = 50               # 重叠字符数

class SemanticChunker:
    """语义分块器——基于句子嵌入相似度检测断点。"""

    def __init__(self, embeddings, config: ChunkConfig = ChunkConfig()):
        self.embeddings = embeddings
        self.config = config

    def _split_sentences(self, text: str) -> list[str]:
        """将文本拆分为句子。"""
        import re
        # 按中英文句号、换行、问号、感叹号拆分
        sentences = re.split(r'(?<=[。！？\.\!\?\n])\s*', text)
        return [s.strip() for s in sentences if s.strip()]

    async def _embed_batch(self, sentences: list[str]) -> np.ndarray:
        """批量嵌入句子。"""
        vectors = await self.embeddings.aembed_documents(sentences)
        return np.array(vectors)

    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        """计算余弦相似度。"""
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-8))

    async def chunk(self, text: str) -> list[Document]:
        """语义分块。"""
        sentences = self._split_sentences(text)
        if len(sentences) <= 1:
            return [Document(page_content=text)]

        # 嵌入所有句子
        embeddings = await self._embed_batch(sentences)

        # 计算相邻句子相似度
        chunks = []
        current_sentences = [sentences[0]]
        current_size = len(sentences[0])

        for i in range(1, len(sentences)):
            sim = self._cosine_similarity(embeddings[i-1], embeddings[i])
            sentence = sentences[i]
            sentence_len = len(sentence)

            # 相似度低=语义断点，或者超过最大长度
            if sim < self.config.similarity_threshold or current_size + sentence_len > self.config.max_chunk_size:
                # 检查当前块是否足够大
                if current_size >= self.config.min_chunk_size:
                    chunk_text = "".join(current_sentences)
                    chunks.append(Document(
                        page_content=chunk_text,
                        metadata=&#123;"chunk_type": "semantic", "sentence_count": len(current_sentences)&#125;,
                    ))
                    current_sentences = [sentence]
                    current_size = sentence_len
                else:
                    # 当前块太小，继续追加
                    current_sentences.append(sentence)
                    current_size += sentence_len
            else:
                current_sentences.append(sentence)
                current_size += sentence_len

        # 最后一个块
        if current_sentences:
            chunk_text = "".join(current_sentences)
            chunks.append(Document(
                page_content=chunk_text,
                metadata=&#123;"chunk_type": "semantic", "sentence_count": len(current_sentences)&#125;,
            ))

        return chunks


class RecursiveChunker:
    """递归分块器——按分隔符层级递归切分。"""

    SEPARATORS = [
        "\n\n\n",  # 段落级
        "\n\n",    # 多行段落
        "\n",      # 单行
        "。",      # 中文句号
        "；",      # 中文分号
        "，",      # 中文逗号
        " ",       # 空格
        "",        # 逐字符
    ]

    def __init__(self, config: ChunkConfig = ChunkConfig()):
        self.config = config

    def chunk(self, text: str) -> list[Document]:
        return self._recursive_split(text, self.SEPARATORS)

    def _recursive_split(self, text: str, separators: list[str]) -> list[Document]:
        """递归切分。"""
        if len(text) <= self.config.max_chunk_size:
            return [Document(page_content=text, metadata=&#123;"chunk_type": "recursive"&#125;)]

        if not separators:
            # 无分隔符可用，强制按大小切
            return self._force_split(text)

        sep = separators[0]
        if sep == "":
            return self._force_split(text)

        parts = text.split(sep)
        chunks = []
        current = ""

        for part in parts:
            candidate = current + sep + part if current else part
            if len(candidate) <= self.config.max_chunk_size:
                current = candidate
            else:
                if current and len(current) >= self.config.min_chunk_size:
                    chunks.append(Document(page_content=current, metadata=&#123;"chunk_type": "recursive"&#125;))
                # 递归处理超长部分
                if len(part) > self.config.max_chunk_size:
                    sub_chunks = self._recursive_split(part, separators[1:])
                    chunks.extend(sub_chunks)
                    current = ""
                else:
                    current = part

        if current and len(current) >= self.config.min_chunk_size:
            chunks.append(Document(page_content=current, metadata=&#123;"chunk_type": "recursive"&#125;))

        return chunks

    def _force_split(self, text: str) -> list[Document]:
        """强制按大小切分。"""
        chunks = []
        for i in range(0, len(text), self.config.max_chunk_size):
            chunk = text[i:i + self.config.max_chunk_size]
            chunks.append(Document(page_content=chunk, metadata=&#123;"chunk_type": "force"&#125;))
        return chunks


class AdaptiveChunker:
    """自适应分块器——根据文档类型选择策略。"""

    def __init__(self, semantic: SemanticChunker, recursive: RecursiveChunker):
        self.semantic = semantic
        self.recursive = recursive

    async def chunk(self, text: str, doc_type: str = "auto") -> list[Document]:
        """自适应选择分块策略。"""
        if doc_type == "auto":
            doc_type = self._detect_type(text)

        if doc_type == "qa":
            # QA类文档用语义分块（每个Q&A是一个语义单元）
            return await self.semantic.chunk(text)
        elif doc_type == "code":
            # 代码文档用递归分块（保持代码块完整）
            return self.recursive.chunk(text)
        elif doc_type == "table":
            # 表格类用递归分块（分隔符在换行处）
            return self.recursive.chunk(text)
        else:
            # 默认用语义分块
            return await self.semantic.chunk(text)

    def _detect_type(self, text: str) -> str:
        """简单检测文档类型。"""
        if "```" in text or "def " in text or "function " in text:
            return "code"
        if "|" in text and "---" in text:
            return "table"
        if "问：" in text or "答：" in text or "Q:" in text:
            return "qa"
        return "text"
```

## 三、分块策略对比

| 策略 | 原理 | 优点 | 缺点 | 延迟 | 适用场景 |
|------|------|------|------|------|----------|
| 固定大小 | 每500字符切一刀 | 最快、最简单 | 语义断裂 | 极低 | 快速原型 |
| 递归分块 | 按分隔符层级递归 | 保持段落完整 | 依赖分隔符 | 低 | 通用文档 |
| 语义分块 | 嵌入相似度检测断点 | 语义边界精确 | 需嵌入调用 | 中 | 高质量RAG |
| 自适应分块 | 按文档类型选策略 | 兼顾质量与速度 | 实现复杂 | 中 | 生产环境 |
| 父子分块 | 大块检索，小块送入LLM | 上下文完整 | 存储翻倍 | 中 | 长文档 |

## 四、分块质量评估

```python
@dataclass
class ChunkQuality:
    """分块质量评估。"""
    avg_chunk_size: float
    size_variance: float
    semantic_coherence: float  # 块内语义一致性
    boundary_quality: float    # 边界质量

def evaluate_chunks(chunks: list[Document]) -> ChunkQuality:
    """评估分块质量。"""
    sizes = [len(c.page_content) for c in chunks]
    avg = sum(sizes) / len(sizes) if sizes else 0
    variance = sum((s - avg) ** 2 for s in sizes) / len(sizes) if sizes else 0
    # 块大小越均匀，方差越小
    coherence = 1.0 - min(variance / (avg ** 2 + 1e-8), 1.0) if avg > 0 else 0
    return ChunkQuality(
        avg_chunk_size=round(avg, 1),
        size_variance=round(variance, 1),
        semantic_coherence=round(coherence, 2),
        boundary_quality=0.0,  # 需LLM评估
    )
```

## 五、最佳实践

| 实践 | 说明 | 优先级 |
