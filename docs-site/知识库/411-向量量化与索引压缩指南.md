# 向量量化与索引压缩指南

> 一百万条文档 × 1536 维向量 × float32 = 6GB 内存。这在生产环境是个沉重负担。向量量化通过降低精度来压缩存储，用 5-10% 的精度损失换取 4-32 倍的内存节省，让大规模向量检索变得可行。

---

## 1. 为什么需要向量量化

### 内存问题

```
文档量    维度    精度        内存
10 万     1536   float32    0.6 GB
100 万    1536   float32    6.0 GB
1000 万   1536   float32    60 GB  ← 单机放不下
1 亿      1536   float32    600 GB ← 需要集群
```

### 量化后的对比

```
文档量    维度    精度         内存      召回率    速度
100 万    1536   float32     6.0 GB    100%    基准
100 万    1536   float16     3.0 GB    ~100%   更快
100 万    1536   int8(PQ)    0.4 GB    95%     更快
100 万    1536   binary      0.2 GB    85%     最快
```

## 2. 量化方法详解

### 方法一：float16 半精度

最简单的量化——把 float32 降为 float16，精度几乎无损。

```python
import numpy as np

def quantize_to_float16(vectors: np.ndarray) -> np.ndarray:
    """float32 → float16 量化"""
    return vectors.astype(np.float16)

def dequantize_from_float16(vectors: np.ndarray) -> np.ndarray:
    """float16 → float32 反量化"""
    return vectors.astype(np.float32)

# 使用
original = np.random.randn(1_000_000, 1536).astype(np.float32)
print(f"float32 内存: &#123;original.nbytes / 1024**3:.2f&#125; GB")  # 5.72 GB

quantized = quantize_to_float16(original)
print(f"float16 内存: &#123;quantized.nbytes / 1024**3:.2f&#125; GB")  # 2.86 GB

# 精度损失
dequantized = dequantize_from_float16(quantized)
error = np.mean(np.abs(original - dequantized))
print(f"平均误差: &#123;error:.6f&#125;")  # 极小
```

| 指标 | float32 | float16 |
|------|---------|---------|
| 内存 | 100% | 50% |
| 精度损失 | 0% | < 0.1% |
| 召回率 | 100% | ~100% |
| 适用 | 基准 | 通用推荐 |

### 方法二：标量量化（Scalar Quantization / SQ8）

把 float32 的连续值映射到 0-255 的 uint8，每个维度从 4 字节降到 1 字节。

```python
import numpy as np

class ScalarQuantizer:
    """SQ8 标量量化器"""

    def fit(self, vectors: np.ndarray, sample_size: int = 100_000):
        """从训练数据中学习量化参数"""
        # 采样以加速
        if len(vectors) > sample_size:
            idx = np.random.choice(len(vectors), sample_size, replace=False)
            sample = vectors[idx]
        else:
            sample = vectors

        # 计算每维的最小值和最大值
        self.mins = sample.min(axis=0)  # shape: (dim,)
        self.maxs = sample.max(axis=0)
        # 避免除零
        self.ranges = np.maximum(self.maxs - self.mins, 1e-8)

        return self

    def quantize(self, vectors: np.ndarray) -> np.ndarray:
        """量化为 uint8"""
        # 归一化到 [0, 255]
        normalized = (vectors - self.mins) / self.ranges
        return np.clip(normalized * 255, 0, 255).astype(np.uint8)

    def dequantize(self, quantized: np.ndarray) -> np.ndarray:
        """反量化为 float32"""
        return quantized.astype(np.float32) / 255 * self.ranges + self.mins

    def search(
        self,
        query: np.ndarray,
        quantized_db: np.ndarray,
        top_k: int = 10,
    ) -> list[int]:
        """在量化后的向量上搜索"""
        # 量化查询
        q_norm = (query - self.mins) / self.ranges
        q_uint8 = np.clip(q_norm * 255, 0, 255).astype(np.uint8)

        # 用 uint8 计算距离（更快）
        # L2 距离：sum((a - b)^2)
        diffs = q_uint8.astype(np.int16) - quantized_db.astype(np.int16)
        distances = np.sum(diffs ** 2, axis=1)

        # 返回 top_k 索引
        return np.argsort(distances)[:top_k].tolist()


# 使用
db = np.random.randn(1_000_000, 1536).astype(np.float32)
print(f"原始内存: &#123;db.nbytes / 1024**3:.2f&#125; GB")  # 5.72 GB

quantizer = ScalarQuantizer().fit(db)
db_sq8 = quantizer.quantize(db)
print(f"SQ8 内存: &#123;db_sq8.nbytes / 1024**3:.2f&#125; GB")  # 1.43 GB (4x 压缩)

# 搜索
query = np.random.randn(1536).astype(np.float32)
results = quantizer.search(query, db_sq8, top_k=10)
```

### 方法三：乘积量化（Product Quantization / PQ）

最常用的向量量化方法。把高维向量切分成子向量，每个子向量独立聚类编码。

```python
import numpy as np
from sklearn.cluster import KMeans

class ProductQuantizer:
    """乘积量化器"""

    def __init__(self, m: int = 8, k: int = 256):
        """
        Args:
            m: 子向量数量（将 D 维切分成 m 段）
            k: 每段的聚类中心数（256 = uint8 编码）
        """
        self.m = m
        self.k = k
        self.codebooks: list[np.ndarray] = []
        self.dim: int = 0
        self.sub_dim: int = 0

    def fit(self, vectors: np.ndarray, sample_size: int = 100_000):
        """训练码本"""
        self.dim = vectors.shape[1]
        assert self.dim % self.m == 0, f"维度 &#123;self.dim&#125; 不能被 m=&#123;self.m&#125; 整除"
        self.sub_dim = self.dim // self.m

        # 采样
        if len(vectors) > sample_size:
            idx = np.random.choice(len(vectors), sample_size, replace=False)
            sample = vectors[idx]
        else:
            sample = vectors

        # 对每个子空间训练 KMeans
        self.codebooks = []
        for i in range(self.m):
            start = i * self.sub_dim
            end = start + self.sub_dim
            sub_vectors = sample[:, start:end]

            kmeans = KMeans(n_clusters=self.k, random_state=42, n_init=3)
            kmeans.fit(sub_vectors)
            self.codebooks.append(kmeans.cluster_centers_)

        return self

    def quantize(self, vectors: np.ndarray) -> np.ndarray:
        """编码为 PQ 码（每个向量 m 字节）"""
        n = len(vectors)
        codes = np.zeros((n, self.m), dtype=np.uint8)

        for i in range(self.m):
            start = i * self.sub_dim
            end = start + self.sub_dim
            sub_vectors = vectors[:, start:end]

            # 找最近的聚类中心
            # 距离矩阵: (n, k)
            dists = np.linalg.norm(
                sub_vectors[:, np.newaxis] - self.codebooks[i][np.newaxis, :],
                axis=2
            )
            codes[:, i] = np.argmin(dists, axis=1).astype(np.uint8)

        return codes

    def search(
        self,
        query: np.ndarray,
        codes: np.ndarray,
        top_k: int = 10,
    ) -> list[int]:
        """使用 ADC（Asymmetric Distance Computation）搜索"""
        # 预计算查询与所有码本的距离表: (m, k)
        distance_tables = np.zeros((self.m, self.k), dtype=np.float32)
        for i in range(self.m):
            start = i * self.sub_dim
            end = start + self.sub_dim
            sub_query = query[start:end]
            distance_tables[i] = np.sum(
                (self.codebooks[i] - sub_query) ** 2, axis=1
            )

        # 查表计算每个向量的近似距离
        # codes: (n, m), distance_tables: (m, k)
        # 对每个向量：sum(distance_tables[i, codes[j, i]] for i in range(m))
        distances = np.zeros(len(codes), dtype=np.float32)
        for i in range(self.m):
            distances += distance_tables[i, codes[:, i]]

        return np.argsort(distances)[:top_k].tolist()


# 使用
db = np.random.randn(1_000_000, 1536).astype(np.float32)
# 1536 维 / m=8 = 192 维子向量

pq = ProductQuantizer(m=8, k=256)
pq.fit(db)
codes = pq.quantize(db)
print(f"PQ 内存: &#123;codes.nbytes / 1024**3:.2f&#125; GB")  # 0.001 GB (760x 压缩!)

query = np.random.randn(1536).astype(np.float32)
results = pq.search(query, codes, top_k=10)
```

### 方法四：二值量化（Binary Quantization）

把每个分量量化为 1 bit（正/负），压缩 32 倍，用汉明距离搜索。

```python
import numpy as np

class BinaryQuantizer:
    """二值量化器"""

    def quantize(self, vectors: np.ndarray) -> np.ndarray:
        """量化为二值（packbits 压缩）"""
        # 取符号位：正→1，负→0
        binary = (vectors > 0).astype(np.uint8)
        # 打包为 bit 数组（每 8 个值压缩为 1 字节）
        packed = np.packbits(binary, axis=1)
        return packed

    def search(
        self,
        query: np.ndarray,
        packed_db: np.ndarray,
        top_k: int = 10,
    ) -> list[int]:
        """用汉明距离搜索"""
        query_binary = (query > 0).astype(np.uint8)
        query_packed = np.packbits(query_binary)

        # 汉明距离 = popcount(XOR)
        xor_result = query_packed[np.newaxis, :] ^ packed_db
        # popcount：计算每行中 1 的个数
        hamming = np.unpackbits(xor_result, axis=1).sum(axis=1)

        return np.argsort(hamming)[:top_k].tolist()


# 使用
db = np.random.randn(1_000_000, 1536).astype(np.float32)
bq = BinaryQuantizer()
packed = bq.quantize(db)
print(f"Binary 内存: &#123;packed.nbytes / 1024**3:.2f&#125; GB")  # 0.18 GB (32x 压缩)
```

## 3. 量化方法对比

| 方法 | 压缩比 | 召回率 | 速度 | 内存 | 适用场景 |
|------|--------|--------|------|------|----------|
| float32 | 1x | 100% | 基准 | 6 GB | 小规模精确搜索 |
| float16 | 2x | ~100% | 更快 | 3 GB | 通用推荐 |
| SQ8 | 4x | 95-98% | 快 | 1.5 GB | 中等规模 |
| PQ | 32-128x | 90-95% | 很快 | 0.1-0.5 GB | 大规模召回 |
| Binary | 32x | 80-90% | 最快 | 0.2 GB | 粗筛/海量 |

    @staticmethod

        query_vectors: np.ndarray,

        k: int = 10,
    ) -> dict:
        """对比量化前后的检索质量"""
        results = &#123;"recall": [], "precision": [], "latency_ms": []&#125;

        import time

            relevant = relevant_indices[i]

            # 原始搜索（暴力）
            start = time.time()
            orig_dists = np.dot(original_db, query)
            orig_top_k = set(np.argsort(orig_dists)[-k:].tolist())
            orig_time = (time.time() - start) * 1000

## 6. 配置选择决策

```
文档量 < 10 万     → float32 或 float16（不需要量化）
文档量 10 万-100 万 → float16 或 SQ8
文档量 100 万-1000 万 → SQ8 或 PQ
文档量 > 1000 万   → PQ + 二值粗筛两阶段

延迟敏感           → Binary + float16 两阶段
精度敏感           → SQ8 或 float16
内存受限           → PQ（最高压缩比）
```

| 场景 | 推荐方法 | 理由 |
|------|----------|------|
| 开发环境 | float16 | 简单、精度无损 |
| 中等规模生产 | SQ8 | 4x 压缩、95%+ 召回 |
| 大规模生产 | PQ (m=8) | 32x 压缩、90%+ 召回 |
| 超低延迟 | Binary + float16 | 两阶段、最快 |
| 精度优先 | float16 | 几乎无损 |
