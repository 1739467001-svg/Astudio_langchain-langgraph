## 二、配置中心实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Any, Callable, Optional
import asyncio
import json

class ConfigType(str, Enum):
    FLAG = "flag"        # 布尔开关
    VALUE = "value"      # 动态参数
    JSON = "json"        # 复杂配置
    SECRET = "secret"    # 敏感配置

@dataclass
class ConfigEntry:
    """配置项。"""
    key: str
    config_type: ConfigType
    value: Any
    default: Any
    description: str = ""
    updated_at: str = field(default_factory=lambda: datetime.now().isoformat())
    updated_by: str = "system"
    version: int = 1

    def get_typed_value(self) -> Any:
        """获取类型化值。"""
        if self.config_type == ConfigType.FLAG:
            return bool(self.value)
        elif self.config_type == ConfigType.VALUE:
            return self.value
        elif self.config_type == ConfigType.JSON:
            if isinstance(self.value, str):
                try:
                    return json.loads(self.value)
                except json.JSONDecodeError:
                    return self.default
            return self.value
        elif self.config_type == ConfigType.SECRET:
            return self.value  # 实际应用中应解密
        return self.value

class ConfigCenter:
    """动态配置中心。"""

    def __init__(self, refresh_interval: float = 30.0):
        self._configs: dict[str, ConfigEntry] = &#123;&#125;
        self._callbacks: dict[str, list[Callable]] = &#123;&#125;
        self._local_cache: dict[str, Any] = &#123;&#125;
        self._refresh_interval = refresh_interval
        self._last_refresh: float = 0

    def register(self, key: str, config_type: ConfigType, default: Any, description: str = "") -> ConfigEntry:
        """注册配置项。"""
        entry = ConfigEntry(
            key=key, config_type=config_type,
            value=default, default=default, description=description,
        )
        self._configs[key] = entry
        self._local_cache[key] = entry.get_typed_value()
        return entry

    def get(self, key: str, default: Any = None) -> Any:
        """获取配置值。"""
        if key in self._local_cache:
            return self._local_cache[key]
        return default

    def set(self, key: str, value: Any, updated_by: str = "admin") -> bool:
        """更新配置值——触发回调。"""
        entry = self._configs.get(key)
        if not entry:
            return False

        old_value = entry.value
        entry.value = value
        entry.version += 1
        entry.updated_at = datetime.now().isoformat()
        entry.updated_by = updated_by

        # 更新缓存
        self._local_cache[key] = entry.get_typed_value()

        # 触发回调
        for callback in self._callbacks.get(key, []):
            try:
                callback(old_value, entry.get_typed_value())
            except Exception as e:
                print(f"配置回调失败 &#123;key&#125;: &#123;e&#125;")

        return True

    def on_change(self, key: str, callback: Callable[[Any, Any], None]):
        """注册配置变更回调。"""
        self._callbacks.setdefault(key, []).append(callback)

    def is_enabled(self, key: str) -> bool:
        """快捷检查特征开关。"""
        return self.get(key, False) is True

    def list_configs(self) -> list[dict]:
        """列出所有配置。"""
        return [
            &#123;
                "key": e.key,
                "type": e.config_type.value,
                "value": str(e.value)[:100] if e.config_type == ConfigType.SECRET else e.value,
                "default": e.default,
                "version": e.version,
                "updated_at": e.updated_at,
                "updated_by": e.updated_by,
            &#125;
            for e in self._configs.values()
        ]

    def export_snapshot(self) -> dict:
        """导出配置快照（不含secret）。"""
        return &#123;
            key: &#123;
                "type": e.config_type.value,
                "value": e.value if e.config_type != ConfigType.SECRET else "***",
                "version": e.version,
            &#125;
            for key, e in self._configs.items()
        &#125;


# 全局配置中心
config_center = ConfigCenter()

# 预注册常用配置
config_center.register("model.temperature", ConfigType.VALUE, 0.0, "LLM温度参数")
config_center.register("model.max_tokens", ConfigType.VALUE, 2000, "最大输出Token数")
config_center.register("tool.web_search.enabled", ConfigType.FLAG, True, "网页搜索工具开关")
config_center.register("tool.code_exec.enabled", ConfigType.FLAG, False, "代码执行工具开关")
config_center.register("rag.top_k", ConfigType.VALUE, 5, "RAG检索Top-K")
config_center.register("agent.max_iterations", ConfigType.VALUE, 10, "Agent最大迭代次数")
config_center.register("safety.content_filter", ConfigType.FLAG, True, "内容安全过滤")
config_center.register("route.rules", ConfigType.JSON, &#123;"default": "gpt-4o-mini"&#125;, "路由规则")
config_center.register("api.openai_key", ConfigType.SECRET, "", "OpenAI API Key")
```

### 特征开关使用

```python
from langchain_core.tools import tool
from langgraph.prebuilt import create_react_agent
from langchain_openai import ChatOpenAI

@tool
async def web_search(query: str) -> dict:
    """搜索网页。"""
    return &#123;"results": [f"搜索: &#123;query&#125;"]&#125;

@tool
async def code_execute(code: str) -> dict:
    """执行代码。"""
    return &#123;"output": f"执行结果: &#123;code[:50]&#125;"&#125;

async def build_agent():
    """根据配置中心动态构建Agent。"""
    # 动态读取模型参数
    temperature = config_center.get("model.temperature", 0.0)
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=temperature)

    # 根据特征开关选择工具
    tools = []
    if config_center.is_enabled("tool.web_search.enabled"):
        tools.append(web_search)
    if config_center.is_enabled("tool.code_exec.enabled"):
        tools.append(code_execute)

    # 根据配置选择迭代上限
    max_iter = config_center.get("agent.max_iterations", 10)

    return create_react_agent(llm, tools, prompt="你是智能助手。")

# 注册变更回调
def on_tool_toggle(old, new):
    print(f"工具开关变更: &#123;old&#125; -> &#123;new&#125;，需重建Agent")

config_center.on_change("tool.web_search.enabled", on_tool_toggle)
config_center.on_change("tool.code_exec.enabled", on_tool_toggle)

# 运行时热更新
config_center.set("model.temperature", 0.3, updated_by="admin")
config_center.set("tool.code_exec.enabled", True, updated_by="admin")
config_center.set("rag.top_k", 8, updated_by="admin")
```

## 三、配置类型对比

| 类型 | 值类型 | 热更新 | 用途 | 示例 |
|------|--------|--------|------|------|
| FLAG | 布尔 | 是 | 开关功能 | web_search=true |
| VALUE | 数值/字符串 | 是 | 调参数 | temperature=0.3 |
| JSON | 结构化 | 是 | 复杂配置 | 路由规则 |
| SECRET | 加密字符串 | 受控 | 密钥 | API Key |

## 四、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 预注册+默认值 | 启动时全部注册 | ★★★ |
| 本地缓存+定期刷新 | 减少读取延迟 | ★★★ |
| 变更回调通知 | 配置变更即时生效 | ★★★ |
| 版本号追踪 | 每次更新+1 | ★★☆ |
| Secret脱敏 | 导出时打码 | ★★☆ |
| 配置快照 | 可回滚到任意版本 | ★★☆ |
