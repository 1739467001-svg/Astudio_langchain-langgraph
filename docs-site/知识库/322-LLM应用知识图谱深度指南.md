## 二、实体消歧

```python
from dataclasses import dataclass, field
from collections import defaultdict

@dataclass
class Entity:
    name: str
    entity_type: str
    aliases: list[str] = field(default_factory=list)
    description: str = ""
    confidence: float = 1.0

class EntityDisambiguator:
    """实体消歧器——同名不同实体要区分。"""

    def __init__(self):
        self.entities: dict[str, list[Entity]] = defaultdict(list)

    def add(self, entity: Entity):
        self.entities[entity.name].append(entity)

    def disambiguate(self, name: str, context: str = "") -> Entity | None:
        """根据上下文消歧。"""
        candidates = self.entities.get(name, [])
        if not candidates:
            return None
        if len(candidates) == 1:
            return candidates[0]
        # 简化：用上下文关键词匹配
        best = max(candidates, key=lambda e: sum(1 for w in context.split() if w in e.description))
        return best


class GraphEnhancedRetriever:
    """图谱增强检索——用图谱关系扩展向量结果。"""

    def __init__(self, vectorstore, graph_store):
        self.vector = vectorstore
        self.graph = graph_store

    async def retrieve(self, query: str, k: int = 5) -> list:
        """向量检索+图谱扩展。"""
        # 1. 向量检索
        vec_docs = await self.vector.asimilarity_search(query, k=k)

        # 2. 提取实体并图谱扩展
        for doc in vec_docs:
            entities = self._extract_entities(doc.page_content)
            for entity in entities:
                neighbors = self.graph.get_neighbors(entity)
                # 将邻居信息追加到文档
                if neighbors:
                    extra = "; ".join(f"&#123;n['relation']&#125;→&#123;n['target']&#125;" for n in neighbors[:3])
                    doc.page_content += f"\n[图谱扩展] &#123;entity&#125;: &#123;extra&#125;"

        return vec_docs

    @staticmethod
    def _extract_entities(text: str) -> list[str]:
        """简化：用已知的实体名匹配。"""
        # 实际用NER或LLM提取
        return []
```
