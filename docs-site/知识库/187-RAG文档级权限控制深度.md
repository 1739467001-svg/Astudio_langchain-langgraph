# RAG 文档级权限控制深度

> 企业知识库不是所有文档都能给所有人看——财务文档只有管理层能查、技术文档只有工程师能查。RAG 检索时必须过滤掉用户无权访问的文档。

## 二、权限模型

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Optional

class AccessLevel(str, Enum):
    PUBLIC = "public"        # 所有人可读
    INTERNAL = "internal"    # 内部员工可读
    DEPARTMENT = "department" # 部门内可读
    CONFIDENTIAL = "confidential"  # 机密，需特定权限
    RESTRICTED = "restricted"      # 限制级，需逐文档授权

@dataclass
class DocumentACL:
    """文档访问控制列表。"""
    doc_id: str
    access_level: AccessLevel = AccessLevel.PUBLIC
    allowed_departments: list[str] = field(default_factory=list)
    allowed_users: list[str] = field(default_factory=list)
    excluded_users: list[str] = field(default_factory=list)

@dataclass
class UserContext:
    """用户上下文。"""
    user_id: str
    department: str = ""
    roles: list[str] = field(default_factory=list)  # admin/manager/employee
    accessible_doc_ids: set[str] = field(default_factory=set)  # 逐文档授权

class PermissionChecker:
    """权限检查器。"""

    @staticmethod
    def can_access(user: UserContext, acl: DocumentACL) -> bool:
        """检查用户是否能访问文档。"""
        # 排除列表优先
        if user.user_id in acl.excluded_users:
            return False

        # 公开文档所有人可读
        if acl.access_level == AccessLevel.PUBLIC:
            return True

        # 逐文档授权
        if acl.doc_id in user.accessible_doc_ids:
            return True

        # 管理员可读所有
        if "admin" in user.roles:
            return True

        # 部门级
        if acl.access_level == AccessLevel.DEPARTMENT:
            if user.department in acl.allowed_departments:
                return True

        # 指定用户
        if user.user_id in acl.allowed_users:
            return True

        # 内部级（有角色的员工）
        if acl.access_level == AccessLevel.INTERNAL:
            if user.roles:  # 有任何角色=内部员工
                return True

        return False

    @staticmethod
    def filter_documents(
        user: UserContext,
        docs: list,  # list[Document]
        acl_map: dict[str, DocumentACL],
    ) -> list:
        """过滤出用户有权访问的文档。"""
        result = []
        for doc in docs:
            doc_id = doc.metadata.get("doc_id", "")
            acl = acl_map.get(doc_id)
            if not acl:
                # 无ACL默认公开
                result.append(doc)
            elif PermissionChecker.can_access(user, acl):
                result.append(doc)
        return result
```

## 三、检索前过滤（Metadata 过滤）

```python
class PreRetrievalFilter:
    """检索前过滤：用metadata限制检索范围。"""

    @staticmethod
    def build_filter(user: UserContext) -> dict:
        """根据用户权限构建检索过滤器。"""
        conditions = []

        # 公开文档
        conditions.append(&#123;"access_level": "public"&#125;)

        # 部门文档
        if user.department:
            conditions.append(&#123;
                "access_level": "department",
                "department": user.department,
            &#125;)

        # 管理员看所有
        if "admin" in user.roles:
            return &#123;&#125;  # 无过滤

        # 构建OR条件（向量库的filter语法）
        if len(conditions) == 1:
            return conditions[0]
        return &#123;"$or": conditions&#125;

    @staticmethod
    async def search_with_permission(
        vectorstore,
        query: str,
        user: UserContext,
        k: int = 10,
    ) -> list:
        """带权限的向量检索。"""
        filter_dict = PreRetrievalFilter.build_filter(user)

        # 用filter限制检索范围
        docs = await vectorstore.asimilarity_search(
            query, k=k, filter=filter_dict,
        )
        return docs
```

```python
class PostRetrievalVerifier:
    """检索后权限验证器。"""

    def __init__(self, acl_map: dict[str, DocumentACL]):
        self.acl_map = acl_map

    def verify_and_filter(
        self,
        user: UserContext,
        docs: list,
        min_results: int = 3,
    ) -> list:
        """验证并过滤。"""
        filtered = PermissionChecker.filter_documents(
            user, docs, self.acl_map
        )

        # 如果过滤后结果太少，记录警告
        if len(filtered) < min_results:
            # 可触发扩展检索
            pass

        return filtered
```

## 五、权限缓存

```python
from datetime import datetime, timedelta

class PermissionCache:
    """权限缓存——避免每次检索都查ACL。"""

    def __init__(self, ttl_seconds: int = 300):
        self.cache: dict[str, tuple[bool, datetime]] = &#123;&#125;
        self.ttl = timedelta(seconds=ttl_seconds)

    def get(self, user_id: str, doc_id: str) -> bool | None:
        """获取缓存的权限。"""
        key = f"&#123;user_id&#125;:&#123;doc_id&#125;"
        if key in self.cache:
            allowed, cached_at = self.cache[key]
            if datetime.now() - cached_at < self.ttl:
                return allowed
            else:
                del self.cache[key]
        return None

    def set(self, user_id: str, doc_id: str, allowed: bool):
        """缓存权限。"""
        key = f"&#123;user_id&#125;:&#123;doc_id&#125;"
        self.cache[key] = (allowed, datetime.now())

    def invalidate_user(self, user_id: str):
        """失效某用户的所有缓存。"""
        keys_to_remove = [k for k in self.cache if k.startswith(f"&#123;user_id&#125;:")]
        for k in keys_to_remove:
            del self.cache[k]
```

## 六、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 检索前用metadata过滤 | 减少无权限文档的计算 | ★★★ |
| 检索后二次验证 | 防止metadata被绕过 | ★★★ |
| 权限缓存 | 减少ACL查询开销 | ★★☆ |
| 权限变更时清缓存 | 保证一致性 | ★★★ |
| 文档入库时打权限标签 | metadata中记录access_level | ★★★ |
