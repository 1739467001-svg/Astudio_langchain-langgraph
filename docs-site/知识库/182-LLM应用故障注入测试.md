## 三、实现

```python
import asyncio
import time
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, Callable, Awaitable

class FaultType(str, Enum):
    LLM_DELAY = "llm_delay"
    LLM_ERROR = "llm_error"
    LLM_HALLUCINATION = "llm_hallucination"
    TOOL_TIMEOUT = "tool_timeout"
    TOOL_ERROR = "tool_error"
    FORMAT_ERROR = "format_error"
    VECTOR_DB_DOWN = "vector_db_down"
    MEMORY_PRESSURE = "memory_pressure"

@dataclass
class FaultInjection:
    """故障注入配置。"""
    fault_type: FaultType
    target: str               # 注入目标
    params: dict = field(default_factory=dict)
    duration_seconds: float = 30
    blast_radius: float = 1.0  # 影响比例

@dataclass
class FaultTestResult:
    """故障测试结果。"""
    fault: FaultInjection
    passed: bool              # 系统是否正确处理了故障
    response_time_ms: float
    error_handled: bool       # 错误是否被优雅处理
    fallback_triggered: bool  # 是否触发了降级
    details: str = ""

class FaultInjector:
    """故障注入器。"""

    def __init__(self):
        self.active_faults: dict[str, FaultInjection] = &#123;&#125;
        self.results: list[FaultTestResult] = []

    def inject(self, fault: FaultInjection):
        """激活故障注入。"""
        self.active_faults[fault.target] = fault

    def remove(self, target: str):
        """移除故障注入。"""
        self.active_faults.pop(target, None)

    async def execute_with_fault(
        self,
        target: str,
        normal_func: Callable,
        *args,
        **kwargs,
    ) -> Any:
        """在故障环境下执行函数。"""
        fault = self.active_faults.get(target)

        if not fault:
            return await normal_func(*args, **kwargs)

        start = time.time()

        try:
            if fault.fault_type == FaultType.LLM_DELAY:
                # 注入延迟
                delay = fault.params.get("delay", 5)
                await asyncio.sleep(delay)
                result = await normal_func(*args, **kwargs)

            elif fault.fault_type == FaultType.LLM_ERROR:
                # 注入错误
                raise Exception(f"[FAULT] &#123;fault.params.get('error', 'Injected error')&#125;")

            elif fault.fault_type == FaultType.TOOL_TIMEOUT:
                # 工具超时
                timeout = fault.params.get("timeout", 30)
                await asyncio.sleep(timeout)
                raise asyncio.TimeoutError(f"[FAULT] Tool timeout (&#123;timeout&#125;s)")

            elif fault.fault_type == FaultType.FORMAT_ERROR:
                # 格式错误
                return fault.params.get("bad_response", "这不是有效的JSON格式")

            elif fault.fault_type == FaultType.LLM_HALLUCINATION:
                # 幻觉注入
                return fault.params.get("hallucination", "地球是平的，这是科学事实。")

            elif fault.fault_type == FaultType.VECTOR_DB_DOWN:
                # 向量库不可用
                raise ConnectionError("[FAULT] Vector database unavailable")

            else:
                result = await normal_func(*args, **kwargs)

            elapsed = (time.time() - start) * 1000
            self._record_result(fault, True, elapsed, True, False, "正常执行")
            return result

        except Exception as e:
            elapsed = (time.time() - start) * 1000
            self._record_result(fault, False, elapsed, False, False, str(e))
            raise

    def _record_result(
        self, fault, passed, elapsed, handled, fallback, details
    ):
        self.results.append(FaultTestResult(
            fault=fault, passed=passed, response_time_ms=round(elapsed, 2),
            error_handled=handled, fallback_triggered=fallback, details=details,
        ))

    def summary(self) -> dict:
        """测试摘要。"""
        total = len(self.results)
        passed = sum(1 for r in self.results if r.passed)
        return &#123;
            "total_tests": total,
            "passed": passed,
            "failed": total - passed,
            "pass_rate": round(passed / total, 4) if total else 0,
            "results": [
                &#123;
                    "fault": r.fault.fault_type.value,
                    "passed": r.passed,
                    "latency_ms": r.response_time_ms,
                    "error_handled": r.error_handled,
                    "fallback": r.fallback_triggered,
                &#125;
                for r in self.results
            ],
        &#125;
```

## 四、预定义测试套件

```python
class FaultTestSuite:
    """预定义故障测试套件。"""

    @staticmethod
    def get_test_cases() -> list[FaultInjection]:
        return [
            FaultInjection(
                fault_type=FaultType.LLM_DELAY,
                target="llm",
                params=&#123;"delay": 10&#125;,
                duration_seconds=30,
            ),
            FaultInjection(
                fault_type=FaultType.LLM_ERROR,
                target="llm",
                params=&#123;"error": "429 Rate Limited"&#125;,
                duration_seconds=10,
            ),
            FaultInjection(
                fault_type=FaultType.TOOL_TIMEOUT,
                target="tool:search",
                params=&#123;"timeout": 30&#125;,
                duration_seconds=30,
            ),
            FaultInjection(
                fault_type=FaultType.FORMAT_ERROR,
                target="llm",
                params=&#123;"bad_response": "这不是JSON"&#125;,
                duration_seconds=10,
            ),
            FaultInjection(
                fault_type=FaultType.LLM_HALLUCINATION,
                target="llm",
                params=&#123;"hallucination": "这是一个编造的事实"&#125;,
                duration_seconds=10,
            ),
            FaultInjection(
                fault_type=FaultType.VECTOR_DB_DOWN,
                target="vector_db",
                params=&#123;&#125;,
                duration_seconds=30,
            ),
        ]

    @staticmethod
    async def run_suite(
        injector: FaultInjector,
        test_func: Callable,
    ) -> dict:
        """运行完整测试套件。"""
        cases = FaultTestSuite.get_test_cases()
        results = []

        for case in cases:
            injector.inject(case)
            try:
                await injector.execute_with_fault(case.target, test_func)
            except Exception:
                pass  # 预期可能失败
            finally:
                injector.remove(case.target)
                await asyncio.sleep(1)

        return injector.summary()
```

## 五、稳态验证

```python
class SteadyStateChecker:
    """稳态验证器。"""

    @staticmethod
    def check(metrics: dict) -> dict:
        """检查系统在故障下是否保持稳态。"""
        checks = &#123;
            "response_returned": metrics.get("has_response", False),
            "no_crash": metrics.get("crashed", False) is False,
            "latency_acceptable": metrics.get("latency_ms", 0) < 30000,
            "error_handled": metrics.get("error_handled", False),
            "no_data_loss": metrics.get("data_lost", False) is False,
        &#125;

        all_passed = all(checks.values())

        return &#123;
            "steady_state_maintained": all_passed,
            "checks": checks,
            "failed_checks": [k for k, v in checks.items() if not v],
        &#125;
```

## 六、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 故障注入有明确标准 | 通过/失败标准清晰 | ★★★ |
| 先在预发环境测试 | 生产环境需授权 | ★★★ |
| 每次只注入一个故障 | 便于归因 | ★★★ |
| 必须有自动恢复 | 注入后自动移除 | ★★☆ |
| 记录完整结果 | 便于分析 | ★★☆ |
| 定期执行 | 系统变更后重测 | ★★☆ |
