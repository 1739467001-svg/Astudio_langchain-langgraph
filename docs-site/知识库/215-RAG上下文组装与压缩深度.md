## 二、实现

```python
from langchain_core.documents import Document
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
from dataclasses import dataclass
import tiktoken

@dataclass
class ContextAssemblyConfig:
    """上下文组装配置。"""
    max_tokens: int = 3000
    max_documents: int = 5
    include_sources: bool = True
    compress: bool = True
    separator: str = "\n\n---\n\n"

class ContextAssembler:
    """上下文组装器。"""

    def __init__(self, config: ContextAssemblyConfig = ContextAssemblyConfig()):
        self.config = config

    def assemble(self, documents: list[Document], query: str = "") -> str:
        """组装上下文文本。"""
        # 1. 按相关度排序
        docs = sorted(
            documents,
            key=lambda d: d.metadata.get("relevance_score", 0),
            reverse=True,
        )

        # 2. 截断到最大文档数
        docs = docs[:self.config.max_documents]

        # 3. Token预算控制
        assembled = []
        used_tokens = 0

        for doc in docs:
            content = doc.page_content
            doc_tokens = self._count_tokens(content)

            if used_tokens + doc_tokens > self.config.max_tokens:
                # 截断当前文档
                remaining = self.config.max_tokens - used_tokens
                if remaining > 100:
                    content = self._truncate_to_tokens(content, remaining)
                    assembled.append(self._format_doc(content, doc))
                break

            assembled.append(self._format_doc(content, doc))
            used_tokens += doc_tokens

        return self.config.separator.join(assembled)

    def _format_doc(self, content: str, doc: Document) -> str:
        """格式化单个文档。"""
        if self.config.include_sources:
            source = doc.metadata.get("source", "未知")
            return f"[来源: &#123;source&#125;]\n&#123;content&#125;"
        return content

    @staticmethod
    def _count_tokens(text: str) -> int:
        """估算Token数。"""
        try:
            encoder = tiktoken.encoding_for_model("gpt-4o")
            return len(encoder.encode(text))
        except Exception:
            return len(text) // 3

    @classmethod
    def _truncate_to_tokens(cls, text: str, max_tokens: int) -> str:
        """截断到指定Token数。"""
        try:
            encoder = tiktoken.encoding_for_model("gpt-4o")
            tokens = encoder.encode(text)[:max_tokens]
            return encoder.decode(tokens) + "\n[内容已截断]"
        except Exception:
            return text[:max_tokens * 3] + "..."


class ContextCompressor:
    """上下文压缩器——用LLM压缩。"""

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def compress(self, context: str, query: str) -> str:
        """压缩上下文。"""
        if self._count_tokens(context) < 1000:
            return context  # 不需要压缩

        prompt = f"""请压缩以下信息，只保留与问题相关的内容。保持事实准确。

问题: &#123;query&#125;

原始信息:
&#123;context[:3000]&#125;

压缩后（保留关键信息）:"""

        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content

    @staticmethod
    def _count_tokens(text: str) -> int:
        try:
            encoder = tiktoken.encoding_for_model("gpt-4o")
            return len(encoder.encode(text))
        except Exception:
            return len(text) // 3
```
