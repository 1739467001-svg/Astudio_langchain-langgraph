## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from langchain_core.documents import Document

@dataclass
class MetadataSchema:
    """元数据Schema——标准化的元数据字段。"""
    doc_id: str = ""
    source: str = ""
    doc_type: str = ""           # manual/policy/api/faq
    access_level: str = "public" # public/internal/confidential
    department: str = ""
    tags: list[str] = field(default_factory=list)
    created_at: str = ""
    updated_at: str = ""
    language: str = "zh"
    version: str = "1.0"
    author: str = ""

class MetadataManager:
    """元数据管理器。"""

    def __init__(self):
        self.metadata_store: dict[str, dict] = &#123;&#125;

    def tag_document(self, doc: Document, metadata: dict):
        """标注文档元数据。"""
        doc_id = metadata.get("doc_id", str(id(doc))
        self.metadata_store[doc_id] = metadata
        doc.metadata.update(metadata)
        return doc

    def build_filter(self, user_context: dict) -> dict:
        """根据用户上下文构建元数据过滤器。"""
        conditions = [&#123;"access_level": "public"&#125;]
        if user_context.get("department"):
            conditions.append(&#123;
                "access_level": "department",
                "department": user_context["department"],
            &#125;)
        if "admin" in user_context.get("roles", []):
            return &#123;&#125;  # 管理员无过滤
        return &#123;"$or": conditions&#125; if len(conditions) > 1 else conditions[0]

    def get_metadata(self, doc_id: str) -> dict:
        """获取文档元数据。"""
        return self.metadata_store.get(doc_id, &#123;&#125;)

    def search_by_metadata(self, filters: dict) -> list[str]:
        """按元数据搜索文档ID。"""
        results = []
        for doc_id, meta in self.metadata_store.items():
            match = all(meta.get(k) == v for k, v in filters.items())
            if match:
                results.append(doc_id)
        return results
```

## 三、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
