## 三、效果验证

```python
class ColdStartValidator:
    """冷启动效果验证器。"""

    MIN_STANDARDS = &#123;
        "min_docs": 50,
        "min_recall": 0.7,
        "min_faithfulness": 0.8,
        "min_coverage": 0.8,  # 测试问题覆盖率
    &#125;

    @classmethod
    def check_ready(cls, stats: dict) -> dict:
        """检查是否准备好上线。"""
        checks = &#123;
            "文档数量": stats.get("doc_count", 0) >= cls.MIN_STANDARDS["min_docs"],
            "召回率": stats.get("recall", 0) >= cls.MIN_STANDARDS["min_recall"],
            "忠实度": stats.get("faithfulness", 0) >= cls.MIN_STANDARDS["min_faithfulness"],
            "覆盖率": stats.get("coverage", 0) >= cls.MIN_STANDARDS["min_coverage"],
        &#125;
        return &#123;
            "ready": all(checks.values()),
            "checks": checks,
            "action": "可以上线" if all(checks.values()) else "需要补充数据",
        &#125;
```
