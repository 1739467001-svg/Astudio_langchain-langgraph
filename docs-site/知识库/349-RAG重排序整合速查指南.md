## 二、快速实现

```python
class RerankSpeedGuide:
    """重排序速查。"""

    @staticmethod
    def cohere(api_key: str):
        """Cohere API方案。"""
        import cohere
        client = cohere.Client(api_key)
        def rerank(query, docs, k=5):
            resp = client.rerank(model="rerank-multilingual-v3.0", query=query, documents=docs, top_n=k)
            return [&#123;"index": r.index, "score": r.relevance_score&#125; for r in resp.results]
        return rerank

    @staticmethod
    def bge_local():
        """BGE本地方案。"""
        from sentence_transformers import CrossEncoder
        model = CrossEncoder("BAAI/bge-reranker-large")
        def rerank(query, docs, k=5):
            pairs = [[query, d] for d in docs]
            scores = model.predict(pairs)
            ranked = sorted(enumerate(scores), key=lambda x: x[1], reverse=True)
            return [&#123;"index": i, "score": float(s)&#125; for i, s in ranked[:k]]
        return rerank

    @staticmethod
    def two_stage(vectorstore, reranker, query, recall_k=20, final_k=5):
        """两阶段检索。"""
        # 阶段1: 向量召回
        docs = vectorstore.similarity_search(query, k=recall_k)
        if len(docs) <= final_k:
            return docs
        # 阶段2: 重排序
        texts = [d.page_content for d in docs]
        ranked = reranker(query, texts, k=final_k)
        return [docs[r["index"]] for r in ranked]
```
