## 三、实现

### 3.1 时间戳检测

```python
from datetime import datetime, timedelta
from langchain_core.documents import Document

class DocumentFreshnessChecker:
    """文档新鲜度检查器"""
    def __init__(self, max_age_days: int = 90):
        self.max_age_days = max_age_days

    def check_freshness(self, docs: list[Document]) -> dict:
        """检查文档新鲜度"""
        now = datetime.now()
        cutoff = now - timedelta(days=self.max_age_days)

        fresh = []
        stale = []

        for doc in docs:
            updated = doc.metadata.get("updated_at", "")
            if updated:
                try:
                    doc_date = datetime.fromisoformat(updated)
                    if doc_date < cutoff:
                        doc.metadata["stale"] = True
                        stale.append(doc)
                    else:
                        fresh.append(doc)
                except ValueError:
                    fresh.append(doc)  # 无法解析日期→当作新鲜
            else:
                fresh.append(doc)  # 无日期→当作新鲜

        return &#123;
            "fresh_count": len(fresh),
            "stale_count": len(stale),
            "stale_docs": [
                &#123;"source": d.metadata.get("source"), "updated": d.metadata.get("updated_at")&#125;
                for d in stale
            ],
        &#125;

    def filter_fresh(self, docs: list[Document]) -> list[Document]:
        """只返回新鲜文档"""
        result = self.check_freshness(docs)
        return [d for d in docs if not d.metadata.get("stale", False)]
```

### 3.2 版本对比检测

```python
import hashlib
import os
import json

class DocumentVersionTracker:
    """文档版本跟踪器"""
    def __init__(self, manifest_path: str = "data/doc_versions.json"):
        self.manifest_path = manifest_path
        self.versions = self._load_manifest()

    def _load_manifest(self) -> dict:
        try:
            with open(self.manifest_path) as f:
                return json.load(f)
        except FileNotFoundError:
            return &#123;&#125;

    def _save_manifest(self):
        with open(self.manifest_path, "w") as f:
            json.dump(self.versions, f, indent=2, ensure_ascii=False)

    def _file_hash(self, filepath: str) -> str:
        """计算文件hash"""
        with open(filepath, "rb") as f:
            return hashlib.md5(f.read()).hexdigest()

    def check_changes(self, source_dir: str) -> dict:
        """检查哪些文档发生了变化"""
        changed = []
        new = []
        deleted = []

        current_files = &#123;&#125;
        for filename in os.listdir(source_dir):
            filepath = os.path.join(source_dir, filename)
            if os.path.isfile(filepath):
                current_files[filename] = self._file_hash(filepath)

        # 检查变化和新增
        for filename, file_hash in current_files.items():
            if filename not in self.versions:
                new.append(filename)
            elif self.versions[filename] != file_hash:
                changed.append(filename)

        # 检查删除
        for filename in self.versions:
            if filename not in current_files:
                deleted.append(filename)

        # 更新版本
        self.versions = current_files
        self._save_manifest()

        return &#123;"changed": changed, "new": new, "deleted": deleted&#125;
```
