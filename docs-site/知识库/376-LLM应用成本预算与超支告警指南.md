## 二、成本预算实现

```python
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import Optional
import asyncio
from collections import defaultdict

class AlertLevel(str, Enum):
    INFO = "info"         # <60%
    WARNING = "warning"   # 60-80%
    CRITICAL = "critical" # 80-100%
    EXCEEDED = "exceeded" # >100%

class DegradeAction(str, Enum):
    NONE = "none"
    MODEL_DOWNGRADE = "model_downgrade"
    USE_CACHE = "use_cache"
    REJECT = "reject"

@dataclass
class ModelPricing:
    """模型定价。"""
    model_name: str
    input_cost_per_1k: float   # 每千Token输入成本（美元）
    output_cost_per_1k: float  # 每千Token输出成本
    cached_discount: float = 0.5  # 缓存命中折扣

    def calculate(self, input_tokens: int, output_tokens: int, cached: bool = False) -> float:
        cost = (input_tokens / 1000 * self.input_cost_per_1k +
                output_tokens / 1000 * self.output_cost_per_1k)
        if cached:
            cost *= self.cached_discount
        return round(cost, 6)

@dataclass
class CostRecord:
    """成本记录。"""
    timestamp: str
    model: str
    input_tokens: int
    output_tokens: int
    cost_usd: float
    cached: bool = False

@dataclass
class BudgetConfig:
    """预算配置。"""
    daily_budget_usd: float = 50.0
    monthly_budget_usd: float = 1000.0
    warning_threshold: float = 0.6   # 60%告警
    critical_threshold: float = 0.8   # 80%严重
    exceed_action: DegradeAction = DegradeAction.MODEL_DOWNGRADE
    downgrade_model: str = "gpt-4o-mini"
    alert_webhook: str = ""

PRICING = &#123;
    "gpt-4o": ModelPricing("gpt-4o", 0.0025, 0.01),
    "gpt-4o-mini": ModelPricing("gpt-4o-mini", 0.00015, 0.0006),
    "gpt-4-turbo": ModelPricing("gpt-4-turbo", 0.01, 0.03),
    "gpt-3.5-turbo": ModelPricing("gpt-3.5-turbo", 0.0005, 0.0015),
&#125;


class CostBudgetManager:
    """成本预算管理器。"""

    def __init__(self, config: BudgetConfig = BudgetConfig()):
        self.config = config
        self._daily_spend: float = 0.0
        self._monthly_spend: float = 0.0
        self._records: list[CostRecord] = []
        self._by_model: dict[str, float] = defaultdict(float)
        self._alert_history: list[dict] = []
        self._last_alert_level: AlertLevel = AlertLevel.INFO
        self._date = datetime.now().date()

    def _check_date_reset(self):
        """日重置。"""
        today = datetime.now().date()
        if today != self._date:
            self._daily_spend = 0.0
            self._date = today
        # 月重置
        if today.day == 1 and self._daily_spend > 0:
            self._monthly_spend = 0.0

    def check_budget(self, estimated_cost: float = 0.0) -> dict:
        """检查预算。"""
        self._check_date_reset()
        daily_pct = (self._daily_spend + estimated_cost) / self.config.daily_budget_usd
        monthly_pct = (self._monthly_spend + estimated_cost) / self.config.monthly_budget_usd

        level = AlertLevel.INFO
        action = DegradeAction.NONE

        if daily_pct >= 1.0 or monthly_pct >= 1.0:
            level = AlertLevel.EXCEEDED
            action = self.config.exceed_action
        elif daily_pct >= self.config.critical_threshold or monthly_pct >= self.config.critical_threshold:
            level = AlertLevel.CRITICAL
        elif daily_pct >= self.config.warning_threshold or monthly_pct >= self.config.warning_threshold:
            level = AlertLevel.WARNING

        return &#123;
            "allowed": level != AlertLevel.EXCEEDED or action != DegradeAction.REJECT,
            "level": level.value,
            "action": action.value,
            "daily_spend": round(self._daily_spend, 4),
            "daily_budget": self.config.daily_budget_usd,
            "daily_pct": round(daily_pct * 100, 1),
            "monthly_spend": round(self._monthly_spend, 4),
            "monthly_pct": round(monthly_pct * 100, 1),
            "downgrade_model": self.config.downgrade_model if action == DegradeAction.MODEL_DOWNGRADE else None,
        &#125;

    def record(self, model: str, input_tokens: int, output_tokens: int, cached: bool = False):
        """记录一次调用成本。"""
        self._check_date_reset()
        pricing = PRICING.get(model)
        if not pricing:
            return

        cost = pricing.calculate(input_tokens, output_tokens, cached)
        self._daily_spend += cost
        self._monthly_spend += cost
        self._by_model[model] += cost

        self._records.append(CostRecord(
            timestamp=datetime.now().isoformat(),
            model=model,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            cost_usd=cost,
            cached=cached,
        ))

        # 检查告警
        self._check_alerts()

    def _check_alerts(self):
        """检查并触发告警。"""
        check = self.check_budget()
        level = AlertLevel(check["level"])

        if level != self._last_alert_level:
            self._alert_history.append(&#123;
                "timestamp": datetime.now().isoformat(),
                "level": level.value,
                "daily_pct": check["daily_pct"],
                "monthly_pct": check["monthly_pct"],
                "message": f"预算告警: 日&#123;check['daily_pct']&#125;%, 月&#123;check['monthly_pct']&#125;%",
            &#125;)
            self._last_alert_level = level

    def get_report(self) -> dict:
        """获取成本报告。"""
        check = self.check_budget()
        return &#123;
            **check,
            "by_model": &#123;k: round(v, 4) for k, v in self._by_model.items()&#125;,
            "total_calls": len(self._records),
            "avg_cost_per_call": round(self._daily_spend / max(len(self._records), 1), 6),
            "alert_history": self._alert_history[-5:],
        &#125;


class CostAwareLLMWrapper:
    """成本感知的LLM调用包装器。"""

    def __init__(self, llm, budget_manager: CostBudgetManager, model_name: str):
        self.llm = llm
        self.budget = budget_manager
        self.model_name = model_name

    async def invoke(self, messages: list) -> dict:
        """带预算检查的LLM调用。"""
        # 预估成本（粗略：输入Token / 4）
        est_input = sum(len(str(m.get("content", ""))) // 4 for m in messages)
        pricing = PRICING.get(self.model_name)
        est_cost = pricing.calculate(est_input, 500) if pricing else 0.01

        check = self.budget.check_budget(est_cost)

        if not check["allowed"]:
            return &#123;"error": "budget_exceeded", "level": check["level"], "message": "预算超限"&#125;

        # 降级模型
        model_to_use = self.model_name
        if check.get("downgrade_model"):
            model_to_use = check["downgrade_model"]

        # 执行调用
        actual_model = ChatOpenAI(model=model_to_use, temperature=0)
        response = await actual_model.ainvoke(messages)

        # 记录实际成本
        input_tokens = est_input
        output_tokens = len(str(response.content)) // 4
        self.budget.record(model_to_use, input_tokens, output_tokens)

        return &#123;"response": response, "model_used": model_to_use, "budget_check": check&#125;
```

## 三、使用示例

```python
from langchain_openai import ChatOpenAI
import asyncio

async def main():
    budget = CostBudgetManager(BudgetConfig(
        daily_budget_usd=5.0,      # 日预算5美元
        monthly_budget_usd=100.0,
        warning_threshold=0.6,
        critical_threshold=0.8,
        exceed_action=DegradeAction.MODEL_DOWNGRADE,
        downgrade_model="gpt-4o-mini",
    ))

    llm = ChatOpenAI(model="gpt-4o", temperature=0)
    wrapper = CostAwareLLMWrapper(llm, budget, "gpt-4o")

    # 模拟多次调用
    for i in range(5):
        result = await wrapper.invoke([&#123;"role": "user", "content": f"问题&#123;i&#125;: 什么是AI?"&#125;])
        if "error" in result:
            print(f"调用&#123;i&#125;: 被拦截 - &#123;result['message']&#125;")
        else:
            print(f"调用&#123;i&#125;: 模型=&#123;result['model_used']&#125;, 预算=&#123;result['budget_check']['daily_pct']&#125;%")

    print(f"\n成本报告: &#123;budget.get_report()&#125;")

asyncio.run(main())
```
