## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum

class KnowledgeType(str, Enum):
    FAQ = "faq"
    BEST_PRACTICE = "best_practice"
    FAILURE_CASE = "failure_case"
    TERMINOLOGY = "terminology"

@dataclass
class KnowledgeEntry:
    id: str
    type: KnowledgeType
    title: str
    content: str
    tags: list[str] = field(default_factory=list)
    usage_count: int = 0
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())

class KnowledgeBase:
    """知识沉淀库。"""

    def __init__(self):
        self.entries: dict[str, KnowledgeEntry] = &#123;&#125;

    def add(self, entry: KnowledgeEntry):
        self.entries[entry.id] = entry

    def search(self, query: str, k: int = 3) -> list[KnowledgeEntry]:
        """搜索（关键词匹配）。"""
        query_lower = query.lower()
        scored = []
        for entry in self.entries.values():
            score = 0
            if query_lower in entry.title.lower(): score += 3
            if query_lower in entry.content.lower(): score += 2
            if any(query_lower in t.lower() for t in entry.tags): score += 1
            if score > 0: scored.append((entry, score))
        scored.sort(key=lambda x: x[1], reverse=True)
        return [e for e, _ in scored[:k]]

    def record_usage(self, entry_id: str):
        if entry_id in self.entries:
            self.entries[entry_id].usage_count += 1

    def get_context_for_prompt(self, query: str) -> str:
        """为Prompt提供知识上下文。"""
        results = self.search(query)
        if not results: return ""
        parts = ["## 相关知识"]
        for entry in results:
            parts.append(f"### &#123;entry.title&#125;\n&#123;entry.content[:200]&#125;")
        return "\n\n".join(parts)


class KnowledgeExtractor:
    """知识提取器——从交互中自动沉淀。"""

    EXTRACT_PROMPT = """从以下对话中提取可沉淀的知识。

对话:
&#123;conversation&#125;

输出JSON数组:
```json
[&#123;&#123;"type": "faq", "title": "...", "content": "...", "tags": ["..."]&#125;&#125;]
```
无则返回空数组。"""

    @staticmethod
    async def extract(llm, conversation: str) -> list[dict]:
        from langchain_core.messages import HumanMessage
        import re, json
        prompt = KnowledgeExtractor.EXTRACT_PROMPT.format(conversation=conversation[:1000])
        response = await llm.ainvoke([HumanMessage(content=prompt)])
        match = re.search(r'\[.*\]', response.content, re.DOTALL)
        if match:
            return json.loads(match.group())
        return []
```
