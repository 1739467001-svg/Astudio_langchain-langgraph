## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from pathlib import Path
import yaml, time

@dataclass
class DynamicConfig:
    """动态配置。"""
    model: str = "gpt-4o-mini"
    temperature: float = 0.7
    max_tokens: int = 1000
    cache_enabled: bool = True
    cache_ttl: int = 3600
    rate_limit_qps: int = 10
    rag_k: int = 5
    rerank: bool = True

class HotReloadConfigManager:
    """热更新配置管理器。"""

    def __init__(self, config_path: str):
        self.path = Path(config_path)
        self.config: DynamicConfig = self._load()
        self._last_modified: float = self.path.stat().st_mtime
        self._callbacks: list = []

    def _load(self) -> DynamicConfig:
        """加载配置。"""
        if not self.path.exists():
            return DynamicConfig()
        with open(self.path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f) or &#123;&#125;
        return DynamicConfig(**data)

    def on_change(self, callback):
        """注册变更回调。"""
        self._callbacks.append(callback)

    def check_and_reload(self) -> bool:
        """检查文件变更并热加载。"""
        current_mtime = self.path.stat().st_mtime
        if current_mtime > self._last_modified:
            old = self.config
            self.config = self._load()
            self._last_modified = current_mtime
            for cb in self._callbacks:
                try:
                    cb(old, self.config)
                except Exception:
                    pass
            return True
        return False

    def get(self) -> DynamicConfig:
        return self.config


class FeatureFlagManager:
    """特性开关管理器——秒级控制功能启停。"""

    def __init__(self):
        self.flags: dict[str, dict] = &#123;&#125;

    def set(self, name: str, enabled: bool, percentage: float = 1.0):
        self.flags[name] = &#123;"enabled": enabled, "percentage": percentage&#125;

    def is_enabled(self, name: str, user_id: str = "") -> bool:
        flag = self.flags.get(name)
        if not flag or not flag["enabled"]:
            return False
        if flag["percentage"] >= 1.0:
            return True
        import hashlib
        h = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 100
        return h < flag["percentage"] * 100

    def disable(self, name: str):
        """秒级禁用（不需重新部署）。"""
        if name in self.flags:
            self.flags[name]["enabled"] = False
```
