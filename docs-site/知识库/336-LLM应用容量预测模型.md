## 二、实现

```python
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from collections import defaultdict

@dataclass
class CapacityForecast:
    """容量预测结果。"""
    predicted_qps: float
    predicted_daily_tokens: int
    predicted_daily_cost: float
    current_capacity_qps: float
    needs_scale: bool
    scale_by: float = 1.0  # 需要扩容倍数
    days_until_limit: int = 0  # 距离容量上限天数

class CapacityPredictor:
    """容量预测器——基于历史数据预测未来需求。"""

    def __init__(self, history: list[dict] = None):
        self.history = history or []

    def predict(self, weeks: int = 4) -> CapacityForecast:
        """预测未来N周容量需求。"""
        if len(self.history) < 7:
            return CapacityForecast(
                predicted_qps=10, predicted_daily_tokens=500000,
                predicted_daily_cost=0.5, current_capacity_qps=100,
                needs_scale=False,
            )

        # 1. 计算增长趋势
        recent = self.history[-7:]
        older = self.history[-14:-7] if len(self.history) >= 14 else recent
        avg_recent = sum(h.get("qps", 0) for h in recent) / len(recent)
        avg_older = sum(h.get("qps", 0) for h in older) / len(older)
        growth_rate = (avg_recent - avg_older) / max(avg_older, 1)

        # 2. 预测未来
        predicted_qps = avg_recent * (1 + growth_rate) ** weeks
        predicted_tokens = int(predicted_qps * 86400 * 500)  # 500 token/请求
        predicted_cost = predicted_tokens / 1e6 * 0.15  # mini定价

        # 3. 检查是否需要扩容
        current_capacity = 100  # 当前最大QPS
        needs_scale = predicted_qps > current_capacity * 0.8
        scale_by = predicted_qps / current_capacity if needs_scale else 1.0

        # 4. 计算距上限天数
        if growth_rate > 0 and avg_recent < current_capacity:
            days_to_limit = int(
                (current_capacity * 0.8 / avg_recent - 1) / growth_rate * 7
            )
        else:
            days_to_limit = 0 if needs_scale else 999

        return CapacityForecast(
            predicted_qps=round(predicted_qps, 2),
            predicted_daily_tokens=predicted_tokens,
            predicted_daily_cost=round(predicted_cost, 2),
            current_capacity_qps=current_capacity,
            needs_scale=needs_scale,
            scale_by=round(scale_by, 2),
            days_until_limit=max(0, days_to_limit),
        )

    def report(self, forecast: CapacityForecast) -> str:
        """生成预测报告。"""
        report = f"""# 容量预测报告

## 预测结果（未来4周）
- 预测QPS: &#123;forecast.predicted_qps&#125;
- 预测日Token: &#123;forecast.predicted_daily_tokens:,&#125;
- 预测日成本: $&#123;forecast.predicted_daily_cost&#125;

## 当前容量
- 最大QPS: &#123;forecast.current_capacity_qps&#125;
- 距上限: &#123;forecast.days_until_limit&#125;天

## 建议
"""
        if forecast.needs_scale:
            report += f"⚠️ 需要扩容&#123;forecast.scale_by&#125;倍\n"
            report += f"预计&#123;forecast.days_until_limit&#125;天后达到容量上限\n"
        else:
            report += "✅ 当前容量充足\n"

        return report
```
