# LLM 评测工具链集成指南

> 你改了一版 Prompt，怎么知道新版本比旧版本好？你换了一个模型，怎么确认质量没下降？你上线了 RAG，怎么评估准确率？"感觉好一点"不靠谱，"看起来差不多"更危险。本指南系统讲解 DeepEval、Ragas、Promptfoo、LangSmith 四大评测工具的集成方案，帮你建立可量化的 LLM 评测体系。

---

## 1. 评测工具全景

### 四大工具定位

| 工具 | 定位 | 核心能力 | 适用场景 |
|------|------|---------|---------|
| DeepEval | 单元测试框架 | LLM 行为断言、CI 集成 | Agent/Prompt 单元测试 |
| Ragas | RAG 专项评估 | 忠实度/相关性/召回率 | RAG 系统评估 |
| Promptfoo | Prompt 对比测试 | A/B 测试、批量评测 | Prompt/模型选型对比 |
| LangSmith | 全链路可观测+评测 | 追踪、数据集、评估器 | 生产监控+持续评估 |

### 选型建议

```
我要测什么？
├─ RAG 系统质量 → Ragas
├─ Agent 行为是否符合预期 → DeepEval
├─ Prompt/模型哪个更好 → Promptfoo
├─ 生产环境持续监控 → LangSmith
└─ 全都要 → 组合使用（下面详解）
```

## 2. DeepEval：LLM 单元测试

### 安装与基本用法

```python
# pip install deepeval

from deepeval import assert_test
from deepeval.test_case import LLMTestCase
from deepeval.metrics import (
    AnswerRelevancyMetric,
    FaithfulnessMetric,
    GEval,
)

# === 定义测试用例 ===
test_case = LLMTestCase(
    input="什么是 RAG？",
    actual_output="RAG 是检索增强生成技术，通过检索相关文档来增强 LLM 的回答质量。",
    expected_output="RAG 通过检索外部知识来增强生成质量",
    retrieval_context=[
        "RAG（Retrieval-Augmented Generation）结合检索和生成..."
    ],
)

# === 定义评估指标 ===
# 答案相关性
answer_relevancy = AnswerRelevancyMetric(threshold=0.7)

# 忠实度（是否有幻觉）
faithfulness = FaithfulnessMetric(threshold=0.8)

# 自定义评估
custom_eval = GEval(
    name="准确性",
    criteria="判断回答是否准确、完整、无误导",
    evaluation_params=["actual_output", "expected_output"],
    threshold=0.75,
)

# 运行评估
assert_test(test_case, [answer_relevancy, faithfulness, custom_eval])
```

## 3. Ragas：RAG 专项评估

### 核心指标

```python
# pip install ragas

from ragas import evaluate
from ragas.metrics import (
    faithfulness,           # 忠实度：回答是否基于检索文档
    answer_relevancy,       # 答案相关性：回答是否切题
    context_precision,      # 上下文精确率：检索的文档是否相关
    context_recall,         # 上下文召回率：是否检索到所有相关文档
    context_entity_recall,  # 实体召回率
    answer_correctness,     # 答案正确性：与参考答案对比
)
from ragas.dataset_schema import EvaluationDataset, SingleTurnSample

# === 准备评测数据 ===
test_data = [
    &#123;
        "user_input": "LangChain 的 LCEL 是什么？",
        "response": "LCEL 是 LangChain 表达式语言，用于以声明式方式组合链。",
        "reference": "LCEL 是 LangChain Expression Language，允许用管道符组合组件。",
        "retrieved_contexts": [
            "LCEL (LangChain Expression Language) 是 LangChain 的核心特性...",
            "使用管道符 | 可以将 Prompt、Model、Parser 串联..."
        ],
    &#125;,
    &#123;
        "user_input": "RAG 的全称是什么？",
        "response": "RAG 是 Retrieval-Augmented Generation 的缩写。",
        "reference": "Retrieval-Augmented Generation",
        "retrieved_contexts": [
            "RAG 通过检索增强生成..."
        ],
    &#125;,
]

# === 转换为 Ragas 格式 ===
samples = []
for d in test_data:
    sample = SingleTurnSample(
        user_input=d["user_input"],
        response=d["response"],
        reference=d["reference"],
        retrieved_contexts=d["retrieved_contexts"],
    )
    samples.append(sample)

dataset = EvaluationDataset(samples=samples)

# === 运行评估 ===
results = evaluate(
    dataset,
    metrics=[
        faithfulness,
        answer_relevancy,
        context_precision,
        context_recall,
        answer_correctness,
    ],
)

print(results)
# 输出示例：
# &#123;'faithfulness': 0.92, 'answer_relevancy': 0.88,
#  'context_precision': 0.85, 'context_recall': 0.90,
#  'answer_correctness': 0.82&#125;
```

### 与 LangChain RAG 管线集成

```python
from langchain.chains import RetrievalQA
from langchain_openai import ChatOpenAI
from langchain_community.vectorstores import Chroma
from ragas import evaluate
from ragas.metrics import faithfulness, answer_relevancy, context_precision

# 构建待评估的 RAG 管线
llm = ChatOpenAI(model="gpt-4o-mini")
vectorstore = Chroma(persist_directory="./chroma_db", embedding_function=embeddings)
rag_chain = RetrievalQA.from_chain_type(
    llm=llm,
    retriever=vectorstore.as_retriever(search_kwargs=&#123;"k": 5&#125;),
    return_source_documents=True,
)

# 评测数据集
eval_questions = [
    "LangChain 的核心组件有哪些？",
    "如何使用 LCEL 创建链？",
    "LangGraph 和 LangChain 的区别？",
    "RAG 的分块策略有哪些？",
    "Agent 的记忆机制是什么？",
]
eval_answers = [...]  # 人工标注的参考答案

# 批量运行 RAG 并收集结果
results = []
for q, ref in zip(eval_questions, eval_answers):
    response = rag_chain.invoke(&#123;"query": q&#125;)
    results.append(&#123;
        "user_input": q,
        "response": response["result"],
        "reference": ref,
        "retrieved_contexts": [doc.page_content for doc in response["source_documents"]],
    &#125;)

# 用 Ragas 评估
samples = [SingleTurnSample(**r) for r in results]
dataset = EvaluationDataset(samples=samples)
eval_results = evaluate(dataset, metrics=[faithfulness, answer_relevancy, context_precision])
print(f"忠实度: &#123;eval_results['faithfulness']:.2%&#125;")
print(f"相关性: &#123;eval_results['answer_relevancy']:.2%&#125;")
print(f"精确率: &#123;eval_results['context_precision']:.2%&#125;")
```

### 指标解读与优化

| 指标 | 含义 | 低分原因 | 优化方向 |
|------|------|---------|---------|
| Faithfulness | 回答是否忠于文档 | 幻觉/编造 | 加强 Prompt 约束、增加事实校验 |
| Answer Relevancy | 回答是否切题 | 答非所问 | 查询理解、上下文组装优化 |
| Context Precision | 检索文档是否相关 | 检索噪声多 | 重排序、调优 Top-K |
| Context Recall | 是否检索到所有信息 | 检索遗漏 | 增加分块数、改用混合检索 |
| Answer Correctness | 与参考答案对比 | 整体偏差大 | Prompt 调优、换模型 |

### Python API 调用

```python
# promptfoo 也支持 Python API
import subprocess
import json

def run_promptfoo_eval(config_path: str) -> dict:
    """运行 Promptfoo 评测"""
    result = subprocess.run(
        ["promptfoo", "eval", "-c", config_path, "--output", "json"],
        capture_output=True, text=True
    )
    return json.loads(result.stdout)

# 在 CI/CD 中自动判断
def ci_prompt_check():
    """CI/CD 中的 Prompt 质量门禁"""
    results = run_promptfoo_eval("promptfooconfig.yaml")

    # 检查通过率
    for result in results["results"]["results"]:
        if not result["success"]:
            print(f"❌ 测试失败: &#123;result['description']&#125;")
            if result["passRate"] < 0.8:
                return False  # 阻止部署

    return True
```

from langsmith.evaluation import evaluate as ls_evaluate

def answer_length_evaluator(run, example):
    """评估回答长度"""
    prediction = run.outputs.get("answer", "")
    if len(prediction) < 10:
        return &#123;"key": "answer_length", "score": 0, "comment": "回答过短"&#125;
    elif len(prediction) > 2000:
        return &#123;"key": "answer_length", "score": 0.5, "comment": "回答过长"&#125;
    return &#123;"key": "answer_length", "score": 1, "comment": "长度合适"&#125;

def citation_evaluator(run, example):
    """评估是否有引用来源"""
    prediction = run.outputs.get("answer", "")
    has_citation = "[" in prediction and "]" in prediction
    return &#123;
        "key": "has_citation",
        "score": 1 if has_citation else 0,
        "comment": "有引用" if has_citation else "缺少引用"
    &#125;

# 运行评估
def target_func(inputs):
    """被评估的目标函数"""
    return &#123;"answer": rag_chain.invoke(&#123;"query": inputs["question"]&#125;)["result"]&#125;

results = ls_evaluate(
    target_func,
    data="RAG 质量评估",
    evaluators=[answer_length_evaluator, citation_evaluator],
)
