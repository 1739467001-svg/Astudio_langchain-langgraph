# RAG 离线在线分离架构

> RAG 系统天然适合读写分离——离线建库（文档处理+嵌入+索引）和在线检索（查询+生成）完全不同的工作负载。把两者分开部署，各自优化，是生产级 RAG 的基础架构。

```python
class OfflineIndexer:
    """离线建库服务。"""

    def __init__(self, vectorstore, embeddings, batch_size=100):
        self.vectorstore = vectorstore
        self.embeddings = embeddings
        self.batch_size = batch_size

    async def build_index(self, documents: list[str], metadata: list[dict] = None):
        """离线批量建库。"""
        from langchain_text_splitters import RecursiveCharacterTextSplitter
        from langchain_core.documents import Document
        import time

        start = time.time()

        # 1. 批量分块
        splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
        all_chunks = []
        for i, doc_text in enumerate(documents):
            chunks = splitter.split_text(doc_text)
            meta = metadata[i] if metadata and i < len(metadata) else &#123;&#125;
            for chunk in chunks:
                all_chunks.append(Document(page_content=chunk, metadata=meta))

        # 2. 批量嵌入+入库
        for i in range(0, len(all_chunks), self.batch_size):
            batch = all_chunks[i:i + self.batch_size]
            await self.vectorstore.aadd_documents(batch)

        elapsed = time.time() - start
        return &#123;
            "total_docs": len(documents),
            "total_chunks": len(all_chunks),
            "elapsed_seconds": round(elapsed, 2),
            "throughput": round(len(all_chunks) / elapsed, 1),
        &#125;

    async def incremental_update(self, new_docs: list, modified_docs: list, deleted_ids: list):
        """增量更新——不影响在线服务。"""
        # 新增
        if new_docs:
            await self.build_index(new_docs)
        # 修改=删除旧的+添加新的
        if modified_docs:
            # await self.vectorstore.delete(filter=&#123;"doc_id": ...&#125;)
            await self.build_index(modified_docs)
        # 删除
        if deleted_ids:
            # await self.vectorstore.delete(filter=&#123;"doc_id": ...&#125;)
            pass

class OnlineRetriever:
    """在线检索服务。"""

    def __init__(self, vectorstore, llm, cache=None):
        self.vectorstore = vectorstore
        self.llm = llm
        self.cache = cache

    async def query(self, question: str) -> dict:
        """在线查询：检索+生成。"""
        import time
        start = time.time()

        # 1. 缓存检查
        if self.cache:
            cached = await self.cache.lookup(question)
            if cached:
                return &#123;"answer": cached, "cached": True, "latency_ms": 1&#125;

        # 2. 向量检索
        docs = await self.vectorstore.asimilarity_search(question, k=5)

        # 3. 生成
        context = "\n".join(d.page_content for d in docs)
        from langchain_core.messages import HumanMessage
        response = await self.llm.ainvoke([HumanMessage(
            content=f"基于以下信息回答:\n&#123;context[:2000]&#125;\n\n问题: &#123;question&#125;"
        )])

        # 4. 缓存
        if self.cache:
            await self.cache.store(question, response.content)

        return &#123;
            "answer": response.content,
            "cached": False,
            "latency_ms": round((time.time() - start) * 1000, 2),
            "retrieved_docs": len(docs),
        &#125;
```

## 五、缓存一致性

```mermaid

    subgraph 一致性 &#123;"离线更新→在线缓存一致性"&#125;
        U1["离线建库完成"] --> INVALIDATE["通知在线服务<br/>失效相关缓存"]
        INVALIDATE --> REFRESH["在线服务<br/>下次查询重建缓存"]
    end

    style 一致性 fill:#FFF9C4
```

```python
class CacheInvalidationManager:
    """缓存失效管理器。"""

    def __init__(self):
        self.invalidation_queue: list[dict] = []

    def notify_index_updated(self, doc_ids: list[str] = None):
        """离线建库完成后通知在线服务。"""
        self.invalidation_queue.append(&#123;
            "action": "invalidate",
            "doc_ids": doc_ids or [],
            "reason": "index_updated",
        &#125;)

    async def process_invalidation(self, cache):
        """处理缓存失效。"""
        while self.invalidation_queue:
            msg = self.invalidation_queue.pop(0)
            if msg["doc_ids"]:
                for doc_id in msg["doc_ids"]:
                    cache.invalidate_pattern(f"*&#123;doc_id&#125;*")
            else:
                # 全量失效
                cache.clear()
```
