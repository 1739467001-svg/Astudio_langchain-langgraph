## 三、with_structured_output（推荐）

### 3.1 基础用法

```python
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

# 定义输出结构
class Person(BaseModel):
    name: str = Field(description="姓名")
    age: int = Field(description="年龄")
    hobbies: list[str] = Field(description="爱好列表")

# 绑定结构化输出
structured_llm = llm.with_structured_output(Person)

# 调用：直接返回Pydantic对象
result = structured_llm.invoke("张三今年28岁，喜欢打篮球和看电影")
print(type(result))        # <class 'Person'>
print(result.name)          # "张三"
print(result.age)           # 28
print(result.hobbies)       # ["打篮球", "看电影"]
```

### 3.2 嵌套结构

```python
class Address(BaseModel):
    city: str = Field(description="城市")
    street: str = Field(description="街道")

class PersonDetail(BaseModel):
    name: str = Field(description="姓名")
    age: int = Field(description="年龄")
    address: Address = Field(description="地址")
    contacts: dict[str, str] = Field(description="联系方式", default=&#123;&#125;)

# 使用嵌套结构
structured_llm = llm.with_structured_output(PersonDetail)
result = structured_llm.invoke("李四30岁，住在北京朝阳区，手机13800138000")
print(result.address.city)  # "北京"
```

### 3.3 枚举和约束

```python
from enum import Enum
from pydantic import field_validator

class Sentiment(str, Enum):
    POSITIVE = "positive"
    NEGATIVE = "negative"
    NEUTRAL = "neutral"

class SentimentResult(BaseModel):
    text: str = Field(description="原始文本")
    sentiment: Sentiment = Field(description="情感倾向")
    confidence: float = Field(description="置信度0-1", ge=0, le=1)

    @field_validator("confidence")
    def check_confidence(cls, v):
        if not 0 <= v <= 1:
            raise ValueError("置信度必须在0-1之间")
        return v

structured_llm = llm.with_structured_output(SentimentResult)
result = structured_llm.invoke("这个产品太棒了，强烈推荐！")
print(result.sentiment)      # Sentiment.POSITIVE
print(result.confidence)     # 0.95
```

## 四、OutputParser 方式

### 4.1 JsonOutputParser

```python
from langchain_core.output_parsers import JsonOutputParser
from langchain_core.prompts import ChatPromptTemplate

parser = JsonOutputParser(pydantic_object=Person)

prompt = ChatPromptTemplate.from_template(
    "提取人物信息。\n&#123;format_instructions&#125;\n\n文本：&#123;text&#125;"
)

chain = prompt | llm | parser
result = chain.invoke(&#123;
    "text": "张三28岁喜欢打篮球",
    "format_instructions": parser.get_format_instructions(),
&#125;)
# result 是 dict（不是Pydantic对象）
```

### 4.2 get_format_instructions 输出

```
The output should be formatted as a JSON instance that conforms to the JSON schema below.
&#123;
  "name": "string",       // 姓名
  "age": "integer",        // 年龄
  "hobbies": ["string"]   // 爱好列表
&#125;
```

## 五、安全解析（容错处理）

```python
import json
import re

def safe_json_parse(text: str) -> dict:
    """三层容错的JSON解析"""
    # Layer 1: 直接解析
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        pass

    # Layer 2: 提取```json代码块
    match = re.search(r'```json\s*(.*?)\s*```', text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group(1))
        except json.JSONDecodeError:
            pass

    # Layer 3: 找第一个&#123;到最后一个&#125;
    start = text.find('&#123;')
    end = text.rfind('&#125;')
    if start != -1 and end != -1:
        try:
            return json.loads(text[start:end+1])
        except json.JSONDecodeError:
            pass

    raise ValueError(f"无法解析JSON: &#123;text[:100]&#125;")
```
