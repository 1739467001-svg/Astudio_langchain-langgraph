# LLM 应用多环境管理

> 开发在本地、测试在 staging、生产在 prod——三套环境的 Prompt、模型、知识库配置都不同。手动切换容易出错。这份指南讲透多环境管理的最佳实践。

## 二、环境配置管理

```python
from dataclasses import dataclass, field
from enum import Enum
import os

class Environment(str, Enum):
    DEVELOPMENT = "development"
    STAGING = "staging"
    PRODUCTION = "production"

@dataclass
class EnvironmentConfig:
    """环境配置。"""
    env: Environment
    llm_model: str
    llm_fallback: str
    embedding_model: str
    vectorstore_url: str
    cache_enabled: bool
    cache_ttl: int
    max_tokens: int
    rate_limit_qps: int
    enable_tracing: bool
    enable_red_team: bool  # 红队测试
    api_keys: dict = field(default_factory=dict)

# 各环境配置
ENV_CONFIGS = &#123;
    Environment.DEVELOPMENT: EnvironmentConfig(
        env=Environment.DEVELOPMENT,
        llm_model="gpt-4o-mini",
        llm_fallback="gpt-4o-mini",
        embedding_model="BAAI/bge-small-zh",
        vectorstore_url="http://localhost:19530",
        cache_enabled=False,
        cache_ttl=60,
        max_tokens=500,
        rate_limit_qps=10,
        enable_tracing=False,
        enable_red_team=False,
        api_keys=&#123;"openai": os.getenv("OPENAI_API_KEY_DEV", "")&#125;,
    ),
    Environment.STAGING: EnvironmentConfig(
        env=Environment.STAGING,
        llm_model="gpt-4o",
        llm_fallback="gpt-4o-mini",

    ),

        env=Environment.PRODUCTION,
        llm_model="gpt-4o",
        llm_fallback="gpt-4o-mini",
        embedding_model="BAAI/bge-large-zh",
        vectorstore_url="http://prod-milvus:19530",
        cache_enabled=True,

        max_tokens=2000,
        rate_limit_qps=100,

class EnvironmentManager:
    """环境管理器。"""

    @staticmethod
    def get_config() -> EnvironmentConfig:
        """获取当前环境配置。"""
        env_name = os.getenv("APP_ENV", "development")
        env = Environment(env_name)
        return ENV_CONFIGS.get(env, ENV_CONFIGS[Environment.DEVELOPMENT])

    @staticmethod
    def is_production() -> bool:
        return os.getenv("APP_ENV") == "production"

    @staticmethod
    def is_staging() -> bool:
        return os.getenv("APP_ENV") == "staging"
```

```python
class EnvironmentSyncManager:
    """环境间同步管理器。"""

    @staticmethod
    def sync_knowledge_base(
        from_env: Environment,
        to_env: Environment,
        doc_ids: list[str] = None,
    ):
        """同步知识库。"""
        return &#123;
            "from": from_env.value,
            "to": to_env.value,
            "docs_synced": len(doc_ids) if doc_ids else "all",
            "status": "synced",
        &#125;

    @staticmethod
    def promote_config(from_env: Environment, to_env: Environment):
        """配置提升（staging→prod）。"""
        from_config = ENV_CONFIGS[from_env]
        to_config = ENV_CONFIGS[to_env]

        changes = []
        if from_config.llm_model != to_config.llm_model:
            changes.append(f"模型: &#123;to_config.llm_model&#125; → &#123;from_config.llm_model&#125;")
        if from_config.cache_ttl != to_config.cache_ttl:
            changes.append(f"缓存TTL: &#123;to_config.cache_ttl&#125; → &#123;from_config.cache_ttl&#125;")

        return &#123;
            "changes": changes,
            "requires_approval": to_env == Environment.PRODUCTION,
        &#125;
```

## 四、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| API Key分环境 | 不同环境不同Key | ★★★ |
| 开发用小模型 | 省成本 | ★★★ |
| Staging要追踪 | 测试可观测性 | ★★★ |
| 配置用环境变量 | 不硬编码 | ★★★ |
| Staging→Prod要审批 | 不能直接推 | ★★☆ |
| 知识库定期同步 | 保持一致 | ★★☆ |
