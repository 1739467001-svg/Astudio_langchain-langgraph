# 输出护栏与内容安全

> LLM 可能生成不当内容。护栏（Guardrails）是在 LLM 输入和输出之间设置的安全过滤层。

---

## 二、输入护栏

### 2.1 Prompt 注入检测

```python
import re

def input_guard(text: str) -> tuple[str, bool, str]:
    """输入护栏：返回(处理后文本, 是否拦截, 原因)"""
    # 检测Prompt注入
    injection_patterns = [
        (r"ignore\s+(previous|above|all)\s+(instructions?|prompts?)", "忽略指令攻击"),
        (r"忽略(之前|上面|所有)(的)?(指令|提示|规则)", "中文忽略指令"),
        (r"(system|系统)\s*(prompt|提示)", "尝试获取系统提示"),
        (r"(reveal|show|print)\s+(your|the)\s+(instructions?|prompt)", "尝试泄露指令"),
        (r"(act|pretend)\s+(as|like)\s+(a|an)\s+(hacker|admin)", "角色劫持"),
        (r"(jailbreak|越狱|突破限制)", "越狱尝试"),
    ]

    for pattern, reason in injection_patterns:
        if re.search(pattern, text, re.IGNORECASE):
            return ("检测到可疑输入，请正常提问。", True, reason)

    # PII脱敏
    text = re.sub(r'1[3-9]\d&#123;9&#125;', '[手机号]', text)
    text = re.sub(r'[\w.-]+@[\w.-]+', '[邮箱]', text)

    return (text, False, "")
```

### 2.2 输入长度和格式验证

```python
def validate_input(text: str, max_length: int = 2000) -> tuple[str, bool, str]:
    """输入验证"""
    if not text or not text.strip():
        return ("", True, "输入为空")

    if len(text) > max_length:
        return ("", True, f"输入过长（&#123;len(text)&#125;/&#123;max_length&#125;）")

    # 检测非文本内容
    if text.count('\n') > 100:
        return ("", True, "输入包含过多换行，可能是异常输入")

    return (text, False, "")
```

## 三、输出护栏

### 3.1 不当内容检测

```python
def output_guard(output: str) -> tuple[str, bool, str]:
    """输出护栏：返回(处理后输出, 是否拦截, 原因)"""

    # 1. 检测系统提示泄露
    leak_patterns = [
        "我的系统提示是", "我的指令是", "我被要求",
        "my system prompt", "my instructions are",
    ]
    for pattern in leak_patterns:
        if pattern.lower() in output.lower():
            return ("抱歉，我无法回答这个问题。", True, "系统提示泄露")

    # 2. 检测敏感话题
    sensitive = ["如何制造武器", "如何黑入", "盗取密码的方法"]
    for pattern in sensitive:
        if pattern in output:
            return ("抱歉，我无法提供此类信息。", True, "敏感内容")

    # 3. PII泄露检测
    import re
    if re.search(r'1[3-9]\d&#123;9&#125;', output):
        output = re.sub(r'1[3-9]\d&#123;9&#125;', '[手机号]', output)

    if re.search(r'[\w.-]+@[\w.-]+\.\w+', output):
        output = re.sub(r'[\w.-]+@[\w.-]+\.\w+', '[邮箱]', output)

    # 4. 检测代码执行内容
    dangerous_code = ["os.system", "os.remove", "subprocess.call", "__import__"]
    for code in dangerous_code:
        if code in output:
            return ("输出包含潜在危险代码，已过滤。", True, "危险代码")

    return (output, False, "")
```

### 3.2 输出格式验证

```python
def validate_output(output: str, expected_format: str = None) -> tuple[str, bool, str]:
    """输出格式验证"""
    # 空输出
    if not output or len(output.strip()) < 5:
        return ("", True, "输出过短，可能是异常")

    # JSON格式验证
    if expected_format == "json":
        import json
        try:
            json.loads(output)
        except json.JSONDecodeError:
            # 尝试提取JSON块
            import re
            match = re.search(r'\&#123;.*\&#125;', output, re.DOTALL)
            if match:
                try:
                    json.loads(match.group())
                    return (match.group(), False, "")
                except json.JSONDecodeError:
                    pass
            return ("", True, "JSON格式错误")

    return (output, False, "")
```

## 四、完整护栏管线

```python
def guarded_invoke(chain, user_input: str) -> dict:
    """带完整护栏的LLM调用"""
    # === 输入护栏 ===
    text, blocked, reason = validate_input(user_input)
    if blocked:
        return &#123;"answer": text, "blocked": True, "stage": "input", "reason": reason&#125;

    text, blocked, reason = input_guard(text)
    if blocked:
        return &#123;"answer": text, "blocked": True, "stage": "input_guard", "reason": reason&#125;

    # === LLM 处理 ===
    try:
        output = chain.invoke(&#123;"input": text&#125;)
        if hasattr(output, "content"):
            output = output.content
    except Exception as e:
        return &#123;"answer": "服务暂时不可用", "blocked": True, "stage": "llm", "reason": str(e)&#125;

    # === 输出护栏 ===
    output, blocked, reason = output_guard(output)
    if blocked:
        return &#123;"answer": output, "blocked": True, "stage": "output_guard", "reason": reason&#125;

    return &#123;"answer": output, "blocked": False&#125;
```
