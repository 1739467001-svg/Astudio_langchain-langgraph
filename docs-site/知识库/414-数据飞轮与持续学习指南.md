# 数据飞轮与持续学习指南

> LLM 应用上线只是起点。用户会发现问题、提供反馈、纠正错误——这些数据如果白白流失，系统永远不会进步。数据飞轮把用户反馈变成改进动力：收集反馈 → 标注数据 → 优化检索/Prompt → 更好的回答 → 更多用户 → 更多反馈，形成正向循环。

---

## 1. 数据飞轮原理

### 飞轮循环

```
                    ┌──────────────┐
                    │   系统上线     │
                    └──────┬───────┘
                           ▼
               ┌───────────────────────┐
               │  用户使用 → 产生交互数据  │
               └───────────┬───────────┘
                           ▼
               ┌───────────────────────┐
               │  收集反馈 → 标注质量     │
               │  （赞/踩/修正/投诉）    │
               └───────────┬───────────┘
                           ▼
               ┌───────────────────────┐
               │  分析问题 → 定位根因     │
               │  （检索失败/Prompt差/   │
               │   知识缺失/模型能力）   │
               └───────────┬───────────┘
                           ▼
               ┌───────────────────────┐
               │  改进优化 → 部署上线     │
               │  （更新知识库/调Prompt/  │
               │   加Few-Shot/微调）    │
               └───────────┬───────────┘
                           ▼
               ┌───────────────────────┐
               │  更好的回答 → 更多用户   │ ← 飞轮加速
               └───────────┬───────────┘
                           │
                           └──→ 回到"用户使用"
```

### 飞轮的核心：反馈数据

| 反馈类型 | 来源 | 价值 | 采集难度 |
|----------|------|------|----------|
| 显式反馈 | 用户点赞/踩 | 高 | 低 |
| 隐式反馈 | 用户复制/重问 | 中 | 中 |
| 修正数据 | 用户编辑回答 | 极高 | 低 |
| 投诉报告 | 用户提交问题 | 高 | 低 |
| 专家标注 | 人工审核 | 最高 | 高 |
| A/B 对比 | 灰度测试 | 中高 | 中 |

## 2. 反馈收集系统

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import Any
import time
import json

class FeedbackType(Enum):
    THUMBS_UP = "thumbs_up"           # 点赞
    THUMBS_DOWN = "thumbs_down"      # 踩
    EDIT = "edit"                     # 用户修正回答
    REGENERATE = "regenerate"        # 要求重新生成
    REPORT = "report"                # 投诉举报
    RATING = "rating"                # 评分
    COMMENT = "comment"              # 文字反馈


class IssueCategory(Enum):
    RETRIVAL_MISS = "retrieval_miss"          # 检索失败
    RETRIVAL_NOISE = "retrieval_noise"       # 检索噪声
    HALLUCINATION = "hallucination"           # 幻觉
    INCOMPLETE = "incomplete"                 # 不完整
    WRONG_FORMAT = "wrong_format"            # 格式错误
    OUTDATED = "outdated"                     # 知识过时
    OFF_TOPIC = "off_topic"                  # 答非所问
    SAFETY = "safety"                         # 安全问题


@dataclass
class Feedback:
    """用户反馈"""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    # 关联
    conversation_id: str = ""
    message_id: str = ""             # 哪条 AI 回复的反馈
    user_id: str = ""
    # 反馈内容
    type: FeedbackType = FeedbackType.THUMBS_DOWN
    rating: int = 0                  # 1-5 评分
    comment: str = ""
    # 问题分类（人工或 LLM 标注）
    issue_category: IssueCategory | None = None
    issue_description: str = ""
    # 用户修正（如果有）
    user_correction: str = ""
    # 上下文快照
    query: str = ""
    ai_response: str = ""
    retrieved_docs: list[dict] = field(default_factory=list)
    # 元数据
    timestamp: float = field(default_factory=time.time)
    processed: bool = False


class FeedbackCollector:
    """反馈收集器"""

    def __init__(self):
        self.feedbacks: list[Feedback] = []
        self.conversations: dict[str, dict] = &#123;&#125;  # 快照存储

    def record_conversation(
        self,
        conversation_id: str,
        query: str,
        ai_response: str,
        retrieved_docs: list[dict],
    ):
        """记录对话上下文快照（用于后续分析）"""
        self.conversations[conversation_id] = &#123;
            "query": query,
            "ai_response": ai_response,
            "retrieved_docs": retrieved_docs,
            "timestamp": time.time(),
        &#125;

    def collect(self, feedback: Feedback):
        """收集用户反馈"""
        # 补充上下文快照
        if feedback.conversation_id in self.conversations:
            conv = self.conversations[feedback.conversation_id]
            if not feedback.query:
                feedback.query = conv["query"]
            if not feedback.ai_response:
                feedback.ai_response = conv["ai_response"]
            if not feedback.retrieved_docs:
                feedback.retrieved_docs = conv["retrieved_docs"]

        self.feedbacks.append(feedback)

    def get_negative_feedbacks(self) -> list[Feedback]:
        """获取负面反馈（需要改进的）"""
        return [
            f for f in self.feedbacks
            if f.type in (FeedbackType.THUMBS_DOWN, FeedbackType.REPORT)
            or (f.type == FeedbackType.RATING and f.rating <= 2)
            or f.user_correction
        ]

    def get_corrections(self) -> list[Feedback]:
        """获取用户修正（最有价值的反馈）"""
        return [f for f in self.feedbacks if f.user_correction]

    def stats(self) -> dict:
        """反馈统计"""
        total = len(self.feedbacks)
        positive = sum(1 for f in self.feedbacks if f.type == FeedbackType.THUMBS_UP)
        negative = sum(1 for f in self.feedbacks if f.type == FeedbackType.THUMBS_DOWN)
        corrections = len(self.get_corrections())

        return &#123;
            "total": total,
            "positive": positive,
            "negative": negative,
            "satisfaction_rate": positive / max(total, 1),
            "corrections": corrections,
            "by_category": self._by_category(),
        &#125;

    def _by_category(self) -> dict[str, int]:
        from collections import Counter
        cats = Counter()
        for f in self.feedbacks:
            if f.issue_category:
                cats[f.issue_category.value] += 1
        return dict(cats)
```

## 3. 反馈分析与根因定位

```python
from langchain_core.prompts import ChatPromptTemplate
from langchain_openai import ChatOpenAI

class FeedbackAnalyzer:
    """反馈分析器：定位问题根因"""

    def __init__(self, llm: ChatOpenAI | None = None):
        self.llm = llm or ChatOpenAI(temperature=0, model="gpt-4o-mini")

    def analyze(self, feedback: Feedback) -> dict:
        """分析单个负面反馈的根因"""
        prompt = ChatPromptTemplate.from_messages([
            ("system", """你是 LLM 应用问题诊断专家。
根据用户反馈，分析问题的根因。

问题分类（选一个最匹配的）：
- retrieval_miss: 检索结果中没有正确答案
- retrieval_noise: 检索到不相关内容干扰了回答
- hallucination: 回答包含编造的信息
- incomplete: 回答不完整，缺少关键信息
- wrong_format: 回答格式不符合预期
- outdated: 知识库信息过时
- off_topic: 答非所问
- safety: 安全问题

输出 JSON: &#123;
  "category": "...",
  "root_cause": "...",
  "improvement": "...",
  "priority": "high/medium/low"
&#125;"""),
            ("human", """用户查询：&#123;query&#125;
AI 回答：&#123;response&#125;
检索到的文档：&#123;docs&#125;
用户反馈类型：&#123;feedback_type&#125;
用户评论：&#123;comment&#125;
用户修正：&#123;correction&#125;"""),
        ])

        chain = prompt | self.llm
        result = chain.invoke(&#123;
            "query": feedback.query,
            "response": feedback.ai_response[:500],
            "docs": str(feedback.retrieved_docs)[:500],
            "feedback_type": feedback.type.value,
            "comment": feedback.comment,
            "correction": feedback.user_correction,
        &#125;)

        try:
            import json
            data = json.loads(result.content)
            feedback.issue_category = IssueCategory(data["category"])
            feedback.issue_description = data.get("root_cause", "")
            feedback.processed = True
            return data
        except (json.JSONDecodeError, KeyError) as e:
            return &#123;"error": str(e)&#125;

    def batch_analyze(self, feedbacks: list[Feedback]) -> dict:
        """批量分析反馈，输出改进建议"""
        results = []
        for f in feedbacks:
            if not f.processed:
                results.append(self.analyze(f))

        from collections import Counter
        categories = Counter(r.get("category") for r in results if r)
        return &#123;
            "total_analyzed": len(results),
            "category_distribution": dict(categories),
            "top_issues": categories.most_common(5),
            "improvement_suggestions": self._aggregate_improvements(results),
        &#125;

    def _aggregate_improvements(self, results: list[dict]) -> list[str]:
        """聚合改进建议"""
        suggestions = []
        for r in results:
            if r and r.get("improvement"):
                suggestions.append(r["improvement"])
        # 去重并取前 10
        return list(dict.fromkeys(suggestions))[:10]
```

## 4. 持续优化引擎

```python
class ContinuousImprovementEngine:
    """持续优化引擎：将反馈转化为具体改进"""

    def __init__(
        self,
        feedback_collector: FeedbackCollector,
        analyzer: FeedbackAnalyzer,
    ):
        self.collector = feedback_collector
        self.analyzer = analyzer

    def run_cycle(self) -> dict:
        """执行一轮飞轮循环"""
        # 1. 收集负面反馈
        negatives = self.collector.get_negative_feedbacks()
        corrections = self.collector.get_corrections()

        if not negatives and not corrections:
            return &#123;"status": "no_data", "improvements": 0&#125;

        # 2. 分析根因
        analysis = self.analyzer.batch_analyze(negatives)

        # 3. 生成改进计划
        improvements = self._plan_improvements(analysis, corrections)

        # 4. 执行改进
        executed = self._execute_improvements(improvements)

        return &#123;
            "status": "completed",
            "feedbacks_analyzed": len(negatives),
            "improvements_planned": len(improvements),
            "improvements_executed": len(executed),
            "details": executed,
        &#125;

    def _plan_improvements(
        self,
        analysis: dict,
        corrections: list[Feedback],
    ) -> list[dict]:
        """根据分析结果生成改进计划"""
        plans = []

        # 检索问题 → 更新知识库
        retrieval_issues = [
            c for c in analysis.get("category_distribution", &#123;&#125;)
            if "retrieval" in c
        ]
        if retrieval_issues:
            plans.append(&#123;
                "type": "update_knowledge_base",
                "description": "检索失败率高，需要补充知识库内容",
                "priority": "high",
                "data": corrections,  # 用用户修正作为新知识来源
            &#125;)

        # 幻觉问题 → 添加约束提示
        if "hallucination" in analysis.get("category_distribution", &#123;&#125;):
            plans.append(&#123;
                "type": "update_prompt",
                "description": "出现幻觉，加强回答约束",
                "priority": "high",
                "change": "添加'如果不确定请说明，不要编造信息'到系统提示",
            &#125;)

        # 格式问题 → 添加 Few-Shot
        if "wrong_format" in analysis.get("category_distribution", &#123;&#125;):
            plans.append(&#123;
                "type": "add_few_shot",
                "description": "格式不符合预期，添加格式示例",
                "priority": "medium",
            &#125;)

        # 过时知识 → 更新文档
        if "outdated" in analysis.get("category_distribution", &#123;&#125;):
            plans.append(&#123;
                "type": "refresh_documents",
                "description": "知识库内容过时，需要更新",
                "priority": "high",
            &#125;)

        # 用户修正 → 添加到训练数据
        if corrections:
            plans.append(&#123;
                "type": "add_to_eval_set",
                "description": f"将 &#123;len(corrections)&#125; 条用户修正添加到评估集",
                "priority": "medium",
                "data": corrections,
            &#125;)

        return plans

    def _execute_improvements(self, plans: list[dict]) -> list[dict]:
        """执行改进计划"""
        executed = []
        for plan in plans:
            if plan["type"] == "update_knowledge_base":
                # 从用户修正中提取知识，添加到向量库
                for corr in plan.get("data", []):
                    # 实际实现：将 correction 内容添加到向量库
                    pass
                executed.append(&#123;
                    "type": "knowledge_base_updated",
                    "items_added": len(plan.get("data", [])),
                &#125;)

            elif plan["type"] == "update_prompt":
                executed.append(&#123;
                    "type": "prompt_updated",
                    "change": plan["change"],
                &#125;)

            elif plan["type"] == "add_few_shot":
                executed.append(&#123;
                    "type": "few_shot_added",
                &#125;)

            elif plan["type"] == "add_to_eval_set":
                executed.append(&#123;
                    "type": "eval_set_updated",
                    "items_added": len(plan.get("data", [])),
                &#125;)

        return executed
```

## 5. 隐式反馈收集

```python
from dataclasses import dataclass
from typing import Any

@dataclass
class ImplicitSignal:
    """隐式反馈信号"""
    conversation_id: str
    signal_type: str   # copy / regenerate / dwell_time / follow_up
    value: Any
    timestamp: float


class ImplicitFeedbackCollector:
    """隐式反馈收集器"""

    def __init__(self):
        self.signals: list[ImplicitSignal] = []

    def on_copy(self, conversation_id: str, copied_text: str):
        """用户复制了 AI 回复（正面信号）"""
        self.signals.append(ImplicitSignal(
            conversation_id=conversation_id,
            signal_type="copy",
            value=len(copied_text),
            timestamp=time.time(),
        ))

    def on_regenerate(self, conversation_id: str):
        """用户要求重新生成（负面信号）"""
        self.signals.append(ImplicitSignal(
            conversation_id=conversation_id,
            signal_type="regenerate",
            value=1,
            timestamp=time.time(),
        ))

    def on_dwell_time(self, conversation_id: str, seconds: float):
        """用户停留时间"""
        self.signals.append(ImplicitSignal(
            conversation_id=conversation_id,
            signal_type="dwell_time",
            value=seconds,
            timestamp=time.time(),
        ))

    def on_follow_up(self, conversation_id: str, follow_up_query: str):
        """用户追问（可能是正面深入了解，也可能是没答好）"""
        self.signals.append(ImplicitSignal(
            conversation_id=conversation_id,
            signal_type="follow_up",
            value=follow_up_query,
            timestamp=time.time(),
        ))

    def score_conversation(self, conversation_id: str) -> float:
        """根据隐式信号给对话打分（-1 到 1）"""
        conv_signals = [
            s for s in self.signals
            if s.conversation_id == conversation_id
        ]

        score = 0
        for s in conv_signals:
            if s.signal_type == "copy":
                score += 0.3  # 复制 = 正面
            elif s.signal_type == "regenerate":
                score -= 0.5  # 重新生成 = 负面
            elif s.signal_type == "dwell_time":
                if s.value > 30:
                    score += 0.2  # 长时间阅读 = 正面
                elif s.value < 3:
                    score -= 0.1  # 快速离开 = 可能负面

        return max(-1, min(1, score))
```

## 6. 飞轮指标看板

```python
class FlywheelDashboard:
    """数据飞轮指标看板"""

    def __init__(
        self,
        collector: FeedbackCollector,
        implicit: ImplicitFeedbackCollector,
    ):
        self.collector = collector
        self.implicit = implicit

    def render(self) -> dict:
        """渲染看板数据"""
        fb_stats = self.collector.stats()

        return &#123;
            "feedback": &#123;
                "total": fb_stats["total"],
                "satisfaction_rate": f"&#123;fb_stats['satisfaction_rate']:.1%&#125;",
                "negative_count": fb_stats["negative"],
                "corrections": fb_stats["corrections"],
                "top_issues": fb_stats["by_category"],
            &#125;,
            "implicit": &#123;
                "total_signals": len(self.implicit.signals),
                "copy_events": sum(1 for s in self.implicit.signals if s.signal_type == "copy"),
                "regenerate_events": sum(1 for s in self.implicit.signals if s.signal_type == "regenerate"),
            &#125;,
            "flywheel_health": self._flywheel_health(fb_stats),
            "recommendations": self._recommendations(fb_stats),
        &#125;

    def _flywheel_health(self, stats: dict) -> str:
        """飞轮健康度"""
        rate = stats["satisfaction_rate"]
        if rate > 0.9:
            return "🟢 优秀"
        elif rate > 0.7:
            return "🟡 良好"
        elif rate > 0.5:
            return "🟠 需改进"
        else:
            return "🔴 需紧急改进"

    def _recommendations(self, stats: dict) -> list[str]:
        """改进建议"""
        recs = []
        if stats["satisfaction_rate"] < 0.7:
            recs.append("满意度低于 70%，建议紧急排查高频负面问题")
        if stats["corrections"] > 10:
            recs.append(f"有 &#123;stats['corrections']&#125; 条用户修正，建议提取为训练数据")
        top_issues = stats["by_category"]
        if top_issues:
            top_issue = max(top_issues, key=top_issues.get)
            recs.append(f"最高频问题：&#123;top_issue&#125;（&#123;top_issues[top_issue]&#125;次）")
        return recs
```
