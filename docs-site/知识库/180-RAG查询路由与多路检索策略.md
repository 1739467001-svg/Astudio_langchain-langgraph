# RAG 查询路由与多路检索策略

> 不是所有查询都应该走同一条检索路径——简单事实问答直接向量检索就好，复杂分析可能需要知识图谱查询+向量检索+Web搜索。查询路由根据查询特征自动选择最佳检索路径。

## 二、路由策略

```python
from enum import Enum
from dataclasses import dataclass
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage

class RetrievalRoute(str, Enum):
    VECTOR = "vector"           # 向量检索
    KEYWORD = "keyword"          # 关键词检索
    HYBRID = "hybrid"           # 混合检索
    GRAPH = "graph"             # 图谱查询
    WEB = "web"                  # Web搜索
    MULTI = "multi"              # 多路融合
    DIRECT = "direct"            # 直接LLM（无需检索）

ROUTE_PROMPT = """判断以下查询应该使用哪种检索策略。

查询: &#123;query&#125;

可选策略:
- vector: 语义相似的文档检索（"什么是RAG"）
- keyword: 精确关键词匹配（"订单号ABC123"）
- hybrid: 混合检索（语义+关键词）
- graph: 实体关系查询（"张三的同事"）
- web: 需要最新信息（"今天的新闻"）
- multi: 复杂分析需要多源（"对比A和B"）
- direct: 通用知识无需检索（"你好"）

只输出策略名称:"""

@dataclass
class RouteDecision:
    """路由决策。"""
    route: RetrievalRoute
    reason: str = ""
    confidence: float = 0.8

class QueryRouter:
    """查询路由器。"""

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def route(self, query: str) -> RouteDecision:
        """决定检索路径。"""
        # 简单规则先判断
        route = self._rule_based_route(query)
        if route:
            return RouteDecision(route=route, reason="规则匹配", confidence=0.9)

        # LLM判断
        prompt = ROUTE_PROMPT.format(query=query)
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        result = response.content.strip().lower()

        for r in RetrievalRoute:
            if r.value in result:
                return RouteDecision(route=r, reason="LLM判断", confidence=0.7)

        return RouteDecision(route=RetrievalRoute.VECTOR, reason="默认向量检索", confidence=0.5)

    @staticmethod
    def _rule_based_route(query: str) -> RetrievalRoute | None:
        """基于规则的路由。"""
        # 问候/闲聊→直接回答
        if any(w in query for w in ["你好", "谢谢", "再见"]) and len(query) < 10:
            return RetrievalRoute.DIRECT

        # 包含ID/编号→关键词检索
        import re
        if re.search(r'[A-Z]&#123;2,&#125;\d+', query):
            return RetrievalRoute.KEYWORD

        # 需要最新→Web搜索
        if any(w in query for w in ["今天", "最新", "实时", "现在"]):
            return RetrievalRoute.WEB

        # 关系查询→图谱
        if any(w in query for w in ["的同事", "的关系", "的上级", "的下属"]):
            return RetrievalRoute.GRAPH

        # 对比分析→多路
        if any(w in query for w in ["对比", "比较", "区别", "vs"]):
            return RetrievalRoute.MULTI

        return None
```

## 三、多路检索与融合

```mermaid
graph TB
    Q["查询"] --> ROUTE&#123;"路由决策"&#125;
    ROUTE -->|vector| V["向量检索"]
    ROUTE -->|keyword| K["关键词检索"]
    ROUTE -->|graph| G["图谱查询"]
    ROUTE -->|web| W["Web搜索"]
    ROUTE -->|multi| M["多路并行"]
    M --> V
    M --> K
    M --> G
    M --> W
    V & K & G & W --> FUSE["RRF融合排序"]
    FUSE --> LLM["LLM生成"]

    style ROUTE fill:#FFF9C4,stroke:#F9A825,stroke-width:3px
    style FUSE fill:#C8E6C9
```

```python
import asyncio
from langchain_core.documents import Document

class MultiRouteRetriever:
    """多路检索器。

    根据路由决策选择检索路径，
    多路结果用RRF融合。
    """

    def __init__(
        self,
        vectorstore=None,
        bm25_retriever=None,
        graph_store=None,
        web_search=None,
    ):
        self.vectorstore = vectorstore
        self.bm25 = bm25_retriever
        self.graph = graph_store
        self.web = web_search

    async def retrieve(
        self,
        query: str,
        route: RetrievalRoute,
        k: int = 5,
    ) -> list[Document]:
        """根据路由执行检索。"""
        if route == RetrievalRoute.DIRECT:
            return []  # 不需要检索

        elif route == RetrievalRoute.VECTOR:
            return await self._vector_retrieve(query, k)

        elif route == RetrievalRoute.KEYWORD:
            return self._keyword_retrieve(query, k)

        elif route == RetrievalRoute.HYBRID:
            vec_docs = await self._vector_retrieve(query, k)
            kw_docs = self._keyword_retrieve(query, k)
            return self._rrf_fuse([vec_docs, kw_docs], final_k=k)

        elif route == RetrievalRoute.GRAPH:
            return self._graph_retrieve(query, k)

        elif route == RetrievalRoute.WEB:
            return await self._web_retrieve(query, k)

        elif route == RetrievalRoute.MULTI:
            return await self._multi_retrieve(query, k)

        return []

    async def _vector_retrieve(self, query: str, k: int) -> list[Document]:
        if not self.vectorstore:
            return []
        return await self.vectorstore.asimilarity_search(query, k=k)

    def _keyword_retrieve(self, query: str, k: int) -> list[Document]:
        if not self.bm25:
            return []
        return self.bm25.search(query, k=k)

    def _graph_retrieve(self, query: str, k: int) -> list[Document]:
        if not self.graph:
            return []
        # 简化：返回空，实际调用图谱查询
        return [Document(page_content=f"图谱结果: &#123;query&#125;")]

    async def _web_retrieve(self, query: str, k: int) -> list[Document]:
        if not self.web:
            return []
        # 简化：返回空，实际调用搜索API
        return [Document(page_content=f"Web结果: &#123;query&#125;")]

    async def _multi_retrieve(self, query: str, k: int) -> list[Document]:
        """多路并行检索+RRF融合。"""
        tasks = [
            self._vector_retrieve(query, k),
            asyncio.coroutine(lambda: self._keyword_retrieve(query, k))(),
            asyncio.coroutine(lambda: self._graph_retrieve(query, k))(),
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        valid_results = [r for r in results if isinstance(r, list)]
        return self._rrf_fuse(valid_results, final_k=k)

    @staticmethod
    def _rrf_fuse(result_lists: list[list[Document]], k: int = 60, final_k: int = 5) -> list[Document]:
        """RRF融合排序。"""
        scores = &#123;&#125;
        doc_map = &#123;&#125;
        for result_list in result_lists:
            for rank, doc in enumerate(result_list):
                key = hash(doc.page_content[:200])
                scores[key] = scores.get(key, 0) + 1.0 / (k + rank + 1)
                if key not in doc_map:
                    doc_map[key] = doc
        sorted_keys = sorted(scores.keys(), key=lambda x: scores[x], reverse=True)
        return [doc_map[key] for key in sorted_keys[:final_k]]
```

## 四、完整管线

```python
class SmartRAGPipeline:
    """智能RAG管线：路由→检索→生成。"""

    def __init__(self, llm, router: QueryRouter, retriever: MultiRouteRetriever):
        self.llm = llm
        self.router = router
        self.retriever = retriever

    async def answer(self, query: str) -> dict:
        """完整问答流程。"""
        # 1. 路由
        route_decision = await self.router.route(query)

        # 2. 检索
        docs = await self.retriever.retrieve(query, route_decision.route)

        # 3. 生成
        if not docs:
            # 无检索结果（如DIRECT路由）
            from langchain_core.messages import HumanMessage
            response = await self.llm.ainvoke([HumanMessage(content=query)])
            answer = response.content
        else:
            context = "\n\n".join(d.page_content for d in docs)
            from langchain_core.messages import HumanMessage
            prompt = f"基于以下信息回答问题。\n\n信息:\n&#123;context[:3000]&#125;\n\n问题: &#123;query&#125;"
            response = await self.llm.ainvoke([HumanMessage(content=prompt)])
            answer = response.content

        return &#123;
            "answer": answer,
            "route": route_decision.route.value,
            "retrieved_docs": len(docs),
            "reason": route_decision.reason,
        &#125;
```

class RouteEvaluator:
    """路由效果评估器。"""

    @staticmethod
    async def evaluate(
        pipeline: SmartRAGPipeline,
        test_cases: list[dict],  # [&#123;query, expected_route&#125;]
    ) -> dict:
        """评估路由准确率。"""
        correct = 0
        route_distribution = &#123;&#125;

        for tc in test_cases:
            decision = await pipeline.router.route(tc["query"])
            route_dist[decision.route.value] = route_distribution.get(decision.route.value, 0) + 1
            if decision.route.value == tc["expected_route"]:
                correct += 1

        return &#123;
            "total": len(test_cases),
            "correct": correct,
            "accuracy": round(correct / len(test_cases), 4) if test_cases else 0,
            "route_distribution": route_distribution,
        &#125;

## 六、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 规则+LLM双重路由 | 规则快，LLM补 | ★★★ |
| 默认走向量检索 | 80%场景够用 | ★★★ |
| 多路结果必须RRF融合 | 不要简单拼接 | ★★☆ |
| DIRECT路由省成本 | 闲聊不需要检索 | ★★☆ |
| 评估路由准确率 | 防止路由错误 | ★★☆ |
