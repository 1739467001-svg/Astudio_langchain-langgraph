# RAG 引用溯源与来源追踪

> RAG 生成的答案如果没有来源标注，用户怎么知道是真信息还是幻觉？引用溯源让每个陈述都能追溯到具体文档段落，是建立信任的关键。这份指南覆盖引用标注、来源追踪和置信度展示的实现。

## 三、实现

### 3.1 带引用的检索

```python
from langchain_core.documents import Document
from langchain_core.vectorstores import VectorStore
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage, SystemMessage
from dataclasses import dataclass
from typing import Optional

@dataclass
class SourceCitation:
    """引用来源。"""
    citation_id: int          # [1], [2]...
    doc_id: str
    source: str                # 文件名/URL
    page: Optional[int] = None  # 页码
    content: str = ""           # 被引用的内容片段
    relevance_score: float = 0  0  # 相关度

class CitationRAG:
    """带引用溯源的RAG。

    生成答案时标注引用编号，
    每个编号关联到具体文档来源。
    """

    def __init__(self, llm: BaseChatModel, vectorstore: VectorStore):
        self.llm = llm
        self.vectorstore = vectorstore

    async def answer_with_citations(self, question: str, k: int = 5) -> dict:
        """生成带引用的回答。"""
        # 1. 检索（保留来源信息）
        docs = await self.vectorstore.asimilarity_search(question, k=k)
        citations = []
        for i, doc in enumerate(docs):
            citations.append(SourceCitation(
                citation_id=i + 1,
                doc_id=doc.metadata.get("doc_id", str(i)),
                source=doc.metadata.get("source", "未知"),
                page=doc.metadata.get("page"),
                content=doc.page_content[:200],
                relevance_score=doc.metadata.get("score", 0),
            ))

        # 2. 构建带引用编号的上下文
        context_parts = []
        for cite in citations:
            context_parts.append(
                f"[&#123;cite.citation_id&#125;] (来源: &#123;cite.source&#125;"
                f"&#123;f' 第&#123;cite.page&#125;页' if cite.page else ''&#125;)\n&#123;cite.content&#125;"
            )
        context = "\n\n".join(context_parts)

        # 3. LLM生成（指示标注引用）
        prompt = f"""基于以下信息回答问题。

## 参考信息
&#123;context&#125;

## 回答要求
1. 基于参考信息回答
2. 在陈述事实时标注引用编号，如"公司成立于2019年 [1]"
3. 如果信息来自多个来源，标注多个编号，如"营收增长20% [1][3]"
4. 如果参考信息中没有相关内容，说明"参考信息中未找到相关内容"
5. 不要编造信息

## 问题
&#123;question&#125;

## 回答:"""

        response = await self.llm.ainvoke([
            SystemMessage(content="你是严谨的问答助手，必须标注信息来源。"),
            HumanMessage(content=prompt),
        ])

        # 4. 提取并验证引用
        answer = response.content
        used_citations = self._extract_citations(answer)

        return &#123;
            "answer": answer,
            "citations": [
                &#123;
                    "id": c.citation_id,
                    "source": c.source,
                    "page": c.page,
                    "content": c.content,
                &#125;
                for c in citations if c.citation_id in used_citations
            ],
            "total_sources": len(citations),
            "used_sources": len(used_citations),
        &#125;

    @staticmethod
    def _extract_citations(text: str) -> set[int]:
        """从回答中提取引用编号。"""
        import re
        # 匹配 [1], [2], [1][3] 等模式
        matches = re.findall(r'\[(\d+)\]', text)
        return &#123;int(m) for m in matches&#125;
```

### 3.2 流式输出带引用

```python
class StreamingCitationRAG(CitationRAG):
    """流式输出+引用溯源。"""

    async def stream_answer_with_citations(self, question: str, k: int = 5):
        """流式输出生成，最后附引用列表。"""
        docs = await self.vectorstore.asimilarity_search(question, k=k)
        citations = [
            SourceCitation(
                citation_id=i + 1,
                doc_id=doc.metadata.get("doc_id", str(i)),
                source=doc.metadata.get("source", "未知"),
                page=doc.metadata.get("page"),
                content=doc.page_content[:200],
            )
            for i, doc in enumerate(docs)
        ]

        context = "\n\n".join(
            f"[&#123;c.citation_id&#125;] &#123;c.source&#125;\n&#123;c.content&#125;" for c in citations
        )

        prompt = f"基于以下信息回答问题（标注引用编号）：\n\n&#123;context&#125;\n\n问题: &#123;question&#125;"

        # 流式输出
        async for chunk in self.llm.astream([HumanMessage(content=prompt)]):
            if chunk.content:
                yield &#123;"type": "token", "content": chunk.content&#125;

        # 最后输出引用列表
        yield &#123;"type": "citations", "citations": [
            &#123;"id": c.citation_id, "source": c.source, "page": c.page&#125;
            for c in citations
        ]&#125;

        yield &#123;"type": "done"&#125;
```

## 四、引用验证

```python
class CitationValidator:
    """引用验证器：确保引用真实存在。"""

    @staticmethod
    def validate(
        answer: str,
        available_citations: list[SourceCitation],
    ) -> dict:
        """验证回答中的引用是否合法。"""
        import re
        # 提取回答中的所有引用
        cited_ids = set(int(m) for m in re.findall(r'\[(\d+)\]', answer))
        valid_ids = &#123;c.citation_id for c in available_citations&#125;

        # 找出非法引用（引用了不存在的编号）
        invalid = cited_ids - valid_ids
        # 找出未引用的来源
        unused = valid_ids - cited_ids

        return &#123;
            "cited_count": len(cited_ids),
            "valid_citations": len(cited_ids & valid_ids),
            "invalid_citations": list(invalid),
            "unused_sources": list(unused),
            "citation_coverage": round(
                len(cited_ids & valid_ids) / max(len(valid_ids), 1), 4
            ),
            "is_valid": len(invalid) == 0,
        &#125;
```

## 六、置信度展示

```python
class ConfidenceScorer:
    """回答置信度评分。"""

    @staticmethod
    def score(
        used_citations: int,
        total_available: int,
        avg_relevance: float,
        has_unsupported: bool = False,
    ) -> dict:
        """计算回答置信度。"""
        # 引用覆盖率（引用了多少比例的可用来源）
        coverage = used_citations / max(total_available, 1)

        # 综合：来源数量 * 相关度 * 覆盖率
        base_score = (used_citations / 5) * avg_relevance * (0.5 + coverage * 0.5)

        # 有无支撑的内容降低置信度
        if has_unsupported:
            base_score *= 0.7

        # 归一化到0-1
        confidence = min(1.0, base_score)

        if confidence >= 0.8:
            level = "🟢 高置信度"
        elif confidence >= 0.5:
            level = "🟡 中置信度，建议核实"
        else:
            level = "🔴 低置信度，信息可能不完整"

        return &#123;
            "score": round(confidence, 2),
            "level": level,
            "sources_used": used_citations,
            "avg_relevance": round(avg_relevance, 2),
        &#125;
```

## 七、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 每个事实陈述标注引用 | [1][2]格式 | ★★★ |
| 引用关联到具体页码 | 可定位到原文位置 | ★★★ |
| 验证引用合法性 | 不能引用不存在的编号 | ★★☆ |
| 展示置信度 | 让用户判断可信程度 | ★★☆ |
| 悬停显示来源 | 好体验 | ★★☆ |
| 无来源信息要说明 | 不能编造 | ★★★ |
