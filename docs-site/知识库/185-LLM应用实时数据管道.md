## 二、数据源接入

```python
from dataclasses import dataclass, field
from enum import Enum
from datetime import datetime
import asyncio

class DataSource(str, Enum):
    WEBHOOK = "webhook"       # Webhook推送
    FILE_WATCH = "file_watch" # 文件监听
    DATABASE = "database"     # 数据库CDC
    API_POLL = "api_poll"     # API轮询
    MANUAL = "manual"         # 手动添加

@dataclass
class DataEvent:
    """数据事件。"""
    event_id: str
    source: DataSource
    event_type: str          # create/update/delete
    content: str = ""
    metadata: dict = field(default_factory=dict)
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())

class DataEventQueue:
    """数据事件队列。"""
    def __init__(self):
        self._queue: asyncio.Queue = asyncio.Queue()

    async def push(self, event: DataEvent):
        await self._queue.put(event)

    async def pop(self) -> DataEvent:
        return await self._queue.get()

    def size(self) -> int:
        return self._queue.qsize()

class WebhookReceiver:
    """Webhook接收器。"""
    def __init__(self, queue: DataEventQueue):
        self.queue = queue

    async def receive(self, content: str, source: str = "webhook"):
        event = DataEvent(
            event_id=f"wh_&#123;datetime.now().strftime('%Y%m%d%H%M%S')&#125;",
            source=DataSource.WEBHOOK,
            event_type="create",
            content=content,
            metadata=&#123;"source_url": source&#125;,
        )
        await self.queue.push(event)
        return &#123;"status": "queued", "event_id": event.event_id&#125;

class FileWatcher:
    """文件监听器。"""
    def __init__(self, queue: DataEventQueue):
        self.queue = queue
        self._watching: dict[str, float] = &#123;&#125;  # 文件→上次修改时间

    async def check_changes(self, directory: str):
        """检查文件变更。"""
        from pathlib import Path
        for file_path in Path(directory).rglob("*"):
            if not file_path.is_file():
                continue
            mtime = file_path.stat().st_mtime
            if str(file_path) not in self._watching:
                self._watching[str(file_path)] = mtime
                await self._notify("create", str(file_path))
            elif self._watching[str(file_path)] != mtime:
                self._watching[str(file_path)] = mtime
                await self._notify("update", str(file_path))

    async def _notify(self, event_type: str, file_path: str):
        await self.queue.push(DataEvent(
            event_id=f"fw_&#123;file_path&#125;",
            source=DataSource.FILE_WATCH,
            event_type=event_type,
            content="",
            metadata=&#123;"file_path": file_path&#125;,
        ))
```

## 三、流处理器

```python
class StreamProcessor:
    """流处理器：消费事件→处理→写入向量库。"""

    def __init__(self, queue: DataEventQueue, vectorstore, embeddings):
        self.queue = queue
        self.vectorstore = vectorstore
        self.embeddings = embeddings
        self.processed: list[dict] = []

    async def start(self, max_events: int = 100):
        """启动流处理。"""
        for _ in range(max_events):
            event = await self.queue.pop()
            result = await self._process_event(event)
            self.processed.append(result)

    async def _process_event(self, event: DataEvent) -> dict:
        """处理单个事件。"""
        if event.event_type == "create":
            return await self._handle_create(event)
        elif event.event_type == "update":
            return await self._handle_update(event)
        elif event.event_type == "delete":
            return await self._handle_delete(event)
        return &#123;"status": "unknown_event"&#125;

    async def _handle_create(self, event: DataEvent) -> dict:
        """处理新增。"""
        # 清洗+分块+嵌入+入库
        from langchain_text_splitters import RecursiveCharacterTextSplitter
        from langchain_core.documents import Document

        splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
        chunks = splitter.split_text(event.content)

        docs = [Document(page_content=c, metadata=event.metadata) for c in chunks]
        await self.vectorstore.aadd_documents(docs)

        return &#123;"event_id": event.event_id, "status": "created", "chunks": len(chunks)&#125;

    async def _handle_update(self, event: DataEvent) -> dict:
        """处理更新。"""
        # 先删除旧的，再添加新的
        file_path = event.metadata.get("file_path")
        if file_path:
            # 删除该来源的所有文档
            # await self.vectorstore.delete(filter=&#123;"file_path": file_path&#125;)
            pass
        # 重新处理
        if event.content:
            return await self._handle_create(event)
        return &#123;"status": "updated"&#125;

    async def _handle_delete(self, event: DataEvent) -> dict:
        """处理删除。"""
        doc_id = event.metadata.get("doc_id")
        if doc_id:
            # await self.vectorstore.delete(filter=&#123;"doc_id": doc_id&#125;)
            pass
        return &#123;"status": "deleted", "doc_id": doc_id&#125;

    def stats(self) -> dict:
        """处理统计。"""
        from collections import Counter
        status_counts = Counter(p["status"] for p in self.processed)
        return &#123;
            "total_processed": len(self.processed),
            "by_status": dict(status_counts),
            "queue_remaining": self.queue.size(),
        &#125;
```

## 五、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 增量而非全量 | 只处理变更 | ★★★ |
| 事件队列缓冲 | 削峰填谷 | ★★★ |
| 并行处理 | 加速吞吐 | ★★☆ |
| 更新通知在线 | 缓存一致性 | ★★☆ |
| 监控处理延迟 | 保证实时性 | ★★☆ |
