# 查询理解与意图识别

> 用户输入"帮我看看这个"——什么意思？看什么？意图是什么？查询理解是 RAG 和 Agent 的第一步：先理解用户在问什么，再决定怎么检索和回答。这份指南覆盖意图分类、查询改写和槽位提取。

## 二、意图识别

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage
from enum import Enum
from dataclasses import dataclass
import json, re

class QueryIntent(str, Enum):
    SEARCH = "search"           # 信息搜索
    QA = "qa"                   # 问答
    TASK = "task"               # 执行任务
    CHITCHAT = "chitchat"       # 闲聊
    CLARIFICATION = "clarify"   # 需要澄清
    TRANSLATION = "translate"   # 翻译
    CODE = "code"               # 代码相关
    COMPLAINT = "complaint"     # 投诉

INTENT_PROMPT = """判断用户输入的意图。

用户输入: &#123;input&#125;

可选意图:
- search: 搜索信息（"帮我查...", "有没有..."）
- qa: 问答（"什么是...", "怎么..."）
- task: 执行任务（"帮我发...", "创建..."）
- chitchat: 闲聊（"你好", "谢谢"）
- clarify: 模糊需要澄清（"帮我看看这个"）
- translate: 翻译
- code: 代码相关
- complaint: 投诉

只输出意图名称:"""

@dataclass
class IntentResult:
    intent: QueryIntent
    confidence: float
    sub_intent: str = ""

class IntentClassifier:
    """意图分类器。"""

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def classify(self, user_input: str) -> IntentResult:
        """识别用户意图。"""
        prompt = INTENT_PROMPT.format(input=user_input)
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        result = response.content.strip().lower()

        # 匹配到已知意图
        for intent in QueryIntent:
            if intent.value in result:
                return IntentResult(intent=intent, confidence=0.9)

        # 未知意图
        return IntentResult(intent=QueryIntent.CLARIFICATION, confidence=0.3)
```

```python
REWRITE_PROMPT = """请改写以下用户查询，使其更适合检索。

原始查询: &#123;query&#125;
对话历史: &#123;history&#125;

改写要求：
1. 消除指代词（"这个""那个"→具体内容）
2. 口语化转为书面化
3. 补全省略的上下文
4. 保持原意

直接输出改写后的查询:"""

class QueryRewriter:
    """查询改写器。"""

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def rewrite(
        self,
        query: str,
        history: list[dict] = None,
    ) -> str:
        """改写查询以提升检索效果。"""
        # 简化历史
        history_text = ""
        if history:
            recent = history[-4:]  # 最近2轮
            history_text = "\n".join(
                f"&#123;'用户' if m['role'] == 'user' else '助手'&#125;: &#123;m['content'][:100]&#125;"
                for m in recent
            )

        prompt = REWRITE_PROMPT.format(query=query, history=history_text)
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content.strip()

    async def split_multi_intent(
        self,
        query: str,
    ) -> list[str]:
        """拆分多意图查询。

        "查天气和订机票" → ["查天气", "订机票"]
        """
        prompt = f"""判断以下查询是否包含多个意图。如果是，拆分为多个独立查询。

查询: &#123;query&#125;

如果只有一个意图，直接输出原查询。
如果多个意图，每行输出一个。

输出:"""

        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        lines = [l.strip() for l in response.content.split("\n") if l.strip()]
        return lines if len(lines) > 1 else [query]
```

## 四、槽位提取

```python
SLOT_EXTRACTION_PROMPT = """从用户输入中提取关键信息槽位。

用户输入: &#123;input&#125;

已知意图: &#123;intent&#125;

请提取与该意图相关的关键信息。如果未提及则留空。

输出JSON格式:
```json
&#123;&#123;
  "slot1": "值或空",
  "slot2": "值或空"
&#125;&#125;
```"""

class SlotExtractor:
    """槽位提取器。"""

    # 各意图对应的槽位模板
    INTENT_SLOTS = &#123;
        QueryIntent.SEARCH: ["query", "date_range", "source"],
        QueryIntent.TASK: ["action", "target", "parameters"],
        QueryIntent.QA: ["question", "context"],
        QueryIntent.COMPLAINT: ["order_id", "issue", "expected_resolution"],
    &#125;

    def __init__(self, llm: BaseChatModel):
        self.llm = llm

    async def extract(
        self,
        user_input: str,
        intent: QueryIntent,
    ) -> dict:
        """提取槽位。"""
        slots = self.INTENT_SLOTS.get(intent, ["query"])

        prompt = SLOT_EXTRACTION_PROMPT.format(
            input=user_input,
            intent=intent.value,
        )
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])

        json_match = re.search(r'\&#123;.*\&#125;', response.content, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        return &#123;&#125;

    def get_missing_slots(
        self,
        extracted: dict,
        intent: QueryIntent,
    ) -> list[str]:
        """获取未填充的槽位。"""
        required = self.INTENT_SLOTS.get(intent, [])
        return [s for s in required if not extracted.get(s)]
```

## 五、完整查询理解管线

```python
class QueryUnderstandingPipeline:
    """查询理解管线：意图→改写→槽位。"""

    def __init__(self, llm: BaseChatModel):
        self.intent_classifier = IntentClassifier(llm)
        self.rewriter = QueryRewriter(llm)
        self.slot_extractor = SlotExtractor(llm)

    async def understand(
        self,
        user_input: str,
        history: list[dict] = None,
    ) -> dict:
        """完整的查询理解流程。"""
        # 1. 意图识别
        intent_result = await self.intent_classifier.classify(user_input)

        # 2. 查询改写
        rewritten = await self.rewriter.rewrite(user_input, history)

        # 3. 槽位提取
        slots = await self.slot_extractor.extract(user_input, intent_result.intent)

        # 4. 缺失槽位
        missing = self.slot_extractor.get_missing_slots(
            slots, intent_result.intent
        )

        return &#123;
            "original_query": user_input,
            "intent": intent_result.intent.value,
            "confidence": intent_result.confidence,
            "rewritten_query": rewritten,
            "slots": slots,
            "missing_slots": missing,
            "needs_clarification": len(missing) > 0,
            "clarification_question": self._generate_clarification(missing) if missing else "",
        &#125;

    @staticmethod
    def _generate_clarification(missing_slots: list[str]) -> str:
        """生成澄清问题。"""
        if not missing_slots:
            return ""
        slot_names = &#123;
            "order_id": "订单号",
            "date_range": "日期范围",
            "query": "具体查询内容",
        &#125;
        names = [slot_names.get(s, s) for s in missing_slots]
        return f"请提供&#123; '、'.join(names) &#125;。"
```

## 七、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 先识别意图再决定流程 | 不同意图不同处理 | ★★★ |
| 模糊查询要改写 | 指代消解提升检索 | ★★★ |
| 多意图要拆分 | 分别处理再合并 | ★★☆ |
| 缺槽位要追问 | 不要猜测用户意图 | ★★☆ |
| 意图用小模型分类 | 省成本 | ★★☆ |
