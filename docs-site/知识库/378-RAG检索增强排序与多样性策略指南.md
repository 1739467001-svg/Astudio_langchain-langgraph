## 二、MMR 与多样性实现

```python
from dataclasses import dataclass, field
from langchain_core.documents import Document
from langchain_openai import OpenAIEmbeddings
import numpy as np
from typing import Optional

@dataclass
class DiversityConfig:
    """多样性配置。"""
    lambda_param: float = 0.7    # 0=最大多样性, 1=最大相关性
    top_k_retrieval: int = 20     # 检索召回数
    top_n_final: int = 5         # 最终选取数
    similarity_threshold: float = 0.85  # 去重阈值
    min_topic_diversity: int = 3  # 最少覆盖不同主题数

class MMRReranker:
    """Maximal Marginal Relevance 重排序器。"""

    def __init__(self, embeddings: OpenAIEmbeddings, config: DiversityConfig = DiversityConfig()):
        self.embeddings = embeddings
        self.config = config

    async def rerank(self, query: str, documents: list[Document]) -> list[Document]:
        """MMR重排序。"""
        if len(documents) <= self.config.top_n_final:
            return documents

        # 1. 嵌入查询和文档
        query_emb = np.array(await self.embeddings.aembed_query(query))
        doc_embs = np.array(await self.embeddings.aembed_documents([d.page_content for d in documents]))

        # 2. 去重（基于内容相似度）
        documents, doc_embs = self._dedup(documents, doc_embs)

        # 3. MMR选择
        selected_indices = []
        remaining = list(range(len(documents)))

        # 第一轮选最高相关的
        relevances = [self._cosine(query_emb, doc_embs[i]) for i in remaining]
        first = remaining[np.argmax(relevances)]
        selected_indices.append(first)
        remaining.remove(first)

        # 后续选MMR最大的
        while len(selected_indices) < self.config.top_n_final and remaining:
            mmr_scores = []
            for i in remaining:
                rel = self._cosine(query_emb, doc_embs[i])
                max_sim = max(self._cosine(doc_embs[i], doc_embs[j]) for j in selected_indices)
                mmr = self.config.lambda_param * rel - (1 - self.config.lambda_param) * max_sim
                mmr_scores.append((i, mmr))

            best = max(mmr_scores, key=lambda x: x[1])
            selected_indices.append(best[0])
            remaining.remove(best[0])

        return [documents[i] for i in selected_indices]

    def _dedup(self, documents: list[Document], embeddings: np.ndarray) -> tuple:
        """基于嵌入相似度去重。"""
        if len(documents) <= 1:
            return documents, embeddings

        kept = [0]
        kept_embs = [embeddings[0]]

        for i in range(1, len(documents)):
            is_dup = False
            for j in kept:
                sim = self._cosine(embeddings[i], embeddings[j])
                if sim > self.config.similarity_threshold:
                    is_dup = True
                    break
            if not is_dup:
                kept.append(i)
                kept_embs.append(embeddings[i])

        return [documents[i] for i in kept], np.array(kept_embs)

    def _cosine(self, a: np.ndarray, b: np.ndarray) -> float:
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-8))


class ContextAssembler:
    """上下文组装器——把检索结果组织成最优上下文。"""

    def __init__(self, max_tokens: int = 4000, tokens_per_char: float = 0.25):
        self.max_tokens = max_tokens
        self.tokens_per_char = tokens_per_char

    def assemble(self, documents: list[Document], query: str = "") -> str:
        """组装上下文文本。"""
        # 按相关性排序（假设已排序）
        # 逐个添加，直到达到Token上限
        parts = []
        current_tokens = 0

        for i, doc in enumerate(documents):
            content = doc.page_content
            doc_tokens = len(content) * self.tokens_per_char

            if current_tokens + doc_tokens > self.max_tokens:
                # 截断最后一个文档
                remaining = int((self.max_tokens - current_tokens) / self.tokens_per_char)
                if remaining > 100:  # 至少保留100字符
                    content = content[:remaining] + "..."
                    parts.append(f"[文档&#123;i+1&#125;] &#123;content&#125;")
                    current_tokens += remaining * self.tokens_per_char
                break

            source = doc.metadata.get("source", f"文档&#123;i+1&#125;")
            parts.append(f"[&#123;source&#125;] &#123;content&#125;")
            current_tokens += doc_tokens

        return "\n\n---\n\n".join(parts)

    def estimate_tokens(self, text: str) -> int:
        return int(len(text) * self.tokens_per_char)


class DiversityEvaluator:
    """多样性评估器。"""

    @staticmethod
    def evaluate(documents: list[Document]) -> dict:
        """评估结果集的多样性。"""
        sources = set(d.metadata.get("source", "unknown") for d in documents)
        # 简单的内容前缀多样性
        prefixes = set(d.page_content[:50] for d in documents)
        lengths = [len(d.page_content) for d in documents]
        avg_len = sum(lengths) / max(len(lengths), 1)

        return &#123;
            "total_docs": len(documents),
            "unique_sources": len(sources),
            "unique_prefixes": len(prefixes),
            "avg_doc_length": round(avg_len, 0),
            "diversity_score": round(len(prefixes) / max(len(documents), 1), 2),
        &#125;
```

### 使用示例

```python
import asyncio

async def main():
    # 模拟检索结果
    docs = [
        Document(page_content=f"文档&#123;i&#125;: 关于AI的内容...", metadata=&#123;"source": f"src_&#123;i&#125;"&#125;)
        for i in range(20)
    ]

    # MMR重排序
    embedder = OpenAIEmbeddings()
    reranker = MMRReranker(embedder, DiversityConfig(lambda_param=0.6, top_n_final=5))
    reranked = await reranker.rerank("什么是AI", docs)

    # 上下文组装
    assembler = ContextAssembler(max_tokens=2000)
    context = assembler.assemble(reranked)
    print(f"上下文长度: &#123;len(context)&#125; 字符, 约&#123;assembler.estimate_tokens(context)&#125; tokens")
    print(f"多样性: &#123;DiversityEvaluator.evaluate(reranked)&#125;")

asyncio.run(main())
```

## 三、多样性策略对比

| 策略 | 原理 | 优点 | 缺点 | 适用 |
|------|------|------|------|------|
| MMR | 相关性-冗余度平衡 | 信息密度高 | 需嵌入计算 | 通用 |
| 去重 | 相似度阈值过滤 | 简单快速 | 可能误删 | 文档多 |
| 主题聚类 | 按主题分组选代表 | 覆盖面广 | 需聚类模型 | 大规模 |
| 时间衰减 | 旧文档降权 | 时效性好 | 不适用通用 | 新闻类 |
| 源加权 | 按来源可信度 | 可信度高 | 主观 | 生产 |

## 四、lambda 参数效果

| lambda值 | 行为 | 结果特点 | 适用 |
|----------|------|----------|------|
| 0.3 | 重多样性 | 覆盖面广，信息丰富 | 综述类 |
| 0.5 | 平衡 | 相关+多样 | 通用 |
| 0.7 | 重相关性 | 精准命中 | 事实问答 |
| 0.9 | 几乎只看相关性 | 与不用MMR类似 | 精确检索 |

## 五、最佳实践

| 实践 | 说明 | 优先级 |
