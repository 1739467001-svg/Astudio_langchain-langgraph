## 二、三种处理器

```python
from langchain_core.callbacks import BaseCallbackHandler

class QuickLoggingCallback(BaseCallbackHandler):
    """快速日志回调——记录关键事件。"""
    def __init__(self):
        self.events = []
    def on_llm_start(self, serialized, prompts, **kwargs):
        self.events.append(&#123;"type": "llm_start"&#125;)
    def on_tool_start(self, serialized, input_str, **kwargs):
        self.events.append(&#123;"type": "tool_start", "name": serialized.get("name", "")&#125;)
    def on_tool_end(self, output, **kwargs):
        self.events.append(&#123;"type": "tool_end", "output": str(output)[:50]&#125;)
    def on_error(self, error, **kwargs):
        self.events.append(&#123;"type": "error", "error": str(error)[:100]&#125;)

class QuickMetricsCallback(BaseCallbackHandler):
    """快速指标回调——采集性能。"""
    def __init__(self):
        self.llm_calls = 0
        self.tool_calls = 0
        self.errors = 0
    def on_llm_start(self, serialized, prompts, **kwargs):
        self.llm_calls += 1
    def on_tool_start(self, serialized, input_str, **kwargs):
        self.tool_calls += 1
    def on_error(self, error, **kwargs):
        self.errors += 1
    def summary(self):
        return &#123;"llm_calls": self.llm_calls, "tool_calls": self.tool_calls, "errors": self.errors&#125;

class QuickTracingCallback(BaseCallbackHandler):
    """快速追踪回调——执行轨迹。"""
    def __init__(self):
        self.trace = []
        self._depth = 0
    def on_chain_start(self, serialized, inputs, **kwargs):
        self.trace.append(&#123;"event": "start", "name": serialized.get("name", ""), "depth": self._depth&#125;)
        self._depth += 1
    def on_chain_end(self, outputs, **kwargs):
        self._depth = max(0, self._depth - 1)
        self.trace.append(&#123;"event": "end", "depth": self._depth&#125;)
    def get_trace_text(self):
        return "\n".join(f"&#123;'  '*t['depth']&#125;&#123;t['event']&#125;: &#123;t.get('name','')&#125;" for t in self.trace)
```
