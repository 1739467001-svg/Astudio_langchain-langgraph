```mermaid

    end

    L1 --> L2 --> L3
    L1 --> L4

    style L2 fill:'#FFF9C4'
```

| 事件 | 触发时机 | 典型用途 |
|------|---------|---------|
| on_chain_start | Chain开始执行 | 记录输入 |
| on_chain_end | Chain执行完成 | 记录输出 |
| on_llm_start | LLM API调用前 | 记录prompt |
| on_llm_end | LLM API返回后 | 记录response/Token/耗时 |
| on_llm_new_token | 流式输出的每个token | 实时显示 |
| on_tool_start | 工具执行前 | 记录工具调用 |
| on_tool_end | 工具执行后 | 记录工具结果 |
| on_retriever_start | 检索前 | 记录查询 |
| on_retriever_end | 检索后 | 记录检索结果 |
| on_error | 任何环节出错 | 记录错误 |

## 三、内置回调

```python
from langchain.callbacks import StdOutCallbackHandler
from langchain_openai import ChatOpenAI

# 内置：打印到控制台
handler = StdOutCallbackHandler()
llm = ChatOpenAI(model="gpt-4o-mini", callbacks=[handler])
response = llm.invoke("你好")
# 控制台会打印完整的调用信息
```

## 四、自定义回调处理器

### 4.1 计时与Token统计

```python
import time
from langchain_core.callbacks import BaseCallbackHandler

class MetricsHandler(BaseCallbackHandler):
    """计时+Token统计回调"""

    def __init__(self):
        self.metrics = []
        self._start_time = None

    def on_llm_start(self, serialized, prompts, **kwargs):
        self._start_time = time.time()

    def on_llm_end(self, response, **kwargs):
        elapsed = time.time() - self._start_time if self._start_time else 0
        usage = response.llm_output or &#123;&#125;
        token_usage = usage.get("token_usage", &#123;&#125;)

        self.metrics.append(&#123;
            "latency": round(elapsed, 2),
            "input_tokens": token_usage.get("prompt_tokens", 0),
            "output_tokens": token_usage.get("completion_tokens", 0),
            "total_tokens": token_usage.get("total_tokens", 0),
        &#125;)

    def summary(self):
        if not self.metrics:
            return "无数据"
        n = len(self.metrics)
        total = sum(m["total_tokens"] for m in self.metrics)
        avg_latency = sum(m["latency"] for m in self.metrics) / n
        return f"调用次数:&#123;n&#125; 平均延迟:&#123;avg_latency:.2f&#125;s 总Token:&#123;total&#125;"

# 使用
handler = MetricsHandler()
llm = ChatOpenAI(model="gpt-4o-mini", callbacks=[handler])
llm.invoke("你好")
llm.invoke("解释AI")
print(handler.summary())
```

### 4.2 流式输出回调

```python
class StreamHandler(BaseCallbackHandler):
    """流式输出到自定义位置"""
    def __init__(self):
        self.tokens = []

    def on_llm_new_token(self, token, **kwargs):
        print(token, end="", flush=True)
        self.tokens.append(token)

    def get_full_text(self):
        return "".join(self.tokens)
```

### 4.3 工具调用追踪

```python
class ToolTracer(BaseCallbackHandler):
    """追踪Agent的工具调用链"""
    def __init__(self):
        self.tool_calls = []

    def on_tool_start(self, serialized, input_str, **kwargs):
        tool_name = serialized.get("name", "unknown")
        self.tool_calls.append(&#123;"tool": tool_name, "input": input_str, "status": "started"&#125;)
        print(f"🔧 调用工具: &#123;tool_name&#125;(&#123;input_str[:50]&#125;)")

    def on_tool_end(self, output, **kwargs):
        print(f"   → 结果: &#123;str(output)[:100]&#125;")
        if self.tool_calls:
            self.tool_calls[-1]["output"] = output
            self.tool_calls[-1]["status"] = "completed"
```

### 4.4 错误捕获回调

```python
class ErrorCapture(BaseCallbackHandler):
    """捕获并记录所有错误"""
    def __init__(self):
        self.errors = []

    def on_llm_error(self, error, **kwargs):
        self.errors.append(&#123;"stage": "llm", "error": str(error)&#125;)
        print(f"❌ LLM错误: &#123;error&#125;")

    def on_tool_error(self, error, **kwargs):
        self.errors.append(&#123;"stage": "tool", "error": str(error)&#125;)
        print(f"❌ 工具错误: &#123;error&#125;")

    def on_chain_error(self, error, **kwargs):
        self.errors.append(&#123;"stage": "chain", "error": str(error)&#125;)
        print(f"❌ Chain错误: &#123;error&#125;")
```
