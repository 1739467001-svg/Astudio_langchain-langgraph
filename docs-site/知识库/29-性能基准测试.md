## 三、延迟基准测试

### 3.1 测量方法

```python
import time
import statistics
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
chain = (
    ChatPromptTemplate.from_template("用一句话解释&#123;concept&#125;")
    | llm | StrOutputParser()
)

def benchmark_latency(chain, inputs: list[dict], warmup: int = 1) -> dict:
    """延迟基准测试"""
    # 预热（第一次调用可能较慢）
    for _ in range(warmup):
        chain.invoke(inputs[0])

    latencies = []
    for inp in inputs:
        start = time.time()
        result = chain.invoke(inp)
        elapsed = time.time() - start
        latencies.append(elapsed)

    latencies.sort()
    n = len(latencies)

    return &#123;
        "count": n,
        "mean": statistics.mean(latencies),
        "median_p50": latencies[n // 2],
        "p95": latencies[int(n * 0.95)] if n > 20 else max(latencies),
        "p99": latencies[int(n * 0.99)] if n > 100 else max(latencies),
        "min": min(latencies),
        "max": max(latencies),
        "latencies": latencies,
    &#125;

# 运行基准测试
inputs = [&#123;"concept": "AI"&#125;, &#123;"concept": "区块链"&#125;, &#123;"concept": "量子计算"&#125;] * 5
result = benchmark_latency(chain, inputs, warmup=2)

print(f"延迟基准:")
print(f"  平均: &#123;result['mean']:.2f&#125;s")
print(f"  P50:  &#123;result['median_p50']:.2f&#125;s")
print(f"  P95:  &#123;result['p95']:.2f&#125;s")
print(f"  范围: &#123;result['min']:.2f&#125;s - &#123;result['max']:.2f&#125;s")
```

### 3.2 首字延迟（流式输出）

```python
async def benchmark_ttft(chain, input_data: dict) -> dict:
    """测量首字延迟(Time To First Token)"""
    start = time.time()
    first_token_time = None
    total_time = None

    async for chunk in chain.astream(input_data):
        if first_token_time is None:
            first_token_time = time.time() - start  # TTFT
            break

    # 完整流式
    full_text = ""
    start = time.time()
    async for chunk in chain.astream(input_data):
        full_text += chunk
    total_time = time.time() - start

    return &#123;
        "ttft": first_token_time,  # 首字延迟
        "total": total_time,       # 总时间
        "text_length": len(full_text),
        "tokens_per_second": len(full_text) / total_time if total_time > 0 else 0,
    &#125;
```

## 四、成本基准测试

```python
def benchmark_cost(llm, questions: list[str]) -> dict:
    """成本基准测试"""
    total_input = 0
    total_output = 0
    results = []

    for q in questions:
        response = llm.invoke(q)
        usage = response.usage_metadata

        input_t = usage.get("input_tokens", 0)
        output_t = usage.get("output_tokens", 0)
        total_input += input_t
        total_output += output_t

        results.append(&#123;
            "question": q[:50],
            "input_tokens": input_t,
            "output_tokens": output_t,
            "total_tokens": input_t + output_t,
        &#125;)

    n = len(questions)
    # GPT-4o-mini 定价
    input_cost = total_input * 0.15 / 1_000_000
    output_cost = total_output * 0.60 / 1_000_000

    return &#123;
        "total_calls": n,
        "total_input_tokens": total_input,
        "total_output_tokens": total_output,
        "avg_input_per_call": total_input / n,
        "avg_output_per_call": total_output / n,
        "total_cost": input_cost + output_cost,
        "avg_cost_per_call": (input_cost + output_cost) / n,
        "estimated_monthly_cost": (input_cost + output_cost) / n * 1000,  # 假设每天1000次
        "details": results,
    &#125;
```

## 五、吞吐量基准测试

### 5.1 并发 vs 串行

```python
import asyncio
import time

async def benchmark_throughput(chain, inputs: list[dict]) -> dict:
    """吞吐量基准测试"""
    # 串行
    start = time.time()
    serial_results = []
    for inp in inputs:
        r = await chain.ainvoke(inp)
        serial_results.append(r)
    serial_time = time.time() - start

    # 并发
    start = time.time()
    tasks = [chain.ainvoke(inp) for inp in inputs]
    concurrent_results = await asyncio.gather(*tasks)
    concurrent_time = time.time() - start

    n = len(inputs)
    return &#123;
        "count": n,
        "serial_time": serial_time,
        "concurrent_time": concurrent_time,
        "serial_qps": n / serial_time,
        "concurrent_qps": n / concurrent_time,
        "speedup": serial_time / concurrent_time,
    &#125;
```

### 5.2 batch 效率

```python
def benchmark_batch(chain, inputs: list[dict]) -> dict:
    """batch vs 循环invoke对比"""
    # 循环invoke
    start = time.time()
    for inp in inputs:
        chain.invoke(inp)
    loop_time = time.time() - start

    # batch
    start = time.time()
    chain.batch(inputs)
    batch_time = time.time() - start

    return &#123;
        "count": len(inputs),
        "loop_time": loop_time,
        "batch_time": batch_time,
        "speedup": loop_time / batch_time,
    &#125;
```

## 六、模型对比基准

```python
def compare_models(models: dict, test_question: str) -> list:
    """对比不同模型的性能和成本"""
    results = []
    for name, llm in models.items():
        start = time.time()
        response = llm.invoke(test_question)
        elapsed = time.time() - start

        usage = response.usage_metadata
        input_t = usage.get("input_tokens", 0)
        output_t = usage.get("output_tokens", 0)

        results.append(&#123;
            "model": name,
            "latency": elapsed,
            "input_tokens": input_t,
            "output_tokens": output_t,
            "total_tokens": input_t + output_t,
            "answer_length": len(response.content),
            "answer_preview": response.content[:100],
        &#125;)
    return results

# 使用
from langchain_openai import ChatOpenAI

models = &#123;
    "gpt-4o-mini": ChatOpenAI(model="gpt-4o-mini"),
    "gpt-4o": ChatOpenAI(model="gpt-4o"),
&#125;

comparison = compare_models(models, "解释量子计算")
for r in comparison:
    print(f"\n&#123;r['model']&#125;:")
    print(f"  延迟: &#123;r['latency']:.2f&#125;s")
    print(f"  Token: &#123;r['total_tokens']&#125;")
    print(f"  回答: &#123;r['answer_preview']&#125;...")
```

## 七、基准测试报告

```python
def generate_benchmark_report(chain, test_cases: list[str]) -> str:
    """生成完整的基准测试报告"""
    report = []
    report.append("=" * 55)
    report.append("  性能基准测试报告")
    report.append("=" * 55)

    # 延迟基准
    lat_results = benchmark_latency(chain, [&#123;"concept": tc&#125; for tc in test_cases])
    report.append(f"\n📊 延迟基准 (&#123;lat_results['count']&#125; 次调用):")
    report.append(f"   平均: &#123;lat_results['mean']:.2f&#125;s")
    report.append(f"   P50:  &#123;lat_results['median_p50']:.2f&#125;s")
    report.append(f"   P95:  &#123;lat_results['p95']:.2f&#125;s")
    report.append(f"   范围: &#123;lat_results['min']:.2f&#125;s - &#123;lat_results['max']:.2f&#125;s")

    # 成本基准
    cost_results = benchmark_cost(chain, test_cases[:5])
    report.append(f"\n💰 成本基准 (&#123;cost_results['total_calls']&#125; 次调用):")
    report.append(f"   总Token: &#123;cost_results['total_input_tokens'] + cost_results['total_output_tokens']&#125;")
    report.append(f"   总成本: $&#123;cost_results['total_cost']:.4f&#125;")
    report.append(f"   每次成本: $&#123;cost_results['avg_cost_per_call']:.4f&#125;")
    report.append(f"   月度估算(1000次/天): $&#123;cost_results['estimated_monthly_cost']:.2f&#125;")

    # 吞吐量基准
    report.append(f"\n⚡ 吞吐量:")
    report.append(f"   串行QPS: &#123;1/lat_results['mean']:.1f&#125;")

    report.append("\n" + "=" * 55)
    return "\n".join(report)

# 运行
report = generate_benchmark_report(
    chain,
    ["AI", "区块链", "量子计算", "云计算", "物联网"]
)
print(report)
```
