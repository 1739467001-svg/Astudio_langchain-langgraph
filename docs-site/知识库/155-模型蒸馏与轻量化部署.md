## 三、蒸馏数据生成

```python
from langchain_core.language_models import BaseChatModel
from langchain_core.messages import HumanMessage, SystemMessage
import json
from dataclasses import dataclass
from typing import Any

@dataclass
class DistillExample:
    """蒸馏训练样本。"""
    input: str
    output: str
    teacher_model: str = ""

class DistillationDataGenerator:
    """蒸馏数据生成器。

    用大模型（老师）生成高质量输入-输出对，
    作为小模型（学生）的微调数据。
    """

    def __init__(self, teacher_model: BaseChatModel):
        self.teacher = teacher_model

    async def generate_examples(
        self,
        task_description: str,
        num_examples: int = 100,
        batch_size: int = 20,
    ) -> list[DistillExample]:
        """生成蒸馏训练数据。

        Args:
            task_description: 任务描述
            num_examples: 目标样本数
            batch_size: 每批生成数

        Returns:
            蒸馏训练样本列表
        """
        import asyncio
        all_examples = []
        num_batches = (num_examples + batch_size - 1) // batch_size

        for batch_idx in range(num_batches):
            actual_size = min(batch_size, num_examples - len(all_examples))
            if actual_size <= 0:
                break

            prompt = f"""请为以下任务生成&#123;actual_size&#125;个多样化的输入-输出样本。

任务描述: &#123;task_description&#125;

要求：
1. 输入要多样化，覆盖不同场景
2. 输出要高质量，是"标准答案"
3. 输出JSON数组格式

```json
[
  &#123;&#123;"input": "...", "output": "..."&#125;&#125;,
  ...
]
```"""

            response = await self.teacher.ainvoke([
                SystemMessage(content="你是高质量训练数据生成器。"),
                HumanMessage(content=prompt),
            ])

            import re
            json_match = re.search(r'\[.*\]', response.content, re.DOTALL)
            if json_match:
                data = json.loads(json_match.group())
                for item in data:
                    all_examples.append(DistillExample(
                        input=item.get("input", ""),
                        output=item.get("output", ""),
                        teacher_model="gpt-4o",
                    ))

        return all_examples[:num_examples]

    async def generate_from_real_queries(
        self,
        real_queries: list[str],
    ) -> list[DistillExample]:
        """用真实用户查询生成蒸馏数据。

        这是最高质量的蒸馏方式：
        用真实查询+老师模型的高质量回答。
        """
        examples = []
        for query in real_queries:
            response = await self.teacher.ainvoke([
                HumanMessage(content=query),
            ])
            examples.append(DistillExample(
                input=query,
                output=response.content,
                teacher_model="gpt-4o",
            ))
        return examples
```

## 四、学生模型微调

```python
class StudentTrainer:
    """学生模型微调器。"""

    @staticmethod
    def prepare_training_data(
        examples: list[DistillExample],
        output_path: str = "training_data.jsonl",
    ) -> str:
        """准备微调数据。

        转换为OpenAI微调API支持的JSONL格式。
        """
        with open(output_path, "w", encoding="utf-8") as f:
            for ex in examples:
                record = &#123;
                    "messages": [
                        &#123;"role": "user", "content": ex.input&#125;,
                        &#123;"role": "assistant", "content": ex.output&#125;,
                    ]
                &#125;
                f.write(json.dumps(record, ensure_ascii=False) + "\n")

        return output_path

    @staticmethod
    async def fine_tune_openai(
        training_file_path: str,
        model: str = "gpt-4o-mini",
        epochs: int = 3,
    ) -> str:
        """用OpenAI微调API微调学生模型。"""
        from openai import OpenAI
        client = OpenAI()

        # 1. 上传训练数据
        with open(training_file_path, "rb") as f:
            file_obj = client.files.create(file=f, purpose="fine-tune")

        # 2. 创建微调任务
        job = client.fine_tuning.jobs.create(
            training_file=file_obj.id,
            model=model,
            hyperparameters=&#123;"n_epochs": epochs&#125;,
        )

        return job.id  # 返回微调任务ID

    @staticmethod
    def prepare_lora_training(
        examples: list[DistillExample],
        output_path: str = "lora_data.json",
    ):
        """准备LoRA微调数据（用于开源模型）。"""
        data = [
            &#123;"instruction": ex.input, "output": ex.output&#125;
            for ex in examples
        ]
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        return output_path
```

## 五、效果评估

```python
class DistillationEvaluator:
    """蒸馏效果评估器。"""

    @staticmethod
    async def evaluate(
        teacher: BaseChatModel,
        student: BaseChatModel,
        test_cases: list[str],
    ) -> dict:
        """对比老师和学生模型的效果。"""
        import asyncio

        teacher_answers = []
        student_answers = []

        for query in test_cases:
            t_resp = await teacher.ainvoke([HumanMessage(content=query)])
            s_resp = await student.ainvoke([HumanMessage(content=query)])
            teacher_answers.append(t_resp.content)
            student_answers.append(s_resp.content)

        # 计算相似度（简化版）
        similarities = []
        for t, s in zip(teacher_answers, student_answers):
            # 用Jaccard相似度做简化评估
            t_words = set(t.split())
            s_words = set(s.split())
            if t_words or s_words:
                sim = len(t_words & s_words) / len(t_words | s_words)
                similarities.append(sim)

        avg_sim = sum(similarities) / len(similarities) if similarities else 0

        return &#123;
            "test_count": len(test_cases),
            "avg_similarity": round(avg_sim, 4),
            "retention_rate": f"&#123;avg_sim * 100:.1f&#125;%",
            "cost_reduction": "约90%",  # 小模型成本低10倍
            "latency_improvement": "约5-10x",
        &#125;
```
