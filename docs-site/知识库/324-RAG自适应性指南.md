## 二、实现

```python
from dataclasses import dataclass
from enum import Enum

class QueryComplexity(str, Enum):
    SIMPLE = "simple"
    COMPLEX = "complex"
    RELATIONAL = "relational"
    FACTOID = "factoid"

class AdaptiveRAGStrategy:
    """自适应RAG——根据查询特征选策略。"""

    @staticmethod
    def analyze(query: str) -> QueryComplexity:
        """分析查询复杂度。"""
        if any(w in query for w in ["的同事", "的关系", "属于", "管理"]):
            return QueryComplexity.RELATIONAL
        if len(query) > 50 or any(w in query for w in ["对比", "分析", "研究", "综合"]):
            return QueryComplexity.COMPLEX
        if len(query) < 15:
            return QueryComplexity.FACTOID
        return QueryComplexity.SIMPLE

    @staticmethod
    def get_config(complexity: QueryComplexity) -> dict:
        """根据复杂度返回RAG配置。"""
        configs = &#123;
            QueryComplexity.SIMPLE: &#123;"k": 3, "rerank": False, "rewrite": False, "verify": False&#125;,
            QueryComplexity.FACTOID: &#123;"k": 5, "rerank": True, "rewrite": False, "verify": True&#125;,
            QueryComplexity.COMPLEX: &#123;"k": 5, "rerank": True, "rewrite": True, "verify": True&#125;,
            QueryComplexity.RELATIONAL: &#123;"k": 3, "rerank": True, "rewrite": True, "verify": True, "use_graph": True&#125;,
        &#125;
        return configs.get(complexity, configs[QueryComplexity.SIMPLE])


class AdaptiveRAGPipeline:
    """自适应RAG管线。"""

    def __init__(self, llm, vectorstore, graph_store=None):
        self.llm = llm
        self.vectorstore = vectorstore
        self.graph = graph_store
        self.strategy = AdaptiveRAGStrategy()

    async def answer(self, query: str) -> dict:
        """自适应回答。"""
        # 1. 分析查询
        complexity = self.strategy.analyze(query)
        config = self.strategy.get_config(complexity)

        # 2. 查询重写（如需）
        search_query = query
        if config.get("rewrite"):
            search_query = await self._rewrite(query)

        # 3. 检索
        docs = await self.vectorstore.asimilarity_search(search_query, k=config["k"])

        # 4. 图谱查询（如需）
        graph_info = ""
        if config.get("use_graph") and self.graph:
            graph_info = self._query_graph(query)

        # 5. 重排序（如需）
        if config.get("rerank"):
            docs = self._rerank(docs, query)

        # 6. 生成
        context = "\n".join(d.page_content for d in docs)
        if graph_info:
            context += f"\n[图谱] &#123;graph_info&#125;"

        from langchain_core.messages import HumanMessage
        response = await self.llm.ainvoke([HumanMessage(
            content=f"基于以下信息回答:\n&#123;context[:2000]&#125;\n\n问题: &#123;query&#125;"
        )])

        # 7. 验证（如需）
        answer = response.content
        if config.get("verify"):
            answer = await self._verify(query, answer, context)

        return &#123;
            "answer": answer,
            "complexity": complexity.value,
            "config": config,
            "docs_retrieved": len(docs),
        &#125;

    async def _rewrite(self, query: str) -> str:
        """查询重写。"""
        from langchain_core.messages import HumanMessage
        response = await self.llm.ainvoke([HumanMessage(content=f"重写查询使其更适合检索: &#123;query&#125;\n只输出重写后的查询")])
        return response.content.strip()

    @staticmethod
    def _rerank(docs: list, query: str) -> list:
        """简化重排序——关键词重叠。"""
        query_words = set(query.lower().split())
        return sorted(docs, key=lambda d: len(query_words & set(d.page_content.lower().split())), reverse=True)

    def _query_graph(self, query: str) -> str:
        """图谱查询。"""
        return f"[图谱结果: &#123;query&#125;]"

    async def _verify(self, query: str, answer: str, context: str) -> str:
        """验证回答。"""
        from langchain_core.messages import HumanMessage
        prompt = f"检查回答是否正确基于上下文:\n问题:&#123;query&#125;\n回答:&#123;answer[:200]&#125;\n上下文:&#123;context[:200]&#125;\n有误请修正，无误原样返回:"
        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content
```

## 三、最佳实践

| 复杂度 | 策略 | 延迟 |
|--------|------|------|
| 简单 | 快速RAG | 低 |
| 事实 | 检索+重排+验证 | 中 |
| 复杂 | 重写+重排+验证 | 高 |
| 关系 | 图谱+重排+验证 | 高 |
