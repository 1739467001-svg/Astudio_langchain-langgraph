# 模型降级链与断路器集成指南

> gpt-4o 限流了→切 gpt-4o-mini→还是失败→切 gpt-3.5-turbo→再失败→返回缓存。这条降级链是 LLM 应用的高可用基石。这篇指南讲透降级链编排、断路器模式集成和自动恢复。

---

## 一、降级链+断路器架构

```mermaid
graph TB
    REQ["调用请求"] --> CB1&#123;"断路器: gpt-4o<br/>状态?"&#125;
    CB1 -->|CLOSED| M1["gpt-4o"]
    CB1 -->|OPEN| SKIP1["跳过→下一级"]
    M1 --> R1&#123;"成功?"&#125;
    R1 -->|是| OK["返回结果"]
    R1 -->|否| TRIP1["熔断+记录"]

    SKIP1 & TRIP1 --> CB2&#123;"断路器: gpt-4o-mini"&#125;
    CB2 -->|CLOSED| M2["gpt-4o-mini"]
    CB2 -->|OPEN| SKIP2["跳过→下一级"]
    M2 --> R2&#123;"成功?"&#125;
    R2 -->|是| OK
    R2 -->|否| TRIP2["熔断+记录"]

    SKIP2 & TRIP2 --> CB3&#123;"断路器: gpt-3.5"&#125;
    CB3 -->|CLOSED| M3["gpt-3.5-turbo"]
    CB3 -->|OPEN| CACHE["返回缓存/默认值"]

    style CB1 fill:#FFF9C4,stroke:#F9A825,stroke-width:3px
    style OK fill:#C8E6C9
    style CACHE fill:#FFE0B2,stroke:#E65100
```

断路器三种状态：CLOSED（正常放行）、OPEN（熔断，直接跳过）、HALF_OPEN（半开，试探性放行一个请求）。

## 二、降级链实现

```python
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import Enum
from typing import Any, Optional, Callable, Awaitable
import asyncio
import time

class CircuitState(str, Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"           # 熔断
    HALF_OPEN = "half_open" # 半开恢复

@dataclass
class CircuitConfig:
    """断路器配置。"""
    failure_threshold: int = 3        # 连续失败次数阈值
    recovery_timeout: float = 30.0    # 熔断后恢复等待(秒)
    half_open_max_calls: int = 1      # 半开状态试探请求数
    success_threshold: int = 2        # 半开状态连续成功恢复阈值

class CircuitBreaker:
    """断路器。"""

    def __init__(self, name: str, config: CircuitConfig = CircuitConfig()):
        self.name = name
        self.config = config
        self.state = CircuitState.CLOSED
        self._failure_count = 0
        self._success_count = 0
        self._last_failure_time: float = 0
        self._half_open_calls = 0

    def can_execute(self) -> bool:
        """是否可以执行。"""
        if self.state == CircuitState.CLOSED:
            return True
        elif self.state == CircuitState.OPEN:
            # 检查是否到了恢复时间
            if time.monotonic() - self._last_failure_time > self.config.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self._half_open_calls = 0
                return True
            return False
        else:  # HALF_OPEN
            if self._half_open_calls < self.config.half_open_max_calls:
                self._half_open_calls += 1
                return True
            return False

    def record_success(self):
        """记录成功。"""
        if self.state == CircuitState.HALF_OPEN:
            self._success_count += 1
            if self._success_count >= self.config.success_threshold:
                self._reset()
        else:
            self._failure_count = 0

    def record_failure(self):
        """记录失败。"""
        self._last_failure_time = time.monotonic()
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.OPEN
            self._failure_count = 0
            self._success_count = 0
        else:
            self._failure_count += 1
            if self._failure_count >= self.config.failure_threshold:
                self.state = CircuitState.OPEN

    def _reset(self):
        self.state = CircuitState.CLOSED
        self._failure_count = 0
        self._success_count = 0
        self._half_open_calls = 0

    def get_state(self) -> dict:
        return &#123;
            "name": self.name,
            "state": self.state.value,
            "failures": self._failure_count,
            "successes": self._success_count,
        &#125;


@dataclass
class FallbackStep:
    """降级链步骤。"""
    name: str
    handler: Callable[[], Awaitable[Any]]
    circuit: CircuitBreaker
    is_cache: bool = False  # 是否为缓存兜底

class FallbackChain:
    """降级链。"""

    def __init__(self):
        self._steps: list[FallbackStep] = []
        self._execution_log: list[dict] = []

    def add(self, name: str, handler: Callable[[], Awaitable[Any]],
            circuit_config: CircuitConfig = CircuitConfig()):
        """添加降级步骤。"""
        circuit = CircuitBreaker(name, circuit_config)
        self._steps.append(FallbackStep(name=name, handler=handler, circuit=circuit))
        return self

    def add_cache(self, name: str, handler: Callable[[], Awaitable[Any]]):
        """添加缓存兜底（无断路器）。"""
        circuit = CircuitBreaker(name, CircuitConfig(failure_threshold=999))
        self._steps.append(FallbackStep(name=name, handler=handler, circuit=circuit, is_cache=True))
        return self

    async def execute(self) -> dict:
        """执行降级链。"""
        for step in self._steps:
            if not step.circuit.can_execute():
                self._log(step.name, "skipped", "断路器开启，跳过")
                continue

            try:
                start = time.monotonic()
                result = await step.handler()
                latency = (time.monotonic() - start) * 1000

                step.circuit.record_success()
                self._log(step.name, "success", f"耗时&#123;latency:.0f&#125;ms")
                return &#123;
                    "success": True,
                    "used_model": step.name,
                    "result": result,
                    "latency_ms": round(latency, 1),
                    "is_cache": step.is_cache,
                &#125;
            except Exception as e:
                step.circuit.record_failure()
                self._log(step.name, "failed", str(e)[:100])
                continue

        return &#123;"success": False, "error": "所有降级步骤均失败", "log": self._execution_log&#125;

    def _log(self, step: str, status: str, detail: str):
        entry = &#123;"step": step, "status": status, "detail": detail, "timestamp": datetime.now().isoformat()&#125;
        self._execution_log.append(entry)

    def get_circuit_states(self) -> list[dict]:
        return [s.circuit.get_state() for s in self._steps]

    def get_log(self) -> list[dict]:
        return self._execution_log
```

### 使用示例

```python
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

# 模拟三个模型调用
async def call_gpt4o():
    llm = ChatOpenAI(model="gpt-4o", temperature=0)
    return await llm.ainvoke([HumanMessage(content="回答: 什么是AI")])

async def call_gpt4o_mini():
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
    return await llm.ainvoke([HumanMessage(content="回答: 什么是AI")])

async def call_gpt35():
    llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)
    return await llm.ainvoke([HumanMessage(content="回答: 什么是AI")])

async def return_cache():
    return "AI是人工智能的缩写，指通过计算机模拟人类智能行为的技术。（缓存）"

# 构建降级链
chain = FallbackChain()
chain.add("gpt-4o", call_gpt4o, CircuitConfig(failure_threshold=3, recovery_timeout=30))
chain.add("gpt-4o-mini", call_gpt4o_mini, CircuitConfig(failure_threshold=5, recovery_timeout=20))
chain.add("gpt-3.5-turbo", call_gpt35, CircuitConfig(failure_threshold=5, recovery_timeout=15))
chain.add_cache("cache", return_cache)

# 执行
result = await chain.execute()
print(f"使用: &#123;result['used_model']&#125;, 缓存: &#123;result.get('is_cache', False)&#125;")
print(f"断路器状态: &#123;chain.get_circuit_states()&#125;")
```

## 三、降级策略对比

| 策略 | 方式 | 恢复速度 | 数据质量 | 适用 |
|------|------|----------|----------|------|
| 模型降级 | gpt-4o→mini→3.5 | 快 | 中高 | 通用 |
| 缓存兜底 | 返回历史结果 | 极快 | 中 | 重复查询 |
| 队列重试 | 放入队列稍后重试 | 慢 | 高 | 非实时 |
| 降级功能 | 去掉部分功能 | 快 | 低 | 可接受降级 |
| 拒绝服务 | 返回503 | 即时 | 无 | 最后手段 |

## 四、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 每模型独立断路器 | 一个熔断不影响其他 | ★★★ |
| 缓存作为最终兜底 | 至少返回个东西 | ★★★ |
| 半开试探恢复 | 不直接全量恢复 | ★★★ |
| 降级链有日志 | 可追溯走了哪条路 | ★★☆ |
| 恢复时间递减 | 高级模型恢复快 | ★★☆ |
| 监控各段成功率 | 评估降级链有效性 | ★★☆ |
