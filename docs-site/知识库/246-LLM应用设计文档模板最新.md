## 二、评审检查

```python
class DesignDocReviewer:
    """设计文档评审器。"""

    MUST_HAVE = [
        "目标可量化",
        "有架构图",
        "有技术栈选择",
        "有数据安全方案",
        "有降级策略",
        "有SLO定义",
        "有风险缓解",
        "有里程碑",
    ]

    @classmethod
    def review(cls, doc_content: str) -> dict:
        """检查文档是否包含必要内容。"""
        missing = []
        for item in cls.MUST_HAVE:
            # 简化检查
            keywords = &#123;
                "目标可量化": ["成功标准", "量化", "指标"],
                "有架构图": ["mermaid", "graph", "架构"],
                "有技术栈选择": ["LangGraph", "技术栈", "模型"],
                "有数据安全方案": ["脱敏", "权限", "安全"],
                "有降级策略": ["降级", "fallback", "备用"],
                "有SLO定义": ["SLO", "可用性", "延迟"],
                "有风险缓解": ["风险", "缓解", "措施"],
                "有里程碑": ["里程碑", "MVP", "阶段"],
            &#125;
            found = any(kw in doc_content for kw in keywords.get(item, [item]))
            if not found:
                missing.append(item)

        return &#123;
            "complete": len(missing) == 0,
            "missing": missing,
            "score": round(1 - len(missing) / len(cls.MUST_HAVE), 2),
        &#125;
```
