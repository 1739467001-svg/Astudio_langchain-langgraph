# LLM 应用全链路压测与性能基准指南

> 单接口测通了不等于系统能扛住真实流量。全链路压测模拟真实用户行为——并发请求、混合场景、逐步加压、长时间持续——找出系统在什么负载下开始降级、什么点会崩溃。性能基准则把压测结果固化为可追踪的指标基线。

## 二、压测场景与配置

```python
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from typing import Literal
import asyncio
import time
import random
from collections import defaultdict

class ScenarioType(Enum):
    SIMPLE_QA = "simple_qa"
    RAG_QUERY = "rag_query"
    AGENT_REASONING = "agent_reasoning"
    STREAMING = "streaming"

@dataclass
class LoadTestScenario:
    """压测场景定义"""
    name: str
    scenario_type: ScenarioType
    weight: int = 1              # 权重（占总流量百分比）
    estimated_latency_ms: float = 500.0  # 预估延迟
    estimated_tokens: int = 500  # 预估Token消耗

@dataclass
class LoadTestConfig:
    """压测配置"""
    scenarios: list[LoadTestScenario] = field(default_factory=list)
    ramp_up_seconds: int = 30     # 预热时间（逐步加压）
    hold_seconds: int = 120       # 持续时间
    ramp_down_seconds: int = 10   # 降压时间
    target_concurrency: int = 50  # 目标并发数
    max_requests: int = 10000    # 最大请求数

    @property
    def total_duration(self) -> int:
        return self.ramp_up_seconds + self.hold_seconds + self.ramp_down_seconds

@dataclass
class RequestResult:
    """单次请求结果"""
    scenario: str
    start_time: float
    end_time: float
    latency_ms: float
    success: bool
    error: str = ""
    tokens_used: int = 0

    @property
    def ttfb_ms(self) -> float:
        """Time to First Byte（近似为首字节延迟）"""
        return self.latency_ms * 0.3  # 模拟
```

场景按权重混合，配置控制加压曲线：预热 → 持续 → 降压。

## 三、压测引擎

```python
class LoadTestEngine:
    """全链路压测引擎"""

    def __init__(self, config: LoadTestConfig):
        self.config = config
        self.results: list[RequestResult] = []
        self._running = False

    def _select_scenario(self) -> LoadTestScenario:
        """按权重随机选择场景"""
        total_weight = sum(s.weight for s in self.config.scenarios)
        r = random.uniform(0, total_weight)
        cumulative = 0
        for scenario in self.config.scenarios:
            cumulative += scenario.weight
            if r <= cumulative:
                return scenario
        return self.config.scenarios[0]

    def _current_concurrency(self, elapsed: float) -> int:
        """根据时间计算当前并发数（梯形加压曲线）"""
        ramp = self.config.ramp_up_seconds
        hold = self.config.hold_seconds
        target = self.config.target_concurrency

        if elapsed < ramp:
            return int(target * elapsed / ramp)
        elif elapsed < ramp + hold:
            return target
        else:
            remaining = max(0, 1 - (elapsed - ramp - hold) / self.config.ramp_down_seconds)
            return int(target * remaining)

    async def _simulate_request(self, scenario: LoadTestScenario) -> RequestResult:
        """模拟单次请求（实际应调用目标系统）"""
        start = time.time()
        # 模拟延迟：基础延迟 + 随机抖动
        latency = scenario.estimated_latency_ms + random.uniform(-100, 200)
        latency = max(50, latency)
        await asyncio.sleep(latency / 1000)

        # 模拟错误率：5%失败
        success = random.random() > 0.05
        error = "" if success else "timeout"

        return RequestResult(
            scenario=scenario.name,
            start_time=start,
            end_time=time.time(),
            latency_ms=latency,
            success=success,
            error=error,
            tokens_used=scenario.estimated_tokens if success else 0
        )

    async def run(self) -> dict:
        """执行压测"""
        print(f"压测开始: 目标并发=&#123;self.config.target_concurrency&#125;, "
              f"持续时间=&#123;self.config.total_duration&#125;s")
        self._running = True
        start_time = time.time()
        tasks: set = set()

        while self._running:
            elapsed = time.time() - start_time
            if elapsed > self.config.total_duration:
                break
            if len(self.results) >= self.config.max_requests:
                break

            target = self._current_concurrency(elapsed)
            while len(tasks) < target and len(self.results) + len(tasks) < self.config.max_requests:
                scenario = self._select_scenario()
                task = asyncio.create_task(self._execute_and_record(scenario))
                tasks.add(task)
                task.add_done_callback(tasks.discard)

            await asyncio.sleep(0.05)  # 控制调度频率

        # 等待所有在途请求完成
        if tasks:
            await asyncio.gather(*tasks, return_exceptions=True)

        self._running = False
        return self._generate_report()

    async def _execute_and_record(self, scenario: LoadTestScenario) -> None:
        result = await self._simulate_request(scenario)
        self.results.append(result)

    def _generate_report(self) -> dict:
        """生成压测报告"""
        if not self.results:
            return &#123;"error": "无结果"&#125;

        total = len(self.results)
        success = sum(1 for r in self.results if r.success)
        latencies = sorted(r.latency_ms for r in self.results if r.success)
        tokens = sum(r.tokens_used for r in self.results)

        # 百分位延迟
        def percentile(data, p):
            if not data:
                return 0
            idx = int(len(data) * p / 100)
            return data[min(idx, len(data) - 1)]

        # 按场景分组统计
        scenario_stats = defaultdict(lambda: &#123;"count": 0, "success": 0, "latencies": []&#125;)
        for r in self.results:
            stats = scenario_stats[r.scenario]
            stats["count"] += 1
            if r.success:
                stats["success"] += 1
                stats["latencies"].append(r.latency_ms)

        return &#123;
            "total_requests": total,
            "success_count": success,
            "error_count": total - success,
            "error_rate": round((total - success) / total * 100, 2),
            "total_tokens": tokens,
            "latency_p50_ms": round(percentile(latencies, 50), 1),
            "latency_p90_ms": round(percentile(latencies, 90), 1),
            "latency_p99_ms": round(percentile(latencies, 99), 1),
            "avg_latency_ms": round(sum(latencies) / max(len(latencies), 1), 1),
            "throughput_rps": round(total / self.config.total_duration, 1),
            "scenario_breakdown": &#123;
                name: &#123;
                    "count": s["count"],
                    "success_rate": round(s["success"] / max(s["count"], 1) * 100, 1),
                    "avg_latency": round(sum(s["latencies"]) / max(len(s["latencies"]), 1), 1)
                &#125;
                for name, s in scenario_stats.items()
            &#125;
        &#125;
```

## 四、运行压测

```python
async def main():
    config = LoadTestConfig(
        scenarios=[
            LoadTestScenario("简单问答", ScenarioType.SIMPLE_QA, weight=30, estimated_latency_ms=300),
            LoadTestScenario("RAG检索", ScenarioType.RAG_QUERY, weight=40, estimated_latency_ms=600),
            LoadTestScenario("Agent推理", ScenarioType.AGENT_REASONING, weight=20, estimated_latency_ms=1500),
            LoadTestScenario("流式输出", ScenarioType.STREAMING, weight=10, estimated_latency_ms=800),
        ],
        ramp_up_seconds=5,
        hold_seconds=20,
        ramp_down_seconds=5,
        target_concurrency=20,
        max_requests=500
    )

    engine = LoadTestEngine(config)
    report = await engine.run()

    print("\n=== 压测报告 ===")
    print(f"总请求: &#123;report['total_requests']&#125;")
    print(f"成功率: &#123;100 - report['error_rate']&#125;%")
    print(f"错误率: &#123;report['error_rate']&#125;%")
    print(f"P50延迟: &#123;report['latency_p50_ms']&#125;ms")
    print(f"P90延迟: &#123;report['latency_p90_ms']&#125;ms")
    print(f"P99延迟: &#123;report['latency_p99_ms']&#125;ms")
    print(f"吞吐量: &#123;report['throughput_rps']&#125; RPS")
    print(f"总Token: &#123;report['total_tokens']&#125;")
    print("\n场景分解:")
    for name, stats in report["scenario_breakdown"].items():
        print(f"  &#123;name&#125;: &#123;stats['count']&#125;次, 成功率&#123;stats['success_rate']&#125;%, 平均&#123;stats['avg_latency']&#125;ms")

asyncio.run(main())
```

## 六、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 梯形加压 | 预热→持续→降压 | ★★★ |
| 场景按权重混合 | 模拟真实流量分布 | ★★★ |
| 百分位延迟 | P50/P90/P99而非平均值 | ★★★ |
| 与基线对比 | 每次发布后对比基线 | ★★★ |
| 长时间持续 | 至少5分钟持续压测 | ★★☆ |
| 监控资源水位 | CPU/内存/连接池 | ★★☆ |
