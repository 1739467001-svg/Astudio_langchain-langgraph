## 二、实现

```python
from langchain_core.messages import HumanMessage
from dataclasses import dataclass
from enum import Enum
import re

class Language(str, Enum):
    CHINESE = "zh"
    ENGLISH = "en"
    JAPANESE = "ja"
    KOREAN = "ko"
    MIXED = "mixed"
    UNKNOWN = "unknown"

class LanguageDetector:
    """语言检测器。"""

    @staticmethod
    def detect(text: str) -> Language:
        """检测文本语言。"""
        if not text.strip():
            return Language.UNKNOWN

        chinese = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        japanese = sum(1 for c in text if '\u3040' <= c <= '\u30ff')
        korean = sum(1 for c in text if '\uac00' <= c <= '\ud7af')
        ascii_alpha = sum(1 for c in text if c.isascii() and c.isalpha())

        total = max(sum([chinese, japanese, korean, ascii_alpha]), 1)
        non_zero = sum(1 for c in [chinese, japanese, korean, ascii_alpha] if c / total > 0.2)

        if non_zero > 1:
            return Language.MIXED
        if chinese / total > 0.3:
            return Language.CHINESE
        if japanese / total > 0.3:
            return Language.JAPANESE
        if korean / total > 0.3:
            return Language.KOREAN
        return Language.ENGLISH


class MultilingualRAG:
    """多语言RAG系统。"""

    LANGUAGE_CONFIG = &#123;
        Language.CHINESE: &#123;"embedding": "BAAI/bge-large-zh", "answer_lang": "中文", "instruction": "请用中文回答"&#125;,
        Language.ENGLISH: &#123;"embedding": "text-embedding-3-small", "answer_lang": "English", "instruction": "Please answer in English"&#125;,
        Language.JAPANESE: &#123;"embedding": "text-embedding-3-small", "answer_lang": "日本語", "instruction": "日本語で答えてください"&#125;,
    &#125;

    def __init__(self, llm, vectorstore):
        self.llm = llm
        self.vectorstore = vectorstore

    async def answer(self, query: str) -> str:
        """多语言问答。"""
        # 1. 检测语言
        lang = LanguageDetector.detect(query)
        config = self.LANGUAGE_CONFIG.get(lang, self.LANGUAGE_CONFIG[Language.ENGLISH])

        # 2. 检索（不区分语言）
        docs = await self.vectorstore.asimilarity_search(query, k=3)
        context = "\n".join(d.page_content for d in docs)

        # 3. 用检测到的语言回答
        prompt = f"""&#123;config['instruction']&#125;

参考信息:
&#123;context[:2000]&#125;

问题: &#123;query&#125;

回答:"""

        response = await self.llm.ainvoke([HumanMessage(content=prompt)])
        return response.content
```
