# LLM 应用多租户资源配额与限流指南

> 多租户 LLM 应用中，一个租户突发流量可能耗尽整个系统的 Token 配额和 API 速率限制。这篇指南讲透租户级限流、Token 配额管理和公平调度策略。

## 二、限流与配额实现

```python
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from collections import defaultdict, deque
import asyncio
import time
from enum import Enum

class QuotaType(str, Enum):
    TOKENS_PER_DAY = "tokens_per_day"
    REQUESTS_PER_MINUTE = "requests_per_minute"
    CONCURRENT = "concurrent"

@dataclass
class TenantQuota:
    """租户配额配置。"""
    tenant_id: str
    tier: str = "standard"  # free / standard / premium
    tokens_per_day: int = 1_000_000
    requests_per_minute: int = 60
    max_concurrent: int = 5
    priority: int = 1  # 调度优先级（数字大=高优先级）

@dataclass
class UsageRecord:
    """用量记录。"""
    tenant_id: str
    tokens_used: int
    request_time: float
    completed_at: float | None = None

class TokenBucket:
    """令牌桶——平滑限流。"""

    def __init__(self, capacity: float, refill_rate: float):
        self.capacity = capacity
        self.refill_rate = refill_rate  # tokens per second
        self.tokens = capacity
        self.last_refill = time.monotonic()
        self._lock = asyncio.Lock()

    async def consume(self, tokens: int = 1) -> bool:
        async with self._lock:
            now = time.monotonic()
            elapsed = now - self.last_refill
            self.tokens = min(self.capacity, self.tokens + elapsed * self.refill_rate)
            self.last_refill = now

            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False

class QuotaManager:
    """配额管理器——多维度资源控制。"""

    def __init__(self):
        self._quotas: dict[str, TenantQuota] = &#123;&#125;
        self._daily_usage: dict[str, int] = defaultdict(int)  # tenant_id -> tokens used today
        self._minute_usage: dict[str, deque] = defaultdict(deque)  # tenant_id -> request timestamps
        self._concurrent: dict[str, int] = defaultdict(int)  # tenant_id -> current concurrent count
        self._buckets: dict[str, TokenBucket] = &#123;&#125;
        self._last_reset_date = datetime.now().date()

    def register(self, quota: TenantQuota):
        self._quotas[quota.tenant_id] = quota
        # 令牌桶：容量=每分钟请求数，填充速率=每秒
        self._buckets[quota.tenant_id] = TokenBucket(
            capacity=quota.requests_per_minute,
            refill_rate=quota.requests_per_minute / 60,
        )

    async def check_and_acquire(self, tenant_id: str, estimated_tokens: int) -> dict:
        """检查并获取资源。"""
        quota = self._quotas.get(tenant_id)
        if not quota:
            return &#123;"allowed": False, "reason": "未注册租户"&#125;

        # 日重置检查
        today = datetime.now().date()
        if today != self._last_reset_date:
            self._daily_usage.clear()
            self._last_reset_date = today

        # 日Token配额检查
        if self._daily_usage[tenant_id] + estimated_tokens > quota.tokens_per_day:
            remaining = quota.tokens_per_day - self._daily_usage[tenant_id]
            return &#123;"allowed": False, "reason": f"日Token配额不足，剩余&#123;remaining&#125;"&#125;

        # 每分钟请求检查（令牌桶）
        bucket = self._buckets[tenant_id]
        if not await bucket.consume(1):
            return &#123;"allowed": False, "reason": "请求频率超限"&#125;

        # 并发检查
        if self._concurrent[tenant_id] >= quota.max_concurrent:
            return &#123;"allowed": False, "reason": "并发数超限"&#125;

        # 全部通过——预扣
        self._concurrent[tenant_id] += 1
        self._daily_usage[tenant_id] += estimated_tokens
        self._minute_usage[tenant_id].append(time.monotonic())

        return &#123;"allowed": True, "quota": quota&#125;

    def release(self, tenant_id: str, actual_tokens: int, estimated_tokens: int):
        """释放资源——实际用量与预估的差额。"""
        self._concurrent[tenant_id] -= 1
        # 修正预估偏差
        diff = estimated_tokens - actual_tokens
        self._daily_usage[tenant_id] = max(0, self._daily_usage[tenant_id] - diff)

    def get_usage(self, tenant_id: str) -> dict:
        """获取租户用量。"""
        quota = self._quotas.get(tenant_id)
        if not quota:
            return &#123;&#125;
        return &#123;
            "tenant_id": tenant_id,
            "tier": quota.tier,
            "tokens_used_today": self._daily_usage[tenant_id],
            "tokens_remaining": quota.tokens_per_day - self._daily_usage[tenant_id],
            "current_concurrent": self._concurrent[tenant_id],
            "max_concurrent": quota.max_concurrent,
            "utilization_pct": round(self._daily_usage[tenant_id] / quota.tokens_per_day * 100, 1),
        &#125;


class FairScheduler:
    """公平调度器——按优先级和等待时间排序。"""

    def __init__(self):
        self._queue: asyncio.PriorityQueue = asyncio.PriorityQueue()

    async def submit(self, tenant_id: str, priority: int, request_fn):
        """提交请求到公平队列。"""
        wait_start = time.monotonic()
        await self._queue.put((priority, wait_start, tenant_id, request_fn))

    async def process(self):
        """处理队列中的请求。"""
        priority, wait_start, tenant_id, request_fn = await self._queue.get()
        wait_time = time.monotonic() - wait_start
        result = await request_fn()
        self._queue.task_done()
        return &#123;"tenant_id": tenant_id, "wait_time": wait_time, "result": result&#125;
```

## 三、中间件集成

```python
from langchain_core.tools import tool
import functools

quota_manager = QuotaManager()

# 注册租户配额
quota_manager.register(TenantQuota("tenant_free", tier="free", tokens_per_day=100_000, requests_per_minute=10, max_concurrent=2))
quota_manager.register(TenantQuota("tenant_pro", tier="premium", tokens_per_day=5_000_000, requests_per_minute=120, max_concurrent=20))

async def llm_with_quota(tenant_id: str, messages: list, estimated_tokens: int = 1000):
    """带配额控制的LLM调用。"""
    check = await quota_manager.check_and_acquire(tenant_id, estimated_tokens)
    if not check["allowed"]:
        return &#123;"error": "rate_limited", "reason": check["reason"]&#125;

    try:
        response = await llm.ainvoke(messages)
        actual_tokens = len(str(response)) // 4  # 粗略估算
        quota_manager.release(tenant_id, actual_tokens, estimated_tokens)
        return response
    except Exception as e:
        quota_manager.release(tenant_id, 0, estimated_tokens)
        raise
```

## 四、配额策略对比

| 策略 | 维度 | 优点 | 缺点 | 适用场景 |
|------|------|------|------|----------|
| 日Token配额 | 每日总量 | 控制成本 | 突发流量无法处理 | 通用 |
| 令牌桶 | 每秒填充 | 平滑限流 | 需调参 | API限流 |
| 并发限制 | 同时在执行 | 防资源耗尽 | 不控总量 | 计算密集 |
| 公平调度 | 优先级+等待 | 大租户不饿死小租户 | 实现复杂 | 多租户SaaS |
| 自适应配额 | 动态调整 | 高峰自动放宽 | 调优困难 | 成熟系统 |

## 五、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 预估+修正 | 先预扣后修正实际用量 | ★★★ |
| 令牌桶平滑限流 | 防突发洪水 | ★★★ |
