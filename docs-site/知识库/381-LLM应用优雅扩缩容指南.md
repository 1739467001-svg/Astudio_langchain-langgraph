# LLM 应用优雅扩缩容指南

> 流量高峰时 Agent 响应变慢，低谷时实例闲置浪费。这篇指南讲透基于指标的自动扩缩容、优雅启动/关闭和连接排空，让 LLM 应用弹性且平稳。

---

## 一、扩缩容架构

## 二、扩缩容实现

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Optional
import asyncio
import time
from collections import deque

class InstanceStatus(str, Enum):
    STARTING = "starting"
    HEALTHY = "healthy"
    DRAINING = "draining"
    TERMINATED = "terminated"

@dataclass
class Instance:
    """应用实例。"""
    instance_id: str
    status: InstanceStatus = InstanceStatus.STARTING
    created_at: float = field(default_factory=time.monotonic)
    active_connections: int = 0
    total_requests: int = 0
    cpu_usage: float = 0.0
    memory_usage: float = 0.0

@dataclass
class ScaleConfig:
    """扩缩容配置。"""
    min_instances: int = 1
    max_instances: int = 10
    target_concurrent_per_instance: int = 5
    target_qps_per_instance: float = 10.0
    scale_up_threshold: float = 0.8    # 利用率>80%扩容
    scale_down_threshold: float = 0.3   # 利用率<30%缩容
    scale_up_cooldown: float = 60.0     # 扩容冷却时间
    scale_down_cooldown: float = 120.0  # 缩容冷却时间
    drain_timeout: float = 30.0        # 排空超时
    warmup_time: float = 10.0          # 预热时间

class AutoScaler:
    """自动扩缩容管理器。"""

    def __init__(self, config: ScaleConfig = ScaleConfig()):
        self.config = config
        self._instances: dict[str, Instance] = &#123;&#125;
        self._last_scale_up: float = 0
        self._last_scale_down: float = 0
        self._next_id: int = 0

    def _gen_id(self) -> str:
        self._next_id += 1
        return f"inst-&#123;self._next_id:03d&#125;"

    def initialize(self):
        """初始化最小实例数。"""
        for _ in range(self.config.min_instances):
            inst = Instance(instance_id=self._gen_id())
            inst.status = InstanceStatus.HEALTHY
            self._instances[inst.instance_id] = inst

    async def scale_up(self, count: int = 1) -> list[str]:
        """扩容。"""
        now = time.monotonic()
        if now - self._last_scale_up < self.config.scale_up_cooldown:
            return []  # 冷却中

        current = len([i for i in self._instances.values() if i.status == InstanceStatus.HEALTHY])
        if current + count > self.config.max_instances:
            count = self.config.max_instances - current
        if count <= 0:
            return []

        new_ids = []
        for _ in range(count):
            inst = Instance(instance_id=self._gen_id())
            self._instances[inst.instance_id] = inst
            # 模拟预热
            asyncio.create_task(self._warmup(inst))
            new_ids.append(inst.instance_id)

        self._last_scale_up = now
        return new_ids

    async def _warmup(self, inst: Instance):
        """实例预热。"""
        await asyncio.sleep(self.config.warmup_time)
        inst.status = InstanceStatus.HEALTHY

    async def scale_down(self, count: int = 1) -> list[str]:
        """缩容。"""
        now = time.monotonic()
        if now - self._last_scale_down < self.config.scale_down_cooldown:
            return []

        healthy = [i for i in self._instances.values() if i.status == InstanceStatus.HEALTHY]
        if len(healthy) - count < self.config.min_instances:
            count = len(healthy) - self.config.min_instances
        if count <= 0:
            return []

        # 选连接数最少的实例缩容
        healthy.sort(key=lambda i: i.active_connections)
        to_drain = healthy[:count]

        for inst in to_drain:
            inst.status = InstanceStatus.DRAINING
            asyncio.create_task(self._drain_and_terminate(inst))

        self._last_scale_down = now
        return [i.instance_id for i in to_drain]

    async def _drain_and_terminate(self, inst: Instance):
        """排空连接后终止。"""
        deadline = time.monotonic() + self.config.drain_timeout
        while inst.active_connections > 0 and time.monotonic() < deadline:
            await asyncio.sleep(1.0)

        inst.status = InstanceStatus.TERMINATED

    def decide_scaling(self, metrics: dict) -> dict:
        """根据指标决定是否扩缩容。"""
        healthy = [i for i in self._instances.values() if i.status == InstanceStatus.HEALTHY]
        if not healthy:
            return &#123;"action": "scale_up", "count": self.config.min_instances, "reason": "无健康实例"&#125;

        total_concurrent = sum(i.active_connections for i in healthy)
        avg_utilization = total_concurrent / (len(healthy) * self.config.target_concurrent_per_instance)

        if avg_utilization > self.config.scale_up_threshold:
            needed = int((avg_utilization - self.config.scale_up_threshold) * len(healthy) / self.config.scale_up_threshold) + 1
            return &#123;"action": "scale_up", "count": min(needed, 3), "reason": f"利用率&#123;avg_utilization:.0%&#125;超阈值"&#125;

        if avg_utilization < self.config.scale_down_threshold and len(healthy) > self.config.min_instances:
            excess = int((self.config.scale_down_threshold - avg_utilization) * len(healthy))
            return &#123;"action": "scale_down", "count": min(excess, 2), "reason": f"利用率&#123;avg_utilization:.0%&#125;低于阈值"&#125;

        return &#123;"action": "hold", "count": 0, "reason": f"利用率&#123;avg_utilization:.0%&#125;正常"&#125;

    def get_status(self) -> dict:
        healthy = [i for i in self._instances.values() if i.status == InstanceStatus.HEALTHY]
        draining = [i for i in self._instances.values() if i.status == InstanceStatus.DRAINING]
        starting = [i for i in self._instances.values() if i.status == InstanceStatus.STARTING]
        total_conns = sum(i.active_connections for i in healthy)
        return &#123;
            "total_instances": len(self._instances),
            "healthy": len(healthy),
            "starting": len(starting),
            "draining": len(draining),
            "total_connections": total_conns,
            "avg_utilization": round(total_conns / max(len(healthy) * self.config.target_concurrent_per_instance, 1) * 100, 1),
        &#125;
```

### 使用示例

```python
import asyncio

async def main():
    scaler = AutoScaler(ScaleConfig(min_instances=2, max_instances=8, target_concurrent_per_instance=5))
    scaler.initialize()
    print(f"初始状态: &#123;scaler.get_status()&#125;")

    # 模拟高负载
    for inst in [i for i in scaler._instances.values() if i.status == InstanceStatus.HEALTHY]:
        inst.active_connections = 6  # 超过target

    decision = scaler.decide_scaling(&#123;"concurrent": 12&#125;)
    print(f"扩容决策: &#123;decision&#125;")

    if decision["action"] == "scale_up":
        new_ids = await scaler.scale_up(decision["count"])
        print(f"新增实例: &#123;new_ids&#125;")

    # 等待预热
    await asyncio.sleep(11)
    print(f"扩容后: &#123;scaler.get_status()&#125;")

    # 模拟低负载
    for inst in [i for i in scaler._instances.values() if i.status == InstanceStatus.HEALTHY]:
        inst.active_connections = 0

    decision = scaler.decide_scaling(&#123;"concurrent": 0&#125;)
    print(f"缩容决策: &#123;decision&#125;")

    if decision["action"] == "scale_down":
        drained = await scaler.scale_down(decision["count"])
        print(f"排空实例: &#123;drained&#125;")

    await asyncio.sleep(2)
    print(f"最终: &#123;scaler.get_status()&#125;")

asyncio.run(main())
```

## 三、扩缩容触发条件

| 指标 | 扩容阈值 | 缩容阈值 | 冷却时间 |
|------|----------|----------|----------|
| 并发利用率 | >80% | <30% | 60s/120s |
| P95延迟 | >5s | - | 60s |
| 队列长度 | >50 | <5 | 30s |
| CPU使用率 | >75% | <25% | 60s |
| 错误率 | >2% | - | 30s |

## 四、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 有最小实例数 | 保证基线容量 | ★★★ |
| 冷却时间 | 防止频繁伸缩 | ★★★ |
| 优雅排空 | 缩容前等请求完成 | ★★★ |
| 预热后接流 | 新实例就绪探针通过才接流 | ★★★ |
| 多指标联合 | 不只看单一指标 | ★★☆ |
| 扩容快缩容慢 | 扩容立即，缩容慢 | ★★☆ |
