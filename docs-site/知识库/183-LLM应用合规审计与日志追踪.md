# LLM 应用合规审计与日志追踪

> 企业级 LLM 应用需要满足合规要求：谁问了什么、Agent 回了什么、调了什么工具、用了什么模型。这些都需要可审计、可追踪。这份指南覆盖审计日志、合规检查和数据治理。

## 二、审计日志设计

```python
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
import json
import hashlib

class AuditEventType(str, Enum):
    USER_INPUT = "user_input"       # 用户输入
    LLM_RESPONSE = "llm_response"   # LLM响应
    TOOL_CALL = "tool_call"          # 工具调用
    TOOL_RESULT = "tool_result"      # 工具结果
    MODEL_SELECTION = "model_select" # 模型选择
    ERROR = "error"                  # 错误
    POLICY_VIOLATION = "policy_violation"  # 策略违反
    DATA_ACCESS = "data_access"      # 数据访问

@dataclass
class AuditEvent:
    """审计事件。"""
    event_id: str
    event_type: AuditEventType
    user_id: str
    session_id: str
    timestamp: str
    input_data: str = ""         # 输入内容（脱敏后）
    output_data: str = ""        # 输出内容（脱敏后）
    model_used: str = ""        # 使用的模型
    tool_name: str = ""          # 工具名称
    tool_args: str = ""         # 工具参数（脱敏后）
    metadata: dict = field(default_factory=dict)
    content_hash: str = ""      # 内容哈希（完整性校验）

class AuditLogger:
    """审计日志记录器。

    每个操作都记录为不可变的审计事件，
    支持事后追溯和合规检查。
    """

    def __init__(self):
        self.events: list[AuditEvent] = []

    def log(
        self,
        event_type: AuditEventType,
        user_id: str,
        session_id: str,
        input_data: str = "",
        output_data: str = "",
        model_used: str = "",
        tool_name: str = "",
        tool_args: str = "",
        **metadata,
    ):
        """记录审计事件。"""
        # 脱敏处理
        sanitized_input = self._sanitize(input_data)
        sanitized_output = self._sanitize(output_data)
        sanitized_args = self._sanitize(tool_args)

        # 内容哈希
        content = f"&#123;sanitized_input&#125;&#123;sanitized_output&#125;&#123;sanitized_args&#125;"
        content_hash = hashlib.sha256(content.encode()).hexdigest()[:16]

        event = AuditEvent(
            event_id=f"audit_&#123;len(self.events) + 1:06d&#125;",
            event_type=event_type,
            user_id=user_id,
            session_id=session_id,
            timestamp=datetime.now().isoformat(),
            input_data=sanitized_input[:500],
            output_data=sanitized_output[:500],
            model_used=model_used,
            tool_name=tool_name,
            tool_args=sanitized_args[:200],
            metadata=metadata,
            content_hash=content_hash,
        )
        self.events.append(event)
        return event

    @staticmethod
    def _sanitize(text: str) -> str:
        """脱敏PII。"""
        import re
        text = re.sub(r'\b1[3-9]\d&#123;9&#125;\b', '[PHONE]', text)
        text = re.sub(r'\b\d&#123;17&#125;[\dXx]\b', '[ID]', text)
        text = re.sub(r'sk-[a-zA-Z0-9]&#123;40,&#125;', '[API_KEY]', text)
        return text

    def get_session_audit_trail(self, session_id: str) -> list[AuditEvent]:
        """获取某会话的完整审计轨迹。"""
        return [e for e in self.events if e.session_id == session_id]

    def get_user_audit_trail(self, user_id: str, limit: int = 100) -> list[AuditEvent]:
        """获取某用户的审计轨迹。"""
        user_events = [e for e in self.events if e.user_id == user_id]
        return user_events[-limit:]

    def search(
        self,
        event_type: AuditEventType | None = None,
        user_id: str | None = None,
        session_id: str | None = None,
        start_time: str | None = None,
        end_time: str | None = None,
    ) -> list[AuditEvent]:
        """审计日志搜索。"""
        results = self.events
        if event_type:
            results = [e for e in results if e.event_type == event_type]
        if user_id:
            results = [e for e in results if e.user_id == user_id]
        if session_id:
            results = [e for e in results if e.session_id == session_id]
        if start_time:
            results = [e for e in results if e.timestamp >= start_time]
        if end_time:
            results = [e for e in results if e.timestamp <= end_time]
        return results

    def export_audit_report(self, session_id: str) -> str:
        """导出审计报告。"""
        events = self.get_session_audit_trail(session_id)
        lines = [f"# 审计报告 - 会话 &#123;session_id&#125;\n"]

        for event in events:
            lines.append(f"\n## [&#123;event.event_type.value&#125;] &#123;event.timestamp&#125;")
            lines.append(f"  用户: &#123;event.user_id&#125;")
            if event.input_data:
                lines.append(f"  输入: &#123;event.input_data[:100]&#125;")
            if event.output_data:
                lines.append(f"  输出: &#123;event.output_data[:100]&#125;")
            if event.model_used:
                lines.append(f"  模型: &#123;event.model_used&#125;")
            if event.tool_name:
                lines.append(f"  工具: &#123;event.tool_name&#125;(&#123;event.tool_args[:50]&#125;)")
            lines.append(f"  哈希: &#123;event.content_hash&#125;")

        return "\n".join(lines)
```

## 三、合规检查器

```python
class ComplianceChecker:
    """合规检查器。"""

    CHECKS = &#123;
        "数据保留": [
            "对话历史保留期限已定义",
            "超出保留期自动删除",
            "用户可请求删除其数据",
        ],
        "数据脱敏": [
            "日志中不记录PII",
            "API Key不出现在日志中",
            "审计日志已脱敏",
        ],
        "访问控制": [
            "审计日志有访问控制",
            "用户数据隔离",
            "管理员操作有记录",
        ],
        "可追溯性": [
            "每个操作有审计记录",
            "审计日志不可篡改",
            "支持按用户/会话查询",
        ],
        "模型透明": [
            "记录使用的模型版本",
            "记录Prompt版本",
            "记录工具调用历史",
        ],
    &#125;

    @classmethod
    def audit(cls, status: dict) -> dict:
        """执行合规审计。"""
        results = &#123;&#125;
        total_pass = 0
        total = 0

        for category, checks in cls.CHECKS.items():
            passed = sum(1 for check in checks if status.get(check, False))
            total_pass += passed
            total += len(checks)
            results[category] = &#123;
                "passed": passed,
                "total": len(checks),
                "rate": round(passed / len(checks), 4),
                "items": &#123;check: status.get(check, False) for check in checks&#125;,
            &#125;

        return &#123;
            "overall_compliance": round(total_pass / total * 100, 1),
            "compliant": total_pass == total,
            "categories": results,
        &#125;
```

```python
class DataLifecycleManager:
    """数据生命周期管理器。"""

    RETENTION_DAYS = &#123;
        "audit_logs": 365,      # 审计日志保留1年
        "conversation_history": 90,  # 对话历史90天
        "vector_data": 0,       # 永久（除非手动删除）
        "user_feedback": 180,  # 反馈180天
    &#125;

    @classmethod
    def check_retention(cls, data_type: str, created_at: str) -> dict:
        """检查数据是否超过保留期。"""
        from datetime import datetime, timedelta
        retention = cls.RETENTION_DAYS.get(data_type, 90)

        if retention == 0:
            return &#123;"action": "keep", "reason": "永久保留"&#125;

        created = datetime.fromisoformat(created_at)
        expiry = created + timedelta(days=retention)

        if datetime.now() > expiry:
            return &#123;"action": "delete", "expiry_date": expiry.isoformat()&#125;
        else:
            return &#123;"action": "keep", "expiry_date": expiry.isoformat()&#125;
```

## 五、最佳实践

| 实践 | 说明 | 优先级 |
|------|------|--------|
| 每个操作都有审计日志 | 可追溯 | ★★★ |
| 审计日志脱敏 | PII不记录 | ★★★ |
| 审计日志不可篡改 | 内容哈希校验 | ★★★ |
| 定义数据保留期 | 合规要求 | ★★★ |
| 用户可删除数据 | GDPR权利 | ★★☆ |
| 定期合规审计 | 发现缺口 | ★★☆ |
