## 二、实现

```python
import hashlib
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum

class ChangeType(str, Enum):
    CREATE = "create"
    UPDATE = "update"
    DELETE = "delete"
    NONE = "none"

@dataclass
class FileChange:
    """文件变更记录。"""
    file_path: str
    change_type: ChangeType
    old_hash: str = ""
    new_hash: str = ""
    timestamp: str = field(default_factory=lambda: datetime.now().isoformat())

class IncrementalUpdateManager:
    """增量更新管理器。"""

    def __init__(self, vectorstore, embeddings):
        self.vectorstore = vectorstore
        self.embeddings = embeddings
        self.file_hashes: dict[str, str] = &#123;&#125;  # path→content_hash
        self.changes: list[FileChange] = []

    def detect_changes(self, current_files: dict[str, str]) -> list[FileChange]:
        """检测文件变更。

        Args:
            current_files: &#123;file_path: file_content&#125;
        """
        changes = []
        current_paths = set(current_files.keys())
        stored_paths = set(self.file_hashes.keys())

        # 新增
        for path, content in current_files.items():
            new_hash = hashlib.md5(content.encode()).hexdigest()
            if path not in self.file_hashes:
                changes.append(FileChange(
                    file_path=path, change_type=ChangeType.CREATE,
                    new_hash=new_hash,
                ))
            elif self.file_hashes[path] != new_hash:
                changes.append(FileChange(
                    file_path=path, change_type=ChangeType.UPDATE,
                    old_hash=self.file_hashes[path], new_hash=new_hash,
                ))

        # 删除
        for path in stored_paths - current_paths:
            changes.append(FileChange(
                file_path=path, change_type=ChangeType.DELETE,
                old_hash=self.file_hashes[path],
            ))

        self.changes.extend(changes)
        return changes

    async def apply_changes(self, changes: list[FileChange], content_map: dict[str, str]) -> dict:
        """应用变更。"""
        created, updated, deleted = 0, 0, 0

        for change in changes:
            if change.change_type == ChangeType.CREATE:
                content = content_map.get(change.file_path, "")
                if content:
                    # 嵌入+入库
                    created += 1
                    self.file_hashes[change.file_path] = change.new_hash

            elif change.change_type == ChangeType.UPDATE:
                # 删旧+重新嵌入
                updated += 1
                self.file_hashes[change.file_path] = change.new_hash

            elif change.change_type == ChangeType.DELETE:
                # 删向量
                deleted += 1
                self.file_hashes.pop(change.file_path, None)

        return &#123;
            "created": created,
            "updated": updated,
            "deleted": deleted,
            "total_changes": len(changes),
            "skipped": len(content_map) - created - updated,
        &#125;

    def stats(self) -> dict:
        return &#123;
            "tracked_files": len(self.file_hashes),
            "total_changes": len(self.changes),
            "last_change": self.changes[-1].timestamp if self.changes else None,
        &#125;
```

## 三、缓存一致性

```python
class CacheConsistencyManager:
    """增量更新后的缓存一致性管理。"""

    def __init__(self):
        self.invalidated: int = 0

    def on_documents_updated(self, doc_ids: list[str]):
        """文档更新时失效相关缓存。"""
        self.invalidated += len(doc_ids)

    def on_full_rebuild(self):
        """全量重建时清除所有缓存。"""
        self.invalidated += 1  # 全量标记

    def stats(self) -> dict:
        return &#123;"cache_invalidations": self.invalidated&#125;
```
