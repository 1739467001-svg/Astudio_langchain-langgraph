## 三、State 定义

```python
from typing import TypedDict, Annotated
from langgraph.graph.message import add_messages
from enum import Enum

class Intent(str, Enum):
    FAQ = "faq"
    ORDER_QUERY = "order_query"
    COMPLAINT = "complaint"
    CHITCHAT = "chitchat"
    TRANSFER = "transfer_human"
    UNKNOWN = "unknown"

class CustomerServiceState(TypedDict):
    messages: Annotated[list, add_messages]
    user_id: str
    intent: str
    context: dict          # 对话上下文（订单号、问题类型等）
    need_human: bool       # 是否需要转人工
    satisfaction: str      # 满意度: good/neutral/bad
    turn_count: int
```

## 四、节点实现

### 4.1 意图分类

```python
from langchain_core.messages import HumanMessage, SystemMessage
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

INTENT_PROMPT = """判断用户消息的意图。

用户消息: &#123;message&#125;

可选意图:
- faq: 常见问题咨询（产品介绍、使用方法、政策等）
- order_query: 订单查询（查物流、查状态、查详情）
- complaint: 投诉建议（投诉服务、建议改进、报问题）
- chitchat: 闲聊（问候、感谢、无关对话）
- transfer_human: 明确要求转人工

只输出意图名称:"""

async def classify_intent(state: CustomerServiceState) -> dict:
    """意图分类节点。"""
    last_msg = state["messages"][-1].content
    prompt = INTENT_PROMPT.format(message=last_msg)
    response = await llm.ainvoke([HumanMessage(content=prompt)])
    intent = response.content.strip().lower()

    # 验证意图合法
    valid_intents = [i.value for i in Intent]
    if intent not in valid_intents:
        intent = Intent.UNKNOWN.value

    return &#123;"intent": intent, "turn_count": state.get("turn_count", 0) + 1&#125;
```

### 4.2 FAQ RAG 节点

```python
async def faq_node(state: CustomerServiceState, vectorstore) -> dict:
    """FAQ节点：RAG检索+生成。"""
    question = state["messages"][-1].content

    # 检索知识库
    docs = await vectorstore.asimilarity_search(question, k=3)
    context = "\n\n".join(d.page_content for d in docs)

    # 生成回答
    prompt = f"""你是客服助手。基于以下知识库信息回答用户问题。

知识库:
&#123;context[:2000]&#125;

用户问题: &#123;question&#125;

回答要求：
1. 基于知识库信息回答
2. 如果知识库中没有相关信息，说明并建议转人工
3. 简洁明了"""

    response = await llm.ainvoke([
        SystemMessage(content="你是专业客服助手。"),
        HumanMessage(content=prompt),
    ])

    return &#123;"messages": [response], "context": &#123;"retrieved_docs": len(docs)&#125;&#125;
```

### 4.3 投诉处理节点（DST）

```python
from langgraph.types import interrupt

COMPLAINT_PROMPT = """你是客服投诉处理助手。请按流程处理投诉。

当前已收集信息: &#123;collected&#125;
用户最新消息: &#123;message&#125;

如果信息不完整，请询问缺失的信息：
- 订单号
- 问题描述
- 期望处理方式

如果信息完整，确认并告知处理流程。

回答:"""

async def complaint_node(state: CustomerServiceState) -> dict:
    """投诉处理节点：收集信息→确认→处理。"""
    context = state.get("context", &#123;&#125;)
    collected = context.get("complaint_info", &#123;&#125;)
    message = state["messages"][-1].content

    prompt = COMPLAINT_PROMPT.format(
        collected=str(collected),
        message=message,
    )

    response = await llm.ainvoke([HumanMessage(content=prompt)])

    # 更新收集的信息（简化版）
    if "订单号" not in collected and any(c.isdigit() for c in message):
        collected["order_id"] = message

    # 检查是否需要转人工
    if "人工" in message or "经理" in message:
        return &#123;
            "messages": [response],
            "need_human": True,
            "context": &#123;"complaint_info": collected&#125;,
        &#125;

    return &#123;
        "messages": [response],
        "context": &#123;"complaint_info": collected&#125;,
    &#125;
```

### 4.4 人工转接节点

```python
async def transfer_human_node(state: CustomerServiceState) -> dict:
    """人工转接节点：携带上下文。"""
    # 汇总对话上下文
    messages = state["messages"]
    summary = f"用户ID: &#123;state.get('user_id', '未知')&#125;\n"
    summary += f"意图: &#123;state.get('intent', '未知')&#125;\n"
    summary += f"对话轮数: &#123;state.get('turn_count', 0)&#125;\n"
    summary += f"已收集信息: &#123;state.get('context', &#123;&#125;)&#125;\n"
    summary += f"对话摘要: &#123;messages[-1].content[:200] if messages else '无'&#125;"

    return &#123;
        "messages": [AIMessage(
            content="正在为您转接人工客服，请稍候...\n\n"
                    f"[转接信息已发送给客服]\n&#123;summary&#125;"
        )],
        "need_human": True,
    &#125;
```

### 4.5 满意度收集

```python
async def satisfaction_node(state: CustomerServiceState) -> dict:
    """满意度收集节点。"""
    from langchain_core.messages import AIMessage

    last_msg = state["messages"][-1].content.lower()

    # 检测用户是否表达满意/不满意
    if any(w in last_msg for w in ["谢谢", "好的", "明白了", "解决了"]):
        satisfaction = "good"
    elif any(w in last_msg for w in ["没用", "不行", "没解决", "转人工"]):
        satisfaction = "bad"
    else:
        satisfaction = "neutral"

    return &#123;"satisfaction": satisfaction&#125;
```

## 五、组装图

```python
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import AIMessage

def build_customer_service(vectorstore):
    """构建智能客服图。"""
    graph = StateGraph(CustomerServiceState)

    # 注册节点
    graph.add_node("classify", classify_intent)
    graph.add_node("faq", lambda s: faq_node(s, vectorstore))
    graph.add_node("complaint", complaint_node)
    graph.add_node("transfer", transfer_human_node)
    graph.add_node("satisfaction", satisfaction_node)

    # 路由：分类后按意图分发
    def route_by_intent(state: CustomerServiceState) -> str:
        intent = state.get("intent", "unknown")
        if intent == "faq":
            return "faq"
        elif intent == "complaint":
            return "complaint"
        elif intent == "transfer_human":
            return "transfer"
        else:
            return "faq"  # 兜底走FAQ

    graph.add_edge(START, "classify")
    graph.add_conditional_edges("classify", route_by_intent, &#123;
        "faq": "faq",
        "complaint": "complaint",
        "transfer": "transfer",
    &#125;)

    # FAQ和投诉后检查是否需要转人工
    def check_transfer(state: CustomerServiceState) -> str:
        if state.get("need_human"):
            return "transfer"
        return "satisfaction"

    graph.add_conditional_edges("faq", check_transfer, &#123;
        "transfer": "transfer",
        "satisfaction": "satisfaction",
    &#125;)
    graph.add_conditional_edges("complaint", check_transfer, &#123;
        "transfer": "transfer",
        "satisfaction": "satisfaction",
    &#125;)

    graph.add_edge("satisfaction", END)
    graph.add_edge("transfer", END)

    return graph.compile()

# 创建客服系统
customer_service = build_customer_service(vectorstore)
```

## 六、使用示例

```python
import asyncio

async def main():
    config = &#123;"configurable": &#123;"thread_id": "user-001"&#125;&#125;

    # 第一轮：FAQ
    result = await customer_service.ainvoke(&#123;
        "messages": [&#123;"role": "user", "content": "你们的退货政策是什么？"&#125;],
        "user_id": "user-001",
        "intent": "", "context": &#123;&#125;, "need_human": False,
        "satisfaction": "", "turn_count": 0,
    &#125;, config)
    print(f"客服: &#123;result['messages'][-1].content[:100]&#125;")

    # 第二轮：投诉
    result = await customer_service.ainvoke(&#123;
        "messages": [&#123;"role": "user", "content": "我下的订单还没到，要投诉！"&#125;],
    &#125;, config)
    print(f"客服: &#123;result['messages'][-1].content[:100]&#125;")

    # 第三轮：要求转人工
    result = await customer_service.ainvoke(&#123;
        "messages": [&#123;"role": "user", "content": "我要找人工客服"&#125;],
    &#125;, config)
    print(f"客服: &#123;result['messages'][-1].content[:100]&#125;")
    print(f"需转人工: &#123;result.get('need_human')&#125;")

asyncio.run(main())
```
